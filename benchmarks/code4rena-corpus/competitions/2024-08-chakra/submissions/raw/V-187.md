# Vulnerability details

## Impact
Accepting all ERC20 by the bridge is not safe. Malicious user can create ERC20 token contract that will be harmful for the protocol. For example it is possible to create ERC20 token that will be very gas expensive when making transfer. This can burn out funds out of contract, cause DoS for other users, block bridge etc.

Here is example contract that can simulate malicious ERC20 token
```solidity
// SPDX-License-Identifier: MIT
pragma solidity ^0.8.24;

import "@openzeppelin/contracts/token/ERC20/ERC20.sol";

contract GasExpensiveToken is ERC20 {

    constructor() ERC20("GasExpensiveToken", "GET") {
        _mint(msg.sender, 1e24);
    }

    // Override the transfer function to include gas-heavy logic
    function transfer(address recipient, uint256 amount) public virtual override returns (bool) {
        // Simulate a gas-expensive operation
        for (uint256 i = 0; i < 1000; i++) {
            // Do some work that takes up gas
            uint256 x = i * i;
            x = x / (i + 1);
        }

        // Call the original transfer function
        return super.transfer(recipient, amount);
    }

    // Override the transferFrom function to include gas-heavy logic
    function transferFrom(address sender, address recipient, uint256 amount) public virtual override returns (bool) {
        // Simulate a gas-expensive operation
        for (uint256 i = 0; i < 1000; i++) {
            // Do some work that takes up gas
            uint256 x = i * i;
            x = x / (i + 1);
        }

        // Call the original transferFrom function
        return super.transferFrom(sender, recipient, amount);
    }
}
```
