# Vulnerability details

## Impact
`function verifyECDSA` in `SettlementSignatureVerifier.sol` doesn't verify signatures

## Proof of Concept
First, to make `function verifyECDSA` to verify signatures we should use singed message hash:

    function verifyECDSA(
        bytes32 msgHash,
        bytes calldata signatures
    ) internal view returns (bool) {
        require(
            signatures.length % 65 == 0,
            "Signature length must be a multiple of 65"
        );

        uint256 len = signatures.length;
        uint256 m = 0;
        bytes32 signedMsgHash = MessageHashUtils.toEthSignedMessageHash(msgHash);
        for (uint256 i = 0; i < len; i += 65) {
            bytes memory sig = signatures[i:i + 65];
            if (
                validators[signedMsgHash.recover(sig)] && ++m >=required_validators
            ) {
                return true;
            }
        }

        return false;
    }

Second, even one validator signature been duplicated `requiredValidators` times is enough to bypass verification:

            const [validator1, manager] = await hre.ethers.getSigners();
            const { verifyInstance, verifierOwner } = await loadFixture(deploySettlementFixture)
            await verifyInstance.connect(verifierOwner).add_manager(await manager.getAddress());
            const messageHash = hre.ethers.solidityPackedKeccak256(["address", "string"], [await verifyInstance.getAddress(), "test message"])

            let signatures = Buffer.from([]);

## Recommended Mitigation Steps
Consider implementing duplicate filter:

    function verifyECDSA(
            bytes32 msgHash,
            bytes calldata signatures
        ) internal view returns (bool) {
            require(
                signatures.length % 65 == 0,
                "Signature length must be a multiple of 65"
            );

            uint256 numSignatures = signatures.length / 65;
            uint256 m = 0;
            bytes32[] memory hashedSignatures = new bytes32[](numSignatures);
            bytes32 signedMsgHash = MessageHashUtils.toEthSignedMessageHash(msgHash);
            for (uint256 i = 0; i < numSignatures; i++) {
                bytes memory sig = signatures[i*65:i*65 + 65];
                bytes32 hashedSignature = keccak256(abi.encodePacked(sig));
                for (uint256 n = 0; n < numSignatures; n++) {
                    require(hashedSignatures[n] != hashedSignature, "Signature duplicate");
                }
                hashedSignatures[i] = hashedSignature;
                require(validators[signedMsgHash.recover(sig)], "Not a validator" );
                if (
                    ++m >= required_validators
                ) {
                    return true;
                }
            }

            return false;
        }
