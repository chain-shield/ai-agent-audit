# Vulnerability details

## Impact

A single malicious or compromised validator can self-sign and execute any signature-threshold protected call arbitrary arguments of its choice.

Impacts send_cross_chain_msg, receive_cross_chain_callback, receive_cross_chain_msg, allowing the evil validator to call them with any argument of its choice and in any order.

## Proof of Concept

In the following excerpt form SettlementSignatureVerifier, the line `validators[msgHash.recover(sig)] && ++m >= required_validators` increments and compare the threshold counter if the signature resolves to a `validator`'s address.
However it fails to verify that the signatures resolve to unique validators. For this reason, a single validator can produce enough signatures to match or exceed any required threshold, thus allowing to forge and sign independently any message.

```solidity
    /**
     * @dev Function to verify an ECDSA signature
     * @notice Verifies an ECDSA signature with a given message hash and signatures
     * @param msgHash The hash of the message
     * @param signatures The signatures of the message
     * @return A boolean indicating if the signature is verified
     */
    function verifyECDSA(
        bytes32 msgHash,
        bytes calldata signatures
    ) internal view returns (bool) {
        require(
            signatures.length % 65 == 0,
            "Signature length must be a multiple of 65"
        );

        uint256 len = signatures.length;
        uint256 m = 0;
        for (uint256 i = 0; i < len; i += 65) {
            bytes memory sig = signatures[i:i + 65];
            if (
                validators[msgHash.recover(sig)] && ++m >= required_validators
            ) {
                return true;
            }
        }

        return false;
    }

    /**
     * @dev Function to verify a message
     * @notice Verifies a message with a given hash, signatures, and sign type
     * @param msgHash The hash of the message
     * @param signatures The signatures of the message
     * @param sign_type The type of the signature
     * @return A boolean indicating if the message is verified
     */
    function verify(
        bytes32 msgHash,
        bytes calldata signatures,
        uint8 sign_type
    ) external view returns (bool) {
        if (sign_type == 0) {
            return verifyECDSA(msgHash, signatures);
        } else {
            return false;
        }
    }
```

## Recommended Mitigation Steps

Ensure signatures are form unique signers.

```solidity
/**
 * @dev Function to verify an ECDSA signature
 * @notice Verifies an ECDSA signature with a given message hash and signatures, ensuring unique signers
 * @param msgHash The hash of the message
 * @param signatures The signatures of the message
 * @return A boolean indicating if the signature is verified
 */
function verifyECDSA(
    bytes32 msgHash,
    bytes calldata signatures
) internal view returns (bool) {
    require(
        signatures.length % 65 == 0,
        "Signature length must be a multiple of 65"
    );

    uint256 len = signatures.length;
    uint256 m = 0;
    address[] memory seenSigners = new address[](required_validators); // To track unique signers
    uint256 seenCount = 0;

    for (uint256 i = 0; i < len; i += 65) {
        bytes memory sig = signatures[i:i + 65];
        address signer = msgHash.recover(sig);

        // Ensure signer is a valid validator and hasn't already signed
        if (validators[signer] && !isSignerSeen(seenSigners, seenCount, signer)) {
            seenSigners[seenCount++] = signer;
            if (++m >= required_validators) {
                return true;
            }
        }
    }

    return false;
}

/**
 * @dev Internal function to check if a signer has already been seen
 * @param signers The list of already seen signers
 * @param count The number of signers seen so far
 * @param signer The current signer to check
 * @return A boolean indicating if the signer has already been seen
 */
function isSignerSeen(
    address[] memory signers,
    uint256 count,
    address signer
) internal pure returns (bool) {
    for (uint256 i = 0; i < count; i++) {
        if (signers[i] == signer) {
            return true;
        }
    }
    return false;
}
```
