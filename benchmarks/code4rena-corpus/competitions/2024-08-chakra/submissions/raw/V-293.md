# Vulnerability details

## Impact

Solidity's `settlement.receive_cross_chain_msg` requires the status is `unknown`.      
If the execution of `handler.receive_cross_chain_msg` returns `false`, the status will be set to `failed`, which will cause `settlement.receive_cross_chain_msg` can't execute again, resulting in the loss of funds of the message initiator.     
`settlement.receive_cross_chain_callback` has a similar issue. The function cannot be executed again, resulting in the `MintBurn` type message being unable to execute the burn callback, resulting in additional token issuance.

## Proof of Concept

```solidity
    function is_valid_handler(
        string memory chain_name,
        uint256 handler
    ) public view returns (bool) {
        return handler_whitelist[chain_name][handler];
    }

    function remove_handler(
        string memory chain_name,
        uint256 handler
    ) external onlyOwner {
        handler_whitelist[chain_name][handler] = false;
    }

    function receive_cross_chain_msg(
        uint256 /**txid */,
        string memory from_chain,
        uint256 /**from_address */,
        uint256 from_handler,
        PayloadType payload_type,
        bytes calldata payload,
        uint8 /**sign type */,
        bytes calldata /**signaturs */
    ) external onlySettlement returns (bool) {
        @audit If the handler is no longer supported, this function may return false
        //  from_handler need in whitelist
        if (is_valid_handler(from_chain, from_handler) == false) {
            return false;
        }
```

Since the handler does not check `is_valid_handler` when initiating cross-chain messages, if `to_handler` does not support `from_handler`, `receive_cross_chain_msg` will return false, causing the message to fail to be executed and funds to be lost.  
Besides, the handler's whitelist is variable. If the handler is no longer supported during message execution, the same situation will happen.   

The following is the specific POC. Since the project structure is divided into two folders, you need to put the settlement contract into the handler folder.  
```solidity
// SPDX-License-Identifier: UNLICENSED
pragma solidity ^0.8.13;

import {Test, console2} from "forge-std/Test.sol";
import {ERC1967Proxy} from "@openzeppelin/contracts/proxy/ERC1967/ERC1967Proxy.sol";
import {SettlementSignatureVerifier} from "contracts/SettlementSignatureVerifier.sol";
import {ChakraSettlementHandler} from "contracts/ChakraSettlementHandler.sol";
import {BaseSettlementHandler} from "contracts/BaseSettlementHandler.sol";
import {ChakraSettlement, PayloadType, CrossChainMsgStatus, ISettlementSignatureVerifier} from "contracts/tests/ChakraSettlement.sol";

contract SignatureVerifierTest is Test {
    ChakraSettlement public settlement;
    ChakraSettlementHandler public handler;
    SettlementSignatureVerifier public verifier;
    uint256 constant REQUIRED_VALIDATORS = 2;

    function setUp() public {
        verifier = new SettlementSignatureVerifier();
        verifier.initialize(address(this), REQUIRED_VALIDATORS);
        verifier.add_manager(address(this));
        
        address[] memory managers = new address[](1);
        managers[0] = address(this);

        settlement = new ChakraSettlement();
        settlement.initialize("ChainA", 0, address(this), managers, REQUIRED_VALIDATORS, address(verifier));

        handler = new ChakraSettlementHandler();
        handler.initialize(address(this), BaseSettlementHandler.SettlementMode.MintBurn, "ChainA", address(0), address(verifier), address(0), address(settlement));
    }

    function testReceiveMsgFailedAndBlocked() public {
        uint256 txid = 0;
        string memory from_chain = "ChainB";
        uint256 from_address = 0;
        uint256 from_handler = 0;
        address to_handler = address(handler);
        PayloadType payloadType = PayloadType.ERC20;
        bytes memory payload = "";
        uint8 sign_type = 0;
        bytes32 message_hash = keccak256(
            abi.encodePacked(
                txid,
                from_chain,
                from_address,
                from_handler,
                to_handler,
                keccak256(payload)
            )
        );

        (address validator, uint256 validatorPk) = makeAddrAndKey("validator1");
        verifier.add_validator(validator);

        (uint8 v, bytes32 r, bytes32 s) = vm.sign(validatorPk, message_hash);
        bytes memory signatures = abi.encodePacked(r, s, v);
        for (uint256 i = REQUIRED_VALIDATORS; i > 1; i /= 2) {
            signatures = abi.encodePacked(signatures, signatures);
        }

        (,,,,,,, CrossChainMsgStatus status) = settlement.receive_cross_txs(txid);
        assertEq(uint256(status), uint256(CrossChainMsgStatus.Unknow));
        settlement.receive_cross_chain_msg(txid, from_chain, from_address, from_handler, to_handler, payloadType, payload, sign_type, signatures);
        (,,,,,,, status) = settlement.receive_cross_txs(txid);
        assertEq(uint256(status), uint256(CrossChainMsgStatus.Failed));

        vm.expectRevert("Invalid transaction status");
        settlement.receive_cross_chain_msg(txid, from_chain, from_address, from_handler, to_handler, payloadType, payload, sign_type, signatures);
    }
}
```
