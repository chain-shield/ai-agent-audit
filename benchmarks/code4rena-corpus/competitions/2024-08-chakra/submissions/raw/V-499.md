# Vulnerability details

## Impact

Anyone can block the user callback transaction by incrementing the user `nonce` multiple times.

## Proof of Concept

When a cross-chain tx is initiated via `cross_chain_erc20_settlement`, a `txId` is generated. It contains the source and destination parameters, but mostly we'll focus at the user's `nonce` here.

```solidity
function cross_chain_erc20_settlement(
    string memory to_chain,
    uint256 to_handler,
    uint256 to_token,
    uint256 to,
    uint256 amount
) external {
	...

    {
        // Increment nonce for the sender
        nonce_manager[msg.sender] += 1;  <---------------------------------------
    }

    // Create a new cross chain tx
    uint256 txid = uint256(
        keccak256(
            abi.encodePacked(
                chain,
                to_chain,
                msg.sender, // from address for settlement to calculate txid
                address(this), //  from handler for settlement to calculate txid
                to_handler,
                nonce_manager[msg.sender]
            )
        )
    );
```

`cross_chain_erc20_settlement` calls `ChakraSettlement.send_cross_chain_msg`, which will create the same `txId`, create a `CreatedCrossChainTx` and store it inside `create_cross_txs`, as well as emit a `CrossChainMsg`, which is the event the validators will be waiting for.

```solidity
function send_cross_chain_msg(
    string memory to_chain,
    address from_address,
    uint256 to_handler,
    PayloadType payload_type,
    bytes calldata payload
) external {
    nonce_manager[from_address] += 1;

    address from_handler = msg.sender;

    uint256 txid = uint256(
        keccak256(
            abi.encodePacked(
                contract_chain_name, // from chain
                to_chain,
                from_address, // msg.sender address
                from_handler, // settlement handler address
                to_handler,
                nonce_manager[from_address]
            )
        )
    );

    create_cross_txs[txid] = CreatedCrossChainTx(
        txid,
        contract_chain_name,
        to_chain,
        from_address,
        from_handler,
        to_handler,
        payload,
        CrossChainMsgStatus.Pending
    );

    emit CrossChainMsg(
        txid,
        from_address,
        contract_chain_name,
        to_chain,
        from_handler,
        to_handler,
        payload_type,
        payload
    );
}
```

The abusable problem here is that `ChakraSettlement.send_cross_chain_msg` is **`external` and has no modifier, allowing anyone to call it with any values**. `from_handler` will be the attacker, but his idea is not to create a valid cross-chain transaction from here, but to block any user's callback when `mode = MintBurn`.

This can be achieved by:

1. Attacker calls `ChakraSettlement.send_cross_chain_msg()` with `from_address` = the user he has seen initiate a crosschain transaction in the mempool or whom he wants to block (other parameters do not matter).
2. This way the user nonce will be incremented and the nonce in `ChakraSettlement` and `Handler` will not be the same for this particular user.
3. This will result in different `txIds` in the two contracts when the `cross_chain_erc20_settlement` user call is executed.
4. When the user transaction is executed, inside the `Handler` its transaction will be stored in `create_cross_txs` with the `txId` created there (`txId` with the original `nonce`, not the abused one):
    
    ```solidity
    uint256 txid = uint256(
        keccak256(
            abi.encodePacked(
                chain,
                to_chain,
                msg.sender, // from address for settlement to calculate txid
                address(this), //  from handler for settlement to calculate txid
                to_handler,
                nonce_manager[msg.sender]
            )
        )
    );
    
    {
        // Save the cross chain tx
        create_cross_txs[txid] = CreatedCrossChainTx(
            txid,
            chain,
            to_chain,
            msg.sender,
            to,
            address(this),
            to_token,
            amount,
            CrossChainTxStatus.Pending
        );
    }
    ```
    
5. Inside `ChakraSettlement` another `txId` will be created as the `nonces` are different and this is the `txId` the validators will use to continue the flow on destination chain.
6. Then this `txId` that was created inside `ChakraSettlement` will be passed to `ChakraSettlement.receive_cross_chain_callback` where the `require` inside `processCrossChainCallback()` will pass as this is the `txId` that was stored inside `ChakraSettlement.create_cross_txs`.
    
    ```solidity
    function receive_cross_chain_callback(
        uint256 txid,
        string memory from_chain,
        uint256 from_handler,
        address to_handler,
        CrossChainMsgStatus status,
        uint8 sign_type,
        bytes calldata signatures
    ) external {
        verifySignature(
            txid,
            from_handler,
            to_handler,
            status,
            sign_type,
            signatures
        );
        processCrossChainCallback(
            txid,
            from_chain,
            from_handler,
            to_handler,
            status,
            sign_type,
            signatures
        );
        emitCrossChainResult(txid);
    }
    
    function processCrossChainCallback(
            uint256 txid,
            string memory from_chain,
            uint256 from_handler,
            address to_handler,
            CrossChainMsgStatus status,
            uint8 sign_type,
            bytes calldata signatures
        ) internal {
            require(
                create_cross_txs[txid].status == CrossChainMsgStatus.Pending,
                "Invalid transaction status"
            );
    
            if (
                ISettlementHandler(to_handler).receive_cross_chain_callback(
                    txid,
                    from_chain,
                    from_handler,
                    status,
                    sign_type,
                    signatures
                )
            ) {
                create_cross_txs[txid].status = status;
            } else {
                create_cross_txs[txid].status = CrossChainMsgStatus.Failed;
            }
        }
    ```
    
7. But when `Handler.receive_cross_chain_callback()` is called with the same `txId`, **the function will revert because a different `txId` was initially stored inside `Handler.create_cross_txs`, causing the entire callback flow to always fail and `MintBurn` mode to never be finalized.**
    
    ```solidity
    function receive_cross_chain_callback(
        uint256 txid,
        string memory from_chain,
        uint256 from_handler,
        CrossChainMsgStatus status,
        uint8 /* sign_type */, // validators signature type /  multisig or bls sr25519
        bytes calldata /* signatures */
    ) external onlySettlement returns (bool) {
        //  from_handler need in whitelist
        if (is_valid_handler(from_chain, from_handler) == false) {
            return false;
        }
    
        require(
            create_cross_txs[txid].status == CrossChainTxStatus.Pending,
            "invalid CrossChainTxStatus"
        );
    
        if (status == CrossChainMsgStatus.Success) {
            if (mode == SettlementMode.MintBurn) {
                _erc20_burn(address(this), create_cross_txs[txid].amount);
            }
    
            create_cross_txs[txid].status = CrossChainTxStatus.Settled;
        }
    
        if (status == CrossChainMsgStatus.Failed) {
            create_cross_txs[txid].status = CrossChainTxStatus.Failed;
        }
    
        return true;
    }
    ```
    

> ***NOTE:** Everything described above also applies to the Cairo part and the problem should be fixed there also.*

## Recommended Mitigation Steps

Add a `handler` variable in the `Settlement` contract by storing the `handler` address from the same chain and add `onlyHandler` modifier to `ChakraSettlement.send_cross_chain_msg()`, limiting who can call it.

```diff
function send_cross_chain_msg(
    string memory to_chain,
    address from_address,
    uint256 to_handler,
    PayloadType payload_type,
    bytes calldata payload
) external {
+   require(msg.sender == handler, "Not Handler");

    nonce_manager[from_address] += 1;
```
