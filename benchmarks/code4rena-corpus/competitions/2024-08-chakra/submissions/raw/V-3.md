# Vulnerability details

## Description
In ChakraSettlementHandler contract, a user funds can become permanently locked if a cross-chain transfer fails. When a transfer is initiated, tokens are locked in the contract. However, if the transfer fails, these tokens are not automatically returned to the user.

Vulnerable part of the code:
```
 function receive_cross_chain_callback(
        uint256 txid,
        string memory from_chain,
        uint256 from_handler,
        CrossChainMsgStatus status,
        uint8 /* sign_type */, // validators signature type /  multisig or bls sr25519
        bytes calldata /* signatures */
    ) external onlySettlement returns (bool) {
        //  from_handler need in whitelist
        if (is_valid_handler(from_chain, from_handler) == false) {
            return false;
        }

        require(
            create_cross_txs[txid].status == CrossChainTxStatus.Pending,
            "invalid CrossChainTxStatus"
        );

        if (status == CrossChainMsgStatus.Success) {
            if (mode == SettlementMode.MintBurn) {
                _erc20_burn(address(this), create_cross_txs[txid].amount);
            }

            create_cross_txs[txid].status = CrossChainTxStatus.Settled;
        }

        //@audit no refund logic for failed transfer
        if (status == CrossChainMsgStatus.Failed) {
            create_cross_txs[txid].status = CrossChainTxStatus.Failed;
        }

        return true;
    }
```
As you can see, in the receive_cross_chain_callback function, when a transfer fails (status == CrossChainMsgStatus.Failed), the transaction status is updated, but no action is taken to return the locked funds to the user.


## Impact 
This vulnerability can lead to permanent loss of user funds. If a cross-chain transfer fails, users have no built-in mechanism to recover their locked tokens. This could result in significant financial losses for users.
## Proof of Concept
The vulnerability is demonstrated in the testFlaw() function of the POC contract below. The test shows that after a failed cross-chain transfer:

1. The user's balance remains reduced (900 tokens instead of the initial 1000).
2. The contract retains the locked funds (100 tokens).
3. The transaction is marked as failed, but no refund occurs.

Key test output:
User balance after locking: 900.000000000000000000
User balance after failed transfer: 900.000000000000000000
Contract balance: 100.000000000000000000

// SPDX-License-Identifier: MIT
pragma solidity ^0.8.19;

import "forge-std/Test.sol";

// Shared enums and structs
enum ERC20Method { Transfer }
enum PayloadType { ERC20 }
enum SettlementMode { MintBurn, LockUnlock, LockMint, BurnUnlock }
enum CrossChainTxStatus { Pending, Settled, Failed }
enum CrossChainMsgStatus { Success, Failed }

struct ERC20TransferPayload {
    ERC20Method method;
    uint256 from;
    uint256 to;
    uint256 from_token;
    uint256 to_token;
    uint256 amount;
}

struct CreatedCrossChainTx {
    uint256 txid;
    string from_chain;
    string to_chain;
    uint256 from;
    uint256 to;
    uint256 from_handler;
    uint256 to_token;
    uint256 amount;
    CrossChainTxStatus status;
}

// Mock ERC20 contract
contract MockERC20 {
    mapping(address => uint256) public balanceOf;
    mapping(address => mapping(address => uint256)) public allowance;

    function mint(address to, uint256 amount) public {
        balanceOf[to] += amount;
    }

    function burn(address from, uint256 amount) public {
        require(balanceOf[from] >= amount, "Insufficient balance");
        balanceOf[from] -= amount;
    }

    function transfer(address to, uint256 amount) public returns (bool) {
        require(balanceOf[msg.sender] >= amount, "Insufficient balance");
        balanceOf[msg.sender] -= amount;
        balanceOf[to] += amount;
        return true;
    }

    function transferFrom(address from, address to, uint256 amount) public returns (bool) {
        require(balanceOf[from] >= amount, "Insufficient balance");
        require(allowance[from][msg.sender] >= amount, "Insufficient allowance");
        balanceOf[from] -= amount;
        balanceOf[to] += amount;
        allowance[from][msg.sender] -= amount;
        return true;
    }

    function approve(address spender, uint256 amount) public returns (bool) {
        allowance[msg.sender][spender] = amount;
        return true;
    }
}

// Mock Settlement contract
contract MockSettlement {
    function send_cross_chain_msg(
        string memory, uint256, uint256, PayloadType, bytes memory
    ) public pure returns (bool) {
        return true;
    }
}

// ChakraSettlementHandler contract
contract ChakraSettlementHandler {
    SettlementMode public mode;
    string public chain;
    address public token;
    address public settlement;
    mapping(uint256 => CreatedCrossChainTx) public create_cross_txs;
    mapping(address => uint256) public nonce_manager;
    uint256 public cross_chain_msg_id_counter;

    constructor(SettlementMode _mode, string memory _chain, address _token, address _settlement) {
        mode = _mode;
        chain = _chain;
        token = _token;
        settlement = _settlement;
    }

    function cross_chain_erc20_settlement(
        string memory to_chain,
        uint256 to_handler,
        uint256 to_token,
        uint256 to,
        uint256 amount
    ) external {
        require(amount > 0, "Amount must be greater than 0");
        require(to != 0, "Invalid to address");
        require(to_handler != 0, "Invalid to handler address");
        require(to_token != 0, "Invalid to token address");

        if (mode == SettlementMode.MintBurn) {
            _erc20_lock(msg.sender, address(this), amount);
        } else if (mode == SettlementMode.LockUnlock) {
            _erc20_lock(msg.sender, address(this), amount);
        } else if (mode == SettlementMode.LockMint) {
            _erc20_lock(msg.sender, address(this), amount);
        } else if (mode == SettlementMode.BurnUnlock) {
            _erc20_burn(msg.sender, amount);
        }

        nonce_manager[msg.sender] += 1;

        uint256 txid = uint256(keccak256(abi.encodePacked(
            chain, to_chain, msg.sender, address(this), to_handler, nonce_manager[msg.sender]
        )));

        create_cross_txs[txid] = CreatedCrossChainTx(
            txid, chain, to_chain, uint256(uint160(msg.sender)), to,
            uint256(uint160(address(this))), to_token, amount, CrossChainTxStatus.Pending
        );

        cross_chain_msg_id_counter += 1;
        MockSettlement(settlement).send_cross_chain_msg(to_chain, uint256(uint160(msg.sender)), to_handler, PayloadType.ERC20, "");
    }

    function receive_cross_chain_callback(
        uint256 txid,
        string memory from_chain,
        uint256 from_handler,
        CrossChainMsgStatus status,
        uint8, bytes calldata
    ) external returns (bool) {
        require(
            create_cross_txs[txid].status == CrossChainTxStatus.Pending,
            "invalid CrossChainTxStatus"
        );

        if (status == CrossChainMsgStatus.Success) {
            if (mode == SettlementMode.MintBurn) {
                _erc20_burn(address(this), create_cross_txs[txid].amount);
            }
            create_cross_txs[txid].status = CrossChainTxStatus.Settled;
        }

        if (status == CrossChainMsgStatus.Failed) {
            create_cross_txs[txid].status = CrossChainTxStatus.Failed;
        }

        return true;
    }

    function _erc20_lock(address from, address to, uint256 amount) internal {
        MockERC20(token).transferFrom(from, to, amount);
    }

    function _erc20_burn(address account, uint256 amount) internal {
        MockERC20(token).burn(account, amount);
    }

    function _erc20_mint(address account, uint256 amount) internal {
        MockERC20(token).mint(account, amount);
    }
}

// Test contract
contract ChakraSettlementHandlerTest is Test {
    ChakraSettlementHandler public handler;
    MockERC20 public mockToken;
    MockSettlement public mockSettlement;
    address public user = address(0x1234);

    function setUp() public {
        mockToken = new MockERC20();
        mockSettlement = new MockSettlement();
        handler = new ChakraSettlementHandler(
            SettlementMode.MintBurn,
            "SourceChain",
            address(mockToken),
            address(mockSettlement)
        );

        mockToken.mint(user, 1000 ether);
    }

    function testFlaw() public {
        vm.startPrank(user);
        mockToken.approve(address(handler), 100 ether);
        
        // Initiate cross-chain transfer
        handler.cross_chain_erc20_settlement("DestChain", 1, 2, 3, 100 ether);
        vm.stopPrank();

        uint256 userBalanceAfterLocking = mockToken.balanceOf(user);
        emit log_named_decimal_uint("User balance after locking", userBalanceAfterLocking, 18);

        // Simulate a failed cross-chain transfer
        uint256 txid = uint256(keccak256(abi.encodePacked(
            "SourceChain", "DestChain", user, address(handler), uint256(1), uint256(1)
        )));
        handler.receive_cross_chain_callback(
            txid,
            "DestChain",
            1,
            CrossChainMsgStatus.Failed,
            0,
            ""
        );

        uint256 userBalanceAfterFailure = mockToken.balanceOf(user);
        emit log_named_decimal_uint("User balance after failed transfer", userBalanceAfterFailure, 18);

        uint256 contractBalance = mockToken.balanceOf(address(handler));
        emit log_named_decimal_uint("Contract balance", contractBalance, 18);

        (
            ,,,,,,,, CrossChainTxStatus _status
        ) = handler.create_cross_txs(txid);

        // Demonstrate the vulnerability
        assertTrue(userBalanceAfterFailure < 1000 ether, "VULNERABILITY: User funds are locked despite failed transfer");
        assertTrue(contractBalance > 0, "VULNERABILITY: Contract retains locked funds after failed transfer");
        assertTrue(_status == CrossChainTxStatus.Failed, "VULNERABILITY: Transaction marked as failed but funds not returned");

        emit log_string("Vulnerability successfully demonstrated: Funds remain locked after failed cross-chain transfer");
    }
}

## Recommended Mitigation Steps

Implement an automatic refund mechanism in the receive_cross_chain_callback function:
```
if (status == CrossChainMsgStatus.Failed) {
    create_cross_txs[txid].status = CrossChainTxStatus.Failed;
    //@audit refund the locked tokens to the user
    _erc20_unlock(create_cross_txs[txid].from, create_cross_txs[txid].amount);
}
```
