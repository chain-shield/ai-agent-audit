# Vulnerability details

## Impact
On the call to `loadPrecompilePreimagePart()` an incorrect address is copied to memory `0x80` due to using an incorrect offset on the in-line `calldatacopy()` function, leading to a wrong `key` generated after hashing the `_precompile` & the `_input`:
https://github.com/code-423n4/2024-07-optimism/blob/70556044e5e080930f686c4e5acde420104bb2c4/packages/contracts-bedrock/src/cannon/PreimageOracle.sol#L340C1-L389C6
```soldity
        assembly {
            // we leave solidity slots 0x40 and 0x60 untouched, and everything after as scratch-memory.
            let ptr := 0x80

            // copy precompile address and input into memory
            // len(sig) + len(_partOffset) + address-offset-in-slot
            //@audit supposed to copy from 36 - 0x20 not 48 - 20, 
            calldatacopy(ptr, 48, 20)
            calldatacopy(add(20, ptr), _input.offset, _input.length)
            // compute the hash
            let h := keccak256(ptr, add(20, _input.length))
            // mask out prefix byte, replace with type 6 byte
            key := or(and(h, not(shl(248, 0xFF))), shl(248, 0x06))

            // Call the precompile to get the result.
            res :=
                staticcall(
                    gas(), // forward all gas
                    _precompile,
                    add(20, ptr), // input ptr
                    _input.length,
                    0x0, // Unused as we don't copy anything
                    0x00 // don't copy anything
                )

            size := add(1, returndatasize())
            // revert if part offset >= size+8 (i.e. parts must be within bounds)
            if iszero(lt(_partOffset, add(size, 8))) {
                // Store "PartOffsetOOB()"
                mstore(0, 0xfe254987)
                // Revert with "PartOffsetOOB()"
                revert(0x1c, 4)
            }

            // Reuse the `ptr` to store the preimage part: <sizePrefix ++ precompileStatus ++ returrnData>
            // put size as big-endian uint64 at start of pre-image
            mstore(ptr, shl(192, size))
            ptr := add(ptr, 0x08)

            // write precompile result status to the first byte of `ptr`
            mstore8(ptr, res)
            // write precompile return data to the rest of `ptr`
            returndatacopy(add(ptr, 0x01), 0x0, returndatasize())

            // compute part given ofset
            part := mload(add(sub(ptr, 0x08), _partOffset))
        }
        preimagePartOk[key][_partOffset] = true;
        preimageParts[key][_partOffset] = part;
        preimageLengths[key] = size;
    }
```
The issue starts from the part where the offset 48 is used, however, that isn't the offset in the call data where the precompile is stored so it truncates the leftward bits of the address, and stores the truncated address in memory 0x80, which is later hashed to generate h, via keccak256(ptr, add(20, _input.length)) hashing an incorrect address with the input thereby generating a wrong hash which will later be used to get the key for the mappings.
Given the premimageParts mappings are used in VM.step in the FaultDisputeGame.step, it can lead to incorrect outcomes as the key computed here will be incorrect making the parts stored in the wrong mappings.

## Proof of Concept

pragma solidity 0.8.24;

import "forge-std/Test.sol";
import "forge-std/console.sol";

contract CallsTest is Test {
     Mem public mem;

     function setUp() public {
        mem = new Mem();
     }
    function testRevert() public {
        // create an address and test the functions
        address me = vm.addr(1234);
        // calling the one that uses 48 as offset with truncate the address
        address addr = mem.loadPrecompilePreimagePart(10, me, "");
        //calling the one that uses the right offset preserves the address.
        address addr2 = mem.loadPrecompilePreimagePartCorrect(10, me, "");
        // log to see original address
        console.log(me);
        //log to see truncated address
        console.log(addr);
        // log to see valid address stored in 0x80
        console.log(addr2);
    }
}
contract Mem{
     function loadPrecompilePreimagePart(uint256 _partOffset, address _precompile, bytes calldata _input) external returns(address addr){

        assembly {
            // we leave solidity slots 0x40 and 0x60 untouched, and everything after as scratch-memory.
            let ptr := 0x80
            //@audit  lets load to mem using the offset 48.
            // issue is 48 truncates the bytes of the precompile address causing an invalid address stored in memory
            // as the precompile
            // copy precompile address and input into memory
            // len(sig) + len(_partOffset) + address-offset-in-slot
            calldatacopy(ptr, 48, 20)
            addr := mload(ptr)


        }
    }

       function loadPrecompilePreimagePartCorrect(uint256 _partOffset, address _precompile, bytes calldata _input) external returns(address addr){

        assembly {
            // we leave solidity slots 0x40 and 0x60 untouched, and everything after as scratch-memory.
            let ptr := 0x80
            //@audit  lets load to mem using the offset 36.
            // 4 + 32 = 36. and copies to 0x20, since the evm handles that padding
            // copy precompile address and input into memory
            // len(sig) + len(_partOffset) + address-offset-in-slot
            calldatacopy(ptr, 36, 0x20)
            addr := mload(ptr)


        }
    }
}
