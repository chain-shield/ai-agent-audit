# Vulnerability details

## Impact
This can cause unwanted behaviour in the protocol
This can lead to increased gas spending


## Proof of Concept
Add this to your test suite

in your setup function add these lines
```diff
    function setUp() public {
        lrt = new LRToken();
        token1 = new LRToken();
        token2 = new LRToken();
        token3 = new LRToken();
        lrt.mint(address(this), INITIAL_SUPPLY);
        token2.mint(address(this), INITIAL_SUPPLY);

        address[] storage allowedTokens_ = allowedTokens;
        allowedTokens_.push(address(lrt));
        
+          allowedTokens_.push(address(token1));
+          allowedTokens_.push(address(token2));
+          allowedTokens_.push(address(token3));
        
        // token 2 added 3 times to the allowlist
+          allowedTokens_.push(address(token2));
+          allowedTokens_.push(address(token2));

        prelaunchPoints = new PrelaunchPoints(
            EXCHANGE_PROXY,
            WETH,
            allowedTokens_
        );

        lpETH = new MockLpETH();
        lpETHVault = new MockLpETHVault();
```

```solidity
    function testAllowtoken() public {
```
