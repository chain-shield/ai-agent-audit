The description mentions a point system calculated off-chain based on locked amounts and referrals. However, the contract does not manage or track these points on-chain, which could lead to discrepancies or trust issues if the off-chain system is not transparent or is compromised.

This approach will enhance transparency and reduce the risk of discrepancies between on-chain data and off-chain calculations.

/**
 * @dev Generic internal locking function that updates rewards based on
 *      previous balances, then update balances and points.
 * @param _token       Address of the token to lock
 * @param _amount      Units of ETH or token to add to the users balance
 * @param _receiver    Address of user who will receive the stake
 * @param _referral    Referral code, used to award referral points
 */
function _processLock(address _token, uint256 _amount, address _receiver, bytes32 _referral)
    internal
    onlyBeforeDate(loopActivation)
{
    if (_amount == 0) {
        revert CannotLockZero();
    }
    
    uint256 pointAward = _amount / 1e18; // Example: 1 point per ETH locked
    points[_receiver] += pointAward;
    
    // Referral bonus
    if (_referral != 0) {
        address referrer = _bytes32ToAddress(_referral); // Convert bytes32 to address
        uint256 referralBonus = pointAward / 10; // 10% of the points awarded as referral bonus
        points[referrer] += referralBonus;
        
        emit PointsUpdated(referrer, points[referrer]);
    }
    
    if (_token == ETH) {
        totalSupply += _amount;
        balances[_receiver][ETH] += _amount;
    } else {
        if (!isTokenAllowed[_token]) {
            revert TokenNotAllowed();
        }
        IERC20(_token).safeTransferFrom(msg.sender, address(this), _amount);

        if (_token == address(WETH)) {
            WETH.withdraw(_amount);
            totalSupply += _amount;
            balances[_receiver][ETH] += _amount;
        } else {
            balances[_receiver][_token] += _amount;
        }
    }

    emit Locked(_receiver, _amount, _token, _referral);
    emit PointsUpdated(_receiver, points[_receiver]);
}
