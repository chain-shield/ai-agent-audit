# Vulnerability details

## Impact
The 'configUpdated' function permits reconfiguration of contract settings and is currently accessible to all users. To prevent unauthorized modifications to critical configurations, access to this function should be restricted.

## Proof of Concept

pragma solidity 0.8.25;

import "forge-std/Test.sol";
import "../contracts/LandManager.sol";
import "../interfaces/IConfigStorage.sol";

contract LandManagerTest is Test {
    LandManager landManager;
    IConfigStorage configStorage;

    function setUp() public {
        configStorage = IConfigStorage(address(new MockConfigStorage()));
        landManager = new LandManager();
        landManager.initialize(address(configStorage));
    }

    function testUnauthorizedConfigUpdate() public {
        // Impersonate an unauthorized user
        vm.startPrank(unauthorizedUser);

        // Call the configUpdated function as the unauthorized user
        landManager.configUpdated();

        // Stop impersonation
        vm.stopPrank();

        // Assert that the configuration has been updated

}

contract MockConfigStorage is IConfigStorage {
    mapping(StorageKey => address) public addresses;
    mapping(StorageKey => uint256) public uints;

    constructor() {
        addresses[StorageKey.LockManager] = address(0x1);
        addresses[StorageKey.AccountManager] = address(0x2);
        addresses[StorageKey.MunchNFT] = address(0x3);
        addresses[StorageKey.NFTAttributesManager] = address(0x4);

    }

    function getUint(StorageKey key) external view override returns (uint256) {
        return uints[key];
    }

## Recommended Mitigation Steps
Ensure that this function is accessible only to authorized entities. The `onlyConfigStorage` modifier adds a layer of protection, assuming it effectively restricts access as designed.
