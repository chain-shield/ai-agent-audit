# Vulnerability details

## Impact
The 'triggerPlotMetadata' function is accessible to all users, which may be appropriate based on its intended use. However, it is crucial to ensure that this function does not permit unauthorized alterations to plot metadata.

## Proof of Concept
```solidity
// SPDX-License-Identifier: UNLICENSED
pragma solidity 0.8.25;

import "forge-std/Test.sol";
import "../src/LandManager.sol";
import "../src/interfaces/ILockManager.sol";
import "../src/interfaces/IAccountManager.sol";
import "../src/interfaces/INFTAttributesManager.sol";
import "openzeppelin-contracts/contracts/token/ERC721/IERC721.sol";

contract LandManagerTest is Test {
    LandManager landManager;
    ILockManager lockManager;
    IAccountManager accountManager;
    IERC721 munchNFT;
    INFTAttributesManager nftAttributesManager;
    address configStorage = address(0x1234);
    address unauthorizedUser = address(0x5678);

    function setUp() public {
        landManager = new LandManager();
        landManager.initialize(configStorage);

        // Mocking the necessary interfaces
        lockManager = ILockManager(address(0x1));
        accountManager = IAccountManager(address(0x2));
        munchNFT = IERC721(address(0x3));
        nftAttributesManager = INFTAttributesManager(address(0x4));

        // Setting up the config storage
        vm.mockCall(configStorage, abi.encodeWithSelector(IConfigStorage.getAddress.selector, StorageKey.LockManager), abi.encode(address(lockManager)));
        vm.mockCall(configStorage, abi.encodeWithSelector(IConfigStorage.getAddress.selector, StorageKey.AccountManager), abi.encode(address(accountManager)));
        vm.mockCall(configStorage, abi.encodeWithSelector(IConfigStorage.getAddress.selector, StorageKey.MunchNFT), abi.encode(address(munchNFT)));
        vm.mockCall(configStorage, abi.encodeWithSelector(IConfigStorage.getAddress.selector, StorageKey.NFTAttributesManager), abi.encode(address(nftAttributesManager)));

        // Mocking the account manager to return a registered player
        vm.mockCall(address(accountManager), abi.encodeWithSelector(IAccountManager.getPlayer.selector, unauthorizedUser), abi.encode(unauthorizedUser, MunchablesCommonLib.Player({registrationDate: block.timestamp, unfedSchnibbles: 0, lastPetMunchable: 0, snuggeryRealm: 0})));
    }

    function testUnauthorizedTriggerPlotMetadata() public {
        // Ensure the unauthorized user can call triggerPlotMetadata without reverting
        vm.prank(unauthorizedUser);
        landManager.triggerPlotMetadata();

        // Check if the plot metadata was updated
        (uint256 lastUpdated, uint256 currentTaxRate) = landManager.plotMetadata(unauthorizedUser);
        assertEq(lastUpdated, block.timestamp);
        assertEq(currentTaxRate, landManager.DEFAULT_TAX_RATE());
    }
}
```
