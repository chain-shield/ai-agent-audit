# Vulnerability details

## Impact
**_farmPlots()** doesn't set Plot as dirty if the PlotId is equal to **_getNumPlots()**. Which leads to staker continue to earn rewards for the whole period for this invalid Plot.

## Proof of Concept
Add the following test case to **LandManager/transferToUnoccupiedPlot.test.ts**
``` javascript
it("plot is not marked dirty for case where plotId is equal to no of plots", async () => {
  let noOfPlots = await testContracts.landManagerProxy.contract.read.getNumberOfPlots(
    [alice],
    { account: alice }
  );
  console.log("Checking the number of plots for alice: ", noOfPlots)
  const { request } =
    await testContracts.landManagerProxy.contract.simulate.transferToUnoccupiedPlot([1, 62], {
      account: bob,
    });
  let txHash = await testClient.writeContract(request);
  await assertTxSuccess({ txHash: txHash });

  console.log("The TokenId 1 has been transfered from PlotId 2 to PlotId 62")

  let _data = await testContracts.lockManager.contract.simulate.unlock(
    [zeroAddress, parseEther("0.5")],
    { account: alice, value: parseEther("0") }
  );
  txHash = await testClient.writeContract(_data.request);

  noOfPlots = await testContracts.landManagerProxy.contract.read.getNumberOfPlots(
    [alice],
    { account: alice }
  );
  console.log("Number of plots of alice after unstake: ", noOfPlots)

  let toilerState = await testContracts.landManagerProxy.contract.read.getToilerState(
    [1],
    { account: bob }
  );
  console.log("Dirty state of TokenId: ", toilerState[4])

  _data =
  await testContracts.landManagerProxy.contract.simulate.transferToUnoccupiedPlot([1, 61], {
    account: bob,
  });
  txHash = await testClient.writeContract(_data.request);
  await assertTxSuccess({ txHash: txHash });

  toilerState = await testContracts.landManagerProxy.contract.read.getToilerState(

  console.log("Now the TokenID 1 has been moved from PlotId 62 to PlotId 61. Ideally the tiolerState should've been marked as dirty")
  console.log("Dirty state of TokenId: ", toilerState[4])
});
```

Add a few view functions in solidity contract to read the contract state

``` solidity
function getToilerState(uint256 tokenId) public returns (uint256, uint256, uint256, address, bool) {
    ToilerState memory _toilerState = toilerState[tokenId];
    return (_toilerState.plotId, _toilerState.latestTaxRate, _toilerState.lastToilDate, _toilerState.landlord, _toilerState.dirty);
}

function getNumberOfPlots(address landlord) public returns(uint256) {
    return _getNumPlots(landlord);
}
```

## Recommended Mitigation Steps
Update the check as below in **_farmPlots()**
``` solidity
if (_getNumPlots(landlord) <= _toiler.plotId) {
    timestamp = plotMetadata[landlord].lastUpdated;
    toilerState[tokenId].dirty = true;
}
```
