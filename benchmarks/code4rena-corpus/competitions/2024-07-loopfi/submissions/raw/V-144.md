# Vulnerability details

## Impact
If CDPVault is deployed with a liquidationDiscount > 1e18, positions will have collateral left over after being liquidated and debt fully repaid. 

Additionally, this can cause scenarios where liquidation is impossible. IE, if spotprice is set to 0.75 below, liquidatePosition will revert for badDebt. If liquidatePositionBadDebt is called with the same parameters, it will revert for noBadDebt.

function test_liquidate_full_no_bad_repayGtColl() public {
        // lack of liquidationDiscount check in constructor can lead to liquidationDiscount > 1, 
        // attempting to liquidiate the full debt will leave collateral in the vault for the position
        CDPVault vault = createCDPVault(token, 150 ether, 0, 1.25 ether, 1 ether, 1.1 ether);
        
        createGaugeAndSetGauge(address(vault));

        // create position to liquidate
        _modifyCollateralAndDebt(vault, 100 ether, 80 ether);

        // create second position to increase total debt
        address alice = address(0x123456);
        token.mint(alice, 100 ether);
        vm.startPrank(alice);
        token.approve(address(vault), 100 ether);
        address position2 = address(new PositionOwner(vault));
        vault.deposit(position2, 100 ether);
        vault.borrow(alice, position2, 50 ether);
        uint256 creditAlice = credit(alice);
        assertEq(creditAlice, 50 ether);
        vm.stopPrank();


        // liquidate position
        address position = address(this);
        uint256 repayAmount = 80 ether;
        mockWETH.mint(alice, repayAmount);
        _updateSpot(0.8 ether);
        vm.prank(alice);
        mockWETH.approve(address(vault), repayAmount);
        console.log(vault.totalDebt(), "totalDebt");
        uint256 creditBefore = credit(alice);
        uint256 virtualDebtBefore = virtualDebt(vault, position);
        vm.prank(alice);
        vault.liquidatePosition(position, repayAmount);
        uint256 creditAfter = credit(alice);
        {
        (uint256 collateral, uint256 debtAfter, , , , ) = vault.positions(position);

        // assertEq(debtAfter, virtualDebtBefore - repayAmount); // debt - repayAmount
        assertEq(debtAfter, 0);
        assertEq(creditBefore - creditAfter, 80 ether);
        assertGt(collateral, 0, "collateral");
        assertGt(token.balanceOf(address(vault)), 0);

        // position 1 still has collateral and can withdraw it

        vault.withdraw(address(this), collateral);
        } 

        // now alice pays back debt and withdraws her collateral
        {
        (uint256 collateral, uint256 debtAfter, , , , ) = vault.positions(position2);
        vm.startPrank(alice);
        mockWETH.approve(address(vault), debtAfter);
        vault.repay(alice, position2, debtAfter);
        vault.withdraw(position2, collateral);
        vm.stopPrank();
        }

    }
