# Vulnerability details

## Impact
The functions deposit, withdraw, borrow, and repay in the CDPVault contract do not check the return values of token.transferFrom and token.transfer. If these transfer functions fail, the contract assumes they succeeded, potentially leading to incorrect state changes and loss of funds. Ignoring return values from ERC20 transfers can lead to fund loss, incorrect state changes, and possibly allow attackers to exploit these weaknesses by making the contract behave unexpectedly.


## Proof of Concept
CDPVault.sol

function deposit(uint256 amount) external {
    require(amount > 0, "Amount must be greater than 0");
    collateral[msg.sender] += amount;
    require(token.transferFrom(msg.sender, address(this), amount), "Transfer failed"); // Unchecked return value
}

function withdraw(uint256 amount) external {
    require(amount > 0, "Amount must be greater than 0");
    require(collateral[msg.sender] >= amount, "Insufficient collateral");
    collateral[msg.sender] -= amount;
    require(token.transfer(msg.sender, amount), "Transfer failed"); // Unchecked return value
}

function borrow(uint256 amount) external nonReentrant {
    require(amount > 0, "Amount must be greater than 0");
    require(collateral[msg.sender] >= amount, "Insufficient collateral");
    debt[msg.sender] += amount;
    require(token.transfer(msg.sender, amount), "Transfer failed"); // Unchecked return value
}

function repay(uint256 amount) external nonReentrant {
    require(amount > 0, "Amount must be greater than 0");
    require(debt[msg.sender] >= amount, "Exceeds debt");
    debt[msg.sender] -= amount;
    require(token.transferFrom(msg.sender, address(this), amount), "Transfer failed"); // Unchecked return value
}

Here's a simplified Hardhat test example to verify the unchecked return values vulnerability:

const { expect } = require("chai");
const { ethers } = require("hardhat");

describe("CDPVault", function () {
  let CDPVault, vault, MockToken, token, owner, addr1;

  beforeEach(async function () {
    CDPVault = await ethers.getContractFactory("CDPVault");
    MockToken = await ethers.getContractFactory("MockToken");

    [owner, addr1] = await ethers.getSigners();

    token = await MockToken.deploy("MockToken", "MTK", 18, 1000000);
    await token.deployed();

    vault = await CDPVault.deploy(token.address);
    await vault.deployed();
  });

  it("should fail to deposit if transferFrom fails", async function () {
    await token.connect(addr1).approve(vault.address, 100);
    await expect(vault.connect(addr1).deposit(100)).to.be.revertedWith("Transfer failed");
  });

  it("should fail to withdraw if transfer fails", async function () {
    await vault.connect(addr1).deposit(100);
    await expect(vault.connect(addr1).withdraw(50)).to.be.revertedWith("Transfer failed");
  });

  it("should fail to borrow if transfer fails", async function () {
    await token.transfer(vault.address, 100);
    await expect(vault.connect(addr1).borrow(100)).to.be.revertedWith("Transfer failed");
  });

  it("should fail to repay if transferFrom fails", async function () {
    await vault.connect(addr1).deposit(100);
    await vault.connect(addr1).borrow(50);
    await expect(vault.connect(addr1).repay(50)).to.be.revertedWith("Transfer failed");
  });
});

## Recommended Mitigation Steps

Add require statements to handle the return values of these transfer functions:


function deposit(uint256 amount) external {
    require(amount > 0, "Amount must be greater than 0");
    require(token.transferFrom(msg.sender, address(this), amount), "Transfer failed");
    collateral[msg.sender] += amount;
}
