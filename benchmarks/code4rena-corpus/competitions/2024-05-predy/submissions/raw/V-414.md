# Vulnerability details

## Impact
The identified problem lies in the fact that the function does not explicitly transfer the swapped output amount `amountOut` to the intended recipient. This can lead to unexpected behavior where the swapper does not receive the tokens after the swap is executed.

## Proof of Concept
Place this following code in test/SwapExactInTest.t.sol : 

```solidity

// SPDX-License-Identifier: UNLICENSED
pragma solidity ^0.8.0;

import "./pool/Setup.t.sol";

contract SwapExactInTest is TestPool {

    address swapper = makeAddr("swapper");

    function setUp() public override {
        TestPool.setUp();

    }

function test_swapExactIn_Not_send_the_swap_amount_to_the_swapper() public {
    // Arrange
    // Set initial balances for 
    vm.deal(swapper, 10 ether);
    currency1.transfer(swapper, 10 ether);
    
    // Ensure initial conditions
    assert(currency1.balanceOf(swapper) == 10 ether);
    assert(currency0.balanceOf(swapper) == 0);

    // Act
    // Prank the swapper to call the swap function
    vm.prank(swapper);
    uniswapSettlement.swapExactIn(
        address(currency0),
        address(currency1),
        "",
        1 ether,
        0,
        address(swapper)
    );

    // Assert
    // Check the balances after the swap
    // Assumption: the swap returns 9 ether of currency1 for 1 ether of currency0
    assert(currency1.balanceOf(swapper) == 9 ether);
    assert(currency0.balanceOf(swapper) == 0);
}

```
