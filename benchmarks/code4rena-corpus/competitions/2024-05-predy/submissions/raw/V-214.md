# Vulnerability details

## Root of the bug 
The execLiquidationCall function it’s does not use any checks to verify if the caller is authorized to execute liquidations and this flaw is allows any external address to attempt and potentially succeed in liquidating any vault without proper authorization.
So the the absence of an access control in the execLiquidationCall function make The function is directly calls the LiquidationLogic.liquidate function without verifying if the caller is authorized to perform the liquidation 

```solidity
   */
    function execLiquidationCall(uint256 vaultId, uint256 closeRatio, bytes memory settlementData)
        external
        nonReentrant
        returns (TradeResult memory tradeResult)
    {
        return LiquidationLogic.liquidate(vaultId, closeRatio, globalData, settlementData);
    }

```
- An attacker can exploit this  by calling the execLiquidationCall function with any vault ID, close ratio, and settlement data, leading to unauthorized liquidation

## Impact
An unauthorized user can liquidate vaults, causing unexpected closures of positions and financial losses to vault owners. And this unauthorized action bypasses the intended security so is meant to restrict liquidation to authorized liquidators.


## Proof of Concept
- The documentation say that liquidators are responsible for executing liquidations:

```

 "When a trader's position meets the conditions for liquidation, the Liquidator forcibly closes the position. Liquidators can earn profits from the differences in the liquidation process."

```

And the Violation: The lack of an authorization check in execLiquidationCall contradicts and this role definition, allowing unauthorized users to act as liquidators

- Here is a test : 

```solidity

// SPDX-License-Identifier: UNLICENSED
pragma solidity ^0.8.17;

import "forge-std/Test.sol";
import "../src/PredyPool.sol";
import "../src/libraries/logic/LiquidationLogic.sol"; 

contract ExecLiquidationCallTest is Test {
    PredyPool predyPool;
    address authorizedLiquidator = address(0xAuthorizedLiquidator);
    address unauthorizedUser = address(0xUnauthorizedUser);
    uint256 vaultId = 1;
    uint256 closeRatio = 500; // 50% close ratio, expressed in basis points
    bytes settlementData; // Empty for simplicity

    function setUp() public {
        // Deploying the contract and setting up initial state
        predyPool = new PredyPool();
        
        // Add initial authorized liquidator
        predyPool.addAuthorizedLiquidator(authorizedLiquidator);
        
        // Create a vault with an initial balance
        predyPool.createVault(vaultId, address(this), 1000);
    }

    function testUnauthorizedLiquidation() public {
        // Simulate an unauthorized liquidation attempt
        vm.prank(unauthorizedUser);
        (bool success, bytes memory data) = address(predyPool).call(
            abi.encodeWithSignature(
                "execLiquidationCall(uint256,uint256,bytes)",
                vaultId,
                closeRatio,
                settlementData
            )
        );

        // Decode the response
        (bool callSuccess, string memory reason) = abi.decode(data, (bool, string));
        
        // Assert that the call was unsuccessful and the reason is unauthorized
        assertFalse(callSuccess, "Unauthorized user should not be able to liquidate.");
        assertEq(reason, "Unauthorized", "Expected 'Unauthorized' as reason.");
    }

    function testAuthorizedLiquidation() public {
        // Simulate an authorized liquidation attempt
        vm.prank(authorizedLiquidator);
        (bool success, bytes memory data) = address(predyPool).call(
            abi.encodeWithSignature(
                "execLiquidationCall(uint256,uint256,bytes)",
                vaultId,
                closeRatio,
                settlementData
            )
        );

        // Decode the response
        (bool callSuccess, ) = abi.decode(data, (bool, bytes));

        // Assert that the call was successful
        assertTrue(callSuccess, "Authorized liquidator should be able to liquidate.");
    }
}

```
