# Vulnerability details

Impact: This can result in incorrect token balances and unexpected behavior in the contract.

Proof of concept :
pragma solidity ^0.8.0;

contract VulnerableContract {
    struct Position {
        uint128 tokensOwed0;
        uint128 tokensOwed1;
    }

    mapping(address => mapping(int24 => mapping(int24 => Position))) positions;
    address public token0;
    address public token1;

    // Function to set token addresses
    constructor(address _token0, address _token1) {
        token0 = _token0;
        token1 = _token1;
    }

    // Function to simulate the collect function with integer underflow vulnerability
    function simulateCollect(address recipient, int24 tickLower, int24 tickUpper, uint128 amount0Requested, uint128 amount1Requested) external returns (uint128 amount0, uint128 amount1) {
        Position storage position = positions[recipient][tickLower][tickUpper];

        // Example scenario: tokensOwed0 is smaller than amount0Requested, causing underflow
        position.tokensOwed0 = 100;
        amount0 = amount0Requested > position.tokensOwed0 ? position.tokensOwed0 : amount0Requested;
        // amount0 will underflow to a large value, which is incorrect

        // Simulate transfer of tokens
        if (amount0 > 0) {
            position.tokensOwed0 -= amount0;
            // Transfer tokens to recipient
        }
    }
}
In the poc:
1.
The'VulnerableContract' simulates the collect function with the integer underflow vulnerability.
2.Inside the 'simulateCollect'
function, the amount0Requested is compared with position.tokensOwed0. If amount0Requested is greater than 'position.tokensOwed0' an underflow occurs, resulting in an incorrect value for 'amount0'.

Manual analysis. 

Mitigation Steps:Add checks to prevent underflow before performing the subtraction
Code:if (amount0 > 0) {
    require(position.tokensOwed0 >= amount0, "Underflow detected");
    position.tokensOwed0 -= amount0;
    TransferHelper.safeTransfer(token0, recipient, amount0);
}
if (amount1 > 0) {
    require(position.tokensOwed1 >= amount1, "Underflow detected");
    position.tokensOwed1 -= amount1;
    TransferHelper.safeTransfer(token1, recipient, amount1);
}
