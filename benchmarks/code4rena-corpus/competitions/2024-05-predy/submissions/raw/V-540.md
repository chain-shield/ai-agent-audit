# Vulnerability details

## Impact
The function ```PerpMarket.executeOrderV3L2``` returns a function call ```_executeOrderV3(order, sig, settlementParams, orderId);``` which appears to accept a signature (sig) as an input parameter but does not show any validation logic for this signature. One would ordinarily expect that the verification process likely happens within the ```PerpOrderV3Lib.resolve```, however, ```PerpOrderV3Lib.resolve``` does not perform signature verification. The function simply creates a ```ResolvedOrder``` structure using the input ```perpOrder``` and sig without actually verifying the signature. This unchecked signature can therefore be leveraged by an attacker to carry out Unauthorized actions, Order tampering, Loss of funds and ultimately Systemic risk.

## Proof of Concept
The following PoC demonstrates how an attacker can exploit unchecked signature verification by:
First, Modifying the Order: Changing order details such as the quantity or trader address.
Second, Submitting the Modified Order: Using the original signature to submit the modified order to the vulnerable contract.

The context below is used for brevity:

2. An attacker contract is created created that captures the signed order, modifies it, and submits it to the vulnerable contract.
```solidity
pragma solidity ^0.8.0;

import "./VulnerableContract.sol";

contract AttackerContract {
    VulnerableContract public vulnerableContract;

    constructor(address _vulnerableContract) {
        vulnerableContract = VulnerableContract(_vulnerableContract);
    }

    function attack(
        VulnerableContract.PerpOrderV3 memory capturedOrder,
        bytes memory capturedSignature,
        address newTrader
    ) public {
        // Modify the order (e.g., increase the quantity or change the trader's address)
        capturedOrder.quantity = capturedOrder.quantity * 10;
        capturedOrder.info.trader = newTrader;

        // Submit the modified order with the original signature
        vulnerableContract._executeOrderV3(capturedOrder, capturedSignature);
    }
}
```

To demonstrate the attack, we'll deploy the ```VulnerableContract``` and the ```AttackerContract```, then use the ```AttackerContract``` to execute the attack.

-->Deploy the AttackerContract, passing the address of the deployed VulnerableContract.
-->Execute the attack by calling AttackerContract.attack with a captured order, captured signature, and a new trader address.

## Recommended Mitigation Steps
Mitigation involves the use of a combination of Proper Signature Verification (using EIP712), Use of nonces, Consistent Hashing, Implementing Access Control, Validation of input parameters so that the refactored code looks somewhat like this below:
```solidity
pragma solidity ^0.8.0;

contract SecureContract {
    bytes32 public DOMAIN_SEPARATOR;

    struct OrderInfo {
        address trader;
    }

    struct PerpOrderV3 {
        OrderInfo info;
        uint256 pairId;
        address entryTokenAddress;
        string side;
        uint256 quantity;
        uint256 marginAmount;
        uint256 limitPrice;
        uint256 stopPrice;
        uint256 leverage;
        bool reduceOnly;
        bool closePosition;
        bytes auctionData;
    }

    mapping(address => uint256) public nonces;

    constructor() {
        DOMAIN_SEPARATOR = keccak256(
            abi.encode(
                keccak256("EIP712Domain(string name,string version,uint256 chainId,address verifyingContract)"),
                keccak256(bytes("SecureContract")),
                keccak256(bytes("1")),
                block.chainid,
                address(this)
            )
        );
    }

    function hashOrder(PerpOrderV3 memory order) internal view returns (bytes32) {
        return keccak256(
            abi.encode(
                keccak256("PerpOrderV3(OrderInfo info,uint256 pairId,address entryTokenAddress,string side,uint256 quantity,uint256 marginAmount,uint256 limitPrice,uint256 stopPrice,uint256 leverage,bool reduceOnly,bool closePosition,bytes auctionData)"),
                keccak256(abi.encode(order.info.trader)),
                order.pairId,
                order.entryTokenAddress,
                keccak256(bytes(order.side)),
                order.quantity,
                order.marginAmount,
                order.limitPrice,
                order.stopPrice,
                order.leverage,
                order.reduceOnly,
                order.closePosition,
                keccak256(order.auctionData)
            )
        );
    }

    function _verifySignature(PerpOrderV3 memory order, bytes memory sig) internal view returns (bool) {
        bytes32 digest = keccak256(abi.encodePacked("\x19\x01", DOMAIN_SEPARATOR, hashOrder(order)));
        address signer = recoverSigner(digest, sig);
        return signer == order.info.trader;
    }

    function recoverSigner(bytes32 digest, bytes memory sig) public pure returns (address) {
        (uint8 v, bytes32 r, bytes32 s) = splitSignature(sig);
        return ecrecover(digest, v, r, s);
    }

    function splitSignature(bytes memory sig) public pure returns (uint8 v, bytes32 r, bytes32 s) {
        require(sig.length == 65, "Invalid signature length");
        assembly {
            r := mload(add(sig, 32))
            s := mload(add(sig, 64))
            v := byte(0, mload(add(sig, 96)))
        }
    }

    function _executeOrderV3(PerpOrderV3 memory perpOrder, bytes memory sig) public {
        require(_verifySignature(perpOrder, sig), "Invalid signature");

        // Check nonce
        require(nonces[perpOrder.info.trader] == perpOrder.nonce, "Invalid nonce");
        nonces[perpOrder.info.trader]++; // Increment nonce

        // Order execution logic (omitted for brevity)
    }
}

```
