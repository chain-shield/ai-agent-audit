# Vulnerability details

## Impact
Every pool has a soft cap raised funds that should be met to launch the pool. This is fine but the problem is that pools are launched all at once only by the project manager. The `ILOManager.sol` does not have a function to launch a single pool.
This can lead to the inability to launch other pools that are ready to be launched.

People will not begin receiving rewards at launch time because no pool can launch, leading to potential missed profit opportunity or if a refund period ends to a loss of funds

## Proof of Concept
To replicate the problem, create and prepare a pool for a launch. By preparing I mean, investing funds to meet the soft cap.
Now create another pool that is not meeting that soft cap.
Since the `ILOManager.sol` has a launch function that tries to launch all the pools at once, no pool can be launched even though the first one is ready to be launched.

Here is a whole test file. Create a new test file inside the test directory where other foundry tests sit and place

```javascript
// SPDX-License-Identifier: MIT

pragma solidity =0.7.6;
pragma abicoder v2;

import "./IntegrationTestBase.sol";
import "../src/interfaces/IILOPool.sol";
import "../src/interfaces/IILOWhitelist.sol";
import "../src/interfaces/IILOVest.sol";

contract ILOPoolTest is IntegrationTestBase {
    address iloPool;

    function setUp() external {
        _setupBase();
        iloPool = _initPool(PROJECT_OWNER, _getInitPoolParams());

        vm.prank(PROJECT_OWNER);
        IILOWhitelist(iloPool).setOpenToAll(true);

        vm.prank(DUMMY_ADDRESS);
        IERC20(USDC).approve(iloPool, 1000000000 ether);

        _writeTokenBalance(USDC, DUMMY_ADDRESS, 10 ether);

        vm.prank(DUMMY_ADDRESS);
        vm.warp(SALE_START + 1);
        IILOPool(iloPool).buy(0.1 ether, DUMMY_ADDRESS);

        _prepareBuy();
        _buy(SALE_START + 1, 0.1 ether);
    }

    function _getListAddress() internal pure returns (address[] memory addresses) {
        addresses = new address[](3);
        addresses[0] = INVESTOR;
        addresses[1] = INVESTOR_2;
        addresses[2] = DUMMY_ADDRESS;
    }

    function _getListAddressFromAddress(address addr) internal pure returns (address[] memory addresses) {
        addresses = new address[](1);
        addresses[0] = addr;
    }

    function _prepareLaunch() internal {
        _prepareBuyFor(INVESTOR);
        _buyFor(INVESTOR, SALE_START + 1, 49000 ether);
        _prepareBuyFor(INVESTOR_2);
        _buyFor(INVESTOR_2, SALE_START + 1, 40000 ether);
        _buyFor(INVESTOR, SALE_START + 1, 1000 ether);
        _writeTokenBalance(SALE_TOKEN, iloPool, 95000 * 4 ether);
    }

    function _buy(uint64 buyTime, uint256 buyAmount) internal returns (uint256 tokenId, uint128 liquidity) {
        return _buyFor(INVESTOR, buyTime, buyAmount);
    }

    function _prepareBuy() internal {
        _prepareBuyFor(INVESTOR);
    }

    function _prepareBuyFor(address investor) internal {
        vm.prank(PROJECT_OWNER);
        IILOWhitelist(iloPool).batchWhitelist(_getListAddressFromAddress(investor));

        vm.prank(investor);
        IERC20(USDC).approve(iloPool, 1000000000 ether);

        _writeTokenBalance(USDC, investor, 1000000000 ether);
    }

    function _buyFor(address investor, uint64 buyTime, uint256 buyAmount)
        internal
        returns (uint256 tokenId, uint128 liquidity)
    {
        vm.warp(buyTime);
        vm.prank(investor);
        return IILOPool(iloPool).buy(buyAmount, investor);
    }

    function testRefundedPoolBlocksOtherPools() external {
        /* Prepare the already deployed pool for a launch */
        _prepareLaunch();

        /* Deploy a new pool that does not meet the soft cap */
        IILOManager.InitPoolParams memory params = _getInitPoolParams();
        _initPool(PROJECT_OWNER, _getInitPoolParams());

        /* Try to launch the pools */
        /* Assert that no pool can be launched even though the first pool has reached the soft cap */
        vm.warp(LAUNCH_START + 1);

        vm.expectRevert(bytes("SC"));
        IILOManager(iloManager).launch(params.uniV3Pool);
    }
}
```
