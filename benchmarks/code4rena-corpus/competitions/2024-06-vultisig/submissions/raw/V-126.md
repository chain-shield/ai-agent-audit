# Vulnerability details

## Impact
Admin can deploy an unlaunchable pool by setting the `softCap` higher then the `hardCap`

## Proof of Concept

Due to the following check in `launch()` found [here](https://github.com/code-423n4/2024-06-vultisig/blob/cb72b1e9053c02a58d874ff376359a83dc3f0742/src/ILOPool.sol#L274)

```solidity
require(totalRaised >= saleInfo.softCap, "SC");
```

And the check in `buy()` found [here](https://github.com/code-423n4/2024-06-vultisig/blob/cb72b1e9053c02a58d874ff376359a83dc3f0742/src/ILOPool.sol#L136)

```solidity
require(saleInfo.hardCap - totalRaised >= raiseAmount, "HC");
```

The `launch()` can never be successful because the softcap will never be less then or equal to the amount raised because no one can buy once the `hardCap` is reached. Due to the missing validation when initializing the pool, the pool becomes unlaunchable and traps users funds until they can get a refund.

```solidity
contract Base is Test {
    // Constants from mainnet
    uint256 internal constant _MAINNET_FORK_BLOCK = 20_076_176;
    address public constant WETH = 0xC02aaA39b223FE8D0A0e5C4F27eAD9083C756Cc2;
    address public constant RANDOM_TOKEN = 0x2541A36BE4cD39286ED61a3E6AFC2307602489d6;
    ISwapRouter public constant UNISWAP_V3_ROUTER = ISwapRouter(0xE592427A0AEce92De3Edee1F18E0157C05861564);
    IUniswapFactory public constant UNISWAP_V3_FACTORY = IUniswapFactory(0x1F98431c8aD98523631AE4a59f267346ea31F984);
    IUniswapPool public pool;


    // Mock addresses
    ILOManager public iloManager;
    ILOPool public iloPool;

    address owner = makeAddr('owner');
    address trader = makeAddr('trader');

    function setUp() external {
      vm.createSelectFork('https://rpc.ankr.com/eth', _MAINNET_FORK_BLOCK);

      // Setup vultisig state
      vm.startPrank(owner, owner);
        iloManager = new ILOManager();
        iloPool = new ILOPool();

        uint256 launchTime = block.timestamp + 100;

        IILOManager.InitProjectParams memory params = IILOManager.InitProjectParams({
            saleToken: RANDOM_TOKEN,
            raiseToken: WETH,
            fee: 500,
            initialPoolPriceX96: 1e18,
            launchTime: uint64(launchTime)
        });

        iloManager.initialize(owner, owner, address(iloPool), address(UNISWAP_V3_FACTORY), WETH, 1000, 1000);
        iloManager.initProject(params);

        pool = IUniswapPool(UNISWAP_V3_FACTORY.getPool(WETH, RANDOM_TOKEN, 500));


        IILOVest.VestingConfig[] memory vestingConfigs = new IILOVest.VestingConfig[](1);

        IILOVest.LinearVest[] memory linearVestings = new IILOVest.LinearVest[](1);

        linearVestings[0] = IILOVest.LinearVest({
            shares: 10000,
            start: uint64(launchTime + 100),
            end: uint64(launchTime + 10000)
        });

        vestingConfigs[0] = IILOVest.VestingConfig({
            shares: 10000,
            recipient: address(0),
            schedule: linearVestings
        });
        
        // Should revert the intialization, but it passes
        iloPool = ILOPool(payable(iloManager.initILOPool(IILOManager.InitPoolParams({
            uniV3Pool: address(pool),
            tickLower: -887272,
            tickUpper: 887272,
            hardCap: 100000000000000000,
            softCap: 1000000000000000000000,
            maxCapPerUser: 100000000000000000,
            start: uint64(block.timestamp-1),
            end: uint64(launchTime - 10),
            vestingConfigs: vestingConfigs
        }))));

        iloPool.setOpenToAll(true);
        
      vm.stopPrank();
  
    }
}
```

## Recommended Mitigation Steps

I recommend enforcing a check that `softCap <= hardCap` in `initILOPool` on the `ILOManager` this should ensure that pools config's will be valid
