# Vulnerability details

## Impact

There is no return data from the  `receiveApproval` function this means the `approveAndCall` function will work even if the spender contract doesn't implement the `receiveApproval` function. 

```solidity
    function approveAndCall(address spender, uint256 amount, bytes calldata extraData) external returns (bool) {
        // Approve the spender to spend the tokens
        _approve(msg.sender, spender, amount);

        // Call the receiveApproval function on the spender contract
@->     IApproveAndCallReceiver(spender).receiveApproval(msg.sender, amount, address(this), extraData);

        return true;
    }

```

This call will work with any `spender` contract that has a fallback function.

## Proof of Concept

From the test below, there are two contracts `Vultisig` and `VultisigV2` which contain the current code and the proposed fix respectively.

The `Spender` contract doesn't implement the `receiveApproval` function but it has a `fallback` function that does nothing.

Running the first test will be successful even though the spender doesn't implement this function.

```bash
forge test --mt testSuccessApproval
```

The second test will fail because we are checking for return data

```bash
forge test --mt RevertApproval
```


```solidity
// SPDX-License-Identifier: MIT 

pragma solidity =0.7.6;
pragma abicoder v2;


import {ERC20} from "@openzeppelin/contracts/token/ERC20/ERC20.sol";


interface IApproveAndCallReceiver {
    function receiveApproval(address from, uint256 amount, address token, bytes calldata extraData) external;
}

/**
 * @title ERC20 based Vultisig token contract
 */
contract Vultisig is ERC20 {

    constructor() ERC20("Vultisig Token", "VULT") {
        _mint(_msgSender(), 100_000_000 * 1e18);
    }

    function approveAndCall(address spender, uint256 amount, bytes calldata extraData) external returns (bool) {
        // Approve the spender to spend the tokens
        _approve(msg.sender, spender, amount);

        // Call the receiveApproval function on the spender contract
        IApproveAndCallReceiver(spender).receiveApproval(msg.sender, amount, address(this), extraData);
        return true;
    }
}

interface IApproveAndCallReceiverV2 {
    function receiveApproval(address from, uint256 amount, address token, bytes calldata extraData) external returns (bool);
}

contract VultisigV2 is ERC20 {

    constructor() ERC20("Vultisig Token", "VULT") {
        _mint(_msgSender(), 100_000_000 * 1e18);
    }

    function approveAndCall(address spender, uint256 amount, bytes calldata extraData) external returns (bool) {
        // Approve the spender to spend the tokens
        _approve(msg.sender, spender, amount);

        // Call the receiveApproval function on the spender contract
        return IApproveAndCallReceiverV2(spender).receiveApproval(msg.sender, amount, address(this), extraData);
    }
}

contract Spender {
    fallback () external {

    }
}

contract VultisigTest {

    Vultisig vultisig;
    VultisigV2 vultisigV2;
    Spender spender;

    function setUp() external {
        vultisig = new Vultisig();
        vultisigV2 = new VultisigV2();
        spender  = new Spender();
    }


    function testSuccessApproval() public {
        vultisig.approveAndCall(address(spender), 1 ether, "0x");
    }

    function testRevertApproval() public {
        vultisigV2.approveAndCall(address(spender), 1 ether, "0x");
    }
}
```
