# Vulnerability details

## Impact
During the whitelist period of when a `Vultisig` contract launches with whitelisting, the user will be able to sell into the Uniswap V3 pool, but this is not accounted for and could block them from buying tokens after they sell if they reached the `maxAddressCap`

## Proof of Concept

```solidity
// SPDX-License-Identifier: MIT
pragma solidity ^0.8.0;

import "forge-std/Test.sol";
import {VultisigWhitelisted }  from '../poc/VultisigWhitelisted.sol';
import {Whitelist} from '../poc/Whitelist.sol';


contract Base is Test {
    VultisigWhitelisted vultisig;
    Whitelist whitelist;

    // Mock addresses
    address public owner = address(100);
    address public pool = address(200);
    address public oracle = address(300);
    address public user = address(500);

    function setUp() external {
      vm.startPrank(owner);
        whitelist = new Whitelist();
        vultisig = new VultisigWhitelisted();
        vultisig.setWhitelistContract(address(whitelist));

        // Configure the whitelist
        whitelist.setLocked(false);
        whitelist.setPool(pool);
        whitelist.setAllowedWhitelistIndex(1000);
        whitelist.setOracle(oracle);
        whitelist.setMaxAddressCap(1000);
        whitelist.setVultisig(address(vultisig));

        // User is whitelisted
        whitelist.addWhitelistedAddress(user);

        // Transfer from the owner to our mock pool
        vultisig.transfer(pool, 1000);
      vm.stopPrank();
  
    }
}


contract PoC is Base {
    function testUserCannotRebuyTokensIfSold() external {
      // Mock oracle returning 1000 which is the max amount per address
      vm.mockCall(oracle, abi.encodeWithSignature('peek(uint256)', 1000), abi.encode(1000));

      // In a real world scenario this would be the user buying tokens as the pool is transfering tokens to the user
      // Setting it up like this for simplicity
      vm.prank(pool);
      vultisig.transfer(user, 1000);

      // User sells their tokens into the pool
      vm.prank(user);
      vultisig.transfer(pool, 1000);

      // User tries to buy tokens again but it will revert
      vm.expectRevert(Whitelist.MaxAddressCapOverflow.selector);
      vm.prank(pool);
      vultisig.transfer(user, 1000);
    }
}
```

## Recommended Mitigation Steps

There are two possible ways to mitigate this issue

1. Make the pools "buy only" during the whitelist phase and prevent selling into the pool to avoid this issue by adding an extra if statement in `checkWhitelist()`

```solidity
if(to == _pool && from != owner()) revert();
```

2. Account for the amount sold by adding a check in `checkWhitelist()` that reduces the amount `_contributed[address(user)]` by the amount sold

```solidity
        if (to == _pool && from != owner()) {
          
            // Other checks


            uint256 estimatedETHAmount = IOracle(_oracle).peek(amount);

            _contributed[from] -= estimatedETHAmount;
        }
```
