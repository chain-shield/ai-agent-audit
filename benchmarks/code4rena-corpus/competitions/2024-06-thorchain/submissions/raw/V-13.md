# Vulnerability details

## Impact
The `transferOut` function is designed to transfer assets (either ETH or ERC20 tokens) from the contract to a specified recipient. However, the function originally reduces the allowance for `ERC20` tokens before confirming the transfer's success. This could lead to inconsistencies if the transfer fails. Updating the allowance before confirming the transfer may cause discrepancies and potential issues in the contract's state, leading to unexpected behavior and potential loss of funds.

## Recommended Mitigation Steps

Update the `transferOut` function to only modify the allowance after confirming a successful ERC20 token transfer. Here’s the revised code:

```solidity
function transferOut(
    address payable to,
    address asset,
    uint amount,
    string memory memo
) public payable nonReentrant {
    uint safeAmount;
    if (asset == address(0)) {
        // Handle ETH transfer
        safeAmount = msg.value;
        bool success = to.send(safeAmount); // Attempt to send ETH
        if (!success) {
            // If sending ETH fails, return ETH to the sender
            payable(address(msg.sender)).transfer(safeAmount);
        }
    } else {
        // Attempt the ERC20 transfer
        (bool success, bytes memory data) = asset.call(
            abi.encodeWithSignature("transfer(address,uint256)", to, amount)
        );

        // Check for successful transfer
        require(success && (data.length == 0 || abi.decode(data, (bool))));

       // Only reduce the allowance if the transfer was successful
        _vaultAllowance[msg.sender][asset] -= amount;

        safeAmount = amount;
    }
    emit TransferOut(msg.sender, to, asset, safeAmount, memo);
}
```
