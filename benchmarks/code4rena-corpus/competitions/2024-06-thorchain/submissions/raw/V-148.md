# Vulnerability details

## Impact
Gas asset may be stuck in the `THORChain_Router`.
The following functions: `THORChain_Router::transferOut`, `THORChain_Router::_transferOutV5`, `THORChain_Router::_transferOutAndCallV5` are missing an assertion that guarantees that no gas asset is stored inside the Router. Such assertion is available in `THORChain_Router::_deposit`, but not in the described functions.

## Proof of Concept
Please add a new test file called for example `4_Missing_msg.value_0_assertion.js` and paste the code from below.
The code asserts that you may stuck gas asset in the router by expecting the balance after to be greater than the balance before the calls to: `THORChain_Router::transferOut`, `THORChain_Router::_transferOutV5`, `THORChain_Router::_transferOutAndCallV5`

```javascript
const Router = artifacts.require('THORChain_Router');
const Aggregator = artifacts.require('THORChain_Aggregator');
const SushiRouter = artifacts.require('SushiRouterSmol');
const Token = artifacts.require('ERC20Token');
const Weth = artifacts.require('WETH');
const BigNumber = require('bignumber.js');
const { expect } = require('chai');
function BN2Str(BN) {
  return new BigNumber(BN).toFixed();
}
function getBN(BN) {
  return new BigNumber(BN);
}

var ROUTER, ASGARD, AGG, WETH, SUSHIROUTER, TOKEN, WETH, USER1;
var ETH = '0x0000000000000000000000000000000000000000';
const _10 = '10000000000000000000';

describe('Missing assertion for gas asset being 0 in transfer out function', function () {
  let accounts;

  before(async function () {
    accounts = await web3.eth.getAccounts();
    ROUTER = await Router.new();
    TOKEN = await Token.new();
    USER1 = accounts[0];
    ASGARD = accounts[3];
    WETH = await Weth.new();
    SUSHIROUTER = await SushiRouter.new(WETH.address);
    AGG = await Aggregator.new(WETH.address, SUSHIROUTER.address);
  });

  it('Should stuck gas asset in the router', async function () {
    /* Get the starting router balance */
    const startingRouterBalance = await web3.eth.getBalance(ROUTER.address);

    /* Log the router starting balance */
    console.log(`ROUTER starting balance: ${BN2Str(startingRouterBalance)}`);

    /* Send the gas asset to the router with transferOut */
    await ROUTER.transferOut(USER1, TOKEN.address, '0', '', {
      from: ASGARD,
      value: _10,
    });

    /* Send the gas asset to the router with transferOutV5 */
    await ROUTER.transferOutV5([USER1, TOKEN.address, '0', ''], {
      from: ASGARD,
      value: _10,
    });

    /* Send the gas asset to the router with transferOutAndCallV5 */
    await ROUTER.transferOutAndCallV5(
      [
        AGG.address,
        TOKEN.address,
        '0',
        ETH,
        USER1,
        '0',
        'MEMO',
        '0x', // empty payload
        'bc123', // dummy address
      ],
      {
        from: ASGARD,
        value: _10,
      }
    );

    /* Get the ending router balance */
    const endingRouterBalance = await web3.eth.getBalance(ROUTER.address);

    /* Log the router ending balance */
    console.log(`ROUTER ending balance: ${BN2Str(endingRouterBalance)}`);

    /* Assert that the router has stuck the gas asset */
    expect(
      getBN(endingRouterBalance).isGreaterThan(getBN(startingRouterBalance))
    ).to.equal(true);
  });
});
```

## Recommended Mitigation Steps
Add require statements to functions `THORChain_Router::transferOut`, `THORChain_Router::_transferOutV5`, `THORChain_Router::_transferOutAndCallV5` to ensure that `msg.value` inside the functions is 0. This can be placed just before reducing the allowance, similar to the deposit
