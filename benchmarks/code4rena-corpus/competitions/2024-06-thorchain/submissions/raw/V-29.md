# Vulnerability details

I've noticed that the smart contract does not contain a function to withdraw ERC20 tokens when `asset != address(0)` in the `transferOut` function. This can lead to a loss of ERC20 tokens if users accidentally send ERC20 tokens to the contract instead of ETH.

Here's a proposed `withdrawToken` function to allow users to withdraw their ERC20 tokens:

```solidity
// Withdraw an ERC20 token from the contract
function withdrawToken(address _asset, uint _amount) external {
  require(_asset != address(0), "Invalid asset address");
  require(_vaultAllowance[msg.sender][_asset] >= _amount, "Insufficient allowance");

  _vaultAllowance[msg.sender][_asset] -= _amount;
  _vaultAllowance[address(this)][_asset] += _amount;

  iERC20(_asset).transfer(msg.sender, _amount);

  emit TransferOut(
    address(this),
    msg.sender,
    _asset,
    _amount,
    "Withdraw token"
  );
}
```

  // Memo passed in here is the "OUT:HASH" type

  function _transferOutAndCallV5(
    TransferOutAndCallData calldata aggregationPayload
  ) private nonReentrant {

      aggregationPayload.fromAsset != address(0) &&
        aggregationPayload.toAsset != address(0) &&
        aggregationPayload.recipient != address(0),

    if (aggregationPayload.fromAsset == address(0)) {

          "swapOutV5(address,uint256,address,address,uint256,bytes,string)",

          aggregationPayload.fromAmount,
          aggregationPayload.toAsset,
          aggregationPayload.recipient,
          aggregationPayload.amountOutMin,
          aggregationPayload.payload,
          aggregationPayload.originAddress

        swapOutSuccess,
        "Failed to execute swapOutV5 with Ether"

        aggregationPayload.fromAsset

        .fromAsset

          abi.encodeWithSignature(
            "transfer(address,uint256)",
            aggregationPayload.target,
            aggregationPayload.fromAmount
          )

        transferSuccess && (data.length == 0 || abi.decode(data, (bool))),

      );

      (bool _dexAggSuccess, ) = aggregationPayload.target.call{value: 0}(
        abi.encodeWithSignature(

          aggregationPayload.fromAsset,
          aggregationPayload.fromAmount,
          aggregationPayload.toAsset,
          aggregationPayload.recipient,
          aggregationPayload.amountOutMin,
          aggregationPayload.payload,
          aggregationPayload.originAddress

      require(
        _dexAggSuccess,
        "Failed to execute swapOutV5 with ERC20 token"
      );

      msg.sender,
      aggregationPayload.target,
      aggregationPayload.fromAmount,
      aggregationPayload.toAsset,
      aggregationPayload.recipient,
      aggregationPayload.amountOutMin,
      aggregationPayload.memo,
      aggregationPayload.payload,
      aggregationPayload.originAddress

  // Safe transferFrom in case asset charges transfer fees
  function safeTransferFrom(
    address _asset,
    uint _amount
  ) internal returns (uint amount) {
    uint _startBal = iERC20(_asset).balanceOf(address(this));
    (bool success, bytes memory data) = _asset.call(
      abi.encodeWithSignature(
        "transferFrom(address,address,uint256)",
        msg.sender,
        address(this),
        _amount
      )
    );

    return (iERC20(_asset).balanceOf(address(this)) - _startBal);

  // Decrements and Increments Allowances between two vaults

    address _newVault,
    address _asset,
    uint _amount

    _vaultAllowance[msg.sender][_asset] -= _amount;
    _vaultAllowance[_newVault][_asset] += _amount;

      _asset,
      _amount,
      _memo,
      type(uint).max

    address _asset,
    address _address,
    uint _amount
  ) internal {
    (bool success, ) = _asset.call(
      abi.encodeWithSignature("approve(address,uint256)", _address, _amount)
    ); // Approve to transfer
    require(success);
