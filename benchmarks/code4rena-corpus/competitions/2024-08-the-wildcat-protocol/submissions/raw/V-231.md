# Vulnerability details

## Proof of Concept
In the current implementation, the logic relies on the `getEscrowAddress` function to return the appropriate escrow contract address based on the parameters provided (`borrower`, `account`, and `asset`). If the address returned is not an actual escrow contract but instead points to an unrelated contract, the function’s design could lead to unintended behaviour. 
Here are some considerations:

Potential Scenarios
Existing Non-Escrow Contract:

If `getEscrowAddress` returns an address that is valid (non-zero) but not related to a `WildcatSanctionsEscrow` contract, the check `escrowContract.code.length != 0` will still return true, and the function will return that address.
This could lead to scenarios where the protocol mistakenly believes it has a valid escrow in place, potentially allowing actions that should only be performed by a true escrow contract.
Address Collisions:





## Recommended Mitigation Steps
To mitigate this risk consider implementing any of the following:

1. Introduce a mechanism to ensure the contract at the returned address is indeed a `WildcatSanctionsEscrow` contract. This could involve checking for specific functions that should exist in the escrow contract. For example, you might use interface checks or a specific method signature to validate that the contract is correct.
Contract Metadata:

2. Maintain a registry of valid escrow contract addresses. This would allow the function to verify against known good contracts rather than relying solely on the code size check.

```javascript 
// SPDX-License-Identifier: MIT
pragma solidity ^0.8.0;

contract EscrowRegistry {
    // Mapping to keep track of valid escrow contract addresses
    mapping(address => bool) public validEscrowContracts;

    // Event to emit when a new escrow is registered
    event EscrowRegistered(address escrowContract);

    // Function to register a valid escrow contract
    function registerEscrow(address escrowContract) external {
        require(escrowContract != address(0), "Invalid address");
        validEscrowContracts[escrowContract] = true;
        emit EscrowRegistered(escrowContract);
    }

    // Function to check if an escrow is valid
    function isValidEscrow(address escrowContract) external view returns (bool) {
        return validEscrowContracts[escrowContract];
    }
}

contract WildcatSanctions {
    EscrowRegistry public escrowRegistry;

    // Constructor to set the escrow registry
    constructor(address _escrowRegistry) {
        escrowRegistry = EscrowRegistry(_escrowRegistry);
    }

    event NewSanctionsEscrow(address borrower, address account, address asset);
    event SanctionOverride(address borrower, address escrowContract);

    // Main function to create or retrieve an escrow contract
    function createEscrow(
        address borrower,
        address account,
        address asset
    ) public returns (address escrowContract) {
        escrowContract = getEscrowAddress(borrower, account, asset);

        // Skip creation if the address code size is non-zero and it's a valid escrow contract
        if (escrowContract.code.length != 0 && escrowRegistry.isValidEscrow(escrowContract)) {
            return escrowContract;
        }

        // Create new escrow contract if it doesn't exist
        escrowContract = address(new WildcatSanctionsEscrow{ salt: _deriveSalt(borrower, account, asset) }());

        // Register the new escrow contract
        escrowRegistry.registerEscrow(escrowContract);

        emit NewSanctionsEscrow(borrower, account, asset);
        emit SanctionOverride(borrower, escrowContract);
    }

}
```
