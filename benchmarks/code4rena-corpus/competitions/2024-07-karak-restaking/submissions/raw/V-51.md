# Vulnerability details

## Impact
After the entire balance of a vault is slashed, the vault can no longer be used, as the deposit/mint function will stop working correctly.
This can be an issue as the operator may still be active in the DSS, and they won't be able to simply re stake back assets into the vault, but they will have create and stake a new vault, waiting either 9 or 18 days to get back to provide their services/rewards as before the slashing event.

## Proof of Concept
The [Vault](https://github.com/code-423n4/2024-07-karak/blob/f5e52fdcb4c20c4318d532a9f08f7876e9afb321/src/Vault.sol#L29C10-L29C15) contract is loosely based on a ERC4626, meaning that the *totalAssets()* will be used, as a dividend or multiplier, to compute both the amount of shares to allocate to the user when depositing, and the amount of assets to get when minting.
Having 0 total assets, while having more than 0 shares present in the vault, will cause huge inconsistencies in the amounts being deposited/minted, making the vault unusable in the future.

It is possible to slash the total assets present in the vault, either by setting *dssMaxSlashablePercentage* to 100% or if there are enough redeem queued before finalizing the slashing. Also depending on the context, slashing the full amount to a vault doesn't imply that the operator is no longer part of the DSS, as they might have multiple vaults (up to 32), or, as stated above, they might be slashed for less than 100%, but get the full amount due to how the amount being slashed is computed in [slashAssets](https://github.com/code-423n4/2024-07-karak/blob/f5e52fdcb4c20c4318d532a9f08f7876e9afb321/src/Vault.sol#L193C14-L193C25):
```
transferAmount = Math.min(totalAssets(), totalAssetsToSlash)
```
Since the vault will become unusable (due to inconsistencies in the deposits, while having no longer assets stake into it), if the operator wants to maintain the same throughput of services provided/rewards obtained, they will have to create an stake a new vault, which will be affected by the 9 days cooldown (or 18, if they have reached the maximum amount of vaults, and will have to unstake and restake).
Causing more losses than intended to the operator, as they will lose the incentives for 9/18 days on top of the slashed amount.

Here is a foundry script that shows the above mentioned inconsistencies in the deposit after a 100% slashing event.
Place it in the `/test/core/` folder to preserve dependencies:
```
// SPDX-License-Identifier: SEE LICENSE IN LICENSE
pragma solidity ^0.8.25;

import "forge-std/Test.sol";
import "forge-std/console.sol";

import "solady/src/utils/SafeTransferLib.sol";

import "../../src/Core.sol";
import "../../src/Vault.sol";
import "../../src/entities/VaultLib.sol";
import "../../src/entities/Operator.sol";
import "../../src/entities/Withdraw.sol";
import "../../src/interfaces/Errors.sol";
import "../../src/interfaces/IVault.sol";
import "../helpers/ProxyDeployment.sol";
import "../../src/interfaces/Constants.sol";
import "../helpers/contracts/ERC20Mintable.sol";
import "../helpers/OperatorHelper.sol";



import "../../src/SlashingHandler.sol";
import "../../src/Querier.sol";

import "./../helpers/ProxyDeployment.sol";
import "./../helpers/contracts/ERC20Mintable.sol";
import "./../helpers/contracts/MockDSS.sol";
import "./../helpers/OperatorContract.sol";

contract VaultTest is Test {
    Vault vault;
    Core core;
    IDSS dss = IDSS(address(new MockDSS()));
    ERC20Mintable depositToken;

    address manager = address(11);
    address notOwner = address(12);
    address operator = address(13);
    address proxyAdmin = address(14);
    address beneficiary = address(15);
    address slashingHandler = address(16);
    address slashingVetoCommittee = address(17);

    function setUp() public {
        uint32 hookCallGasLimit = 500_000;
        uint32 hookGasBuffer = 40_000;
        uint32 supportsInterfaceGasLimit = 20_000;
        // Setup core
        depositToken = new ERC20Mintable();
        depositToken.initialize("Test", "TST", 18);
        address vaultImpl = address(new Vault());
        core = Core(ProxyDeployment.factoryDeploy(address(new Core()), proxyAdmin));
        core.initialize(
            vaultImpl, manager, slashingVetoCommittee, hookCallGasLimit, supportsInterfaceGasLimit, hookGasBuffer
        );

        // Allowlist asset
        address[] memory assets = new address[](1);
        assets[0] = address(depositToken);

        // Slashing Handler for the asset
        IERC20[] memory supportedAssets = new IERC20[](1);
        supportedAssets[0] = IERC20(address(depositToken));
        slashingHandler = ProxyDeployment.deployTransparentUpgradeable(address(new SlashingHandler()), proxyAdmin);
        ISlashingHandler(slashingHandler).initialize(proxyAdmin, supportedAssets);


        address[] memory slashingHandlers = new address[](1);
        slashingHandlers[0] = slashingHandler;
        core.allowlistAssets(assets, slashingHandlers);

        // Deploy Vaults
        VaultLib.Config[] memory vaultConfigs = new VaultLib.Config[](1);
        vaultConfigs[0] = VaultLib.Config({
            asset: address(depositToken),
            decimals: 18,
            operator: operator,
            name: "TestVault",
            symbol: "TV",
            extraData: bytes("")
        });


        vm.prank(address(dss));

        core.registerDSS(100 * Constants.ONE_WAD);

        vm.startPrank(operator);
        IKarakBaseVault[] memory vaults = core.deployVaults(vaultConfigs, address(0));
        vault = Vault(address(vaults[0]));
        core.registerOperatorToDSS(dss, "");

        Operator.StakeUpdateRequest memory req = Operator.StakeUpdateRequest(address(vault), dss, true);
        Operator.QueuedStakeUpdate memory queued = core.requestUpdateVaultStakeInDSS(req);
        vm.warp(block.timestamp + Constants.MIN_STAKE_UPDATE_DELAY + 1);
        core.finalizeUpdateVaultStakeInDSS(queued);

        vm.stopPrank();
    }

    function mintAndDepositFor(uint256 amount, address user) public returns (uint256 shares) {
        vm.startPrank(user);
        depositToken.mint(user, amount);
        depositToken.approve(address(vault), amount);
        uint256 expectedShares = vault.convertToShares(amount);
        shares = vault.deposit(amount, user, expectedShares);
        vm.stopPrank();
    }

    function test_vault() public {
 

        address depositor1 = address(0x123);
        address depositor2 = address(0x456);


        address[] memory  vaultsAdd  = new address[](1);
        vaultsAdd[0] = address(vault);

        uint96[] memory  percentage  = new uint96[](1);
        percentage[0] = uint96(100 * Constants.ONE_WAD);

     
        mintAndDepositFor(1000, depositor1);


        vm.startPrank(address(dss));
        SlasherLib.SlashRequest memory slashingReq = SlasherLib.SlashRequest({
            operator: operator,
            slashPercentagesWad: percentage,
            vaults: vaultsAdd
        });
        
        SlasherLib.QueuedSlashing memory queuedSlashing = core.requestSlashing(slashingReq);

        vm.warp(block.timestamp + Constants.SLASHING_VETO_WINDOW + 1);
        vm.stopPrank();



        core.finalizeSlashing(queuedSlashing);

        assertEq(depositToken.balanceOf(address(vault)), 0);  

        mintAndDepositFor(1000, depositor2);

        console.log(vault.balanceOf(depositor1));
        console.log(vault.balanceOf(depositor2));

        //Altough the two deposits are identical, the amounts being minted are extremely different
        assertEq(1000000 + vault.balanceOf(depositor1), vault.balanceOf(depositor2));
    }


}
```
