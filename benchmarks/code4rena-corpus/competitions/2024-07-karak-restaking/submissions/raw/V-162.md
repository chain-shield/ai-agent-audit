# Vulnerability details

Description
The Querier contract has a constructor that initializes the core state variable with a provided address. However, the constructor does not include a check to prevent the initialization with a zero address (address(0)).

Vulnerable Code:
constructor(address coreAddress) {
    core = ICore(coreAddress);
}

Issue: If the coreAddress is set to address(0), it may lead to unexpected behavior and security vulnerabilities. In many cases, smart contracts use zero addresses as placeholders or indicators of invalid states, which can lead to issues such as:

Unintended Behavior: Functions that rely on the core address might fail or behave unexpectedly when it is zero.
Security Risks: Contracts interacting with the Querier might assume the core address is valid and attempt operations based on this assumption, which could lead to security vulnerabilities or loss of funds.


Proof of Vulnerability:
create a test file name it QuerierTest.t.sol and paste this code inside


// SPDX-License-Identifier: SEE LICENSE IN LICENSE
pragma solidity ^0.8.25;

import "forge-std/Test.sol";
import "../src/Querier.sol";
import "../src/Core.sol"; 

contract QuerierTest is Test {
    Querier querier;
    Core core;

    function setUp() public {
        // Deploy a Core contract
        core = new Core();
    }

    function testInitializeWithZeroCoreAddress() public {
        // Try initializing with address(0) for coreAddress
        vm.expectRevert("Invalid address: zero core address");
        new Querier(address(0));
    }

    function testInitializeWithValidCoreAddress() public {
        // Initialize with a valid core address
        querier = new Querier(address(core));

        // Check that the core address is set correctly
        assertEq(address(querier.core()), address(core));
    }
}
