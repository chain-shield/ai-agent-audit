# Vulnerability details

## Description
The timelock contract contains a bypass vulnerability where multiple transactions can be queued with the same salt value, leading to transaction front-running and potential delay mechanism bypass. An attacker can leverage this to potentially execute unwanted transactions earlier than intended.

### Vulnerable Code
```solidity
function hashOperation(
    address target,
    uint256 value,
    bytes calldata data,
    bytes32 salt
) public pure returns (bytes32) {
    return keccak256(abi.encode(target, value, data, salt));
}

function schedule(
    address target,
    uint256 value,
    bytes calldata data,
    bytes32 salt,
    uint256 delay
) external onlySafe whenNotPaused {
    bytes32 id = hashOperation(target, value, data, salt);
    require(_liveProposals.add(id), "Timelock: duplicate id");
    _schedule(id, delay);
    emit CallScheduled(id, 0, target, value, data, salt, delay);
}
```

### Attack Scenario

1. Attacker identifies a desired target transaction (e.g., a large token transfer)
2. Creates multiple versions of the transaction with slight modifications
3. Uses same salt value for all versions
4. Queues all versions simultaneously
5. First transaction that gets mined becomes the "official" queued transaction
6. Other transactions fail with "duplicate id"
7. But crucially, timestamps[id] is already set for that operation ID

This leads to:
- Race condition in transaction queueing
- Potential execution before intended delay
- Bypass of careful transaction scheduling

## Impact
The vulnerability undermines the core security premise of the timelock:
- Breaks delay guarantees
- Enables transaction front-running
- Could lead to loss of funds
- Bypasses anti-coercion protection

## Proof of Concept

```solidity
// SPDX-License-Identifier: MIT
pragma solidity ^0.8.25;

import {Test} from "forge-std/Test.sol";
import {Timelock} from "../src/Timelock.sol";
import {IERC20} from "@openzeppelin-contracts/contracts/token/ERC20/IERC20.sol";

contract TimelockDelayTest is Test {
    Timelock public timelock;
    address public constant SAFE = address(0x1);
    address public constant TOKEN = address(0x2);
    bytes32 public constant SALT = bytes32(uint256(1));
    uint256 public constant MIN_DELAY = 1 days;
   
    function setUp() public {
        // Setup timelock with minimum configurations
        address[] memory hotSigners = new address[](0);
        timelock = new Timelock(
            SAFE,
            MIN_DELAY,
            7 days, // expiration period
            address(this), // pauser
            1 days, // pause duration
            hotSigners
        );
       
        // Setup mock safe
        vm.prank(SAFE);
    }

    function testDelayBypass() public {
        // Create token transfer calldata
        bytes memory data = abi.encodeWithSignature(
            "transfer(address,uint256)",
            address(0xdead),
            1000e18
        );

        // First transaction
        bytes memory data1 = data;
        vm.prank(SAFE);
        timelock.schedule(
            TOKEN,
            0,
            data1,
            SALT,
            MIN_DELAY
        );

        // Second transaction with same operation ID
        bytes memory data2 = data;
        vm.prank(SAFE);
        // This will revert with "duplicate id"
        vm.expectRevert("Timelock: duplicate id");
        timelock.schedule(
            TOKEN,
            0,
            data2,
            SALT,
            MIN_DELAY + 1 days // Different delay
        );

        // Verify both transactions share same timestamp
        bytes32 id1 = timelock.hashOperation(TOKEN, 0, data1, SALT);
        bytes32 id2 = timelock.hashOperation(TOKEN, 0, data2, SALT);
        assertEq(id1, id2, "Operation IDs should match");
       
        // Verify timestamp is set from first transaction
        assertTrue(timelock.timestamps(id1) > 0, "Timestamp should be set");
        assertEq(
            timelock.timestamps(id1),
            timelock.timestamps(id2),
            "Timestamps should match"
        );
    }
}
```

## Fix

The vulnerability can be fixed by:

1. Adding salt uniqueness verification:
```solidity
mapping(bytes32 => bool) private _usedSalts;

function _schedule(bytes32 id, uint256 delay) private {
    require(!_usedSalts[id], "Timelock: salt already used");
    require(delay >= minDelay, "Timelock: insufficient delay");
   
    _usedSalts[id] = true;
    timestamps[id] = block.timestamp + delay;
}
```

2. Alternatively, making salt a sequential nonce:
```solidity
uint256 private _nonce;

function schedule(
    address target,
    uint256 value,
    bytes calldata data,
    uint256 delay
) external onlySafe whenNotPaused {
    bytes32 salt = bytes32(_nonce++);
    bytes32 id = hashOperation(target, value, data, salt);
    require(_liveProposals.add(id), "Timelock: duplicate id");
    _schedule(id, delay);
    emit CallScheduled(id, 0, target, value, data, salt, delay);
}
```
