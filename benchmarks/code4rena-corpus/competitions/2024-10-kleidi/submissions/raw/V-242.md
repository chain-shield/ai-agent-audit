# Vulnerability details

## Description
The RecoverySpell contract is designed to allow recovery of a Gnosis Safe by transferring ownership after a delay period and with sufficient signatures from new owners. The `recoveryThreshold` parameter specifies how many valid signatures are required to execute the recovery.

## Vulnerability Details
The contract validates that enough signatures are provided and that each signature is valid, but it fails to ensure that enough UNIQUE valid signatures are present. The current check only verifies that the signature array length meets the threshold, but after duplicate and invalid signatures are rejected, the actual count of valid signatures could be below the required threshold.

## Vulnerable Code Snippets

```solidity
// Initial threshold check only validates array length
require(
    recoveryThreshold <= v.length,
    "RecoverySpell: Not enough signatures"
);

// Later signature validation
bytes32 digest = getDigest();
for (uint256 i = 0; i < v.length; i++) {
    address recoveredAddress = ECDSA.recover(digest, v[i], r[i], s[i]);
    bool valid;

    assembly ("memory-safe") {
        valid := tload(recoveredAddress)
        if eq(valid, 1) { tstore(recoveredAddress, 0) }
    }

    require(
        valid && recoveredAddress != address(0),
        "RecoverySpell: Invalid signature"
    );
}
```

## Impact
An attacker could execute recovery with fewer valid signatures than required by padding the signature arrays with invalid or duplicate signatures. This breaks the fundamental security assumption of the recovery mechanism.

## Attack Scenario

1. Assume `recoveryThreshold = 3` (requiring 3 valid signatures)
2. Attacker has only 2 valid signatures
3. Attacker constructs signature arrays:
   - First two elements are valid signatures
   - Third element is a duplicate of first signature
4. The function validates because:
   - `v.length >= recoveryThreshold` check passes
   - Each signature is technically "valid" 
   - But only 2 unique valid signatures exist

## Fix

```solidity
function executeRecovery(
    address previousModule,
    uint8[] calldata v,
    bytes32[] calldata r,
    bytes32[] calldata s
) external {
    // ... existing checks ...
    
    bytes32 digest = getDigest();
    uint256 validSignatureCount = 0;  // Add counter for valid signatures
    
    for (uint256 i = 0; i < v.length; i++) {
        address recoveredAddress = ECDSA.recover(digest, v[i], r[i], s[i]);
        bool valid;
        
        assembly ("memory-safe") {
            valid := tload(recoveredAddress)
            if eq(valid, 1) { 
                tstore(recoveredAddress, 0)
                // Increment only when signature is valid and not duplicate
                validSignatureCount := add(validSignatureCount, 1)
            }
        }
        
        require(
            valid && recoveredAddress != address(0),
            "RecoverySpell: Invalid signature"
        );
    }
    
    // Add check for valid signature count
    require(
        validSignatureCount >= recoveryThreshold,
        "RecoverySpell: Not enough valid signatures"
    );
    
    // ... rest of the function ...
}
```

## Test



```solidity
// SPDX-License-Identifier: UNLICENSED
pragma solidity ^0.8.25;

import {Test} from "forge-std/Test.sol";
import {RecoverySpell} from "../src/RecoverySpell.sol";

contract RecoverySpellTest is Test {
    RecoverySpell public spell;
    address[] public newOwners;
    uint256[] public privateKeys;
    
    function setUp() public {
        // Create 3 owners with known private keys
        for(uint i = 0; i < 3; i++) {
            uint256 privateKey = uint256(keccak256(abi.encodePacked("owner", i)));
            address owner = vm.addr(privateKey);
            newOwners.push(owner);
            privateKeys.push(privateKey);
        }
        
        spell = new RecoverySpell(
            newOwners,
            address(0x1),  // mock safe address
            2,             // safe threshold
            3,             // recovery threshold - requires 3 signatures
            1 days         // delay
        );
    }
    
    function testVulnerable_InsufficientValidSignatures() public {
        // Advance time past delay
        vm.warp(block.timestamp + 1 days + 1);
        
        // Generate 2 valid signatures
        bytes32 digest = spell.getDigest();
        (uint8[] memory v, bytes32[] memory r, bytes32[] memory s) = generateSignatures(digest, 2);
        
        // Add duplicate of first signature
        v.push(v[0]);
        r.push(r[0]);
        s.push(s[0]);
        
        // This should revert but doesn't - vulnerability
        spell.executeRecovery(address(0), v, r, s);
    }
    
    function testFixed_InsufficientValidSignatures() public {
        // Advance time past delay
        vm.warp(block.timestamp + 1 days + 1);
        
        // Generate 2 valid signatures
        bytes32 digest = spell.getDigest();
        (uint8[] memory v, bytes32[] memory r, bytes32[] memory s) = generateSignatures(digest, 2);
        
        // Add duplicate of first signature
        v.push(v[0]);
        r.push(r[0]);
        s.push(s[0]);
        
        // Should revert with "Not enough valid signatures"
        vm.expectRevert("RecoverySpell: Not enough valid signatures");
        spell.executeRecovery(address(0), v, r, s);
    }
    
    function generateSignatures(bytes32 digest, uint256 count) internal view returns (uint8[] memory v, bytes32[] memory r, bytes32[] memory s) {
        require(count <= privateKeys.length, "Too many signatures requested");
        
        v = new uint8[](count);
        r = new bytes32[](count);
        s = new bytes32[](count);
        
        for(uint i = 0; i < count; i++) {
            (v[i], r[i], s[i]) = vm.sign(privateKeys[i], digest);
        }
    }
}

```

The fix ensures that not only are signatures valid, but there are enough unique valid signatures to meet the recovery threshold. The test demonstrates how the vulnerability could be exploited and verifies that the fix prevents the attack.
