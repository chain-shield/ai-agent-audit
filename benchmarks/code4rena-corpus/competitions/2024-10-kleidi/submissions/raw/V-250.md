# Vulnerability details

## Description

The Timelock contract implements role-based access control through OpenZeppelin's AccessControl. It manages critical roles including DEFAULT_ADMIN_ROLE and HOT_SIGNER_ROLE. The contract overrides role management functions to prevent modification of the DEFAULT_ADMIN_ROLE.

The `revokeRole` function override in the Timelock contract is inconsistent with other role management functions (`grantRole` and `renounceRole`). While the implementation correctly prevents revoking the DEFAULT_ADMIN_ROLE, it fails to override the IAccessControl interface, potentially causing interface compatibility issues and inconsistent role management behavior.

## Vulnerable Code Snippet
```solidity
// Current implementation - missing IAccessControl override
function revokeRole(bytes32 role, address account) public override {
    require(
        role != DEFAULT_ADMIN_ROLE,
        "Timelock: cannot revoke admin role"
    );
    super.revokeRole(role, account);
}

// Correct implementation in other functions
function grantRole(bytes32 role, address account)
    public
    override(AccessControl, IAccessControl)
{
    require(role != DEFAULT_ADMIN_ROLE, "Timelock: cannot grant admin role");
    super.grantRole(role, account);
}

function renounceRole(bytes32 role, address account)
    public
    override(AccessControl, IAccessControl)
{
    require(role != DEFAULT_ADMIN_ROLE, "Timelock: cannot renounce admin role");
    super.renounceRole(role, account);
}
```

## Impact
1. Interface Compatibility:
   - Potential compilation errors in certain development environments
   - Inconsistent interface implementation
   - Could cause issues with external contracts relying on IAccessControl interface

2. Contract Behavior:
   - While functionality works, the contract doesn't properly declare its interface compliance
   - Could cause integration issues with systems that check for proper interface implementation

## Attack Scenario
1. External contract tries to interact with Timelock through IAccessControl interface:
```solidity
interface IRoleManager is IAccessControl {
    function revokeRole(bytes32 role, address account) external override;
}

contract ExternalManager {
    IRoleManager public timelock;
    
    function manage(bytes32 role, address account) external {
        // This might fail or behave unexpectedly due to incorrect interface
        timelock.revokeRole(role, account);
    }
}
```

## Fix
Update the `revokeRole` function to properly override both AccessControl and IAccessControl:

```solidity
function revokeRole(bytes32 role, address account)
    public
    override(AccessControl, IAccessControl)
{
    require(
        role != DEFAULT_ADMIN_ROLE,
        "Timelock: cannot revoke admin role"
    );
    super.revokeRole(role, account);
}
```

```solidity
// SPDX-License-Identifier: MIT
pragma solidity 0.8.25;

import {Test} from "forge-std/Test.sol";
import {Timelock} from "../src/Timelock.sol";
import {Safe} from "@safe/Safe.sol";
import {IAccessControl} from "@openzeppelin-contracts/contracts/access/IAccessControl.sol";

contract TimelockRoleTest is Test {
    Timelock timelock;
    Safe safe;
    address[] hotSigners;
    
    function setUp() public {
        safe = new Safe();
        hotSigners = new address[](1);
        hotSigners[0] = address(0x1);
        
        timelock = new Timelock(
            address(safe),
            1 days,
            7 days,
            address(0x2),
            1 days,
            hotSigners
        );
    }
    
    function testInterfaceConsistency() public {
        // Test interface compatibility
        IAccessControl accessControl = IAccessControl(address(timelock));
        
        // Should work through interface
        vm.expectRevert("Timelock: cannot revoke admin role");
        accessControl.revokeRole(timelock.DEFAULT_ADMIN_ROLE(), address(this));
        
        // Test hot signer role revocation
        bytes32 hotSignerRole = timelock.HOT_SIGNER_ROLE();
        address signer = hotSigners[0];
        
        accessControl.revokeRole(hotSignerRole, signer);
        assertFalse(timelock.hasRole(hotSignerRole, signer));
    }
    
    function testExternalInteraction() public {
        ExternalRoleManager external = new ExternalRoleManager();
        external.setTimelock(timelock);
        
        // Should work through external contract
        bytes32 hotSignerRole = timelock.HOT_SIGNER_ROLE();
        address signer = hotSigners[0];
        
        external.manageRole(hotSignerRole, signer);
        assertFalse(timelock.hasRole(hotSignerRole, signer));
    }
}

contract ExternalRoleManager {
    IAccessControl public timelock;
    
    function setTimelock(IAccessControl _timelock) external {
        timelock = _timelock;
    }
    
    function manageRole(bytes32 role, address account) external {
        timelock.revokeRole(role, account);
    }
}
```

## Recommendations
1. Update `revokeRole` to properly override both AccessControl and IAccessControl
2. Maintain consistent interface implementations across all role management functions
