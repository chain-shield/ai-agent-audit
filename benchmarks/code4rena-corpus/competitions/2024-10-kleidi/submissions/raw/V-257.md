# Vulnerability details

## Finding Description
Lines of impacted code:
- https://github.com/code-423n4/2024-10-kleidi/blob/main/src/Timelock.sol#L642 

```executeBatch``` lets someone to send native ether to multiple addresses at once. But if one of the recipients is a contract that can't receive ether or intentionally reverts upon receiving ether the whole tx fails and no recipient gets paid.

## Proof of Concept
Steps to reproduce:
1) create a test file inside the test folder of the project directory.
2) copy and paste the code below.
3) run it with `forge test...` command.

Test Code:
```{solidity}
pragma solidity 0.8.25;

import {Timelock} from "src/Timelock.sol";
import "test/utils/TimelockUnitFixture.sol";

contract Receiver {
    // contract without receive() or fallback()
    // contract that can't accept ether 
}
contract ReceiverThatReverts {
    fallback() external payable {
        revert('Intentionally reverted!');
    }
}

contract TimelockUnitTest is TimelockUnitFixture {
    function testGriefBatchExecution() public {
        bytes32 salt = bytes32(0);
        uint256 newDelay = MINIMUM_DELAY + 3 days;
        address[] memory targets = new address[](3);
        targets[0] = address(0x11111);
        targets[1] = address(new Receiver());   // listing address that can't receive ether
        targets[2] = address(0x22222);

        uint256[] memory values = new uint256[](3);
        values[0] = 1 ether;
        values[1] = 0.1 ether;
        values[2] = 1 ether;

        bytes[] memory datas = new bytes[](3);
        datas[0] = bytes('');
        datas[1] = bytes('');
        datas[2] = bytes('');

        _scheduleBatch({
            caller: address(safe),
            timelock: address(timelock),
            targets: targets,
            values: values,
            payloads: datas,
            salt: salt,
            delay: newDelay
        });

        vm.warp(block.timestamp + newDelay);

        vm.expectRevert();
        Timelock(payable(timelock)).executeBatch{value: 3 ether}(
            targets, values, datas, salt
        );
    }
    function testGriefBatchExecution2() public {
        bytes32 salt = bytes32(0);
        uint256 newDelay = MINIMUM_DELAY + 3 days;
        address[] memory targets = new address[](3);
        targets[0] = address(0x11111);
        targets[1] = address(new ReceiverThatReverts());    // listing address that reverts upon receiving ether
        targets[2] = address(0x22222);

        uint256[] memory values = new uint256[](3);
        values[0] = 1 ether;
        values[1] = 0.1 ether;
        values[2] = 1 ether;

        bytes[] memory datas = new bytes[](3);
        datas[0] = bytes('');
        datas[1] = bytes('');
        datas[2] = bytes('');

        _scheduleBatch({
            caller: address(safe),
            timelock: address(timelock),
            targets: targets,
            values: values,
            payloads: datas,
            salt: salt,
            delay: newDelay
        });

        vm.warp(block.timestamp + newDelay);

        vm.expectRevert();
        Timelock(payable(timelock)).executeBatch{value: 3 ether}(
            targets, values, datas, salt
        );
    }
}
```


## Impact 
* One single recipient can grief all other recipients. 
* Loss of gas since the whole tx will fail.
