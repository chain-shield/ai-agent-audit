# Vulnerability details

## Description
The Timelock contract implements whitelisted function execution and delayed transaction processing for a Safe. It allows execution of whitelisted functions through `executeWhitelisted` and time-delayed arbitrary calls through `execute`. The whitelisting system uses byte range checks to validate calldata.

The current whitelisting system in the Timelock contract is vulnerable to CREATE2 collisions through delegatecall patterns. While the contract implements calldata validation, it cannot detect when a whitelisted call leads to a delegatecall + CREATE2 sequence that can result in contract address collisions. This allows an attacker to replace contracts at predetermined addresses, potentially compromising system security.

## Vulnerable Code Snippets

1. Whitelisted execution without CREATE2/delegatecall detection:
```solidity
function executeWhitelisted(
    address target,
    uint256 value,
    bytes calldata payload
) external payable onlyRole(HOT_SIGNER_ROLE) whenNotPaused {
    // Only checks calldata bytes, doesn't detect delegatecall + CREATE2 patterns
    checkCalldata(target, payload);
    _execute(target, value, payload);
}
```

2. Limited calldata validation:
```solidity
function checkCalldata(address contractAddress, bytes memory data) public view {
    bytes4 selector = data.getFunctionSignature();
    Index[] storage calldataChecks = _calldataList[contractAddress][selector];
    
    for (uint256 i = 0; i < calldataChecks.length; i++) {
        Index storage calldataCheck = calldataChecks[i];
        // Only checks byte ranges, can't detect dangerous patterns
        require(
            calldataCheck.dataHashes.contains(
                data.getSlicedBytesHash(
                    calldataCheck.startIndex, 
                    calldataCheck.endIndex
                )
            ),
            "CalldataList: Calldata does not match expected value"
        );
    }
}
```

## Impact
- Malicious actors can replace contracts at predetermined addresses
- Potential theft of funds if replaced contracts handle assets
- Breaking of system invariants through contract replacement
- Loss of trust in deployed contract addresses

## Attack Scenario

1. Attacker gets a seemingly harmless contract whitelisted that internally uses delegatecall
2. Through the whitelisted contract, attacker deploys Contract A using CREATE2 with a specific salt
3. Later, attacker deploys Contract B with different code but same salt to the same address
4. Contract B now exists at Contract A's address with different behavior

Here's an example:

```solidity
// Step 1: Innocent-looking whitelisted contract
contract WhitelistedProxy {
    function forwardCall(address target, bytes calldata data) external {
        // Hidden delegatecall capability
        (bool success,) = target.delegatecall(data);
        require(success);
    }
}

// Step 2: Factory used via delegatecall
contract CREATE2Factory {
    function deploy(bytes32 salt) external returns (address) {
        bytes memory code = type(ContractA).creationCode;
        address addr;
        assembly {
            addr := create2(0, add(code, 0x20), mload(code), salt)
        }
        return addr;
    }
}

// Step 3: Original contract
contract ContractA {
    function isCompromised() external pure returns (bool) {
        return false;
    }
}

// Step 4: Malicious replacement
contract ContractB {
    function isCompromised() external pure returns (bool) {
        return true;
    }
}
```

## Fix

1. Implement contract deployment tracking:
```solidity
contract Timelock {
    // Track deployed addresses
    mapping(address => bool) private _deployedContracts;
    
    function _beforeExecute(address target, bytes calldata data) private {
        // If target is a contract being created, ensure it hasn't been used before
        if (target.code.length == 0) {
            require(!_deployedContracts[target], "Address previously deployed");
            _deployedContracts[target] = true;
        }
    }

    function executeWhitelisted(
        address target,
        uint256 value,
        bytes calldata payload
    ) external payable onlyRole(HOT_SIGNER_ROLE) whenNotPaused {
        checkCalldata(target, payload);
        _beforeExecute(target, payload);
        _execute(target, value, payload);
    }
}
```

2. Add bytecode analysis in whitelisting:
```solidity
function _addCalldataCheck(
    address contractAddress,
    bytes4 selector,
    uint16 startIndex,
    uint16 endIndex,
    bytes[] memory data
) private {
    // Check for dangerous patterns in contract code
    bytes memory code = contractAddress.code;
    require(!_containsDelegateCall(code), "Contract contains delegatecall");
    require(!_containsCreate2(code), "Contract contains CREATE2");
    
    // Rest of implementation...
}

function _containsDelegateCall(bytes memory code) private pure returns (bool) {
    // Implementation to detect DELEGATECALL opcode
    for (uint i = 0; i < code.length - 1; i++) {
        if (code[i] == 0xF4) return true; // DELEGATECALL opcode
    }
    return false;
}

function _containsCreate2(bytes memory code) private pure returns (bool) {
    // CREATE2 opcode is 0xf5
    for (uint i = 0; i < code.length - 1; i++) {
        // Skip PUSH operations
        if (uint8(code[i]) >= 0x60 && uint8(code[i]) <= 0x7f) {
            uint256 pushSize = uint8(code[i]) - 0x5f;
            i += pushSize;
            continue;
        }
        // Check for CREATE2 opcode
        if (code[i] == 0xf5) return true;
    }
    return false;
}
```
