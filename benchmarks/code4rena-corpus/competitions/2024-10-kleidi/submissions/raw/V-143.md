# Vulnerability details

## root of the bug 

The  cause of the issue is occur In the pause function, When the function is called, the pause guardian is set to address(0), and there is no fallback is present to automatically assign a new guardian, and this is resulting that the system is being unable to pause again after the current pause expires in this line on the contract : 

pauseGuardian = address(0);

Here this line is  removes the pause guardian after a pause is initiated, and there is no mechanism is in place to reassign a new guardian. Without external intervention as result the system will become unmanageable and remains in an unpaused state indefinitely once the pause duration ends.

## impact 
The issue is can make the contract unpaused no one can call the pause function until a new guardian is assigned this  If the protocol requires an urgent pause
And if this is  Relying on external parties that manually reassign a guardian this is gone introduces an  operational inefficiencies and delays, particularly in emergencies
## Proof of Concept
Here is poc that confirm the finding : 
```solidity

pragma solidity 0.8.25;

import {Test, console} from "forge-std/Test.sol";
import {ConfigurablePause} from "../../src/ConfigurablePause.sol";


// Expose the internal _grantGuardian() function for testing
contract ConfigurablePauseTest is ConfigurablePause {
    function grantGuardian(address newGuardian) public {
        _grantGuardian(newGuardian);
    }
}

contract TimelockPoc is Test {
    ConfigurablePauseTest pauseContract;
    address owner = address(1);
    address guardian = address(2);

    function setUp() public {
        // Deploy the contract
        pauseContract = new ConfigurablePauseTest();

        // Simulate the owner assigning a guardian
        vm.prank(owner); // Simulate calling from the owner address
        pauseContract.grantGuardian(guardian);
    }

    function testPauseAndGuardianRemoval() public {
        // Check that guardian is correctly assigned
        vm.prank(guardian);  // Simulate calling from the guardian address
        pauseContract.pause();

        // Ensure the guardian is now address(0) after pausing
        assertEq(pauseContract.pauseGuardian(), address(0), "Guardian should be reset to address(0)");

        // Check that the contract is now paused
        assertEq(pauseContract.paused(), true, "Contract should be paused");

        // Simulate fast-forwarding time to the end of the pause duration
        vm.warp(block.timestamp + 2 days); // Fast-forward 2 days

        // Now that the pause duration has ended, the contract should be unpaused automatically
        assertEq(pauseContract.paused(), false, "Contract should be unpaused after duration ends");

        // Try pausing again but there's no guardian
        vm.expectRevert("ConfigurablePauseGuardian: only pause guardian");
        pauseContract.pause(); // This will fail as the guardian is now address(0)

        // Prove that no one can pause the contract anymore since guardian is address(0)
        assertEq(pauseContract.paused(), false, "Contract remains unpaused as no guardian exists");

        console.log("Pause contract is stuck in unpaused state because no guardian is assigned");
    }

    function testManualGuardianReassignment() public {
        // Manually reassign a new guardian
        address newGuardian = address(3);

        // Only owner should be able to reassign guardian
        vm.prank(owner);
        pauseContract.grantGuardian(newGuardian);

        // Ensure the new guardian is assigned
        assertEq(pauseContract.pauseGuardian(), newGuardian, "New guardian should be assigned");

        // Now the new guardian can pause the contract
        vm.prank(newGuardian);
        pauseContract.pause();

        // Ensure the contract is paused again
        assertEq(pauseContract.paused(), true, "Contract should be paused after new guardian pauses");
    }
}

```
