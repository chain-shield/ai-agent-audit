# Vulnerability details

## Finding Description
Lines of impacted code:
- https://github.com/code-423n4/2024-10-kleidi/blob/main/src/Timelock.sol#L589 
- https://github.com/code-423n4/2024-10-kleidi/blob/main/src/Timelock.sol#L621 

Both ```execute``` and ```executeBatch``` are ```payable``` functions. Tx Senders might need to send fund to the timelock contract when they execute a transaction. Since there is no exact check on how much  fund a sender is allowed to send, Senders accidentally or on-purpose might send funds more than they would like to distribute among recipients. The undistributed funds are not resent back to the senders. 

## Proof of Concept
Steps to reproduce:
1) create a test file inside the test folder of the project directory.
2) copy and paste the code below.
3) run it with `forge test...` command.

Expected output logs:
```
[PASS] testUndistributedFundsLockedViaExecute()
Logs:
  BEFORE:execute():
  Funds to be distributed: 1.1 ether.
  Tx Sender balance: 2 ether.
  Timelock balance: 0.

  AFTER:execute():
  Tx Sender balance: 0.
  Undistributed funds locked in Timelock: 900000000000000000 wei.
```
```
[PASS] testUndistributedFundsLockedViaExecuteBatch()
```

Test Code:
```{solidity}
pragma solidity 0.8.25;

import {Timelock} from "src/Timelock.sol";
import "test/utils/TimelockUnitFixture.sol";

contract TimelockUnitTest is TimelockUnitFixture {
    function testUndistributedFundsLockedViaExecute() public {
        bytes32 salt = bytes32(0);
        bytes memory data = bytes('');
        address target = address(0x11111);
        uint value = 1.1 ether;
        uint256 newDelay = MINIMUM_DELAY + 3 days;

        _schedule({
            caller: address(safe),
            timelock: address(timelock),
            target: target,
            value: value,
            data: data,
            salt: salt,
            delay: newDelay
        });

        vm.deal(address(safe), 2 ether);
        console.log('BEFORE:execute():');
        console.log('Funds to be distributed: 1.1 ether.');
        console.log('Tx Sender balance: %s ether.', address(safe).balance / 1 ether);
        console.log('Timelock balance: %s.\n', address(timelock).balance);

        vm.warp(block.timestamp + newDelay);
        vm.prank(address(safe));
        Timelock(payable(timelock)).execute{value: 2 ether}(   // sending 0.9 ether more
            target, value, data, salt
        );

        assertEq(address(safe).balance, 0); // 0.9 ether was not refunded, tx sender remaining balance is 0.
        assertEq(address(timelock).balance, 0.9 ether); // undistributed funds locked in timelock is 0.9 ether.

        console.log('AFTER:execute():');
        console.log('Tx Sender balance: %s.', address(safe).balance);
        console.log('Undistributed funds locked in Timelock: %s wei.', address(timelock).balance);
    }
    function testUndistributedFundsLockedViaExecuteBatch() public {
        bytes32 salt = bytes32(0);
        uint256 newDelay = MINIMUM_DELAY + 3 days;
        address[] memory targets = new address[](3);
        targets[0] = address(0x11111);
        targets[1] = address(0x22222);
        targets[2] = address(0x33333);

        uint256[] memory values = new uint256[](3);
        values[0] = 1 ether;
        values[1] = 0.5 ether;
        values[2] = 1 ether;

        bytes[] memory datas = new bytes[](3);
        datas[0] = bytes('');
        datas[1] = bytes('');
        datas[2] = bytes('');

        _scheduleBatch({
            caller: address(safe),
            timelock: address(timelock),
            targets: targets,
            values: values,
            payloads: datas,
            salt: salt,
            delay: newDelay
        });

        vm.deal(address(safe), 3 ether);
        console.log('BEFORE:executeBatch():');
        console.log('Funds to be distributed: 2.5 ether.');
        console.log('Tx Sender balance: %s ether.', address(safe).balance / 1 ether);
        console.log('Timelock balance: %s.\n', address(timelock).balance);

        vm.warp(block.timestamp + newDelay);
        vm.prank(address(safe));
        Timelock(payable(timelock)).executeBatch{value: 3 ether}(   // sending 0.5 ether more
            targets, values, datas, salt
        );

        assertEq(address(safe).balance, 0); // 0.5 ether was not refunded, tx sender remaining balance is 0.
        assertEq(address(timelock).balance, 0.5 ether); // undistributed funds locked in timelock is 0.5 ether.

        console.log('AFTER:executeBatch():');
        console.log('Tx Sender balance: %s.', address(safe).balance);
        console.log('Undistributed funds locked in Timelock: %s wei.', address(timelock).balance);
    }
}
```

## Impact 
* Tx senders might loose some fund in the process.
* Undistributed funds might remain locked in the contract temporarily/permanently.

## Tools Used
Foundry

## Recommendation
Either add a check on how much fund a tx sender is supposed to send or refund the undistributed fund back to the sender.
