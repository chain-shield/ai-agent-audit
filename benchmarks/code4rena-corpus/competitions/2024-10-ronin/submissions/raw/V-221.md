# Vulnerability details

## Proof of Concept

The vulnerability can be demonstrated through the following test case that shows how a zero denominator can be set and its implications:

```solidity
// SPDX-License-Identifier: UNLICENSED
pragma solidity ^0.8.0;

import {Test} from "forge-std/Test.sol";
import {KatanaV3Factory} from "../src/KatanaV3Factory.sol";

contract KatanaV3FactoryTest is Test {
    KatanaV3Factory factory;
    address owner = address(1);

    function setUp() public {
        vm.prank(owner);
        factory = new KatanaV3Factory(owner);
    }

    function testEnableFeeAmountWithZeroDenominator() public {
        vm.startPrank(owner);
        
        // Parameters for enableFeeAmount
        uint24 fee = 500;        // 0.05% fee
        int24 tickSpacing = 10;
        uint8 feeProtocolNum = 1;
        uint8 feeProtocolDen = 0; // Zero denominator
        
        // This call will succeed despite invalid denominator
        factory.enableFeeAmount(fee, tickSpacing, feeProtocolNum, feeProtocolDen);
        
        // Future operations using these fee settings would fail
        // For demonstration, we could add assertions here to show the impact
        // on subsequent operations that depend on fee calculations
        
        vm.stopPrank();
    }
}
```

To validate this vulnerability:

1. The test demonstrates that we can set a zero denominator through the `enableFeeAmount` function
2. While the initial setting succeeds, subsequent operations would fail due to division by zero
3. Key sequence of events:
   ```
   Owner -> enableFeeAmount(fee=500, tickSpacing=10, feeProtocolNum=1, feeProtocolDen=0)
   -> Passes require(feeProtocolNum < feeProtocolDen)
   -> Sets invalid fee protocol parameters
   -> Future fee calculations will revert
   ```

## Recommended Mitigation Steps

1. Add Input Validation:
```solidity
function enableFeeAmount(
    uint24 fee,
    int24 tickSpacing,
    uint8 feeProtocolNum,
    uint8 feeProtocolDen
) public override onlyOwner {
    // Add validation for denominator
    require(feeProtocolDen > 0, "KatanaV3Factory: INVALID_FEE_PROTOCOL_DENOMINATOR");
    
    // Existing checks
    require(fee < 1000000, "KatanaV3Factory: FEE_TOO_HIGH");
    require(tickSpacing > 0 && tickSpacing < 16384, "KatanaV3Factory: INVALID_TICK_SPACING");
    require(feeProtocolNum < feeProtocolDen, "KatanaV3Factory: INVALID_FEE_PROTOCOL");
    require(feeAmountTickSpacing[fee] == 0, "KatanaV3Factory: FEE_AMOUNT_ALREADY_ENABLED");

    _enableFeeAmount(fee, tickSpacing, feeProtocolNum, feeProtocolDen);
}
```

2. Additional Safety Measures:

3. Implementation Checklist:
   - Add non-zero denominator validation
   - Implement reasonable bounds for denominator
   - Add validation for numerator
   - Update tests to cover new validation rules
   - Document valid parameter ranges
   - Add events for parameter changes

4. Testing Requirements:
