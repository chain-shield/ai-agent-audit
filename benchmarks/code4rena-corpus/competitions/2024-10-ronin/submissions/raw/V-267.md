# Vulnerability details

## Summary
The `quoteExactInput` function only tracks gas costs for V3 swaps while ignoring V2 swap gas costs, leading to inaccurate gas estimations for mixed routes. This can cause users to receive incomplete or misleading gas estimates for their transactions.

The current implementation only accumulates gas estimates for V3 pools:
```solidity
function quoteExactInput(bytes memory path, uint256 amountIn) public returns (
    uint256 amountOut,
    uint160[] memory v3SqrtPriceX96AfterList,
    uint32[] memory v3InitializedTicksCrossedList,
    uint256 v3SwapGasEstimate  // Only V3 gas tracked
)
```

## Impact
1. **Inaccurate Gas Estimates**:
   - Users receive incomplete gas estimates for mixed routes
   - V2 swap costs are completely ignored
   - Can lead to transaction failures due to insufficient gas

2. **Economic Impact**:
   - Users cannot accurately compare route costs
   - May choose suboptimal routes based on incomplete information
   - Higher than expected transaction costs

## Scenario Example
```solidity
// Route: USDC (V2)-> WETH (V3)-> DAI (V2)-> USDT

// Current gas accounting:
// V2 USDC->WETH: Not counted
// V3 WETH->DAI:  Counted (~100k gas)
// V2 DAI->USDT:  Not counted
// Reported gas: ~100k
// Actual gas needed: ~220k
```

## Fixed Implementation
```solidity
struct GasAccounting {
    uint256 v2SwapGasEstimate;
    uint256 v3SwapGasEstimate;
    uint256 pathGasEstimate;
}

function quoteExactInput(
    bytes memory path,
    uint256 amountIn
) public returns (
    uint256 amountOut,
    uint160[] memory v3SqrtPriceX96AfterList,
    uint32[] memory v3InitializedTicksCrossedList,
    GasAccounting memory gasAccounting
) {
    v3SqrtPriceX96AfterList = new uint160[](path.numPools());
    v3InitializedTicksCrossedList = new uint32[](path.numPools());
    
    // Initialize gas accounting
    gasAccounting = GasAccounting({
        v2SwapGasEstimate: 0,
        v3SwapGasEstimate: 0,
        pathGasEstimate: PATH_BASE_GAS // Base cost for path processing
    });

    uint256 i = 0;
    while (true) {
        uint256 gasStart = gasleft();
        
        (address tokenIn, address tokenOut, uint24 fee) = path.decodeFirstPool();

        if (fee & flagBitmask != 0) {
            // V2 swap with gas tracking
            uint256 v2GasStart = gasleft();
            
            // Estimate approval costs if needed
            if (!isTokenApproved(tokenIn)) {
                gasAccounting.v2SwapGasEstimate += APPROVE_GAS_ESTIMATE;
            }
            
            amountIn = quoteExactInputSingleV2(
                QuoteExactInputSingleV2Params({
                    tokenIn: tokenIn,
                    tokenOut: tokenOut,
                    amountIn: amountIn
                })
            );
            
            // Account for V2 swap gas
            uint256 v2SwapGas = v2GasStart - gasleft();
            gasAccounting.v2SwapGasEstimate += v2SwapGas;
            
            // Add V2 specific costs
            gasAccounting.v2SwapGasEstimate += calculateV2SpecificCosts(
                tokenIn,
                tokenOut
            );
        } else {
            // V3 swap with existing gas tracking
            (uint256 _amountOut, uint160 _sqrtPriceX96After, 
             uint32 _initializedTicksCrossed, uint256 _gasEstimate) =
                quoteExactInputSingleV3(
                    QuoteExactInputSingleV3Params({
                        tokenIn: tokenIn,
                        tokenOut: tokenOut,
                        fee: fee,
                        amountIn: amountIn,
                        sqrtPriceLimitX96: 0
                    })
                );

            v3SqrtPriceX96AfterList[i] = _sqrtPriceX96After;
            v3InitializedTicksCrossedList[i] = _initializedTicksCrossed;
            gasAccounting.v3SwapGasEstimate += _gasEstimate;
            
            // Add V3 specific costs
            gasAccounting.v3SwapGasEstimate += calculateV3SpecificCosts(
                tokenIn,
                tokenOut,
                fee
            );
            
            amountIn = _amountOut;
        }

        // Account for path processing gas
        gasAccounting.pathGasEstimate += (gasStart - gasleft());
        
        i++;
        
        if (!path.hasMultiplePools()) {
            // Add final overhead costs
            gasAccounting.pathGasEstimate += FINAL_OVERHEAD_GAS;
            return (
                amountIn,
                v3SqrtPriceX96AfterList,
                v3InitializedTicksCrossedList,
                gasAccounting
            );
        }
        path = path.skipToken();
    }
}

function calculateV2SpecificCosts(
    address tokenIn,
    address tokenOut
) internal view returns (uint256) {
    uint256 gasEstimate = V2_BASE_SWAP_GAS;
    
    // Add sync costs
    gasEstimate += V2_SYNC_GAS;
    
    // Add transfer costs based on token types
    if (isNativeToken(tokenIn) || isNativeToken(tokenOut)) {
        gasEstimate += WRAP_UNWRAP_GAS;
    }
    
    return gasEstimate;
}

function calculateV3SpecificCosts(
    address tokenIn,
    address tokenOut,
    uint24 fee
) internal view returns (uint256) {
    uint256 gasEstimate = V3_BASE_SWAP_GAS;
    
    // Add costs for crossing initialized ticks
    IKatanaV3Pool pool = getPool(tokenIn, tokenOut, fee);
    (, int24 tick,,,,,,) = pool.slot0();
    gasEstimate += estimateTickCrossingGas(pool, tick);
    
    return gasEstimate;
}
```

The fix provides comprehensive gas estimation for all pool types while maintaining modularity and extendibility.
