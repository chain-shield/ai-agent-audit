# Vulnerability details

## Impact

The distribution of rewards can be permanently blocked when too many curators hold shares of a cred. If the gas cost of executing the `CuratorRewardsDistributor.sol::distribute()` function exceeds the block gas limit, the function reverts, rendering the distribution process non-functional.

## Proof of Concept

The following table outlines the gas costs associated with distributing rewards as the number of curators increase:

| Curators |     Gas    |
|:--------:|:----------:|
|    50    |  1 647 252 |
|    100   |  3 278 569 |
|    250   |  8 180 197 |
|    500   | 16 375 161 |
|    750   | 24 602 108 |
|    850   | 27 901 842 |
|    886   | 29 090 999 |
|    887   | Out of Gas |

This Proof of Concept illustrates how the rewards distribution mechanism can be permanently bricked if the number of curators naturally becomes too large.


```solidity
// SPDX-License-Identifier: MIT
pragma solidity 0.8.25;

import { Settings, console2 as console } from "../helpers/Settings.sol";

import { ECDSA } from "solady/utils/ECDSA.sol";

/**
 * @notice forge test --mt test_distribute_denial_of_service -vvv --gas-limit 30000000
 */
contract DistributorDos is Settings {
    address internal immutable distributor = makeAddr("distributor");
    address internal immutable creator = makeAddr("creator");
    address internal immutable account = makeAddr("account");

    function setUp() public override {
        super.setUp();
    }

    function test_distribute_denial_of_service() public {
        // Stop metering gas.
        vm.pauseGasMetering();

        // Create a new cred.
        uint256 credId = _createCred();

        // Amount to be distributed to the curators of the cred.
        uint256 amountToBeDistributed = 100 ether;

        // Deal ETH to the account depositing rewards to be distributed.
        deal(account, amountToBeDistributed);

        // Account deposits ETH in the distributor.
        vm.prank(account);
        curatorRewardsDistributor.deposit{ value: amountToBeDistributed }(credId, amountToBeDistributed);

        // Buy cred shares with as little value as possible with different accounts.
        uint256 supply = 1;
        for (uint256 i = 1; i < 887; ++i) {
            // Generator a random curator address
            address curator = makeAddr(string(abi.encodePacked("curator", i)));
            
            // Get the price needed to buy one share.
            uint256 buyPrice = bondingCurve.getBuyPriceAfterFee(credId, supply, 1);

            // Deal ETH to the curator to buy cred shares.
            deal(curator, buyPrice);

            // Curator buys one share of the cred.
            vm.prank(curator);
            cred.buyShareCred{ value: buyPrice }(credId, 1, buyPrice);

            // Increment the shares supply already bought
            supply++;
        }

        // Prank the distributor
        vm.startPrank(distributor);

        // Resume gas metering.
        vm.resumeGasMetering();

        // Try to call distribute and expect that it runs out of gas.
        vm.expectRevert();
        curatorRewardsDistributor.distribute(credId);
    }

    function _createCred() internal returns (uint256){
        // Encode the create cred data.
        bytes memory data = abi.encode(block.timestamp + 100, creator, 31_337, address(bondingCurve), "test", "BASIC", "SIGNATURE", 0x0);

        // Compute the hash of the structured data.
        bytes32 structHash = keccak256(data);

        // Compute the digest of the structured data.
        bytes32 digest = ECDSA.toEthSignedMessageHash(structHash);

        // Phi signer signs the create cred digest.
        (uint8 cv, bytes32 cr, bytes32 cs) = vm.sign(claimSignerPrivateKey, digest);
        if (cv != 27) cs = cs | bytes32(uint256(1) << 255);

        // Creator submites the create cred transaction signed by the phi signer.
        vm.startPrank(creator);

        // Get the price needed to buy the first share.
        uint256 credId = 1;
        uint256 supply = 0;
        uint256 amountToBuy = 1;
        uint256 buyPrice = bondingCurve.getBuyPriceAfterFee(credId, supply, amountToBuy);

        // Deal ETH to the creator to buy the first share.
        deal(creator, buyPrice);

        // Create new cred.
        cred.createCred{ value: buyPrice }(creator, data, abi.encodePacked(cr, cs), 100, 100);

        vm.stopPrank();

        // Return the cred id of the newly created cred.
        return credId;
    }
}
```
