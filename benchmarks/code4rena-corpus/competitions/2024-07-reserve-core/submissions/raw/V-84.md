# Vulnerability details

## Impact
The safeApproveFallbackToMax function in the AllowanceLib library does not check the return value of the approve function from the ERC20 token contract. This can lead to situations where the approval operation fails silently, causing the allowance to not be set as intended. This can result in unexpected behavior in downstream contracts that rely on the correct setting of allowances, potentially leading to denial of service or incorrect token transfers.

## Proof of Concept
```
// ExampleToken contract that returns false on approve() for values > 1000
contract ExampleToken is IERC20ApproveOnly {
    mapping(address => mapping(address => uint256)) private _allowances;

    function approve(address spender, uint256 value) external override returns (bool) {
        if (value > 1000) {
            return false; // Fail without revert
        }
        _allowances[msg.sender][spender] = value;
        return true; // Success
    }

    function allowance(address owner, address spender) external view override returns (uint256) {
        return _allowances[owner][spender];
    }
}

// Test contract using AllowanceLib
contract TestAllowance {
    using AllowanceLib for address;

    function testApprove(address tokenAddress, address spender, uint256 value) external {
        tokenAddress.safeApproveFallbackToMax(spender, value);
    }
}

// Deploy ExampleToken and TestAllowance contracts
ExampleToken token = new ExampleToken();
TestAllowance test = new TestAllowance();

// Scenario: Attempt to approve a value greater than 1000
token.approve(address(test), 1001); // This will return false

// Call the testApprove function
test.testApprove(address(token), address(this), 1001);

// Expected behavior: The approval should fail and fallback to max allowance
// Actual behavior: The approval fails silently and may not set the allowance correctly
```

## Recommended Mitigation Steps
Modify the safeApproveFallbackToMax function to include return value checks.
```
library AllowanceLib {
    function safeApproveFallbackToMax(
        address tokenAddress,
        address spender,
        uint256 value
    ) internal {
        IERC20ApproveOnly token = IERC20ApproveOnly(tokenAddress);

        // 1. Set initial allowance to 0
        require(token.approve(spender, 0), "approve to 0 failed");
        require(token.allowance(address(this), spender) == 0, "allowance not 0");

        if (value == 0) return;

        // 2. Try to set the provided allowance
        bool success;
        try token.approve(spender, value) returns (bool result) {
            success = result && token.allowance(address(this), spender) == value;
        } catch {
            success = false;
        }

        // 3. Fall-back to setting a maximum allowance
        if (!success) {
            require(token.approve(spender, type(uint256).max), "approve to max failed");
            require(token.allowance(address(this), spender) >= value, "allowance missing");
        }
    }
}
```
