S-84
Retiring strategies don't send the assets to the vault, causing them to be stuck forever
0xAlix2
82
Sufficient
383d ago
Finding description and impact

Strategy admins call retireStrategy to redeem the third-party protocol shares in return for the underlying assets. On the other hand, when withdrawing from a strategy, it automatically withdraws from the third-party protocols:

function _withdraw(
    address caller_,
    address receiver_,
    address owner_,
    uint256 assets_,
    uint256 shares_
) internal virtual override(ERC4626Upgradeable) whenNotPaused onlyVault {
    //slither-disable-next-line incorrect-equality
    if (shares_ == 0 || assets_ == 0) revert ZeroAmount();

@>  _protocolWithdraw(assets_, shares_);

    _burn(owner_, shares_);
    _handleRewardsOnWithdraw();
    IERC20(asset()).safeTransfer(receiver_, assets_);

    emit Withdraw(caller_, receiver_, owner_, assets_, shares_);
}

So withdrawals can't be done on strategies unless there are some funds in the third-party protocols. When a strategy is retired, all third-party protocol assets are redeemed and any withdrawal call after that will revert, as it doesn't hold any more shares for that protocol.

As a result, retiring a strategy leaves the underlying assets stuck in the strategy forever, as there's no way to withdraw these underlying funds out of the strategy contract. Ultimately, leading to loss of funds.

Proof of Concept

For simplicity, the following test assumes that the vault has no withdrawal Q. Note that the issue exists even if a withdrawal Q is set, but this decision is just to narrow down the problem.

contract PoC is Test {
    using Math for uint256;

    Strategy[] strategies;
    GraduatedFee[] graduatedFees;
    ConcreteMultiStrategyVault vault;
    WithdrawalQueue queue;

    IERC20 USDC = IERC20(0xaf88d065e77c8cC2239327C5EDb3A432268e5831);
    IERC20 WETH = IERC20(0x82aF49447D8a07e3bd95BD0d56f35241523fBab1);
    address feeRecipient = address(0x1111);
    address admin = address(0x4444);
    address hazel = address(0x2222);

    IERC20 asset;

    // Aave //

    AaveV3Strategy aaveStrategy;
    address aaveDataProvider = 0x69FA688f1Dc47d4B5d8029D5a35FB7a548310654;
    ILendingPool lendingPool;
    IAaveIncentives aaveIncentives;
    IAToken aToken;

    function setUp() public {
        vm.selectFork(
            vm.createFork(
                "https://arb-mainnet.g.alchemy.com/v2/THuQ6WJZ6orCusOEvVFb9baHWzk_KG_D",
                223_500_507
            )
        );

        asset = USDC;

        _setUpGraduatedFees();

        vault = ConcreteMultiStrategyVault(
            Clones.clone(address(new ConcreteMultiStrategyVault()))
        );
        vault.initialize(
            IERC20(address(asset)),
            "USDC Vault",
            "USDCV",
            strategies,
            feeRecipient,
            VaultFees({
                depositFee: 0,
                withdrawalFee: 0,
                protocolFee: 3_00,
                performanceFee: graduatedFees
            }),
            type(uint256).max,
            admin
        );

        queue = new WithdrawalQueue(address(vault));

        vm.startPrank(admin);

        // vault.setWithdrawalQueue(address(queue));

        aaveStrategy = _setUpAaveStrategy();
        vault.addStrategy(
            0,
            false,
            Strategy({
                strategy: aaveStrategy,
                allocation: Allocation({index: 0, amount: 100_00})
            })
        );
        vm.stopPrank();
    }

    function _setUpGraduatedFees() internal {
        graduatedFees.push(
            GraduatedFee({lowerBound: 0, upperBound: 500, fee: 300})
        );
        graduatedFees.push(
            GraduatedFee({lowerBound: 501, upperBound: 2000, fee: 1000})
        );
        graduatedFees.push(
            GraduatedFee({lowerBound: 2001, upperBound: 5000, fee: 2000})
        );
        graduatedFees.push(
            GraduatedFee({lowerBound: 5001, upperBound: 10000, fee: 5000})
        );
    }

    function _setUpAaveStrategy() internal returns (AaveV3Strategy strategy) {
        (address _aToken, , ) = IProtocolDataProvider(aaveDataProvider)
            .getReserveTokensAddresses(address(asset));
        aToken = IAToken(_aToken);
        lendingPool = ILendingPool(aToken.POOL());
        aaveIncentives = IAaveIncentives(aToken.getIncentivesController());

        strategy = new AaveV3Strategy(
            asset,
            feeRecipient,
            admin,
            10_00,
            aaveDataProvider,
            address(vault)
        );
    }

    function test_StuckAssetsOnRetireStrategy() public {
        uint256 amount = 100e6;

        _mintAsset(amount, hazel);

        // Hazel holds 100 USDC
        assertEq(asset.balanceOf(hazel), amount);

        // Hazel deposits 100 USDC into the vault
        vm.startPrank(hazel);
        USDC.approve(address(vault), amount);
        vault.deposit(amount);
        vm.stopPrank();

        // Hazel holds 0 USDC
        assertEq(asset.balanceOf(hazel), 0);
        // AaveStrategy holds 0 USDC
        assertEq(asset.balanceOf(address(aaveStrategy)), 0);
        // AaveStrategy holds > 0 aUSDC
        assertGt(aToken.scaledBalanceOf(address(aaveStrategy)), 0);

        // Admin retires the AaveStrategy
        vm.prank(admin);
        aaveStrategy.retireStrategy();

        // AaveStrategy holds 100 USDC
        assertEq(asset.balanceOf(address(aaveStrategy)), amount);
        // AaveStrategy holds 0 aUSDC
        assertEq(aToken.scaledBalanceOf(address(aaveStrategy)), 0);

        // Hazel can't redeem her shares
        vm.startPrank(hazel);
        vault.redeem(vault.balanceOf(hazel)); // InsufficientVaultFunds
        vm.stopPrank();
    }

    function _mintAsset(uint256 amount_, address to_) internal {
        deal(address(asset), to_, amount_);
    }
}
Recommended mitigation steps

After withdrawing the assets from the third-party protocol in retireStrategy, send the withdrawn assets back to the vault.

function retireStrategy() external onlyOwner {
    // ...

+   IERC20(asset()).safeTransfer(_vault, IERC20(asset()).balanceOf(address(this)));
}
