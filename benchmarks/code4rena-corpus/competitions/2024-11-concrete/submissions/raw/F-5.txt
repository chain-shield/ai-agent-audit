S-170
`previewRedeem` and `previewDeposit` don't match `redeem` and `deposit` respectively
0xAlix2
82
Sufficient
383d ago
Finding description and impact

According to ERC4626 standards, previewDeposit allows users to simulate the effects of their deposit at the current block, and previewRedeem allows users to simulate the effects of their redemption at the current block. However, that's not the case in ConcreteMultiStrategyVault, as both previewDeposit and previewRedeem don't cut fees if the sender is the fee recipient:

function previewDeposit(uint256 assets_) public view override returns (uint256) {
    uint256 netAssets = assets_ -
        (
@>          msg.sender != feeRecipient
@>              ? assets_.mulDiv(uint256(fees.depositFee), MAX_BASIS_POINTS, Math.Rounding.Floor)
@>              : 0
        );
    return _convertToShares(netAssets, Math.Rounding.Floor);
}

function previewRedeem(uint256 shares_) public view override returns (uint256) {
    uint256 netShares = shares_ -
        (
@>          msg.sender != feeRecipient
@>              ? shares_.mulDiv(uint256(fees.withdrawalFee), MAX_BASIS_POINTS, Math.Rounding.Floor)
@>              : 0
        );
    return _convertToAssets(netShares, Math.Rounding.Floor);
}

While both deposit and redeem do, this makes the vault incompliant with ERC4626 standards and causes unexpected reverts when the fee recipient is interacting with the vault.

Proof of Concept
contract PoC2 is Test {
    Strategy[] strategies;
    GraduatedFee[] graduatedFees;
    ConcreteMultiStrategyVault vault;
    WithdrawalQueue queue;

    IERC20 USDC = IERC20(0xaf88d065e77c8cC2239327C5EDb3A432268e5831);
    address feeRecipient = address(0x1111);
    address hazel = address(0x11113349);
    address admin = address(0x4444);

    IERC20 asset;

    function setUp() public {
        vm.selectFork(
            vm.createFork(
                "https://arb-mainnet.g.alchemy.com/v2/THuQ6WJZ6orCusOEvVFb9baHWzk_KG_D",
                223_500_507
            )
        );

        asset = USDC;

        vault = ConcreteMultiStrategyVault(
            Clones.clone(address(new ConcreteMultiStrategyVault()))
        );
        vault.initialize(
            IERC20(address(asset)),
            "USDC Vault",
            "USDCV",
            strategies,
            feeRecipient,
            VaultFees({
                depositFee: 3_00,
                withdrawalFee: 3_00,
                protocolFee: 3_00,
                performanceFee: graduatedFees
            }),
            type(uint256).max,
            admin
        );

        queue = new WithdrawalQueue(address(vault));

        vm.prank(admin);
        vault.setWithdrawalQueue(address(queue));
    }

    function test_mismatch() public {
        uint256 amount = 100e6;

        _mintAsset(amount, feeRecipient);
        _mintAsset(amount, hazel);

        vm.startPrank(feeRecipient);
        USDC.approve(address(vault), amount);
        // Deposit 100 USDC into the vault, deposit's return value doesn't match the preview
        assertNotEq(vault.previewDeposit(amount), vault.deposit(amount));

        uint256 shares = vault.balanceOf(feeRecipient);
        // Redeem all shares, redeem's return value doesn't match the preview
        assertNotEq(vault.previewRedeem(shares), vault.redeem(shares));
        vm.stopPrank();

        vm.startPrank(hazel);
        USDC.approve(address(vault), amount);
        // Deposit 100 USDC into the vault, deposit's return value matches the preview
        assertEq(vault.previewDeposit(amount), vault.deposit(amount));

        shares = vault.balanceOf(hazel);
        // Redeem all shares, redeem's return value matches the preview
        assertEq(vault.previewRedeem(shares), vault.redeem(shares));
        vm.stopPrank();
    }

    function _mintAsset(uint256 amount_, address to_) internal {
        deal(address(asset), to_, amount_);
    }
}
Recommended mitigation steps

It should be decided if fees should be taken when the user interacting with the vault is the fee recipient, and then a refactor previewRedeem and previewDeposit to match redeem and deposit respectively, or vice versa.
