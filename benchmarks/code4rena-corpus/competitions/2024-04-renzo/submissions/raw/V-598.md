# Vulnerability details

## Impact
The code is using `OpenZeppelin:SafeERC20::safeApprove` function which reverts if there is previous approval not used or 0.
## Proof of Concept
The Protocol uses OpenZeppeline `safeApprove` function for approvals of token widely:
```solidity
/@openzeppelin/contracts/token/ERC20/utils/SafeERC20.sol
45:     function safeApprove(IERC20 token, address spender, uint256 value) internal {
46:         // safeApprove should only be called when setting an initial allowance,
47:         // or when resetting it to zero. To increase and decrease it, use
48:         // 'safeIncreaseAllowance' and 'safeDecreaseAllowance'
49: @1>        require(
50:             (value == 0) || (token.allowance(address(this), spender) == 0),
51:             "SafeERC20: approve from non-zero to non-zero allowance"
52:         );
53:         _callOptionalReturn(token, abi.encodeWithSelector(token.approve.selector, spender, value));
54:     }

```
`@1>` As it can be seen that if the approval is 0 or there is any previous approval then this function will revert.
In the current implementation of the code it widely used with third parties like blastBridge, egienLayer, connext , chainlinkCCIP and gives approval using `safeApprove` function. 

so if any change occur on these third parties which does not utilize the complete approval then the `safeApprove` will start reverting.

    // SPDX-License-Identifier: MIT
    pragma solidity ^0.8.0;
    
    import {
        ERC20Upgradeable
    } from "@openzeppelin/contracts-upgradeable/token/ERC20/ERC20Upgradeable.sol";
    import "@openzeppelin/contracts/token/ERC20/utils/SafeERC20.sol";
    import "@openzeppelin/contracts-upgradeable/proxy/utils/Initializable.sol";


    contract TestToken is ERC20Upgradeable {

        /// @custom:oz-upgrades-unsafe-allow constructor
        constructor(){
            _disableInitializers();
        }
    function initialize(string memory tokenName , string memory tokenSymbol) external initializer{
        __ERC20_init(tokenName, tokenSymbol);
    }
        function mint(address user) external{
            _mint(user , 100e18);
        }


    }

    contract TestApprovalContract is Initializable
    {
        address token;
        using SafeERC20 for ERC20Upgradeable;
        /// @custom:oz-upgrades-unsafe-allow constructor
        constructor(){
                    _disableInitializers();

        }
        function initialize(address _token) external initializer {
            token = _token;
        }



        function grantApprovals(address spender , uint256 amount) external{
        SafeERC20.safeApprove(IERC20(token) ,spender , amount );
     }
    }

  Add this file inside test folder
  ```javascript
    import { ethers, upgrades } from 'hardhat';
    import {
        TestApproval, TestApprovalContract, TestToken
    } from '../../../typechain-types';

    const { expect } = require("chai");

    describe("Token Approvals contract", function () {
        it("Should fail on 2nd approval", async function () {
            const [owner, addr1, addr2] = await ethers.getSigners();

            const TestTokenFactory = await ethers.getContractFactory('TestToken');
            const deployedTestToken = await upgrades.deployProxy(TestTokenFactory, ["Test Token", "TT"]);

            const testToken = TestTokenFactory.attach(
                await deployedTestToken.getAddress(),
            ) as TestToken;
            const TestApprovalFactory = await ethers.getContractFactory("TestApprovalContract");
            const deployedTestApproval = await upgrades.deployProxy(TestApprovalFactory, [await deployedTestToken.getAddress()]);


            const testApprovalContract = TestApprovalFactory.attach(
                await deployedTestApproval.getAddress(),
            ) as TestApprovalContract;



            // mint tokens 
            testToken.mint(addr1.address);
            expect(testToken.balanceOf(addr1.address), 100e18);
            await testApprovalContract.connect(addr1).grantApprovals(addr2.address, 18);
            await expect(testApprovalContract.connect(addr1).grantApprovals(addr2.address, 18)).to.be.revertedWith("SafeERC20: approve from non-zero to non-zero allowance");

        });
    });
  ```
