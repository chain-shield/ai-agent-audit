# Vulnerability details

## Description
Current function implementation does not perform a full refund of gas given that it does not consider posterior opcode cost requiered for the function.

## Impact
Incomplate gas refund 

## POC
```solidity
    function _refundGas(uint256 initialGas) internal {
        uint256 gasUsed = (initialGas - gasleft()) * tx.gasprice;
        uint256 gasRefund = address(this).balance >= gasUsed ? gasUsed : address(this).balance;
        // @audit Opcodes after this line are not considered in the calculation of gas refund
        (bool success, ) = payable(msg.sender).call{ value: gasRefund }("");
        if (!success) revert TransferFailed();
        emit GasRefunded(msg.sender, gasRefund);
    }
```

## Recommended mitigation
Add a new state variable which saves this extra gas requiered and use it to calculate the gas to refund. 

```diff
+   uint256 public gasToRefundExtra;

+   function setGasToRefundExtra(uint256 _gasToRefundExtra) internal {
+     gasToRefundExtra = _gasToRefundExtra;  
+   }

    function _refundGas(uint256 initialGas) internal {
-       uint256 gasUsed = (initialGas - gasleft()) * tx.gasprice;
+       uint256 gasUsed = (initialGas - gasleft() + _gasToRefundExtra) * tx.gasprice;
        uint256 gasRefund = address(this).balance >= gasUsed ? gasUsed : address(this).balance;
        (bool success, ) = payable(msg.sender).call{ value: gasRefund }("");
```
