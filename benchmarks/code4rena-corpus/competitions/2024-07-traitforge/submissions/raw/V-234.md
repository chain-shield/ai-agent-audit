# Vulnerability details

## Impact
Missing excess payments handler in function `EntityForging::forgeWithListed` makes it impossible for mergers to recover funds accidentally sent over the fee amount set by forger.
## Proof of Concept
The following PoC is coded using Foundry:

import {Test, stdError, console} from "forge-std/Test.sol";
import "../src/EntropyGenerator/EntropyGenerator.sol";

contract EntropyGeneratorTest is Test {
    address public owner = makeAddr("owner");
    address public bob = makeAddr("bob");
    address public alice = makeAddr("alice");

    EntropyGenerator public entropyGenerator;
    DevFund public devFund;
    Airdrop public airdrop;
    TraitForgeNft public traitForgeNft;
    EntityForging public entityForging;
    ERC20Mock public mockERC20;

    function setUp() public {
        vm.startPrank(owner);
        devFund = new DevFund();
        airdrop = new Airdrop();
        traitForgeNft = new TraitForgeNft();
        entropyGenerator = new EntropyGenerator(address(traitForgeNft));
        entityForging = new EntityForging(address(traitForgeNft));
        traitForgeNft.setEntityForgingContract(address(entityForging));
        traitForgeNft.setEntropyGenerator(address(entropyGenerator));
        traitForgeNft.setAirdropContract(address(airdrop));

        mockERC20 = new ERC20Mock("Trait Token", "TT", 18);
        airdrop.setTraitToken(address(mockERC20));
        airdrop.transferOwnership(address(traitForgeNft));
        vm.stopPrank();
    }

    function test_excessPaymentForge() public {
        entropyGenerator.writeEntropyBatch1();
        vm.deal(bob, 10 ether);
        vm.deal(alice, 10 ether);
        bytes32[] memory proof = new bytes32[](1);
        proof[0] = bytes32("proof");
        bytes32 bobHash = keccak256(abi.encodePacked(bob));
        bytes32 rootHash = uint256(proof[0]) < uint256(bobHash)
            ? keccak256(abi.encode(proof[0], bobHash))
            : keccak256(abi.encode(bobHash, proof[0]));
        vm.startPrank(owner);
        traitForgeNft.setRootHash(rootHash);
        vm.stopPrank();
        vm.startPrank(bob);
        traitForgeNft.mintToken{value: 1 ether}(proof);
        uint256 tokenIdBob = 1;
        // If minted token is not forger role or is infertile, keep trying until it is found
        while (
            ((traitForgeNft.tokenEntropy(tokenIdBob) / 10) % 10 == 0) ||
            (traitForgeNft.tokenEntropy(tokenIdBob) % 3 != 0)
        ) {
            traitForgeNft.mintToken{value: 1 ether}(proof);
            tokenIdBob++;
        }
        vm.stopPrank();
        uint8 forgePotentialBob = uint8(
            (traitForgeNft.tokenEntropy(tokenIdBob) / 10) % 10
        );
        console.log(forgePotentialBob); // 7

        bytes32 aliceHash = keccak256(abi.encodePacked(alice));
        rootHash = uint256(proof[0]) < uint256(aliceHash)
            ? keccak256(abi.encode(proof[0], aliceHash))
            : keccak256(abi.encode(aliceHash, proof[0]));

        vm.startPrank(owner);
        traitForgeNft.setRootHash(rootHash);
        vm.stopPrank();

        vm.startPrank(alice);
        traitForgeNft.mintToken{value: 1 ether}(proof);
        uint256 tokenIdAlice = tokenIdBob + 1;
        // Mint until fertile and merger
        while (
            ((traitForgeNft.tokenEntropy(tokenIdAlice) / 10) % 10 == 0) ||
            (traitForgeNft.tokenEntropy(tokenIdAlice) % 3 == 0)
        ) {
            traitForgeNft.mintToken{value: 1 ether}(proof);
            tokenIdAlice++;
        }
        uint256 forgePotentialAlice = uint8(
            (traitForgeNft.tokenEntropy(tokenIdAlice) / 10) % 10
        );
        console.log(forgePotentialAlice); // 9

        vm.startPrank(bob);
        entityForging.listForForging(tokenIdBob, 0.01 ether);
        vm.stopPrank();
        uint256 balanceAliceBefore = alice.balance;
        vm.startPrank(alice);
        entityForging.forgeWithListed{value: 1 ether}(tokenIdBob, tokenIdAlice);
        vm.stopPrank();
        uint256 balanceAliceAfter = alice.balance;
        assertEq(balanceAliceBefore - balanceAliceAfter, 1 ether);
    }
}

if (excessPayment > 0) {
    (bool refundSuccess, ) = msg.sender.call{value: excessPayment}("");
    require(refundSuccess, "Refund of excess payment failed.");
}
