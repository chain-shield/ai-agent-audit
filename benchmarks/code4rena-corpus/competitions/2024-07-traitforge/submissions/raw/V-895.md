# Vulnerability details

## Summary
here the loop condition should use generationMintCounts[currentGeneration] instead of _tokenIds
```solidity
// TraitForgeNft.sol :215
    while (budgetLeft >= mintPrice && _tokenIds < maxTokensPerGen) {
      _mintInternal(msg.sender, mintPrice);
      amountMinted++;
      budgetLeft -= mintPrice;
      mintPrice = calculateMintPrice();
    }
```
mintWithBudget won't work once the first gen was fully generated.

## Proof of Concept
We first generate all entities in the first generation, with at least one forge for the second generation. After all entities in the first generation was generated, mintWithBudget won't work.
run the following foundry test with --gas-limit 10000000000
```solidity
contract EntityForgingTest is Test {
    EntityForging entityForging;
    TraitForgeNft traitForgeNft;
    EntropyGenerator entropyGenerator;
    Airdrop airdrop;
    DevFund devFund;
    NukeFund nukeFund;
    EntityTrading entityTrading;

    address ownerAddr = makeAddr("ownerAddr");
    address player1 = makeAddr("player1");
    address player2 = makeAddr("player2");
    address player3 = makeAddr("player3");

    function setUp() public {
        vm.startPrank(ownerAddr);

        // Deploy TraitForgeNft contract
        traitForgeNft = new TraitForgeNft();

        // Deploy Airdrop contract
        airdrop = new Airdrop();
        traitForgeNft.setAirdropContract(address(airdrop));
        airdrop.transferOwnership(address(traitForgeNft));

        // Deploy entropyGenerator contract
        entropyGenerator = new EntropyGenerator(address(traitForgeNft));
        entropyGenerator.writeEntropyBatch1();
        entropyGenerator.writeEntropyBatch2();
        entropyGenerator.writeEntropyBatch3();
        traitForgeNft.setEntropyGenerator(address(entropyGenerator));

        // Deploy EntityForging contract
        entityForging = new EntityForging(address(traitForgeNft));
        traitForgeNft.setEntityForgingContract(address(entityForging));

        devFund = new DevFund();
        nukeFund = new NukeFund(address(traitForgeNft), address(airdrop), payable(address(devFund)), payable(ownerAddr));
        traitForgeNft.setNukeFundContract(payable(address(nukeFund)));
        entityForging.setNukeFundAddress(payable(address(nukeFund)));

        entityTrading = new EntityTrading(address(traitForgeNft));
        entityTrading.setNukeFundAddress(payable(address(nukeFund)));

        traitForgeNft.setStartPrice(0.001 ether);
        traitForgeNft.setPriceIncrement(0.0001 ether);

        vm.stopPrank();

        vm.deal(player1, 10 ether);
        vm.deal(player2, 10 ether);
        vm.deal(player3, 100000 ether);

        // Used to avoid whitelist proof
        vm.warp(86402);
        vm.roll(86402);
    }

    // test with --gas-limit 10000000000
    function testHighTokenIdDOSMintWithBudget() public {
        bytes32[] memory proof;

        // palyer1 gets token 1-4
        vm.startPrank(player1);
        traitForgeNft.mintWithBudget{value: 0.005 ether}(proof); // mintWithBudget works fine now
        // for (uint256 i = 1; i < 5; i++) {
            // traitForgeNft.mintToken{value: 1 ether}(proof);
        // }

        // player2 mint token 5-9, list token 5 for forging
        vm.startPrank(player2);
        for (uint256 i = 5; i < 10; i++) {
            traitForgeNft.mintToken{value: 1 ether}(proof);
        }
        entityForging.listForForging(5, 0.01 ether);
        vm.stopPrank();

        // player1 forge token 2 and 5 to get token 10
        vm.prank(player1);
        entityForging.forgeWithListed{value: 1 ether}(5, 2);

        vm.startPrank(player3);
        for (uint256 i = 10; i < 10000; i++) {
            traitForgeNft.mintToken{value: 2 ether}(proof);
        }

        assertEq(traitForgeNft.balanceOf(player3), 9990);
        assertEq(traitForgeNft.currentGeneration(), 1);

        traitForgeNft.mintToken{value: 10 ether}(proof);
        assertEq(traitForgeNft.balanceOf(player3), 9991);
        assertEq(traitForgeNft.currentGeneration(), 1);

        // mintWithBudget doesn't work now
        traitForgeNft.mintWithBudget{value: 10 ether}(proof);
        assertEq(traitForgeNft.balanceOf(player3), 9991);
        assertEq(traitForgeNft.currentGeneration(), 1);

        vm.stopPrank();

        // same for player1 and player2
        assertEq(traitForgeNft.balanceOf(player1), 5);
        vm.prank(player1);
        traitForgeNft.mintWithBudget{value: 8 ether}(proof);
        assertEq(traitForgeNft.balanceOf(player1), 5);

        assertEq(traitForgeNft.balanceOf(player2), 5);
        vm.prank(player2);
        traitForgeNft.mintWithBudget{value: 8 ether}(proof);
        assertEq(traitForgeNft.balanceOf(player2), 5);
        
    }
}
```
