# Vulnerability details

## Impact
The invariant of having only 10,000 entities per generation can be broken very casually due to the resetting of the generationMintCount in the TraitForgeNft:`_incrementGeneration` function.

## Proof of Concept
Let's take a scenario where minting has occurred in the first generation up to 9,500 entities and forging has occurred between the first generation entities up to 1,500 entities.

The mapping,
`generationMintCounts[gen1]` would be up to 9,500
`generationMintCounts[gen2]` would be up to 1,500

If an extra 501 mints occur in the traitForgeNFT contract, the `_incrementGeneration` funciton is called because gen1 would have exceeded its entity cap.

Inside the `_incrementGeneration` function, this logic is executed;

`currentGeneration++;`
`generationMintCounts[currentGeneration] = 0;`

Basically incrementing the current generation from gen1 to gen2 and resetting the `generatioMintCounts[gen2]` from 1,500 to 0.
Ignoring the 1,500 entities that already existed and starting from 0.

Eventually when the `generatioMintCounts[gen2]` gets to its max cap. 
The generation 2 would be left with 10,000 + 1,500 entities, which breaks the invariant.


## Tools Used
Foundry in the test files

Make sure to follow the instructions on how to add foundry to a hardhat project [here](https://book.getfoundry.sh/config/hardhat?highlight=hardhat#adding-foundry-to-a-hardhat-project).

To run the POC, create a folder in the test file called foundry-tests and a test file of MyTest.t.sol
Then paste the POC code into it.

You'll notice I transferred the `EntropyGenerator` ownership to the `traitForgeNft` contract to bypass the wrong modifier bug in `initializeAlphaIndices` I submitted in a different issue.

I also added a view function into the traitForgeNft contract that returns the total number of tokenId's created, to aid testing.

Run the test in the terminal with `forge t --mt test_MoreThan10000__EntitiesCanExistInAsingleGeneration -vv`


```solidity
// SPDX-License-Identifier: MIT

pragma solidity ^0.8.20;

import "forge-std/Test.sol";
import "forge-std/console.sol";

// Imports
import {EntropyGenerator} from "../../contracts/EntropyGenerator/EntropyGenerator.sol";
import {DevFund} from "../../contracts/DevFund/DevFund.sol";
import {TraitForgeNft} from "../../contracts/TraitForgeNft/TraitForgeNft.sol";
import {EntityForging} from "../../contracts/EntityForging/EntityForging.sol";
import {EntityTrading} from "../../contracts/EntityTrading/EntityTrading.sol";
import {NukeFund} from "../../contracts/NukeFund/NukeFund.sol";
import {Airdrop} from "../../contracts/Airdrop/Airdrop.sol";
import {Trait} from "../../contracts/Trait/Trait.sol";

contract MyTest is Test {
    // Contracts
    EntropyGenerator entropyGenerator;
    DevFund devFund;
    TraitForgeNft traitForgeNft;
    EntityForging entityForging;
    EntityTrading entityTrading;
    NukeFund nukeFund;
    Airdrop airdrop;
    Trait trait;

    // users
    address owner = makeAddr("owner");
    address tempDaoAddr = makeAddr("DAO");
    address userOne = makeAddr("userOne");
    address userTwo = makeAddr("userTwo");
    address userThree = makeAddr("userThree");

    // token info
    string name = "TRAIT";
    string symbol = "TRAIT";
    uint8 decimals = 18;
    uint256 totalSupply = 1000000e18;
    bytes32[] emptyProof = [keccak256("hey")];
    uint256 startingBal = 10000 ether;

    function setUp() public {
        vm.startPrank(owner);

        // deploy
        traitForgeNft = new TraitForgeNft();
        airdrop = new Airdrop();
        devFund = new DevFund();
        nukeFund = new NukeFund(address(traitForgeNft), address(airdrop), payable(owner), payable(tempDaoAddr));
        entropyGenerator = new EntropyGenerator(address(traitForgeNft));
        entityForging = new EntityForging(address(traitForgeNft));
        entityTrading = new EntityTrading(address(traitForgeNft));
        trait = new Trait(name, symbol, decimals, totalSupply);

        // setters after deploy
        traitForgeNft.setEntropyGenerator(address(entropyGenerator));
        traitForgeNft.setEntityForgingContract(address(entityForging));
        traitForgeNft.setNukeFundContract(payable(address(nukeFund)));
        traitForgeNft.setAirdropContract(address(airdrop));
        entityTrading.setNukeFundAddress(payable(address(nukeFund)));
        entityForging.setNukeFundAddress(payable(address(nukeFund)));
        airdrop.setTraitToken(address(trait));
        airdrop.transferOwnership(address(traitForgeNft));

        // to avoid the issue with reverting on incrementing generation
        entropyGenerator.transferOwnership(address(traitForgeNft));
        // writing the entropy batches
        entropyGenerator.writeEntropyBatch1();
        entropyGenerator.writeEntropyBatch2();
        entropyGenerator.writeEntropyBatch3();

        // move time to after whtelist period
        vm.warp(block.timestamp + 24 hours + 1);

        // deal 3 test accounts ETH
        vm.deal(userOne, startingBal);
        vm.deal(userTwo, startingBal);
        vm.deal(userThree, startingBal);

        vm.stopPrank();
    }

    /* MAIN TEST STARTS HERE */
    function test_MoreThan10000__EntitiesCanExistInAsingleGeneration() public {
        uint256 maxEntitiesPerGen = 10000;

        // the forger and merger token Id to use to forge
        uint256 forgerTokenId;
        uint256 mergerTokenId;
        uint256 expectedForgePotential = 5;

        vm.startPrank(userOne);

        // skip-mint entities to be close to the amount of max entities for the 1st gen
        // mints 9,999 entities
        for (uint256 i = 1; i <= (maxEntitiesPerGen - 1); i++) {
            uint256 tokenId = i;
            // mints entity
            traitForgeNft.mintToken{value: traitForgeNft.calculateMintPrice()}(emptyProof);

            // helper to get a forger and a merger with a high forge potential of >= 5 so i can forge multiple times in test
            bool isTokenForger = traitForgeNft.isForger(tokenId);
            uint256 entropy = traitForgeNft.getTokenEntropy(tokenId);
            uint256 forgePotential = uint8((entropy / 10) % 10);

            // gets forger
            if (isTokenForger && forgePotential > expectedForgePotential && forgerTokenId == 0) {
                console.log("Forger With Forge Potential > 0 Found: ");
                console.log("Forge Potential: ", forgePotential);
                forgerTokenId = tokenId;
                console.log("===============================");
            }

            // gets merger
            if (!isTokenForger && forgePotential > expectedForgePotential && mergerTokenId == 0) {
                console.log("Merger Found: ");
                console.log("Forge Potential: ", forgePotential);
                mergerTokenId = tokenId;
                console.log("===============================");
            }
        }

        // transfer the merger token Id to user two because user one wont be able to forge with himself.
        traitForgeNft.transferFrom(userOne, userTwo, mergerTokenId);

        vm.stopPrank();

        // forge and create 5 tokens in generation 2 due to their parents being gen 1
        for (uint256 i = 1; i <= expectedForgePotential; i++) {
            vm.startPrank(userOne);
            entityForging.listForForging(forgerTokenId, entityForging.minimumListFee());
            vm.stopPrank();

            vm.startPrank(userTwo);
            uint256 newTokenId =
                entityForging.forgeWithListed{value: entityForging.minimumListFee()}(forgerTokenId, mergerTokenId);
            vm.stopPrank();
        }

        // mint the last entity on gen1 and extra entities
        // after the cap is exceeded it triggers the zero reset in the `_incrementGeneration`
        // i.e from gen 1: 9,999 -> gen1: 10,000 -> gen2: 0 (reset new generation)
        // ignoring that there were already forged entities in the generation before the mint in gen1 finished.
        vm.startPrank(userThree);
        // mint and fill up the 2nd generation
        for (uint256 i = 1; i <= maxEntitiesPerGen + 1; i++) {
            traitForgeNft.mintToken{value: traitForgeNft.calculateMintPrice()}(emptyProof);
        }
        vm.stopPrank();

        uint256 generationTwo = 2;

        // assert equal
        assertEq(traitForgeNft.currentGeneration(), generationTwo);
        // assert greater than
        assertGt(_getNumberOfEntitiesInGeneration(generationTwo), maxEntitiesPerGen);

        console.log("Number Of Entities In Gen 1: ", _getNumberOfEntitiesInGeneration(1));
        console.log("Number Of Entities In Gen 2: ", _getNumberOfEntitiesInGeneration(generationTwo));
        console.log("Number Of Entities Created: ", traitForgeNft.getTotalTokensCreated());
        console.log("Current Generation: ", traitForgeNft.getGeneration());
    }

    function _getNumberOfEntitiesInGeneration(uint256 _generation) internal view returns (uint256) {
        uint256 totalEntitiesExisting = traitForgeNft.getTotalTokensCreated();
        uint256 generationCount;

        for (uint256 entity = 1; entity <= totalEntitiesExisting; entity++) {
            uint256 entityGeneration = traitForgeNft.getTokenGeneration(entity);
            if (entityGeneration == _generation) {
                generationCount++;
            }
        }

        return generationCount;
    }
}
```

## Recommended Mitigation Steps
Keeping track of how many entities have been forged before minting increments generation, then picking up the generation mint count from where it stopped rather than from zero.
