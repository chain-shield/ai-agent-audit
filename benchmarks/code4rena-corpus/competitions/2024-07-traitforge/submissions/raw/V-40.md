# Vulnerability details

When Nuking a token the claimAmount is not sent to the token owner, but to caller of function, which can result in lost ETH for owner. 

## Proof of Concept

The owner of a token is being phished and accidentally approves a malicious actor, the malicious actor nukes the token, the claimAmount is then transferred to the caller of the function (malicious actor) instead of the owner of the token. 

```solidity
// the following function is added to TraitForgeNft.sol to receive last minted tokenId
  function getTokenId() external view returns(uint256) {
    return _tokenIds;
  }
  
  // following POC showcase the approved user receives claimAmount instead of token owner
  it('should demonstrate owner of token does not receive claimAmount when nuking', async function () {
    // Mint a token
    await nft.connect(user1).mintToken(merkleInfo.whitelist[1].proof, {
      value: ethers.parseEther('1'),
    });

    // helper function added
    const tokenIdMinted = await nft.getTokenId();

    // accidentally approves wrong message
    await nft.connect(user1).setApprovalForAll(maliciousUser.address, true)
    const result = await nft.isApprovedForAll(user1.address, maliciousUser.address);
    expect(result).to.be.true;

    // get user balance before nuking
    const prevTokenOwnerEthBalance = await ethers.provider.getBalance(
      await user1.getAddress()
    );
    // get malicious user balance before nuking
    const prevMaliciousUserEthBalance = await ethers.provider.getBalance(
      await maliciousUser.getAddress()
    );

    // Ensure the token can be nuked
    expect(await nukeFund.canTokenBeNuked(tokenIdMinted)).to.be.true;

    // malicious user approves nukeFund address
    await nft.connect(maliciousUser).approve(await nukeFund.getAddress(), tokenIdMinted);

    // malicious user nukes the token.
    const finalNukeFactor = await nukeFund.calculateNukeFactor(tokenIdMinted);
    const fund = await nukeFund.getFundBalance();
    // malicious user nukes the token.
    await expect(nukeFund.connect(maliciousUser).nuke(tokenIdMinted))
      .to.emit(nukeFund, 'Nuked')
      .withArgs(maliciousUser, tokenIdMinted, (fund * finalNukeFactor) / 100000n)
      .to.emit(nukeFund, 'FundBalanceUpdated')
      .withArgs(fund - (fund * finalNukeFactor) / 100000n);

    const curTokenOwnerEthBalance = await ethers.provider.getBalance(
      await user1.getAddress()
    );
    // balance is same as before
    expect(curTokenOwnerEthBalance).to.be.eq(prevTokenOwnerEthBalance);
    
    const curMaliciousEthBalance = await ethers.provider.getBalance(
      await maliciousUser.getAddress()
    );
    // claim amount transferred to attacker.
    expect(curMaliciousEthBalance).to.be.gt(prevMaliciousUserEthBalance);
  });
```

## Recommended Mitigation Steps

When nuking a token make sure the owner of the token receives the claimedAmount instead of the function caller. 

```diff
function nuke(uint256 tokenId) public whenNotPaused nonReentrant {
    require(
      nftContract.isApprovedOrOwner(msg.sender, tokenId),
      'ERC721: caller is not token owner or approved'
    );
    require(
      nftContract.getApproved(tokenId) == address(this) ||
        nftContract.isApprovedForAll(msg.sender, address(this)),
      'Contract must be approved to transfer the NFT.'
    );
    require(canTokenBeNuked(tokenId), 'Token is not mature yet');

    uint256 finalNukeFactor = calculateNukeFactor(tokenId); // finalNukeFactor has 5 digits
    uint256 potentialClaimAmount = (fund * finalNukeFactor) / MAX_DENOMINATOR; // Calculate the potential claim amount based on the finalNukeFactor
    uint256 maxAllowedClaimAmount = fund / maxAllowedClaimDivisor; // Define a maximum allowed claim amount as 50% of the current fund size

    // Directly assign the value to claimAmount based on the condition, removing the redeclaration
    uint256 claimAmount = finalNukeFactor > nukeFactorMaxParam
      ? maxAllowedClaimAmount
      : potentialClaimAmount;

    fund -= claimAmount; // Deduct the claim amount from the fund

    nftContract.burn(tokenId); // Burn the token

+   address ownerOfNft = nft.ownerOf(tokenId); // must be above burn
    nftContract.burn(tokenId);
+   (bool success, ) = payable(ownerOfNft).call{ value: claimAmount }('');
-   (bool success, ) = payable(msg.sender).call{ value: claimAmount }('');
    require(success, 'Failed to send Ether');

    emit Nuked(msg.sender, tokenId, claimAmount); // Emit the event with the actual claim amount
    emit FundBalanceUpdated(fund); // Update the fund balance
  }
```
