# Vulnerability details

## Impact
-  If a user forges with a listed entity by calling `forgeWithListed` in `EntityForging.sol` with more Eth than necessary the excess Eth will not be refunded back to them. 
- Observe this line of code in the `forgeWithListed` function in `EntityForging.sol`. It requires that a user provide a `msg.value` that is greater than or equal to the `forgingFee` but there is no implementation to make sure that if the user sends excess Eth it is refunded back to them.
```solidity
require(msg.value >= forgingFee, 'Insufficient fee for forging');
```

## Proof of Concept
#### Setup
 - In `foundry.toml` add `verbosity = 4`  or set it to 4 to view the logs.
 - Since `Airdrop.sol` is not in scope remove the `onlyOwner` modifer from the function `addUserAmount()` for this demonstration.
`Airdrop.sol`
```solidity
function addUserAmount(
      address user,
      uint256 amount
    ) external whenNotPaused nonReentrant  {
      require(!started, 'Already started');
      userInfo[user] += amount;
      totalValue += amount;
    }
```

#### THE POC
- Create a file called `POC.sol` in the test folder and add the below code, then run `forge test --match-test testCheckAliceBalance`.
```solidity
// SPDX-License-Identifier: MIT
pragma solidity ^0.8.20;

import "lib/forge-std/src/Test.sol";
import "lib/forge-std/src/console.sol";
import {TraitForgeNft} from "contracts/TraitForgeNft/TraitForgeNft.sol";
import {EntropyGenerator} from "contracts/EntropyGenerator/EntropyGenerator.sol";
import {NukeFund} from "contracts/NukeFund/NukeFund.sol";
import {EntityForging} from "contracts/EntityForging/EntityForging.sol";
import "contracts/Airdrop/Airdrop.sol";

contract POC is Test {

    TraitForgeNft _traitforgeNft;
    EntropyGenerator _entropygenerator;
    EntityForging _entityforging;
    NukeFund _nukefund;
    Airdrop _airdrop;

    address Bob = makeAddr("Bob");
    address Alice = makeAddr("Alice");
    address Dev1 = makeAddr("Dev1");
    address Dao = makeAddr("Dao");
    address payable _payableDev1 = payable(Dev1);
    address payable _payableDao = payable(Dao);

    function setUp() public {
        _traitforgeNft = new TraitForgeNft();
        _entropygenerator = new EntropyGenerator(address(_traitforgeNft));
        _airdrop = new Airdrop();
        _nukefund = new NukeFund(
            address(_traitforgeNft),
            address(_airdrop),
            _payableDev1,
            _payableDao
        );
        _entityforging = new EntityForging(address(_traitforgeNft));
        // Deal Eth to users
        vm.deal(Bob, 20 ether);
        vm.deal(Alice, 20 ether);
    }

    function setAddress() public {
        address payable _payable_nukefund = payable(_nukefund);
        _traitforgeNft.setNukeFundContract(_payable_nukefund);
        _traitforgeNft.setEntropyGenerator(address(_entropygenerator));
        _traitforgeNft.setEntityForgingContract(address(_entityforging));
        _traitforgeNft.setAirdropContract(address(_airdrop));
    }

    function testCheckAliceBalance() public {
        // Set addresses in TraitForgeNft
        setAddress();
        // initalize entropy values
        _entropygenerator.writeEntropyBatch1();
        // Bob mints 15 Entities
        vm.startPrank(Bob);
        for (uint256 i = 0; i < 15; i++) {
            _traitforgeNft.mintToken{value: 10 ether}();
        }
        vm.stopPrank();

        // Alice mints 15 Entities
        vm.startPrank(Alice);
        for (uint256 i = 0; i < 15; i++) {
            _traitforgeNft.mintToken{value: 10 ether}();
        }
        // Query Alice's balance
        console.log("Alice's balance after minting: ", address(Alice).balance);
        // After minting Alice's balance will be 19.916915 ether
        vm.stopPrank();

        // Bob lists his entity for forging
        vm.startPrank(Bob);
        _entityforging.listForForging(5, 0.01 ether);
        vm.stopPrank();

        // Alice forges with Bobs entity
        vm.startPrank(Alice);
        _entityforging.forgeWithListed{value: 19.916915 ether}(5, 16);
        // Query Alice's balance
        console.log("Alice's balance after forging: ", address(Alice).balance);
        // Alice's balance is now 0
        vm.stopPrank();
        assertEq(address(Alice).balance, 0);
    }

}
```
