# Vulnerability details

[The docs](https://docs.google.com/document/d/1pihtkKyyxobFWdaNU4YfAy56Q7WIMbFJjSHUAfRm6BA/edit) describe pricing of mints with the following statements:

> The Mint fee for generation 1 starts at 0.005 ETH and ends at 0.25ETH after 10,000. Generation 2 starts at 0.005 ETH and ends at 0.3ETH. Generation 3 starts at 0.005 ETH and ends at 0.35 ETH, and so on.

Also, in the Discord chat, the sponsor confirmed that the startPrice for each generation should be 0.005 ETH.

The implemented logic, however, does not cover these requirements.

## Impact
Generation 2 actually does not start at 0.005 ETH. The first mint of Generation 2, Generation 3, and so on would be overpriced and charged the price of the 10,000th entity from the last generation.

Thus, this violates the statement that "Generations start at 0.005 ETH," as after Generation 1, every first-minted entity of the next generations would be overpriced, and the second mint from a new generation would be 0.005 ETH + 1 * (0.0000245 + 0.000005). 

So we won't have a case where the startPrice is just 0.005 ETH, which violates the protocol and the sponsor's requirements.

Additionally its important to mention that sponsor confirmed also that last price of gen should be 9999 * priceForEntityForThisGen, [here](https://discord.com/channels/810916927919620096/1266325696079466609/1267766987925360714), since as mentioned in the docs, the price is incrementing after the first mint, for each Generation.

```solidity
// SPDX-License-Identifier: UNLICENSED
pragma solidity ^0.8.25;

import {Test} from "forge-std/Test.sol";
import {StdInvariant} from "forge-std/StdInvariant.sol";
import {console} from "forge-std/console.sol";
import {EntityForging} from "../contracts/EntityForging/EntityForging.sol";
import {TraitForgeNft} from "../contracts/TraitForgeNft/TraitForgeNft.sol";
import {Airdrop} from "../contracts/Airdrop/Airdrop.sol";
import {DevFund} from "../contracts/DevFund/DevFund.sol";
import {EntityTrading} from "../contracts/EntityTrading/EntityTrading.sol";
import {EntropyGenerator} from "../contracts/EntropyGenerator/EntropyGenerator.sol";
import {NukeFund} from "../contracts/NukeFund/NukeFund.sol";

contract FoundryTest is StdInvariant, Test {
    /* Main Contracts */
    EntropyGenerator public entropyGeneratorContract;
    EntityTrading public entityTradingContract;
    EntityForging public entityForgingContract;
    TraitForgeNft public nftContract;
    NukeFund public nukeFundContract;

    /* Misc Contracts */
    Airdrop public airdropContract;
    DevFund public devFundContract;

    /* User Addresses */
    address NFT_OWNER = makeAddr("nftOwner");
    address SECOND_OWNER = makeAddr("nftOwnerSecond");

    function setUp() public {
        /* Create contracts */
        airdropContract = new Airdrop();
        devFundContract = new DevFund();
        nftContract = new TraitForgeNft();
        entityTradingContract = new EntityTrading(address(nftContract));
        entityForgingContract = new EntityForging(address(nftContract));
        entropyGeneratorContract = new EntropyGenerator(address(nftContract));
        nukeFundContract = new NukeFund(
            address(nftContract),
            address(airdropContract),
            payable(address(devFundContract)),
            payable(address(devFundContract))
        );

        /* Link contracts via setters */
        nftContract.setEntropyGenerator(address(entropyGeneratorContract));
        nftContract.setEntityForgingContract(address(entityForgingContract));
        nftContract.setAirdropContract(address(airdropContract));
        nftContract.setNukeFundContract(payable(address(nukeFundContract)));
        entityTradingContract.setNukeFundAddress(payable(address(nukeFundContract)));
        entityForgingContract.setNukeFundAddress(payable(address(nukeFundContract)));
        devFundContract.addDev(address(this), 1);

        /* Generate entropies */
        entropyGeneratorContract.writeEntropyBatch1();
        entropyGeneratorContract.writeEntropyBatch2();
        entropyGeneratorContract.writeEntropyBatch3();

        // Skip the "24hrs mint whitelist period" for TraitForgeNft, to avoid overloading test with merkle proofs
        skip(25 hours);
    }

    function test_mintWithLowestEntropyAboveZero() public {
        /* Mint 100 NFTs */
        for (uint256 i = 1; i <= 100; i++) {
            hoax(NFT_OWNER, 0.5 ether);
            nftContract.mintToken{value: 0.5 ether}(new bytes32[](0));
            assertEq(nftContract.ownerOf(i), NFT_OWNER);
        }

        /* We set maxTokensPerGen in TraitForgeNft to 100, so next mint should increment generation */
        assertEq(nftContract.currentGeneration(), 1);
        assertEq(nftContract.generationMintCounts(1), 100);

        // Since this is the first mint of the new generation the price should be 0.005 ether
        // However, currently we always overpay for first mint of new generation, since its calculated,
        // before the increment in generation
        vm.expectRevert("Insufficient ETH send for minting.");
        hoax(SECOND_OWNER, 0.005 ether);
        nftContract.mintToken{value: 0.005 ether}(new bytes32[](0));

        // Try mint with price increment, since our maxTokensPerGen is 100, we factor it in the calculation
        uint256 price = 0.005 ether + (0.0000245 ether * 100);
        assertEq(price, nftContract.calculateMintPrice());

        hoax(SECOND_OWNER, price);
        nftContract.mintToken{value: price}(new bytes32[](0));
        assertEq(nftContract.ownerOf(101), SECOND_OWNER);

        // Next price should be 0.005 ether + (0.0000245 ether * 2)
        // But we actually already paid more than proposed, for the first mint of the new generation
        // We don't even have case for just 0.005 ether, which is wrong regarding to the docs.
        assertEq(nftContract.calculateMintPrice(), 0.005 ether + 1 * (0.0000245 ether + 0.000005 ether));
    }
}
```

## Recommendations
Adjust the `calculateMintPrice()` as follows:
```diff
 function calculateMintPrice() public view returns (uint256) {
    uint256 currentGenMintCount = generationMintCounts[currentGeneration];
+      if (currentGenMintCount == maxTokensPerGen) {
+         return startPrice;
+      }
    uint256 priceIncrease = priceIncrement * currentGenMintCount;
    uint256 price = startPrice + priceIncrease;
    return price;
  }
```
