# Vulnerability details

## Description
The `addDev` function does not set the `pendingRewards` value when adding a new developer. This oversight could potentially allow a newly added developer to claim rewards for the period before they were added to the system.

## Current code
```solidity
function addDev(address user, uint256 weight) external onlyOwner {
    DevInfo storage info = devInfo[user];
    require(weight > 0, "Invalid weight");
    require(info.weight == 0, "Already registered");
    info.rewardDebt = totalRewardDebt;
    info.weight = weight;
    totalDevWeight += weight;
    emit AddDev(user, weight);
}
```

## Impact
Unfair reward distribution for devs

## Recommended fix:
Add a line to initialize `pendingRewards` to zero:

```solidity
function addDev(address user, uint256 weight) external onlyOwner {
    DevInfo storage info = devInfo[user];
    require(weight > 0, "Invalid weight");
    require(info.weight == 0, "Already registered");
    info.rewardDebt = totalRewardDebt;
    info.weight = weight;
    info.pendingRewards = 0; // Add this line
    totalDevWeight += weight;
    emit AddDev(user, weight);
}
```

By initializing `pendingRewards` to zero, we ensure that new developers start with no accumulated rewards. This prevents potential unfair distribution of rewards and aligns the function's behavior with the likely intended logic of the contract.



## Poc

```javascript
const { expect } = require("chai");
const { ethers } = require("hardhat");

describe("DevFund", function () {
  let DevFund, devFund, owner, dev1, dev2, user;

  beforeEach(async function () {
    [owner, dev1, dev2, user] = await ethers.getSigners();
    DevFund = await ethers.getContractFactory("DevFund");
    devFund = await DevFund.deploy();
    await devFund.deployed();
  });

  it("should demonstrate the issue in addDev function", async function () {
    await devFund.connect(owner).addDev(dev1.address, 100);

    await user.sendTransaction({
      to: devFund.address,
      value: ethers.utils.parseEther("1"),
    });

    let pendingRewards = await devFund.pendingRewards(dev1.address);
    expect(pendingRewards).to.equal(ethers.utils.parseEther("1"));

    await devFund.connect(owner).addDev(dev2.address, 100);

    await user.sendTransaction({
      to: devFund.address,
      value: ethers.utils.parseEther("1"),
    });

    pendingRewards = await devFund.pendingRewards(dev1.address);
    expect(pendingRewards).to.equal(ethers.utils.parseEther("1.5"));

    pendingRewards = await devFund.pendingRewards(dev2.address);
    expect(pendingRewards).to.equal(ethers.utils.parseEther("0.5"));

    // The issue: dev2 has 0.5 ETH pending rewards, even though they were added after the first ETH was sent

    await devFund.connect(dev1).claim();
    await devFund.connect(dev2).claim();

    const dev1Balance = await ethers.provider.getBalance(dev1.address);
    const dev2Balance = await ethers.provider.getBalance(dev2.address);

    // dev2 incorrectly received rewards for the period before they were added
    expect(dev2Balance.gt(ethers.utils.parseEther("0.4"))).to.be.true;
  });
});
```

The issue is demonstrated by the fact that dev2 receives rewards for the period before they were added to the system. This is shown by dev2 having 0.5 ETH in pending rewards after the second ETH is sent, and their balance being greater than 0.4 ETH after claiming (accounting for gas costs).
