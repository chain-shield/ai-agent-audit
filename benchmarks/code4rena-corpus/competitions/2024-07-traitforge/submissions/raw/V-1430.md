# Vulnerability details

## Impact
DevFund, EntityTrading, EntropyGenerator, TraitForgeNft, NukeFund, EntityForging contracts they all inherits from openzeppelin Pausable contract. OZ Pausable contract has 2 virtual functions named _pause and _unpause. but they are internal and need to be used in external/public functions inside the derived contract for the contract owner to be able to pause and unpause the contracts. 

## Proof of Concept
// SPDX-License-Identifier: GPL-3.0-or-later

pragma solidity 0.8.20;

import "forge-std/Test.sol";

/*
* @test: forge test -C DevFundTest.sol --rpc-url http://127.0.0.1:8545 -vvvv
*/

import '@openzeppelin/contracts/access/Ownable.sol';
import '@openzeppelin/contracts/security/Pausable.sol';

contract DevFund is Ownable, Pausable {
    function pauseContract() external onlyOwner {
        _pause();
    }

    function unPauseContract() external onlyOwner {
        _unpause();
    }
}

contract DevFundTest is Test {
    DevFund devFund;

    function setUp() external {
        devFund = new DevFund();
    }

    function testPause() external {
        assertEq(devFund.paused(), false);
        devFund.pauseContract();
        assertEq(devFund.paused(), true);
    }

    function testUnPause() external {
        devFund.pauseContract();
        assertEq(devFund.paused(), true);
        devFund.unPauseContract();
        assertEq(devFund.paused(), false);
    }
}
