# Vulnerability details

## Impact
User can call `listForForging` https://github.com/code-423n4/2024-07-traitforge/blob/279b2887e3d38bc219a05d332cbcb0655b2dc644/contracts/EntityForging/EntityForging.sol#L67
And then he can either burn or nuke his token.

The user could `listForForging` with very high fee, so that no one would want to merge with that forger. Then after 3 days he can at least get some of his money back if he decides to nuke and not burn his token.

When the burn or nuke happen, the `Listing` will still be `"alive"`, and other users will think that they can merge, but they won't be able to.

## Proof of Concept
I have used the `TestERC721.sol` file that is inside contracts/test folder and slightly modified it.
```javascript
// SPDX-License-Identifier: MIT
pragma solidity ^0.8.20;

import "@openzeppelin/contracts/token/ERC721/ERC721.sol";

contract TestERC721 is ERC721 {
    uint256 private _tokenIds;

    constructor() ERC721("TraitForgeNft", "TFGNFT") {}

    function mintToken(address to) external {
        _tokenIds++;
        _mint(to, _tokenIds);
    }

    function burn(uint256 id) external {
        require(_isApprovedOrOwner(msg.sender, id), "ERC721: caller is not token owner or approved");
        _burn(id);
    }

    function getTokenEntropy(uint256 tokenId) public view returns (uint256) {
        require(ownerOf(tokenId) != address(0), "ERC721: query for nonexistent token");
        return 123456;
    }
}
```

I have also created a test file inside `contracts/test`.
```javascript
// SPDX-License-Identifier: MIT
pragma solidity ^0.8.20;

import {Test} from "../../lib/forge-std/src/Test.sol";
import "../EntityTrading/EntityTrading.sol";
import "./TestERC721.sol";
import "contracts/EntropyGenerator/EntropyGenerator.sol";
import "contracts/EntityForging/EntityForging.sol";
import "contracts/NukeFund/NukeFund.sol";

contract EntityTradingTest is Test {
    EntityTrading entityTrading;
    EntropyGenerator entropyGenerator;
    EntityForging entityForging;
    TestERC721 mockNft;
    NukeFund nukeFund;

    address owner = makeAddr("owner");
    address user1 = makeAddr("user");
    address user2 = makeAddr("user2");

    function setUp() public {
        vm.startPrank(owner);
        mockNft = new TestERC721();
        entropyGenerator = new EntropyGenerator(address(mockNft));
        entityForging = new EntityForging(address(mockNft));
        entityTrading = new EntityTrading(address(mockNft));
        nukeFund = new NukeFund(address(mockNft), address(0), payable(address(0)), payable(address(0)));
        entityTrading.setNukeFundAddress(payable(address(nukeFund)));
        vm.stopPrank();
    }

    function testUserListsForForgingAndThenBurnsToken() public {
        uint256 tokenId = 1;
        uint256 minimumFee = 0.01 ether;

        mockNft.mintToken(user2);
        mockNft.setApprovalForAll(address(mockNft), true);

        vm.prank(user2);
        entityForging.listForForging(tokenId, minimumFee);

        uint256 firstListing = entityForging.listingCount();
        // checking if i have listed. If success it will be 1
        assertEq(firstListing, 1);
        // checking if the listing is active
        assertEq(entityForging.getListings(tokenId).isListed, true);

        vm.prank(user2);
        mockNft.burn(tokenId);

        // checking if the listing is still active after the token burn
        assertEq(entityForging.getListings(tokenId).isListed, true);
    }
}
```
You can run the test with `forge test --mt testUserListsForForgingAndThenBurnsToken`

## Recommended Mitigation Steps
Consider not letting the user burn/nuke his token if he has the token registered for `listingForForge`.

Another solution would be if you want him to burn/nuke his token, just check if that token has been listed and then simply call this function `EntityForging.sol::cancelListingForForging`:
```javascript
    function cancelListingForForging(uint256 tokenId) external whenNotPaused nonReentrant {
        require(
            nftContract.ownerOf(tokenId) == msg.sender || msg.sender == address(nftContract),
            "Caller must own the token"
        );
        require(listings[listedTokenIds[tokenId]].isListed, "Token not listed for forging");

        _cancelListingForForging(tokenId);
    }
```
