# Vulnerability details

**Description:** 
As you can see below `maxGeneration` is set to be a certain number means currentGeneration can't be more than that.
```solidity
  uint256 public maxGeneration = 10;
  uint256 public currentGeneration = 1;
```
And `setMaxGeneration` doing the same.
```solidity
  function setMaxGeneration(uint maxGeneration_) external onlyOwner {
    require(
      maxGeneration_ >= currentGeneration,
      "can't below than current generation"
    );
    maxGeneration = maxGeneration_;
  }
```
But `_incrementGeneration()` function is not checking that `currentGeneration` is equal to the `maxGeneration`. And cause of this `currentGeneration` can be more than `maxGeneration`.

**Proof of Concept:**
1. Function mintToken():
```solidity
 function mintToken(
    bytes32[] calldata proof
  )
    public
    payable
    whenNotPaused
    nonReentrant
    onlyWhitelisted(proof, keccak256(abi.encodePacked(msg.sender)))
  {
    uint256 mintPrice = calculateMintPrice();
    require(msg.value >= mintPrice, 'Insufficient ETH send for minting.');

    _mintInternal(msg.sender, mintPrice);

    uint256 excessPayment = msg.value - mintPrice;
    if (excessPayment > 0) {
      (bool refundSuccess, ) = msg.sender.call{ value: excessPayment }('');
      require(refundSuccess, 'Refund of excess payment failed.');
    }
  }
```
2. Function _mintInternal():
```solidity
  function _mintInternal(address to, uint256 mintPrice) internal {
      //@audit-info Not checking if current gen is == max gen and incrementing gen more than max
    if (generationMintCounts[currentGeneration] >= maxTokensPerGen) {
      _incrementGeneration(); //increment gen and make genMintcount 0
    }

    _tokenIds++;
    uint256 newItemId = _tokenIds;
    _mint(to, newItemId);
    uint256 entropyValue = entropyGenerator.getNextEntropy();

    tokenCreationTimestamps[newItemId] = block.timestamp;
    tokenEntropy[newItemId] = entropyValue;
    tokenGenerations[newItemId] = currentGeneration;
    generationMintCounts[currentGeneration]++;
    initialOwners[newItemId] = to;

    if (!airdropContract.airdropStarted()) {
      airdropContract.addUserAmount(to, entropyValue);
    }

    emit Minted(
      msg.sender,
      newItemId,
      currentGeneration,
      entropyValue,
      mintPrice
    );

    _distributeFunds(mintPrice);
  }
```
3. Function _incrementGeneration():
```solidity
  function _incrementGeneration() private {
    require(
      generationMintCounts[currentGeneration] >= maxTokensPerGen,
      'Generation limit not yet reached'
    );
    currentGeneration++;
    generationMintCounts[currentGeneration] = 0;
    priceIncrement = priceIncrement + priceIncrementByGen;//@Question price is incremented by priceIncrementByGen
    entropyGenerator.initializeAlphaIndices();
    emit GenerationIncremented(currentGeneration);
  }
```
4. As we can see above the `mintToken()` is calling `_mintInternal()` function.
```solidity
_mintInternal(msg.sender, mintPrice);
```
5. And `_mintInternal()` function is checking if generationMintCounts[currentGeneration] >= maxTokensPerGen than increment the generation.
```solidity
    if (generationMintCounts[currentGeneration] >= maxTokensPerGen) {
      _incrementGeneration(); //increment gen and make genMintcount 0
    }
```
6. What if `currentGeneration` is `maxGeneration` and generationMintCounts[currentGeneration] is >= maxTokensPerGen.
7. Than we are calling  `_incrementGeneration()` function which is not checking that `currentGeneration` has hit `maxGeneration` before incrementing the generation.
```solidity
    require(
      generationMintCounts[currentGeneration] >= maxTokensPerGen,
      'Generation limit not yet reached'
    );
    currentGeneration++;
```
8. And due to all this  `currentGeneration` will become more than defined `maxGeneration` which will brake the invariant.

**Impact:** 
The impact of this vulnerability should be medium because it is breaking the invariant.

**Recommended Mitigation:** 
The recommendation is to add a check in `_incrementGeneration()` to check that `currentGeneration` is lesser than `maxGeneration` before incrementing it.

```solidity
  function _incrementGeneration() private {
    require(
      generationMintCounts[currentGeneration] >= maxTokensPerGen,
      'Generation limit not yet reached'
    );
    require(currentGeneration < maxGeneration, "Generation is not lesser than maximum Generation ")
    currentGeneration++;
    generationMintCounts[currentGeneration] = 0;
    priceIncrement = priceIncrement + priceIncrementByGen;//@Question price is incremented by priceIncrementByGen
    entropyGenerator.initializeAlphaIndices();
    emit GenerationIncremented(currentGeneration);
  }
```
