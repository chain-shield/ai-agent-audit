# Vulnerability details



## Impact
After every NFT purchase , the function  [ EntityTrading.sol:buyNFT( )](https://github.com/code-423n4/2024-07-traitforge/blob/279b2887e3d38bc219a05d332cbcb0655b2dc644/contracts/EntityTrading/EntityTrading.sol#L63)  is espected to remove  a listing  ` delete listings[listedTokenIds[tokenId]];` . However listings are not deleted  leading to repurchase of same item . An attacker can leverage this to cheat and steal user funds.



## Proof of Concept
```solidity
import { ethers } from 'hardhat';
import { EntityTrading, TestERC721 } from '../typechain-types';
import { HardhatEthersSigner } from '@nomicfoundation/hardhat-ethers/signers';
import { expect } from 'chai';

// Define some constants for testing
const TOKEN_ID = 0;
const LISTING_PRICE = ethers.parseEther('1.0');

describe('EntityTrading: List & Buy', function () {
  let seller: HardhatEthersSigner;
  let buyer: HardhatEthersSigner;
  let nft: TestERC721;
  let entityTrading: EntityTrading;
  
 
  before(async function () {
    
    [seller, buyer] = await ethers.getSigners();

    nft = await ethers.deployContract('TestERC721');
    entityTrading = await ethers.deployContract('EntityTrading', [
      await nft.getAddress(),
    ]);
});



it('check if listing is cleared after purchase ', async function () {
    // Mint NFT
    await nft
    .connect(seller)
    .mintToken(seller.address);
    // Aprove  Contract to list NFT
    await nft
    .connect(seller)
    .approve(await entityTrading.getAddress(), TOKEN_ID);
// Before Listing 

const listingId = await entityTrading.connect(seller).listedTokenIds(TOKEN_ID); 
const listing = await entityTrading.connect(seller).listings(listingId);

   // Initial Listing State  is false 
    expect(listing.isActive).to.be.false;
    console.log("List isActive before listing ?:", listing.isActive);
 
//List Minted Tokens for sale 
    await entityTrading.connect(seller).listNFTForSale(TOKEN_ID , LISTING_PRICE);
    const after_listingId = await entityTrading.connect(seller).listedTokenIds(TOKEN_ID);
    const after_listing = await entityTrading.connect(seller).listings(after_listingId);
    console.log("List isActive after listing ?:", after_listing.isActive);
    expect(after_listing.seller).to.equal(seller.address);
    expect(after_listing.price).to.equal(LISTING_PRICE);
    
    // Listing State after is true after listing 
    expect(after_listing.isActive).to.be.true;

// Buy the nft with any amount specifying the token ID
    entityTrading.connect(buyer).buyNFT(TOKEN_ID, { value: LISTING_PRICE })

    const after_buy_listingId = await entityTrading.connect(buyer).listedTokenIds(0);
    const after_buy_listing = await entityTrading.connect(buyer).listings(after_buy_listingId);
   // Listing is not cleared and it appears to be active  after purchase 
  
// Returns true
    console.log("List isActive after purchasing  ?:", after_buy_listing.isActive);
 
   });

})

```
run command 
```console
npx hardhat test --grep 'check if listing is cleared after purchase'
```

As you can see `after_buy_listing.isActive` returns true instead of false
