# Vulnerability details

## Impact
`EntityForging::forgeWithListed` function lets mergers send more `ETH` to the contract than the `forgingFee` they intend to pay. These extra funds are not returned to the merger but are kept in the contract which has no method for withdrawing `ETH`. So funds will be stuck in the contract forever.


## Proof of Concept

Foundry test : 
```
// SPDX-License-Identifier: MIT
pragma solidity ^0.8.20;

import { Test, console } from 'forge-std/Test.sol';
import { EntropyGenerator } from '../contracts/EntropyGenerator/EntropyGenerator.sol';
import { DevFund } from '../contracts/DevFund/DevFund.sol';
import { Airdrop } from '../contracts/Airdrop/Airdrop.sol';
import { TraitForgeNft } from '../contracts/TraitForgeNft/TraitForgeNft.sol';
import { Trait } from '../contracts/Trait/Trait.sol';
import { EntityForging } from '../contracts/EntityForging/EntityForging.sol';
import { EntityTrading } from '../contracts/EntityTrading/EntityTrading.sol';
import { NukeFund } from '../contracts/NukeFund/NukeFund.sol';

contract BaseTest is Test {
  address public owner = makeAddr('owner');

  EntropyGenerator public entropyGenerator;
  DevFund public devFund;
  Airdrop public airdrop;
  TraitForgeNft public traitForgeNft;
  EntityForging public entityForging;
  EntityTrading public entityTrading;
  NukeFund public nukeFund;
  address public user = makeAddr('user');
  bytes32[] public proof;

  function setUp() public {
    vm.startPrank(owner);
    devFund = new DevFund();
    airdrop = new Airdrop();
    traitForgeNft = new TraitForgeNft();
    entropyGenerator = new EntropyGenerator(address(traitForgeNft));
    entityForging = new EntityForging(address(traitForgeNft));
    entityTrading = new EntityTrading(address(traitForgeNft));
    address devAddress = makeAddr('devAddress');
    address daoAddress = makeAddr('daoAddress');
    nukeFund = new NukeFund(
      address(traitForgeNft),
      address(airdrop),
      payable(devAddress),
      payable(daoAddress)
    );
    vm.stopPrank();
  }

  function test_refundInForgingEntity() public {
    //Set up
    vm.startPrank(traitForgeNft.owner());
    traitForgeNft.setEntityForgingContract(address(entityForging));
    traitForgeNft.setEntropyGenerator(address(entropyGenerator));
    traitForgeNft.setAirdropContract(address(airdrop));
    vm.stopPrank();
    //Generate entropies
    entropyGenerator.writeEntropyBatch1();

    proof = new bytes32[](1);
    proof[0] = keccak256(abi.encodePacked(user));
    vm.warp(3 days);
    vm.deal(user, 10 ether);
    //user mint some nft
    vm.startPrank(user);
    traitForgeNft.mintToken{ value: 0.5 ether }(proof);
    traitForgeNft.mintToken{ value: 0.5 ether }(proof);
    traitForgeNft.mintToken{ value: 0.5 ether }(proof);
    traitForgeNft.mintToken{ value: 0.5 ether }(proof);
    vm.stopPrank();
    uint256 forgingFee = 0.02 ether;
    proof = new bytes32[](1);
    address forger = makeAddr('forger');
    proof[0] = keccak256(abi.encodePacked(forger));
    vm.warp(3 days);
    vm.deal(forger, 1 ether);
    //forger mint nft then list it for forging
    vm.startPrank(forger);
    traitForgeNft.mintToken{ value: 0.5 ether }(proof);
    entityForging.listForForging(5, forgingFee);
    //check if tokenId 5 is listed
    assertEq(
      entityForging.getListings(entityForging.getListedTokenIds(5)).isListed,
      true
    );
    vm.stopPrank();

    vm.startPrank(user);
    uint256 userBalanceBefore = user.balance;
    uint256 contractBalanceBefore = address(entityForging).balance;
    // user put all its balance
    entityForging.forgeWithListed{ value: userBalanceBefore }(5, 2);
    uint256 userBalanceAfter = user.balance;
    uint256 contractBalanceAfter = address(entityForging).balance;
    //user's balance is 0
    assertEq(userBalanceAfter, 0);
    // entityForging receives all the funds
    assertEq(
      contractBalanceAfter,
      contractBalanceBefore + userBalanceBefore - forgingFee
    );
    vm.stopPrank();
  }
}
```
This test shows that funds are sent to the contract, but the extra funds are not sent back to the merger. Since the `EntityForging` contract has no way to withdraw ETH, the funds will be stuck in it.

## Tools Used
Foundry

## Recommended Mitigation Steps
Change the requirement to be equal to forgingFee
```diff
- require(msg.value >= forgingFee, 'Insufficient fee for forging');
+ require(msg.value = forgingFee, 'Not correct fee for forging');
```
or implement a mechanism to refund the overpayment.
