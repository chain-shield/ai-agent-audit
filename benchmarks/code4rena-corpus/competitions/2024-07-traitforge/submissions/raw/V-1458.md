# Vulnerability details

## Impact
Sending fund to DevFund contract will always fail if the owner is a contract and always reverts upon receiving ether. 

## Proof of Concept
contract DevFund is IDevFund, Ownable, ReentrancyGuard, Pausable {
    uint256 public totalDevWeight;
    uint256 public totalRewardDebt;
    mapping(address => DevInfo) public devInfo; 
    uint public ownerWithdrawalAmount;

    receive() external payable {
        if (totalDevWeight > 0) {
            uint256 amountPerWeight = msg.value / totalDevWeight;   // maintains equal fund distribution 
            uint256 remaining = msg.value - (amountPerWeight * totalDevWeight);
            totalRewardDebt += amountPerWeight;
            if (remaining > 0) {
                // (bool success, ) = payable(owner()).call{ value: remaining }('');
                // require(success, 'Failed to send Ether to owner');
                ownerWithdrawalAmount = ownerWithdrawalAmount + remaining;
            }
        } else {
            // (bool success, ) = payable(owner()).call{ value: msg.value }('');
            // require(success, 'Failed to send Ether to owner');
            ownerWithdrawalAmount = ownerWithdrawalAmount + remaining;
        }

        emit FundReceived(msg.sender, msg.value);
    }

    function withdrawFund() external nonReentrant onlyOwner {
        require(ownerWithdrawalAmount > 0, 'not enough fund');
        (bool success, ) = msg.sender.call{ value: ownerWithdrawalAmount }('');
        require(success, 'Failed to send Ether to owner');
        ownerWithdrawalAmount = 0;
    }
}
