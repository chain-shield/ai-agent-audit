# Vulnerability details

## Impact
When an incorrect dev address is added to the dev fund there is no way to recover the funds that have been assigned to it.

## Proof of Concept
This test can be added at the end of the DevFund.test.ts file:
```javascript 
describe('DevFund Custom', function () {
  let owner: HardhatEthersSigner;
  let user1: HardhatEthersSigner;
  let user2: HardhatEthersSigner;
  let user3: HardhatEthersSigner;
  let devFund: DevFund;

  before(async () => {
    [owner, user1, user2, user3] = await ethers.getSigners();

    devFund = await ethers.deployContract('DevFund');
  });

  it('should add a dev successfully', async () => {
    await expect(devFund.addDev(user1.address, 5))
      .to.emit(devFund, 'AddDev')
      .withArgs(user1.address, 5);

    await expect(devFund.addDev(user2.address, 5))
      .to.emit(devFund, 'AddDev')
      .withArgs(user2.address, 5);
  });

  it('after removing dev funds are stuck', async () => {
    await expect(
      owner.sendTransaction({
        to: await devFund.getAddress(),
        value: ethers.parseEther('1'),
      })
    )
      .to.emit(devFund, 'FundReceived')
      .withArgs(await owner.getAddress(), ethers.parseEther('1'));

    // incorrectly added user1 is removed
    await devFund.removeDev(user1.address);

    // user2 claims it's share of the rewards
    await expect(devFund.connect(user2).claim())
    .to.emit(devFund, 'Claim')
    .withArgs(user2.address, ethers.parseEther('0.5'));

    // another user is added with the same weight
    await devFund.addDev(user3.address, 5);
    
    // This line fails because user3 pending claim equals zero. We now have 0.5 eth stuck in the contract
    await expect(devFund.connect(user3).claim())
      .to.emit(devFund, 'Claim')
      .withArgs(user3.address, ethers.parseEther('0.5'));
  });
})
```

## Tools Used
Manual review and hardhat tests.

## Recommended Mitigation Steps
Add a withdraw function to remove any funds that are left in the contract, this function must also update the ```totalRewardDebt``` value.
