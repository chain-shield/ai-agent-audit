# Vulnerability details

### [H-02] Increment Generation Resets Mint Count Incorrectly Breaking Price Calc And Mint Counts Tracking

**Description:**
After all 10_000 NFTs for the current generation are minted, **_mintInternal()** function will trigger
**_incrementGeneration()**.
(Note: **H-01** shows that  **_incrementGeneration()** cannot ever be invoked because **initializeAlphaIndices** cannot be called when **TraitForgeNft** is the msg.sender**.)
Still, even if **H-01** is fixed and **_incrementGeneration()** is successfully triggered, it would first increase **currentGeneration**
and then immediately on the next execution line it would reset **generationMintCounts[currentGeneration] to 0**.
This would mean that if we already have several NFTs minted from any next generation (that now after **_incrementGeneration()** becomes the **currentGeneration**) using the forge method,
we would lose their track inside the **generationMintCounts** mapping.

**Impact:** Since we have already minted some NFTs from the next generation through forging, now that we reset their **generationMintCounts** after **_incrementGeneration()**,
this will affect price calculation as it uses exactly the same mapping struct to calculate number of NFTs minted per gen. 
What is more, after the minting of all the NFTs for the now new generation is done, we would have more than 10_000 minted of them.


**Proof of Concept:**
1. Comment out the line **entropyGenerator.initializeAlphaIndices();** inside **_incrementGeneration()** (refer to the above note regarding H-01)
2. Comment **addUserAmount** in Airdrop contract as it uses onlyOwner
3.Configure foundry & make a foundry test file
4.Paste this test contract inside the file & run

```solidity
// SPDX-License-Identifier: UNLICENSED
pragma solidity ^0.8.20;

import {Test, console} from "forge-std/Test.sol";
import {EntropyGenerator} from "../../contracts/EntropyGenerator/EntropyGenerator.sol";
import {TraitForgeNft} from "../../contracts/TraitForgeNft/TraitForgeNft.sol";
import {Airdrop} from "../../contracts/Airdrop/Airdrop.sol";
import {EntityForging} from "../../contracts/EntityForging/EntityForging.sol";
import {NukeFund} from "../../contracts/NukeFund/NukeFund.sol";

contract WrongGenerationResetTest is Test {

    address public owner = makeAddr("owner");

    address public alice = makeAddr("alice");
    address public bob = makeAddr("bob");

    EntropyGenerator public entropyGenerator;
    Airdrop public airdrop;
    TraitForgeNft public traitForgeNft;
    EntityForging public entityForging;
    NukeFund public nukeFund;

    function setUp() public {
        vm.deal(alice, 10_000 ether);
        vm.deal(bob, 10_000 ether);
        vm.startPrank(owner);
        traitForgeNft = new TraitForgeNft();
        entropyGenerator = new EntropyGenerator(address(traitForgeNft));
        airdrop = new Airdrop();
        entityForging = new EntityForging(address(traitForgeNft));
        traitForgeNft.setEntityForgingContract(address(entityForging));
        traitForgeNft.setEntropyGenerator(address(entropyGenerator));
        traitForgeNft.setAirdropContract(address(airdrop));
        nukeFund = new NukeFund(address(traitForgeNft), payable(address(airdrop)), payable(alice), payable(owner));
        traitForgeNft.setNukeFundContract(payable(address(nukeFund)));
        nukeFund.setAgeMultplier(1);
        assertEq(nukeFund.getAgeMultiplier(), 1);
        vm.stopPrank();
    }

    function test_writeBatches() public {
        entropyGenerator.writeEntropyBatch1();
        entropyGenerator.writeEntropyBatch2();
        entropyGenerator.writeEntropyBatch3();
    }

    function test_poc_generation_mint_reset_causes_wrong_price_calc() public {
        test_writeBatches();

        vm.startPrank(alice);
        for(uint i = 0; i < 3; ++i) {
            traitForgeNft.mintToken{value: 1e18}();
        }
        vm.stopPrank();
        assertFalse(traitForgeNft.isForger(2));

        vm.startPrank(bob);
        for(uint i = 0; i < 10; ++i) {
            traitForgeNft.mintToken{value: 1e18}();
        }

        assertTrue(traitForgeNft.isForger(5));
        assertTrue(traitForgeNft.isForger(9));

        entityForging.listForForging(5, 1e16); // bob lists his tokenId = 5 for forging
        entityForging.listForForging(9, 1e16); // bob lists his tokenId = 9 for forging
        vm.stopPrank();

        vm.startPrank(alice);
        entityForging.forgeWithListed{value:1e16}(5, 2); // alice's tokenId = 2 forges with bob's 5
        uint256 newTokenIdMinted = 14;
        uint256 newTokenIdMintedGen = traitForgeNft.getTokenGeneration(newTokenIdMinted);
        assertEq(traitForgeNft.currentGeneration(), 1); // current generation is 1
        assertEq(newTokenIdMintedGen, 2); // assert newly minted token is from gen 2 (next gen)

        entityForging.forgeWithListed{value:1e16}(9, 3); // alice's tokenId = 3 forges with bob's 9
        newTokenIdMinted = 15;
        uint256 token15Gen = traitForgeNft.getTokenGeneration(newTokenIdMinted);
        assertEq(token15Gen, 2); // assert newly minted token is from gen 2 (next gen)
        vm.stopPrank();
        // Currently, we have 2 tokens from next gen minted through forging
        // let's mint the rest of the tokens from gen 1
        vm.startPrank(alice);
        for(uint i = 0; i < 9987; ++i) {
            traitForgeNft.mintToken{value: 1e18}();
        }
        uint256 gen1MintCounts = traitForgeNft.generationMintCounts(1);
        assertEq(gen1MintCounts, 10_000);

        uint256 gen2MintCounts = traitForgeNft.generationMintCounts(2);
        assertEq(gen2MintCounts, 2);

        // let's now mint 1 more token that will trigger _incrementGeneration()
        traitForgeNft.mintToken{value: 1e18}();
        vm.stopPrank();

        gen1MintCounts = traitForgeNft.generationMintCounts(1);
        assertEq(gen1MintCounts, 10000); // old generation count is Ok

        // gen 2 count gets reset to 1 even though we already have minted 2 tokens through forging!
        uint256 gen2MintCountsAfter = traitForgeNft.generationMintCounts(2);
        assertEq(gen2MintCountsAfter, 1);

        uint256 currentMintPrice = traitForgeNft.calculateMintPrice();
        uint256 expectedMintPrice = traitForgeNft.priceIncrement() * 2 + traitForgeNft.startPrice();
        assertLt(currentMintPrice, expectedMintPrice); // current mint price is account for 1 minted token instead of 2
    }
}
```

**Recommended Mitigation:** Prior to incrementing the generation count, fetch the already minted tokens from the next generation
and add them to be the starting value of the currentGeneration count:
```solidity
    function _incrementGeneration() private {
        require(
        generationMintCounts[currentGeneration] >= maxTokensPerGen,
        'Generation limit not yet reached'
        );
        uint256 nextGenCount = generationMintCounts[currentGeneration+1];
        currentGeneration++;
        generationMintCounts[currentGeneration] = 0 + nextGenCount;
        priceIncrement = priceIncrement + priceIncrementByGen;
        entropyGenerator.initializeAlphaIndices();
        emit GenerationIncremented(currentGeneration);
    }
```
