# Vulnerability details

## Impact
There is a chance that the function `EntropyGenerator::getNextEntropy` returns the value `999999` when it is not supposed to do so, that is, when `slotIndex` and `numberIndex` do not equal `slotIndexSelectionPoint` and `numberIndexSelectionPoint` respectively.


## Proof of Concept
There is an estimated probability greater than 0.08% (1 in 125) that the pseudo-random values set during deployment in the entropy slots result in the function `EntropyGenerator::getNextEntropy` returning entropy of `999999` when the slotIndex and numberIndex are not what is expected.

This happens because the generation of entropy slots, and the entropy value calculated in [line 181](https://github.com/code-423n4/2024-07-traitforge/blob/main/contracts/EntropyGenerator/EntropyGenerator.sol#L181) do not ensure that the final value is not `999999`.

For example, with the following values the entropy generated will be `999999` in the second entropy generation: 
```solidity
numberIndex = 1;
slotIndex = 0;
entropySlots[0] = 999999 * 10**66;
```

These values are possible, because during the entropy slots initialization there is a check to ensure the values are not equal to `999999`, but it is possible to get `999999` after the division in line 181 with values different than `999999`.

The following foundry test uses the previous values to get the desired entropy:
```solidity
pragma solidity ^0.8.13;

import {VmSafe} from "forge-std/Vm.sol";
import {Test, console} from "forge-std/Test.sol";
import {EntropyGenerator} from "../contracts/EntropyGenerator/EntropyGenerator.sol";

contract EntropyGeneratorTest is Test {
    function test_999999EntropyGeneratedWithTwoAttempts() public {
        EntropyGenerator generator = new EntropyGenerator(address(this));
        generator.writeEntropyBatch1();
        generator.writeEntropyBatch2();
        generator.writeEntropyBatch3();
        generator.initializeAlphaIndices();

        bytes32 slot = bytes32(uint256(1)); // entropySlot[0]
        bytes32 value = bytes32(uint256(999999 * 10**66));

        // store value `999999 * 10**66` in entropySlot[0]
        vm.store(address(generator), slot, value); 

        generator.getNextEntropy(); // numberIndex = 0
        uint256 entropy = generator.getNextEntropy(); // numberIndex = 1

        assertEq(entropy, 999999);
    }
}
```

The following test shows how this could happen in a real scenario. If the deployment is made in block `20679985` (1 month from now approximately), then the 2144 minted token will have an entropy of `999999`:

```solidity
pragma solidity ^0.8.13;

import {VmSafe} from "forge-std/Vm.sol";
import {Test, console} from "forge-std/Test.sol";
import {EntropyGenerator} from "../contracts/EntropyGenerator/EntropyGenerator.sol";

contract EntropyGeneratorTest is Test {
    function test_got999999EntropyUsingSlotsFromFirstBatch() public {
        // block approximately 1 month from now
        // 20679985 = 20463985 + 30*24*60*60/12
        vm.roll(20679985);

        EntropyGenerator generator = new EntropyGenerator(address(this));
        generator.writeEntropyBatch1();
        generator.writeEntropyBatch2();
        generator.writeEntropyBatch3();
        generator.initializeAlphaIndices();

        // this mint uses entropy slots from the first batch
        // 2144 < 13*256
        uint256 goldenMint = 2144;

        for (uint256 i = 0; i < goldenMint; ++i) {
            generator.getNextEntropy();
        }

        uint256 entropy = generator.getNextEntropy();
        assertEq(entropy, 999999);
    }
}
```

Finally, the probability of this happening was estimated to be between 0.08% (1 in 125) and 0.1% (1 in 100). The estimation was made running the following code with different block numbers:

```solidity
pragma solidity ^0.8.13;

import {VmSafe} from "forge-std/Vm.sol";
import {Test, console} from "forge-std/Test.sol";
import {EntropyGenerator} from "../contracts/EntropyGenerator/EntropyGenerator.sol";

contract EntropyGeneratorTest is Test {
    function test_simulation() public {
        uint256 blockNumber = 1;
        uint256 samples = 3000;

        uint256 goldenCount = 0;

        for (uint256 i = blockNumber; i < blockNumber + samples; ++i) {
            vm.roll(i);

            EntropyGenerator generator = new EntropyGenerator(address(this));
            generator.writeEntropyBatch1();
            generator.writeEntropyBatch2();
            generator.writeEntropyBatch3();
            generator.initializeAlphaIndices();

            bool done;
            for (uint256 slotNumber = 0; slotNumber < 770; slotNumber++) {
                if (done) { break; }

                uint256 slotValue = uint256(vm.load(address(generator), bytes32(slotNumber)));

                for (uint256 numberIndex = 0; numberIndex < 13; numberIndex++) {
                    uint256 position = numberIndex * 6;
                    uint256 entropy = (slotValue / (10 ** (72 - position))) % 1000000;

                    if (entropy == 999999) {
                        ++goldenCount;
                        done = true;
                        break;
                    }
                }
            }
        }

        console.log("Probability ~ %d / 3000", goldenCount);
    }
}
```
