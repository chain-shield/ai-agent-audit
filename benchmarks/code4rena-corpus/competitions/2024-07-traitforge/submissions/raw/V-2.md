# Vulnerability details

Checking the `EntropyGenerator` contract it shows that during deployment the `traitForgeNft` contract is set as an allowedCaller, but the method `initializeAlphaIndices` has the onlyOwner modifier. 

When a NFT is minted there is a check if the generation limit is reached, and when the limit is reached it would increment to the next generation. However the line

`entropyGenerator.initializeAlphaIndices();` would always revert because it uses the wrong modifier.

This results in that the `traitForgeNft` contract can’t increment generations and when this cap is reached the contract will always revert.

```
modifier onlyAllowedCaller() {
    require(msg.sender == allowedCaller, 'Caller is not allowed');
    _;
  }
  
  constructor(address _traitForgetNft) {
    allowedCaller = _traitForgetNft;
    initializeAlphaIndices();
  }
  
  // must be onlyAllowedCaller modifier
function initializeAlphaIndices() public whenNotPaused onlyOwner {
    uint256 hashValue = uint256(
      keccak256(abi.encodePacked(blockhash(block.number - 1), block.timestamp))
    );

    uint256 slotIndexSelection = (hashValue % 258) + 512;
    uint256 numberIndexSelection = hashValue % 13;

    slotIndexSelectionPoint = slotIndexSelection;
    numberIndexSelectionPoint = numberIndexSelection;
}
```
## Proof of Concept

For this POC we’ve set the `maxTokensPerGen` to 1 instead of 10K so that we simulate what happens when the NFT must increment the generations and the limit is reached. 

It will reverts because the `onlyOwner` modifier is used instead of the `onlyAllowedCaller` 

```tsx
import { HardhatEthersSigner } from '@nomicfoundation/hardhat-ethers/signers';
import { expect } from 'chai';
import { ethers } from 'hardhat';
import {
  Trait,
  Airdrop,
  DevFund,
  EntityForging,
  EntropyGenerator,
  EntityTrading,
  NukeFund,
  TraitForgeNft,
} from '../typechain-types';

import generateMerkleTree from '../scripts/genMerkleTreeLib';

describe('POC:', function () {
  let token: Trait;

  let entityForging: EntityForging;
  let nft: TraitForgeNft;
  let owner: HardhatEthersSigner;
  let user1: HardhatEthersSigner;
  let user2: HardhatEthersSigner;

  let merkleInfo: any;

  before(async () => {
    [owner, user1, user2] = await ethers.getSigners();
    // Deploy TraitForgeNft contract
    const TraitForgeNft = await ethers.getContractFactory('TraitForgeNft');
    nft = (await TraitForgeNft.deploy()) as TraitForgeNft;

    // Deploy Airdrop contract
    const airdropFactory = await ethers.getContractFactory('Airdrop');
    const airdrop = (await airdropFactory.deploy()) as Airdrop;

    await nft.setAirdropContract(await airdrop.getAddress());
    // TraitForgeNft will be owner of Airdrop Contract
    await airdrop.transferOwnership(await nft.getAddress());

    // Deploy EntityForging contract
    const EntropyGenerator = await ethers.getContractFactory(
      'EntropyGenerator'
    );
    const entropyGenerator = (await EntropyGenerator.deploy(
      await nft.getAddress()
    )) as EntropyGenerator;

    await entropyGenerator.writeEntropyBatch1();
    await nft.setEntropyGenerator(await entropyGenerator.getAddress());

    // Deploy EntityForging contract
    const EntityForging = await ethers.getContractFactory('EntityForging');
    entityForging = (await EntityForging.deploy(
      await nft.getAddress()
    )) as EntityForging;
    await nft.setEntityForgingContract(await entityForging.getAddress());
  });

  it('should demonstrate TraitForgeNft cant increment generation', async() => {
      // for this poc we've set the maxTokensPerGen to 1 instead of 10k
      // in the TraitForgeNft.sol 
      // uint256 public maxTokensPerGen = 1; // 10000
      merkleInfo = generateMerkleTree([
        owner.address,
        user1.address,
        user2.address,
      ]);

      await nft.setRootHash(merkleInfo.rootHash);

      await nft.connect(user1).mintToken(merkleInfo.whitelist[1].proof, {
        value: ethers.parseEther('1'),
      });
      expect(await nft.balanceOf(user1.address)).to.be.eq(1);
      
      // reverts due to wrong modifier
      await expect(
        nft.connect(user2).mintToken(merkleInfo.whitelist[2].proof, {
          value: ethers.parseEther('1'),
        })
      ).to.be.rejectedWith('Ownable: caller is not the owner'); 
  });
});

```

## Recommended mitigation steps

Make changes to the onlyAllowedCaller modifier that the contract and owner has permissions to call the function.

```diff
  modifier onlyAllowedCaller() {
-  require(msg.sender == allowedCaller, 'Caller is not allowed');
+  require((msg.sender == allowedCaller) || (msg.sender == owner()), 'Caller is not allowed');
    _;
  }
  
-function initializeAlphaIndices() public whenNotPaused onlyOwner {
+function initializeAlphaIndices() public whenNotPaused onlyAllowedCaller {
    uint256 hashValue = uint256(
      keccak256(abi.encodePacked(blockhash(block.number - 1), block.timestamp))
    );

    uint256 slotIndexSelection = (hashValue % 258) + 512;
    uint256 numberIndexSelection = hashValue % 13;

    slotIndexSelectionPoint = slotIndexSelection;
    numberIndexSelectionPoint = numberIndexSelection;
  }
```
