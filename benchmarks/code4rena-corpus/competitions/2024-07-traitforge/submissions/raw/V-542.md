# Vulnerability details

## Impact
High as when entropy is 0, NFT's minted become useless and users will basically have useless NFT's that they cannot use in the protocol, or nuke to return some of the spent funds.
## Proof of Concept
I've made a new fresh testing env. in foundry, in order to use it:
1. `foundry init` in an empty folder.
2. Copy all contracts from `contracts` folder in to `src` folder of foundry
3. `forge install openzeppelin/openzeppelin-contracts@v4.9.3 --no-commit` to have same version of OZ as used in the protocol.
4. Make a test file in `test` folder.
5. Paste code:
```
// SPDX-License-Identifier: UNLICENSED
pragma solidity ^0.8.13;

import {Test, console} from "forge-std/Test.sol";
import {Counter} from "../src/Counter.sol";
import {TraitForgeNft} from "../src/TraitForgeNft/TraitForgeNft.sol";
import {EntityForging} from "../src/EntityForging/EntityForging.sol";
import {EntropyGenerator} from "../src/EntropyGenerator/EntropyGenerator.sol";
import {EntityTrading} from "../src/EntityTrading/EntityTrading.sol";
import {Airdrop} from "../src/Airdrop/Airdrop.sol";
import {NukeFund} from "../src/NukeFund/NukeFund.sol";
import {Trait} from "../src/Trait/Trait.sol";
import {DAOFund} from "../src/DAOFund/DAOFund.sol";
import {DevFund} from "../src/DevFund/DevFund.sol";

contract MyTest is Test {
    TraitForgeNft public traitForgeNft;
    EntityForging public entityForging;
    EntropyGenerator public entropyGenerator;
    EntityTrading public entityTrading;
    Airdrop public airdrop;
    NukeFund public nukeFund;
    Trait public traitToken;
    DAOFund public dAOFund;
    DevFund public devFund;
    address public router = makeAddr("router");
    address public minter = makeAddr("1");
    address public owner = makeAddr("owner");
    bytes32 public root = 0xd4dee0beab2d53f2cc83e567171bd2820e49898130a22622b10ead383e90bd77;
    bytes32 public proof = 0xcb3b69276e6ab36698e5ef9b8544b8cdeaeca28720ac727a7791aab800174591;

    function setUp() public {
        traitForgeNft = new TraitForgeNft();

        entropyGenerator = new EntropyGenerator(address(traitForgeNft));
        airdrop = new Airdrop();
        traitToken = new Trait("Trait", "TRT", 18, 1_000_000e18);
        airdrop.setTraitToken(address(traitToken));
        dAOFund = new DAOFund(address(traitToken), address(router));
        devFund = new DevFund();
        nukeFund = new NukeFund(address(traitForgeNft), address(airdrop), payable(devFund), payable(dAOFund));
        entityForging = new EntityForging(address(traitForgeNft));
        entityForging.setNukeFundAddress(payable(nukeFund));
        entityTrading = new EntityTrading(address(traitForgeNft));
        entityTrading.setNukeFundAddress(payable(nukeFund));

        traitForgeNft.setNukeFundContract(payable(nukeFund));
        traitForgeNft.setEntityForgingContract(address(entityForging));
        traitForgeNft.setEntropyGenerator(address(entropyGenerator));
        traitForgeNft.setAirdropContract(address(airdrop));
        traitForgeNft.setRootHash(root);
        vm.deal(minter, 1 ether);
        nukeFund.transferOwnership(owner);
    }

    function test_EntropyIsZeroWhenBatchNotEnabled() public {
        traitToken.transfer(0x1804c8AB1F12E6bbf3894d4083f33e07309d1f38, 5_000e18);

        vm.prank(0x1804c8AB1F12E6bbf3894d4083f33e07309d1f38);
        traitToken.approve(address(airdrop), 5_000e18);
        console.log(address(this));
        airdrop.startAirdrop(1e18);
        vm.startPrank(minter);
        uint256 mintPrice = traitForgeNft.calculateMintPrice();
        bytes32[] memory proofSend = new bytes32[](1);
        proofSend[0] = proof;
        vm.roll(36 * 60 * 6);
        vm.warp(36 hours);
        traitForgeNft.mintToken{value: mintPrice}(proofSend);
        uint256 creationTimestamp = traitForgeNft.getTokenCreationTimestamp(1);
        console.log(creationTimestamp);
        uint256 entropy = traitForgeNft.getTokenEntropy(1);
        console.log(entropy);
        assertEq(entropy,0);
        vm.stopPrank();
    }
}
``` 

## Tools Used
Manual review, foundry
## Recommended Mitigation Steps
Always make sure that Entropy Batches are initialized, in order to protect the first minters of NFT's.
To make it even further safe, a good practise will be to `writeEntropyBatch` in the `constructor` of the EntropyGenerator contract.
