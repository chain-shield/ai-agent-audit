# Vulnerability details

## Impact
`EntropyGenerator::getNextEntropy` function is called while minting a new nft to give to the nft a certain entropy value, but the function does not have a check that `entropySlots` is not empty. 

If nobody calls `writeEntropyBatch1`, `writeEntropyBatch2`, and `writeEntropyBatch3` functions, the array `entropySlots` will remain empty, thus the nfts will have 0 as a value of the entropy, which means forger nft, but infertile nft.

So all minters of nft without batch initialization will have forger infertile nft.

## Proof of Concept
Test with foundry: 
```
// SPDX-License-Identifier: MIT
pragma solidity ^0.8.20;

import { Test, console } from 'forge-std/Test.sol';
import { EntropyGenerator } from '../contracts/EntropyGenerator/EntropyGenerator.sol';
import { DevFund } from '../contracts/DevFund/DevFund.sol';
import { Airdrop } from '../contracts/Airdrop/Airdrop.sol';
import { TraitForgeNft } from '../contracts/TraitForgeNft/TraitForgeNft.sol';
import { Trait } from '../contracts/Trait/Trait.sol';
import { EntityForging } from '../contracts/EntityForging/EntityForging.sol';
import { EntityTrading } from '../contracts/EntityTrading/EntityTrading.sol';
import { NukeFund } from '../contracts/NukeFund/NukeFund.sol';

contract BaseTest is Test {
  address public owner = makeAddr('owner');

  EntropyGenerator public entropyGenerator;
  DevFund public devFund;
  Airdrop public airdrop;
  TraitForgeNft public traitForgeNft;
  EntityForging public entityForging;
  EntityTrading public entityTrading;
  NukeFund public nukeFund;
  address public user = makeAddr('user');
  bytes32[] public proof;

  function setUp() public {
    vm.startPrank(owner);
    devFund = new DevFund();
    traitForgeNft = new TraitForgeNft();
    entropyGenerator = new EntropyGenerator(address(traitForgeNft));
    entityForging = new EntityForging(address(traitForgeNft));
    entityTrading = new EntityTrading(address(traitForgeNft));
    address devAddress = makeAddr('devAddress');
    address daoAddress = makeAddr('daoAddress');
    nukeFund = new NukeFund(
      address(traitForgeNft),
      address(airdrop),
      payable(devAddress),
      payable(daoAddress)
    );
    vm.stopPrank();
    vm.startPrank(address(traitForgeNft));
    airdrop = new Airdrop();
    vm.stopPrank();
  }

function test_mintWithoutBatchInit() public {
    //Set up
    vm.startPrank(traitForgeNft.owner());
    traitForgeNft.setEntityForgingContract(address(entityForging));
    traitForgeNft.setEntropyGenerator(address(entropyGenerator));
    traitForgeNft.setAirdropContract(address(airdrop));
    vm.stopPrank();

    proof = new bytes32[](1);
    proof[0] = keccak256(abi.encodePacked(user));
    //skip whitelist
    vm.warp(3 days);
    vm.deal(user, 10 ether);
    //user mint some nft
    vm.startPrank(user);
    traitForgeNft.mintToken{ value: 0.5 ether }(proof);
    traitForgeNft.mintToken{ value: 0.5 ether }(proof);
    traitForgeNft.mintToken{ value: 0.5 ether }(proof);
    vm.stopPrank();
    //make sure that owner of token id 1, 2, 3 is user
    assertEq(traitForgeNft.ownerOf(1), user);
    assertEq(traitForgeNft.ownerOf(2), user);
    assertEq(traitForgeNft.ownerOf(3), user);
    //entropies generated are 0
    assertEq(traitForgeNft.getTokenEntropy(1), 0);
    assertEq(traitForgeNft.getTokenEntropy(2), 0);
    assertEq(traitForgeNft.getTokenEntropy(3), 0);

    assertEq(traitForgeNft.isForger(1), true);
    assertEq(traitForgeNft.isForger(2), true);
    assertEq(traitForgeNft.isForger(3), true);

    //Extract the 5th digit from the entropy, to know the forge potential
    uint8 forgePotential = uint8((traitForgeNft.getTokenEntropy(1) / 10) % 10);
    // 0 means infertile
    assertEq(forgePotential, 0);

    //Entropy with initialize
    entropyGenerator.writeEntropyBatch1();
    // new user mint after batch entropy
    address newUser = makeAddr('newUser');
    vm.deal(newUser, 10 ether);
    proof = new bytes32[](1);
    proof[0] = keccak256(abi.encodePacked(newUser));
    vm.startPrank(newUser);
    traitForgeNft.mintToken{ value: 0.5 ether }(proof);
    vm.stopPrank();
    //owner of token 4 is the new user
    assertEq(traitForgeNft.ownerOf(4), newUser);
    //entropy generated is not 0
    assertNotEq(traitForgeNft.getTokenEntropy(4), 0);
  }

}
```

## Recommended Mitigation Steps
Add check in `getNextEntropy` to check that all 3 batch entropy are initialized
```diff
  function getNextEntropy() public onlyAllowedCaller returns (uint256) {
+   require(entropySlots[maxSlotIndex-1]!=0, "Need batch initialization");
    //..
}
```
