- Title: Unchecked Transfer Allows Unauthourised Token Transfers
- Severity: High
- Issue Type: Access Control

### Vulnerability Details
The DualVmToken contract’s transfer(uint256 to, uint256 amount) function lacks proper access control checks to ensure that tokens are transferred from the caller’s own balance. This allows any user to transfer tokens from any address to any to address without authorisation.

### Impact
An attacker can transfer tokens from any address to themselves or others without the token owner’s consent, leading to unauthorised token transfers and potential total loss of user funds.

### Vulnerable Code
```Solidity
    /// @return True if the transfer was successful
    function transfer(uint256 to, uint256 amount) external returns (bool) {
        _transfer(to, amount);
        emit Transfer(msg.sender, to, amount);
        return true;
    }
```

### The _transfer function:
```Solidity
    function _transfer(uint256 to, uint256 amount) private {
        if (to >= STARKNET_FIELD_PRIME) {
            revert InvalidStarknetAddress();
        }
        // Split amount in [low, high]
        uint128 amountLow = uint128(amount);
        uint128 amountHigh = uint128(amount >> 128);


        uint256[] memory transferCallData = new uint256[](3);
        transferCallData[0] = to;
        transferCallData[1] = uint256(amountLow);
        transferCallData[2] = uint256(amountHigh);


        starknetToken.delegatecallCairo("transfer", transferCallData);
    }
```

## Recommended Mitigation Steps
Modify the _transfer function to include the from address and ensure that tokens are only transferred from the caller’s own balance. Use the msg.sender to get the caller’s StarkNet address and pass it to the StarkNet token’s transfer function.
```Solidity
function _transfer(uint256 to, uint256 amount) private {
    uint256;
    senderCalldata[0] = uint256(uint160(msg.sender));
    uint256 senderStarknetAddress =
        abi.decode(kakarot.staticcallCairo("get_starknet_address", senderCalldata), (uint256));

    if (to >= STARKNET_FIELD_PRIME || senderStarknetAddress >= STARKNET_FIELD_PRIME) {
        revert InvalidStarknetAddress();
    }

    // Split amount into [low, high]
    uint128 amountLow = uint128(amount);
    uint128 amountHigh = uint128(amount >> 128);

    uint256;
    transferCallData[0] = senderStarknetAddress; // From address
    transferCallData[1] = to;                    // To address
    transferCallData[2] = uint256(amountLow);
    transferCallData[3] = uint256(amountHigh);

    starknetToken.delegatecallCairo("transfer_from", transferCallData);
}
```
